######代码######

import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.metrics import mean_squared_error
from sklearn.impute import SimpleImputer
from scipy import stats
import joblib
from sklearn.linear_model import Ridge
from sklearn.model_selection import GridSearchCV
import seaborn as sns
import matplotlib.pyplot as plt

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Load datasets
train_data = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/07_sustainable_energy/train.csv')
test_data = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/07_sustainable_energy/test.csv')

# Exclude 'Entity' and 'Year' features if they exist
if 'Entity' in train_data.columns:
    train_data = train_data.drop(columns=['Entity'])
if 'Year' in train_data.columns:
    train_data = train_data.drop(columns=['Year'])
if 'Entity' in test_data.columns:
    test_data = test_data.drop(columns=['Entity'])
if 'Year' in test_data.columns:
    test_data = test_data.drop(columns=['Year'])

# Convert string columns with commas to numeric
for col in train_data.select_dtypes(include=['object']).columns:
    train_data[col] = pd.to_numeric(train_data[col].str.replace(',', ''), errors='coerce')
for col in test_data.select_dtypes(include=['object']).columns:
    test_data[col] = pd.to_numeric(test_data[col].str.replace(',', ''), errors='coerce')

# Handle missing values using mean imputation
imputer = SimpleImputer(strategy='mean')
train_data_imputed = pd.DataFrame(imputer.fit_transform(train_data), columns=train_data.columns)
test_data_imputed = pd.DataFrame(imputer.transform(test_data), columns=test_data.columns)

# Handle outliers using Z-score
z_scores = stats.zscore(train_data_imputed.select_dtypes(include=[float, int]))
abs_z_scores = np.abs(z_scores)
filtered_entries = (abs_z_scores < 3).all(axis=1)
train_data_clean = train_data_imputed[filtered_entries]

# Correlation matrix
corr_matrix = train_data_clean.corr()
sns.heatmap(corr_matrix, annot=True, cmap='coolwarm')
plt.show()

# Feature selection using correlation
target = 'Renewable energy share in the total final energy consumption (%)'
features = train_data_clean.drop(columns=[target])
target_variable = train_data_clean[target]

# Train Ridge Regression model
ridge_model = Ridge(alpha=1.0)
ridge_model.fit(features, target_variable)

# Feature importance
feature_importance = ridge_model.coef_
feature_importance_df = pd.DataFrame({'Feature': features.columns, 'Importance': feature_importance})
feature_importance_df = feature_importance_df.sort_values(by='Importance', ascending=False)
print(feature_importance_df)

# Define hyperparameter search space
param_grid = {
    'alpha': [0.01, 0.1, 1.0, 10.0, 100.0]
}

# Perform grid search
grid_search = GridSearchCV(Ridge(), param_grid, cv=5, scoring='neg_mean_squared_error')
grid_search.fit(features, target_variable)

# Best hyperparameters
best_alpha = grid_search.best_params_['alpha']
print(f"Best alpha: {best_alpha}")

# Train Ridge Regression model with best hyperparameters
best_ridge_model = Ridge(alpha=best_alpha)
best_ridge_model.fit(features, target_variable)

# Make predictions
test_features = test_data_imputed.drop(columns=[target])
predictions = best_ridge_model.predict(test_features)

# Calculate RMSE
test_target_variable = test_data_imputed[target]
rmse = np.sqrt(mean_squared_error(test_target_variable, predictions))
print(f"RMSE: {rmse}")

# Save the model
model_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/models/ridge_regression_model.pkl'
joblib.dump(best_ridge_model, model_path)
