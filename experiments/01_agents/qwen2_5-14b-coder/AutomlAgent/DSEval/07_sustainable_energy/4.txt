import os, random, time, json
import numpy as np
import pandas as pd
from sklearn.metrics import mean_squared_error
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
import joblib

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(train_df, test_df):
    # Exclude 'Entity' and 'Year' features if they exist
    if 'Entity' in train_df.columns:
        train_df = train_df.drop(columns=['Entity'])
    if 'Year' in train_df.columns:
        train_df = train_df.drop(columns=['Year'])
    if 'Entity' in test_df.columns:
        test_df = test_df.drop(columns=['Entity'])
    if 'Year' in test_df.columns:
        test_df = test_df.drop(columns=['Year'])

    # Handle missing values (e.g., impute with mean)
    train_df = train_df.fillna(train_df.mean(numeric_only=True))
    test_df = test_df.fillna(test_df.mean(numeric_only=True))

    # Handle outliers (e.g., capping)
    Q1 = train_df.select_dtypes(include=['int64', 'float64']).quantile(0.25)
    Q3 = train_df.select_dtypes(include=['int64', 'float64']).quantile(0.75)
    IQR = Q3 - Q1
    train_df = train_df[~((train_df.select_dtypes(include=['int64', 'float64']) < (Q1 - 1.5 * IQR)) | (train_df.select_dtypes(include=['int64', 'float64']) > (Q3 + 1.5 * IQR))).any(axis=1)]

    # Separate features and target variable
    X_train = train_df.drop(columns=['Renewable energy share in the total final energy consumption (%)'])
    y_train = train_df['Renewable energy share in the total final energy consumption (%)']
    X_test = test_df.drop(columns=['Renewable energy share in the total final energy consumption (%)'])
    y_test = test_df['Renewable energy share in the total final energy consumption (%)']

    # Identify categorical and numerical features
    numerical_features = X_train.select_dtypes(include=['int64', 'float64']).columns
    categorical_features = X_train.select_dtypes(include=['object']).columns

    # Create a preprocessing pipeline
    preprocessor = ColumnTransformer(
        transformers=[
            ('num', StandardScaler(), numerical_features),
            ('cat', OneHotEncoder(handle_unknown='ignore'), categorical_features)
        ])

    # Apply preprocessing
    X_train_preprocessed = preprocessor.fit_transform(X_train)
    X_test_preprocessed = preprocessor.transform(X_test)

    return X_train_preprocessed, y_train, X_test_preprocessed, y_test, preprocessor

def train_model(X_train_preprocessed, y_train):
    # Initialize the Linear Regression model
    model = LinearRegression()

    # Fit the model to the training data
    model.fit(X_train_preprocessed, y_train)

    return model

def evaluate_model(model, X_test_preprocessed, y_test):
    # Predict on the test set
    y_pred = model.predict(X_test_preprocessed)

    # Calculate RMSE
    rmse = np.sqrt(mean_squared_error(y_test, y_pred))

    performance_scores = {
        'RMSE': rmse
    }

    return performance_scores

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    """
    Main function to execute the tabular regression pipeline.
    """

    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/07_sustainable_energy/train.csv')
    test_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/07_sustainable_energy/test.csv')

    # Step 2. Preprocess the data
    X_train_preprocessed, y_train, X_test_preprocessed, y_test, preprocessor = preprocess_data(train_df, test_df)

    # Step 3. Define required model
    model = train_model(X_train_preprocessed, y_train)

    # Step 4. Evaluate the trained model
    model_performance = evaluate_model(model, X_test_preprocessed, y_test)

    # Step 5. Save the trained model and preprocessor
    model_path = "./agent_workspace/trained_models/linear_regression_model.joblib"
    joblib.dump((model, preprocessor), model_path)

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
