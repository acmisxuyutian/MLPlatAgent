import os
import requests
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler
from scipy import stats
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error
import joblib
from flask import Flask, request, jsonify

# Constants
SEED = 42
np.random.seed(SEED)

# Data Retrieval and Loading
def download_dataset(url, path):
    if not os.path.exists(path):
        response = requests.get(url)
        with open(path, 'wb') as file:
            file.write(response.content)

train_url = 'https://example.com/train.csv'
test_url = 'https://example.com/test.csv'

train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/07_sustainable_energy/train.csv'
test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/07_sustainable_energy/test.csv'

download_dataset(train_url, train_path)
download_dataset(test_url, test_path)

# Data Preprocessing
def preprocess_data(train_path, test_path):
    # Load the datasets
    train_data = pd.read_csv(train_path)
    test_data = pd.read_csv(test_path)

    # Remove 'Entity' and 'Year' columns if they exist
    if 'Entity' in train_data.columns:
        train_data = train_data.drop(columns=['Entity'])
    if 'Year' in train_data.columns:
        train_data = train_data.drop(columns=['Year'])

    if 'Entity' in test_data.columns:
        test_data = test_data.drop(columns=['Entity'])
    if 'Year' in test_data.columns:
        test_data = test_data.drop(columns=['Year'])

    # Handle missing values
    imputer = SimpleImputer(strategy='mean')
    train_data = pd.DataFrame(imputer.fit_transform(train_data.select_dtypes(include=[np.number])), columns=train_data.select_dtypes(include=[np.number]).columns)
    test_data = pd.DataFrame(imputer.transform(test_data.select_dtypes(include=[np.number])), columns=test_data.select_dtypes(include=[np.number]).columns)

    # Handle outliers using Z-score
    z_scores = stats.zscore(train_data.select_dtypes(include=[np.number]))
    abs_z_scores = np.abs(z_scores)
    filtered_entries = (abs_z_scores < 3).all(axis=1)
    train_data = train_data[filtered_entries]

    # Normalize features
    scaler = StandardScaler()
    train_data_scaled = scaler.fit_transform(train_data)
    test_data_scaled = scaler.transform(test_data)

    return train_data_scaled, test_data_scaled

# Model Training and Evaluation
def train_model(train_data_scaled, target_column, train_data):
    # Split the training data into training and validation sets
    X_train, X_val, y_train, y_val = train_test_split(train_data_scaled, train_data[target_column], test_size=0.2, random_state=SEED)

    # Train the Linear Regression model
    model = LinearRegression()
    model.fit(X_train, y_train)

    # Evaluate the model on the validation set
    y_pred = model.predict(X_val)
    rmse = mean_squared_error(y_val, y_pred, squared=False)
    print(f'Validation RMSE: {rmse}')

    return model

# Main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_data, test_data = preprocess_data(train_path, test_path)

    # Step 2. Define required model. You may retrieve model from available hub or library along with pretrained weights (if any).
    # If pretrained or predefined model is not available, please create the model according to the given user's requirements below using PyTorch and relevant libraries.
    model = train_model(train_data, 'Renewable energy share in the total final energy consumption (%)', train_data)

    # Step 3. Save the trained model
    model_save_path = '/remote-home/cs_acmis_xyt/AutomlAgent/models/linear_regression_model.pkl'
    joblib.dump(model, model_save_path)

    # Step 4. Deploy the model using Flask
    app = Flask(__name__)

    # Load the model
    model = joblib.load(model_save_path)

    @app.route('/predict', methods=['POST'])
    def predict():
        data = request.json
        features = np.array(data['features']).reshape(1, -1)
        prediction = model.predict(features)
        return jsonify({'prediction': prediction[0]})

    if __name__ == '__main__':
        app.run(host='0.0.0.0', port=5000)

if __name__ == "__main__":
    main()
