# The following code is for "tabular classification" task.

import os, random, time, json
import numpy as np
import pandas as pd
import torch
import torch.nn as nn
import torch.optim as optim
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.metrics import roc_auc_score
from sklearn.impute import SimpleImputer
from sklearn.pipeline import Pipeline
from sklearn.compose import ColumnTransformer
from sklearn.preprocessing import LabelEncoder

# Ensure reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)

# Data paths
train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/02_cardiovascular_diseases/train.csv'
test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/02_cardiovascular_diseases/test.csv'

# Data preprocessing and feature engineering
def preprocess_data(train_df, test_df):
    # Drop missing values
    train_df = train_df.dropna()
    test_df = test_df.dropna()
    
    # Drop duplicates
    train_df = train_df.drop_duplicates()
    test_df = test_df.drop_duplicates()
    
    # Check for 'Weight' and 'Height' columns
    if 'Weight' not in train_df.columns or 'Height' not in train_df.columns:
        raise ValueError("The dataset is missing 'Weight' or 'Height' columns.")
    if 'Weight' not in test_df.columns or 'Height' not in test_df.columns:
        raise ValueError("The dataset is missing 'Weight' or 'Height' columns.")
    
    # Feature engineering: Calculate BMI and create BMI_Category
    train_df['BMI'] = train_df['Weight'] / ((train_df['Height'] / 100) ** 2)
    test_df['BMI'] = test_df['Weight'] / ((test_df['Height'] / 100) ** 2)
    
    bmi_categories = ['Underweight', 'Normal weight', 'Overweight', 'Obesity']
    train_df['BMI_Category'] = pd.cut(train_df['BMI'], bins=[0, 18.5, 24.9, 29.9, np.inf], labels=bmi_categories)
    test_df['BMI_Category'] = pd.cut(test_df['BMI'], bins=[0, 18.5, 24.9, 29.9, np.inf], labels=bmi_categories)
    
    # Categorical encoding
    encoder = OneHotEncoder(sparse=False)
    train_bmi_encoded = encoder.fit_transform(train_df[['BMI_Category']])
    test_bmi_encoded = encoder.transform(test_df[['BMI_Category']])
    
    # Combine encoded features with other features
    train_df = pd.concat([train_df, pd.DataFrame(train_bmi_encoded, columns=encoder.get_feature_names_out(['BMI_Category']))], axis=1)
    test_df = pd.concat([test_df, pd.DataFrame(test_bmi_encoded, columns=encoder.get_feature_names_out(['BMI_Category']))], axis=1)
    
    # Drop original BMI and BMI_Category columns
    train_df = train_df.drop(columns=['Weight', 'Height', 'BMI', 'BMI_Category'])
    test_df = test_df.drop(columns=['Weight', 'Height', 'BMI', 'BMI_Category'])
    
    # Separate features and target
    X_train = train_df.drop(columns=['Heart_Disease'])
    y_train = train_df['Heart_Disease']
    X_test = test_df.drop(columns=['Heart_Disease'])
    y_test = test_df['Heart_Disease']
    
    # Standardize features
    scaler = StandardScaler()
    X_train = scaler.fit_transform(X_train)
    X_test = scaler.transform(X_test)
    
    return X_train, y_train, X_test, y_test

# Define the model architecture
class FeedforwardNN(nn.Module):
    def __init__(self, input_size):
        super(FeedforwardNN, self).__init__()
        self.fc1 = nn.Linear(input_size, 128)
        self.fc2 = nn.Linear(128, 64)
        self.fc3 = nn.Linear(64, 1)
        self.relu = nn.ReLU()
        self.sigmoid = nn.Sigmoid()
    
    def forward(self, x):
        x = self.relu(self.fc1(x))
        x = self.relu(self.fc2(x))
        x = self.sigmoid(self.fc3(x))
        return x

# Train the model
def train_model(model, train_loader, valid_loader, epochs=50, lr=0.001):
    criterion = nn.BCELoss()
    optimizer = optim.Adam(model.parameters(), lr=lr)
    
    for epoch in range(epochs):
        model.train()
        for inputs, labels in train_loader:
            optimizer.zero_grad()
            outputs = model(inputs.float())
            loss = criterion(outputs.squeeze(), labels.float())
            loss.backward()
            optimizer.step()
        
        model.eval()
        with torch.no_grad():
            valid_loss = 0.0
            for inputs, labels in valid_loader:
                outputs = model(inputs.float())
                loss = criterion(outputs.squeeze(), labels.float())
                valid_loss += loss.item()
        
        print(f'Epoch {epoch+1}, Validation Loss: {valid_loss/len(valid_loader)}')
    
    return model

# Evaluate the model
def evaluate_model(model, test_loader):
    model.eval()
    y_true = []
    y_pred = []
    
    with torch.no_grad():
        for inputs, labels in test_loader:
            outputs = model(inputs.float())
            y_true.extend(labels.numpy())
            y_pred.extend(outputs.squeeze().numpy())
    
    auc_score = roc_auc_score(y_true, y_pred)
    performance_scores = {
        'AUC': auc_score
    }
    
    return performance_scores

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    
    # Step 2. Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
    X_train, y_train, X_test, y_test = preprocess_data(train_df, test_df)
    
    # Split train into train and validation
    X_train, X_valid, y_train, y_valid = train_test_split(X_train, y_train, test_size=0.2, random_state=SEED)
    
    # Convert to PyTorch tensors
    train_tensor = torch.utils.data.TensorDataset(torch.tensor(X_train, dtype=torch.float32), torch.tensor(y_train, dtype=torch.float32))
    valid_tensor = torch.utils.data.TensorDataset(torch.tensor(X_valid, dtype=torch.float32), torch.tensor(y_valid, dtype=torch.float32))
    test_tensor = torch.utils.data.TensorDataset(torch.tensor(X_test, dtype=torch.float32), torch.tensor(y_test, dtype=torch.float32))
    
    train_loader = torch.utils.data.DataLoader(train_tensor, batch_size=32, shuffle=True)
    valid_loader = torch.utils.data.DataLoader(valid_tensor, batch_size=32, shuffle=False)
    test_loader = torch.utils.data.DataLoader(test_tensor, batch_size=32, shuffle=False)
    
    # Step 4. Define required model. You may retrieve model from available hub or library along with pretrained weights (if any).
    # If pretrained or predefined model is not available, please create the model according to the given user's requirements below using PyTorch and relevant libraries.
    input_size = X_train.shape[1]
    model = FeedforwardNN(input_size)
    
    # Step 5. train the retrieved/loaded model using the defined "train_model" function
    model = train_model(model, train_loader, valid_loader)
    
    # Step 6. evaluate the trained model using the defined "evaluate_model" function
    model_performance = evaluate_model(model, test_loader)
    
    # Save the trained model
    torch.save(model.state_dict(), './agent_workspace/trained_models/cardiovascular_diseases_model.pth')
    
    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
