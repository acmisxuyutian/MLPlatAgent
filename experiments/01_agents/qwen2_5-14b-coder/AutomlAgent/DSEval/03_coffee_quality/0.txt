import os, random, time, json
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split, GridSearchCV, cross_val_score
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import mean_squared_error
from sklearn.preprocessing import StandardScaler
import joblib

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data retrieval and preparation
def retrieve_data():
    # Load the training dataset
    df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/03_coffee_quality/train.csv')
    
    # Load the test dataset
    test_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/03_coffee_quality/test.csv')
    
    return df, test_df

# Data preprocessing
def preprocess_data(df, test_df):
    # Drop duplicates
    df = df.drop_duplicates()
    test_df = test_df.drop_duplicates()
    
    # Drop unnecessary columns
    columns_to_drop = ['ID', 'ICO Number', 'Owner', 'Region', 'Certification Contact', 'Certification Address', 'Farm Name', 'Lot Number', 'Mill', 'ICO Number', 'Producer', 'Company', 'Expiration', 'Harvest Year', 'Unnamed: 0', 'Number of Bags', 'Bag Weight', 'In-Country Partner', 'Country of Origin', 'Processing Method', 'Grading Date', 'Variety', 'Status', 'Defects', 'Uniformity', 'Clean Cup', 'Sweetness', 'Certification Body']
    df = df.drop(columns=columns_to_drop, errors='ignore')
    test_df = test_df.drop(columns=columns_to_drop, errors='ignore')
    
    # One-hot encoding for 'Color' column
    df = pd.get_dummies(df, columns=['Color'], drop_first=True)
    test_df = pd.get_dummies(test_df, columns=['Color'], drop_first=True)
    
    return df, test_df

# Model training and optimization
def train_model(df):
    # Split the data into features and target
    X = df.drop(columns=['Total Cup Points'])
    y = df['Total Cup Points']
    
    # Split the data into training and testing sets
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=SEED)
    
    # Initialize the model
    model = RandomForestRegressor(random_state=SEED)
    
    # Hyperparameter tuning
    param_grid = {
        'n_estimators': [100, 200, 300],
        'max_depth': [None, 10, 20, 30],
        'min_samples_split': [2, 5, 10]
    }
    grid_search = GridSearchCV(estimator=model, param_grid=param_grid, cv=3, scoring='neg_mean_squared_error', n_jobs=-1)
    grid_search.fit(X_train, y_train)
    best_model = grid_search.best_estimator_
    
    return best_model, X_test, y_test

# Model evaluation
def evaluate_model(best_model, X_test, y_test, X, y):
    # Predict on the test set
    y_pred = best_model.predict(X_test)
    
    # Calculate mean-squared error
    mse = mean_squared_error(y_test, y_pred)
    print(f'Mean-Squared Error: {mse}')
    
    # Cross-validation
    cv_scores = cross_val_score(best_model, X, y, cv=5, scoring='neg_mean_squared_error')
    print(f'Cross-Validation Mean-Squared Error: {-cv_scores.mean()}')
    
    return mse, -cv_scores.mean()

# Main function to orchestrate the pipeline
def main():
    # Step 1: Retrieve data
    df, test_df = retrieve_data()
    
    # Step 2: Preprocess data
    df, test_df = preprocess_data(df, test_df)
    
    # Step 3: Train model
    best_model, X_test, y_test = train_model(df)
    
    # Step 4: Evaluate model
    mse, cv_mse = evaluate_model(best_model, X_test, y_test, df.drop(columns=['Total Cup Points']), df['Total Cup Points'])
    
    # Step 5: Export the model
    model_path = './agent_workspace/trained_models/coffee_quality_model.joblib'
    joblib.dump(best_model, model_path)
    print(f'Model saved to {model_path}')
    
    # Step 6: Make predictions on the test set
    test_X = test_df.drop(columns=['Total Cup Points'])
    test_y_pred = best_model.predict(test_X)
    
    # Step 7: Export predictions
    predictions_df = pd.DataFrame({'Predicted_Total_Cup_Points': test_y_pred})
    predictions_path = './agent_workspace/predictions/coffee_quality_predictions.csv'
    predictions_df.to_csv(predictions_path, index=False)
    print(f'Predictions saved to {predictions_path}')
    
    return mse, cv_mse

if __name__ == "__main__":
    mse, cv_mse = main()
    print("Model Performance on Test Set:", mse)
    print("Cross-Validation Performance:", cv_mse)
