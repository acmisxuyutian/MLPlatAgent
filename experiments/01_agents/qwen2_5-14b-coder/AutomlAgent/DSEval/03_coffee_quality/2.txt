import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.metrics import mean_squared_error
from sklearn.model_selection import train_test_split, GridSearchCV
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import StandardScaler

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Step 1: Retrieve and Load the Dataset
def load_dataset(url):
    return pd.read_csv(url)

# Step 2: Preprocess the Dataset
def preprocess_data(df):
    # Remove duplicates
    df.drop_duplicates(inplace=True)
    
    # Drop irrelevant columns
    columns_to_drop = ['ID', 'ICO Number', 'Owner', 'Region', 'Certification Contact', 'Certification Address', 'Farm Name', 'Lot Number', 'Mill', 'ICO Number', 'Producer', 'Company', 'Expiration', 'Harvest Year', 'Unnamed: 0', 'Number of Bags', 'Bag Weight', 'In-Country Partner', 'Country of Origin', 'Processing Method', 'Grading Date', 'Variety', 'Status', 'Defects', 'Uniformity', 'Clean Cup', 'Sweetness', 'Certification Body']
    df.drop(columns=columns_to_drop, inplace=True)
    
    # Drop rows with missing values
    df.dropna(inplace=True)
    
    # One-hot encoding for the 'Color' column
    df = pd.get_dummies(df, columns=['Color'], drop_first=True)
    
    return df

# Step 3: Train-Test Split
def split_data(df):
    X = df.drop(columns=['Total Cup Points'])
    y = df['Total Cup Points']
    return train_test_split(X, y, test_size=0.2, random_state=SEED)

# Step 4: Hyperparameter Tuning
def tune_hyperparameters(X_train, y_train):
    param_grid = {
        'n_estimators': [200],
        'max_depth': [20],
        'min_samples_split': [5],
        'min_samples_leaf': [2]
    }
    
    grid_search = GridSearchCV(estimator=RandomForestRegressor(random_state=SEED), param_grid=param_grid, cv=5, scoring='neg_mean_squared_error')
    grid_search.fit(X_train, y_train)
    
    return grid_search.best_estimator_

# Main function to orchestrate the pipeline
def main():
    # Step 1: Load the dataset
    url = 'https://example.com/coffee_quality.csv'  # Replace with the actual URL
    try:
        dataset = load_dataset(url)
    except Exception as e:
        print(f"Error loading dataset: {e}")
        return
    
    # Step 2: Preprocess the dataset
    processed_data = preprocess_data(dataset)
    
    # Step 3: Split the dataset
    X_train, X_test, y_train, y_test = split_data(processed_data)
    
    # Step 4: Hyperparameter tuning
    model = tune_hyperparameters(X_train, y_train)
    
    # Step 5: Evaluate the model
    performance_scores = evaluate_model(model, X_test, y_test)
    
    # Save the trained model
    model_path = "./agent_workspace/trained_models/coffee_quality_model.pkl"
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    import joblib
    joblib.dump(model, model_path)
    
    return performance_scores

if __name__ == "__main__":
    model_performance = main()
