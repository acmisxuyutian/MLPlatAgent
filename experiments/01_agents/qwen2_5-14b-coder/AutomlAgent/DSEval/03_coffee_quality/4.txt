import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import mean_squared_error
from sklearn.preprocessing import StandardScaler
import joblib

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Step 1: Retrieve or load the dataset
train_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/03_coffee_quality/train.csv')
test_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/03_coffee_quality/test.csv')

# Step 2: Preprocess the dataset
# Remove duplicate rows
train_df = train_df.drop_duplicates()
test_df = test_df.drop_duplicates()

# Drop unnecessary columns
columns_to_drop = ['ID', 'ICO Number', 'Owner', 'Region', 'Certification Contact', 'Certification Address', 'Farm Name', 'Lot Number', 'Mill', 'ICO Number', 'Producer', 'Company', 'Expiration', 'Harvest Year', 'Number of Bags', 'Bag Weight', 'In-Country Partner', 'Country of Origin', 'Processing Method', 'Grading Date', 'Variety', 'Status', 'Defects', 'Uniformity', 'Clean Cup', 'Sweetness', 'Certification Body']

# Ensure the columns to drop exist in the DataFrame
columns_to_drop = [col for col in columns_to_drop if col in train_df.columns and col in test_df.columns]

train_df = train_df.drop(columns=columns_to_drop)
test_df = test_df.drop(columns=columns_to_drop)

# Apply one-hot encoding to the categorical column 'Color'
train_df = pd.get_dummies(train_df, columns=['Color'])
test_df = pd.get_dummies(test_df, columns=['Color'])

# Ensure both DataFrames have the same columns
train_df = train_df.reindex(columns=test_df.columns, fill_value=0)

# Step 3: Split the dataset into training and validation sets
X_train = train_df.drop('Total Cup Points', axis=1)
y_train = train_df['Total Cup Points']

X_train, X_val, y_train, y_val = train_test_split(X_train, y_train, test_size=0.2, random_state=SEED)

# Step 4: Train the RandomForestRegressor
model = RandomForestRegressor(n_estimators=100, max_depth=20, min_samples_split=5, min_samples_leaf=2, random_state=SEED)
model.fit(X_train, y_train)

# Step 5: Evaluate the model using Mean-Squared Error (MSE)
y_val_pred = model.predict(X_val)
mse_val = mean_squared_error(y_val, y_val_pred)
print(f"Validation MSE: {mse_val}")

# Step 6: Make predictions on the test set
y_test_pred = model.predict(test_df.drop('Total Cup Points', axis=1))

# Step 7: Evaluate the model on the test set
y_test = test_df['Total Cup Points']
mse_test = mean_squared_error(y_test, y_test_pred)
print(f"Test MSE: {mse_test}")

# Step 8: Save the trained model
model_path = './agent_workspace/trained_models/coffee_quality_model.joblib'
joblib.dump(model, model_path)

# Step 9: Save the predictions
predictions_df = pd.DataFrame({'Predicted_Total_Cup_Points': y_test_pred})
predictions_path = './agent_workspace/predictions/coffee_quality_predictions.csv'
predictions_df.to_csv(predictions_path, index=False)
