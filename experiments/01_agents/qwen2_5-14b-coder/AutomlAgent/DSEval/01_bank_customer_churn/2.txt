# The following code is for "tabular classification" task.

import os, random, time, json
import numpy as np
import pandas as pd
from sklearn.metrics import accuracy_score, f1_score
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import roc_auc_score
import xgboost as xgb
import pickle

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(train_df, test_df):
    # Drop specified columns
    train_df = train_df.drop(['CustomerId', 'Complain', 'Surname'], axis=1)
    test_df = test_df.drop(['CustomerId', 'Complain', 'Surname'], axis=1)
    
    # One-hot encoding for categorical variables
    train_df = pd.get_dummies(train_df, columns=['Geography', 'Card Type'])
    test_df = pd.get_dummies(test_df, columns=['Geography', 'Card Type'])
    
    # Ensure both train and test have the same columns
    missing_cols = set(train_df.columns) - set(test_df.columns)
    for col in missing_cols:
        test_df[col] = 0
    
    # Scaling numerical features
    numerical_cols = train_df.select_dtypes(include=['int64', 'float64']).columns
    scaler = StandardScaler()
    train_df[numerical_cols] = scaler.fit_transform(train_df[numerical_cols])
    test_df[numerical_cols] = scaler.transform(test_df[numerical_cols])
    
    return train_df, test_df

def train_model(train_df, test_df):
    # Split dataset into training and validation sets
    X_train, X_val, y_train, y_val = train_test_split(train_df.drop('Exited', axis=1), train_df['Exited'], test_size=0.2, random_state=42)
    
    # Initialize XGBoost model with optimized hyperparameters and enable categorical handling
    model = xgb.XGBClassifier(max_depth=6, learning_rate=0.1, n_estimators=100, subsample=0.8, colsample_bytree=0.8, random_state=42, enable_categorical=True, early_stopping_rounds=10)
    
    # Train the model
    model.fit(X_train, y_train, eval_set=[(X_val, y_val)], verbose=True)
    
    return model

def evaluate_model(model, test_df):
    # Generate predictions on the testing dataset
    y_pred_proba = model.predict_proba(test_df.drop('Exited', axis=1))[:, 1]
    
    # Calculate the AUC-ROC metric
    auc_roc = roc_auc_score(test_df['Exited'], y_pred_proba)
    
    performance_scores = {
        'AUC-ROC': auc_roc
    }
    
    return performance_scores

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    """
    Main function to execute the tabular classification pipeline.
    """
    
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/01_bank_customer_churn/train.csv'
    test_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/01_bank_customer_churn/test.csv'
    train_df = pd.read_csv(train_data_path)
    test_df = pd.read_csv(test_data_path)
    
    # Step 2. Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
    # Here, the train_loader contains 70% of the `dataset`, the valid_loader contains 20% of the `dataset`, and the test_loader contains 10% of the `dataset`.
    train_loader, valid_loader, test_loader = (
        train_df,
        None,
        test_df
    )  # corresponding to 70%, 20%, 10% of `dataset`
    
    # Step 3. With the split dataset, run data preprocessing and feature engineering (if applicable) using the "preprocess_data" function you defined
    processed_train_df, processed_test_df = preprocess_data(train_loader, test_loader)
    
    # Step 4. Define required model. You may retrieve model from available hub or library along with pretrained weights (if any).
    # If pretrained or predefined model is not available, please create the model according to the given user's requirements below using PyTorch and relevant libraries.
    model = train_model(processed_train_df, processed_test_df)
    
    # Step 5. train the retrieved/loaded model using the defined "train_model" function
    # Step 6. evaluate the trained model using the defined "evaluate_model" function
    model_performance = evaluate_model(model, processed_test_df)
    
    # Save the trained model
    model_path = '/remote-home/cs_acmis_xyt/AutomlAgent/models/xgboost_churn_model.pkl'
    with open(model_path, 'wb') as file:
        pickle.dump(model, file)
    
    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
