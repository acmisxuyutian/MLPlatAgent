import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.metrics import mean_squared_error
from sklearn.linear_model import LinearRegression
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
import joblib

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(train_df, test_df):
    # Identify numerical and categorical features
    numerical_features = train_df.select_dtypes(include=['float64', 'int64']).columns.tolist()
    categorical_features = train_df.select_dtypes(include=['object']).columns.tolist()

    # Impute missing values
    train_df[numerical_features] = train_df[numerical_features].fillna(train_df[numerical_features].mean())
    train_df[categorical_features] = train_df[categorical_features].fillna('Unknown')

    test_df[numerical_features] = test_df[numerical_features].fillna(train_df[numerical_features].mean())
    test_df[categorical_features] = test_df[categorical_features].fillna('Unknown')

    # Create a preprocessing pipeline
    preprocessor = ColumnTransformer(
        transformers=[
            ('num', StandardScaler(), numerical_features),
            ('cat', OneHotEncoder(), categorical_features)
        ])

    # Create a pipeline with preprocessing and linear regression
    pipeline = Pipeline(steps=[('preprocessor', preprocessor),
                               ('regressor', LinearRegression())])

    return pipeline, train_df, test_df

def train_model(pipeline, X_train, y_train):
    # Train the pipeline
    pipeline.fit(X_train, y_train)
    return pipeline

def evaluate_model(model, X_val, y_val, X_test, y_test):
    # Validate the model on the validation set
    y_pred_val = model.predict(X_val)
    rmse_val = mean_squared_error(y_val, y_pred_val, squared=False)

    # Evaluate the model on the test set
    y_pred_test = model.predict(X_test)
    rmse_test = mean_squared_error(y_test, y_pred_test, squared=False)

    # Performance scores
    performance_scores = {
        'Validation RMSE': rmse_val,
        'Test RMSE': rmse_test
    }

    return performance_scores

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    """
    Main function to execute the tabular regression pipeline.
    """

    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/09_housing_price/train.csv')
    test_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/09_housing_price/test.csv')

    # Verify that the target variable 'price' is present in both datasets
    if 'price' not in train_df.columns:
        raise ValueError("Target variable 'price' not found in the training dataset.")
    if 'price' not in test_df.columns:
        raise ValueError("Target variable 'price' not found in the test dataset.")

    # Step 2. Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
    X_train, X_val, y_train, y_val = train_test_split(train_df.drop('price', axis=1), train_df['price'], test_size=0.2, random_state=SEED)

    # Step 3. With the split dataset, run data preprocessing and feature engineering (if applicable) using the "preprocess_data" function you defined
    pipeline, train_df, test_df = preprocess_data(train_df, test_df)

    # Step 4. Define required model. You may retrieve model from available hub or library along with pretrained weights (if any).
    # If pretrained or predefined model is not available, please create the model according to the given user's requirements below using PyTorch and relevant libraries.
    model = pipeline

    # Step 5. train the retrieved/loaded model using the defined "train_model" function
    model = train_model(model, X_train, y_train)

    # Step 6. evaluate the trained model using the defined "evaluate_model" function
    model_performance = evaluate_model(model, X_val, y_val, test_df.drop('price', axis=1), test_df['price'])

    # Step 7. Save the trained model
    os.makedirs('./agent_workspace/trained_models', exist_ok=True)
    joblib.dump(model, './agent_workspace/trained_models/linear_regression_model.pkl')

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
