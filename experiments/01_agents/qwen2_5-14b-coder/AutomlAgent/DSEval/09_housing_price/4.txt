import os, random, time, json
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.impute import SimpleImputer
from sklearn.linear_model import Ridge
from sklearn.metrics import mean_squared_error
import math

# Set the random seed for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data retrieval and initialization
def load_data(train_data_path, test_data_path):
    # Load the datasets
    train_df = pd.read_csv(train_data_path)
    test_df = pd.read_csv(test_data_path)

    # Verify the successful loading
    print(train_df.head())
    print(train_df.info())

    return train_df, test_df

# Data exploration and preprocessing
def preprocess_data(train_df, test_df):
    # Handle missing values
    imputer = SimpleImputer(strategy='mean')
    train_df_imputed = pd.DataFrame(imputer.fit_transform(train_df.select_dtypes(include=[np.number])), columns=train_df.select_dtypes(include=[np.number]).columns)
    train_df_imputed = pd.concat([train_df_imputed, train_df.select_dtypes(exclude=[np.number])], axis=1)
    test_df_imputed = pd.DataFrame(imputer.transform(test_df.select_dtypes(include=[np.number])), columns=test_df.select_dtypes(include=[np.number]).columns)
    test_df_imputed = pd.concat([test_df_imputed, test_df.select_dtypes(exclude=[np.number])], axis=1)

    # Handle outliers (example using Z-score)
    z_scores = np.abs((train_df_imputed.select_dtypes(include=[np.number]) - train_df_imputed.select_dtypes(include=[np.number]).mean()) / train_df_imputed.select_dtypes(include=[np.number]).std())
    train_df_cleaned = train_df_imputed[(z_scores < 3).all(axis=1)]

    # Normalize or scale the features
    scaler = StandardScaler()
    numerical_features = train_df_cleaned.select_dtypes(include=[np.number]).columns
    train_df_scaled = train_df_cleaned.copy()
    train_df_scaled[numerical_features] = scaler.fit_transform(train_df_scaled[numerical_features])
    test_df_scaled = test_df_imputed.copy()
    test_df_scaled[numerical_features] = scaler.transform(test_df_scaled[numerical_features])

    # Encode categorical variables
    categorical_features = train_df_cleaned.select_dtypes(exclude=[np.number]).columns
    encoder = OneHotEncoder(sparse_output=False, handle_unknown='ignore')
    train_df_encoded = pd.concat([train_df_scaled, pd.DataFrame(encoder.fit_transform(train_df_scaled[categorical_features]), columns=encoder.get_feature_names_out(categorical_features))], axis=1)
    train_df_encoded.drop(columns=categorical_features, inplace=True)
    test_df_encoded = pd.concat([test_df_scaled, pd.DataFrame(encoder.transform(test_df_scaled[categorical_features]), columns=encoder.get_feature_names_out(categorical_features))], axis=1)
    test_df_encoded.drop(columns=categorical_features, inplace=True)

    return train_df_encoded, test_df_encoded

# Model implementation and training
def train_model(X_train, y_train):
    # Initialize the linear regression model with Ridge regularization
    ridge_model = Ridge(alpha=0.1)

    # Fit the model to the training data
    ridge_model.fit(X_train, y_train)

    return ridge_model

# Model evaluation
def evaluate_model(model, X_val, y_val):
    # Make predictions on the validation set
    y_val_pred = model.predict(X_val)

    # Evaluate the model's performance using the RMSE metric
    rmse = math.sqrt(mean_squared_error(y_val, y_val_pred))
    print(f'RMSE on validation set: {rmse}')

    return rmse

# Model testing
def test_model(model, test_df_encoded):
    # Split the test dataset into features and target
    X_test = test_df_encoded.drop(columns=['price'])
    y_test = test_df_encoded['price']

    # Make predictions on the test dataset
    y_test_pred = model.predict(X_test)

    # Evaluate the model's performance using the RMSE metric
    rmse_test = math.sqrt(mean_squared_error(y_test, y_test_pred))
    print(f'RMSE on test set: {rmse_test}')

    return rmse_test

    train_df_encoded, test_df_encoded = preprocess_data(train_df, test_df)
