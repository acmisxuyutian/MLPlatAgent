######代码######

import os
import random
import time
import json
import numpy as np
import pandas as pd
import requests
from sklearn.metrics import mean_squared_error
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import StandardScaler
import seaborn as sns
import matplotlib.pyplot as plt

# Set random seed for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data Retrieval and Loading
def load_data(train_url, test_url):
    # Download the datasets
    train_response = requests.get(train_url)
    test_response = requests.get(test_url)

    # Save the datasets to local files
    with open('housing_price_prediction_train.csv', 'wb') as f:
        f.write(train_response.content)

    with open('housing_price_prediction_test.csv', 'wb') as f:
        f.write(test_response.content)

    # Load the datasets into dataframes
    train_df = pd.read_csv('housing_price_prediction_train.csv')
    test_df = pd.read_csv('housing_price_prediction_test.csv')

    return train_df, test_df

# Data Cleaning
def clean_data(train_df, test_df):
    # Handle missing values by imputing with the mean
    train_df_clean = train_df.fillna(train_df.select_dtypes(include=[np.number]).mean())
    test_df_clean = test_df.fillna(test_df.select_dtypes(include=[np.number]).mean())

    # Remove duplicates
    train_df_clean = train_df_clean.drop_duplicates()
    test_df_clean = test_df_clean.drop_duplicates()

    return train_df_clean, test_df_clean

# Feature Engineering
def feature_engineering(train_df_clean, test_df_clean):
    # Ensure the required columns exist before creating new features
    required_columns = ['sqft_living', 'sqft_lot', 'bedrooms']
    if not all(col in train_df_clean.columns for col in required_columns) or not all(col in test_df_clean.columns for col in required_columns):
        raise ValueError("Required columns are missing in the dataset.")

    # Create new features
    train_df_engineered = train_df_clean.copy()
    test_df_engineered = test_df_clean.copy()

    train_df_engineered['total_sqft'] = train_df_engineered['sqft_living'] + train_df_engineered['sqft_lot']
    test_df_engineered['total_sqft'] = test_df_engineered['sqft_living'] + test_df_engineered['sqft_lot']

    # Create interaction terms
    train_df_engineered['sqft_living_bedrooms'] = train_df_engineered['sqft_living'] * train_df_engineered['bedrooms']
    test_df_engineered['sqft_living_bedrooms'] = test_df_engineered['sqft_living'] * test_df_engineered['bedrooms']

    return train_df_engineered, test_df_engineered

# Feature Selection
def feature_selection(train_df_engineered, test_df_engineered):
    # Correlation analysis
    corr_matrix = train_df_engineered.corr()
    sns.heatmap(corr_matrix, annot=True)
    plt.show()

    # Select features based on correlation with the target variable 'price'
    correlation_threshold = 0.3
    selected_features = corr_matrix['price'][abs(corr_matrix['price']) > correlation_threshold].index.tolist()
    selected_features.remove('price')  # Remove the target variable from the list

    train_df_selected = train_df_engineered[selected_features + ['price']]
    test_df_selected = test_df_engineered[selected_features + ['price']]

    return train_df_selected, test_df_selected

# Feature Scaling
def feature_scaling(train_df_selected, test_df_selected):
    # Split the data into features and target variable
    X_train = train_df_selected.drop('price', axis=1)
    y_train = train_df_selected['price']
    X_test = test_df_selected.drop('price', axis=1)
    y_test = test_df_selected['price']

    # Initialize the StandardScaler
    scaler = StandardScaler()

    # Fit and transform the training features
    X_train_scaled = scaler.fit_transform(X_train)

    # Transform the test features
    X_test_scaled = scaler.transform(X_test)

    return X_train_scaled, y_train, X_test_scaled, y_test

# Model Training and Evaluation
def train_and_evaluate_model(X_train_scaled, y_train, X_test_scaled, y_test):
    # Initialize the Linear Regression model
    linear_model = LinearRegression()

    # Fit the model on the training data
    linear_model.fit(X_train_scaled, y_train)

    # Make predictions on the test data
    y_pred = linear_model.predict(X_test_scaled)

    # Calculate RMSE
    rmse = np.sqrt(mean_squared_error(y_test, y_pred))
    print(f'RMSE: {rmse}')

    return linear_model, rmse

# Model Serialization
def save_model(model, model_path):
    import joblib
    joblib.dump(model, model_path)

# Main function to orchestrate the entire pipeline
def main():
    # User-provided URLs
    train_url = 'https://remote-home.cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/09_housing_price/train.csv'
    test_url = 'https://remote-home.cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/09_housing_price/test.csv'

    # Load data
    train_df, test_df = load_data(train_url, test_url)

    # Clean data
    train_df_clean, test_df_clean = clean_data(train_df, test_df)

    # Feature engineering
    train_df_engineered, test_df_engineered = feature_engineering(train_df_clean, test_df_clean)

    # Feature selection
    train_df_selected, test_df_selected = feature_selection(train_df_engineered, test_df_engineered)

    # Feature scaling
    X_train_scaled, y_train, X_test_scaled, y_test = feature_scaling(train_df_selected, test_df_selected)

    # Train and evaluate model
    model, rmse = train_and_evaluate_model(X_train_scaled, y_train, X_test_scaled, y_test)

    # Save the model
    model_path = './agent_workspace/trained_models/linear_regression_model.joblib'
    save_model(model, model_path)

    return {'RMSE': rmse}

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
