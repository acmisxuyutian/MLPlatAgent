import os

import json
import numpy as np
import pandas as pd
from sklearn.metrics import accuracy_score, f1_score, roc_auc_score
from sklearn.model_selection import train_test_split, GridSearchCV
from sklearn.ensemble import RandomForestClassifier

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(train_df, test_df):
    # Remove duplicates
    train_df = train_df.drop_duplicates()
    
    # One-hot encoding for categorical features
    train_df = pd.get_dummies(train_df, columns=['gender', 'smoking_history'])
    test_df = pd.get_dummies(test_df, columns=['gender', 'smoking_history'])
    
    # Ensure both datasets have the same columns
    missing_cols = set(train_df.columns) - set(test_df.columns)
    for col in missing_cols:
        test_df[col] = 0
    
    test_df = test_df[train_df.columns]
    
    return train_df, test_df

def train_model(X_train, y_train):
    # Split the training set into training and validation sets
    X_train, X_val, y_train, y_val = train_test_split(X_train, y_train, test_size=0.2, random_state=SEED)
    
    # Train a random forest classifier
    rf = RandomForestClassifier(random_state=SEED)
    rf.fit(X_train, y_train)
    
    # Hyperparameter tuning
    param_grid = {
        'n_estimators': [100],
        'max_depth': [10],
        'min_samples_split': [2],
        'min_samples_leaf': [1]
    }
    
    grid_search = GridSearchCV(estimator=rf, param_grid=param_grid, cv=5, scoring='roc_auc')
    grid_search.fit(X_train, y_train)
    
    best_rf = grid_search.best_estimator_
    
    return best_rf, X_val, y_val
