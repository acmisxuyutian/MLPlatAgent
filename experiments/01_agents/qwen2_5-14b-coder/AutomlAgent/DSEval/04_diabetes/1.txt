import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.metrics import accuracy_score, f1_score, roc_auc_score, confusion_matrix
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import LabelEncoder
import joblib

# Set the random seed for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(train_data, test_data):
    # Drop duplicate rows from the training dataset
    train_data = train_data.drop_duplicates()

    # One-hot encode categorical variables in the training dataset
    train_data = pd.get_dummies(train_data, columns=['gender', 'smoking_history'])

    # One-hot encode categorical variables in the test dataset
    test_data = pd.get_dummies(test_data, columns=['gender', 'smoking_history'])

    # Ensure both datasets have the same columns after one-hot encoding
    missing_cols = set(train_data.columns) - set(test_data.columns)
    for c in missing_cols:
        test_data[c] = 0

    # Align the columns in the same order
    test_data = test_data[train_data.columns]

    return train_data, test_data

def train_model(train_data, valid_data):
    # Separate features and target variable
    X_train = train_data.drop('diabetes', axis=1)
    y_train = train_data['diabetes']
    X_valid = valid_data.drop('diabetes', axis=1)
    y_valid = valid_data['diabetes']

    # Initialize the Random Forest Classifier with optimal hyperparameters
    model = RandomForestClassifier(n_estimators=200, max_depth=20, min_samples_split=5, min_samples_leaf=2)

    # Train the model on the training dataset
    model.fit(X_train, y_train)

    return model

def evaluate_model(model, test_data):
    # Separate features and target variable
    X_test = test_data.drop('diabetes', axis=1)
    y_test = test_data['diabetes']

    # Make predictions on the test dataset
    y_pred = model.predict(X_test)
    y_pred_proba = model.predict_proba(X_test)[:, 1]

    # Compute the AUC-ROC score
    auc_roc = roc_auc_score(y_test, y_pred_proba)

    # Compute the confusion matrix
    conf_matrix = confusion_matrix(y_test, y_pred)

    # Compute accuracy and F1 score
    accuracy = accuracy_score(y_test, y_pred)
    f1 = f1_score(y_test, y_pred)

    # Store performance scores in a dictionary
    performance_scores = {
        'AUC-ROC': auc_roc,
        'Confusion Matrix': conf_matrix,
        'Accuracy': accuracy,
        'F1 Score': f1
    }

    return performance_scores

    # Step 2. Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
    # Here, the train_loader contains 70% of the `dataset`, the valid_loader contains 20% of the `dataset`, and the test_loader contains 10% of the `dataset`.
    train_data, valid_data = train_test_split(train_data, test_size=0.3, random_state=SEED)

    # TODO: on top of the model training, please run hyperparamter optimization based on the suggested hyperparamters and their values before proceeding to the evaluation step to ensure model's optimality
