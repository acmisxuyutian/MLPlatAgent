import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.metrics import accuracy_score, precision_score, recall_score
from sklearn.model_selection import train_test_split, GridSearchCV
from sklearn.feature_selection import SelectKBest, chi2, mutual_info_classif, RFE
from sklearn.linear_model import LogisticRegression
from sklearn.exceptions import NotFittedError

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data retrieval and loading
def load_data():
    train_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/05_patient_profile/train.csv')
    test_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/05_patient_profile/test.csv')
    return train_df, test_df

# Preprocessing
def preprocess_data(train_df, test_df):
    # Check if 'outcome variable' column exists
    if 'outcome variable' not in train_df.columns or 'outcome variable' not in test_df.columns:
        raise ValueError("The 'outcome variable' column is not found in the datasets.")
    
    X_train = train_df.drop(columns=['outcome variable'])
    y_train = train_df['outcome variable']
    X_test = test_df.drop(columns=['outcome variable'])
    y_test = test_df['outcome variable']
    return X_train, y_train, X_test, y_test

# Feature selection
def select_features(X_train, y_train):
    # Chi-Square Test
    chi2_selector = SelectKBest(score_func=chi2, k=5)
    chi2_selector.fit(X_train, y_train)
    chi2_features = X_train.columns[chi2_selector.get_support()]

    # Mutual Information
    mi_selector = SelectKBest(score_func=mutual_info_classif, k=5)
    mi_selector.fit(X_train, y_train)
    mi_features = X_train.columns[mi_selector.get_support()]

    # Recursive Feature Elimination (RFE)
    rfe = RFE(estimator=LogisticRegression(), n_features_to_select=5)
    rfe.fit(X_train, y_train)
    rfe_features = X_train.columns[rfe.support_]

    # Combine feature selection results
    selected_features = set(chi2_features).intersection(mi_features).intersection(rfe_features)
    return selected_features

# Model training
def train_model(X_train, y_train, param_grid):
    model = LogisticRegression()
    grid_search = GridSearchCV(model, param_grid, cv=5, scoring='accuracy')
    grid_search.fit(X_train, y_train)
    return grid_search.best_estimator_

# Model evaluation
def evaluate_model(model, X_test, y_test):
    try:
        y_pred = model.predict(X_test)
        accuracy = accuracy_score(y_test, y_pred)
        precision = precision_score(y_test, y_pred)
        recall = recall_score(y_test, y_pred)
        performance_scores = {
            'Accuracy': accuracy,
            'Precision': precision,
            'Recall': recall
        }
        return performance_scores
    except NotFittedError:
        raise ValueError("The model has not been fitted yet. Please fit the model before evaluation.")

# Main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1: Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_df, test_df = load_data()

    # Step 2: Preprocess the data
    X_train, y_train, X_test, y_test = preprocess_data(train_df, test_df)

    # Step 3: Feature selection
    selected_features = select_features(X_train, y_train)

    # Step 4: Define required model. You may retrieve model from available hub or library along with pretrained weights (if any).
    # If pretrained or predefined model is not available, please create the model according to the given user's requirements below using PyTorch and relevant libraries.
    param_grid = {
        'C': [0.001, 0.01, 0.1, 1, 10],
        'solver': ['liblinear', 'saga'],
        'max_iter': [100, 200, 300]
    }

    # Step 5: Train the retrieved/loaded model using the defined "train_model" function
    # Model 1: All Features
    model1 = train_model(X_train, y_train, param_grid)

    # Model 2: Top 5 Features
    model2 = train_model(X_train[selected_features], y_train, param_grid)

    # Step 6: Evaluate the trained model using the defined "evaluate_model" function
    model1_performance = evaluate_model(model1, X_test, y_test)
    model2_performance = evaluate_model(model2, X_test[selected_features], y_test)

    return model1_performance, model2_performance

if __name__ == "__main__":
    try:
        model1_performance, model2_performance = main()
        print("Model 1 Performance on Test Set:", model1_performance)
        print("Model 2 Performance on Test Set:", model2_performance)
    except Exception as e:
        print(f"An error occurred: {e}")
