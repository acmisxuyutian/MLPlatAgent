import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import OneHotEncoder
from sklearn.feature_selection import RFE
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import accuracy_score, f1_score
from sklearn.pipeline import Pipeline

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(train_data, test_data):
    # Check for missing values
    print("Missing values in train data:\n", train_data.isnull().sum())
    print("Missing values in test data:\n", test_data.isnull().sum())

    # Select numeric columns
    numeric_cols = train_data.select_dtypes(include=[np.number]).columns

    # Fill missing values with the mean of numeric columns
    imputer = SimpleImputer(strategy='mean')
    train_data[numeric_cols] = imputer.fit_transform(train_data[numeric_cols])
    test_data[numeric_cols] = imputer.transform(test_data[numeric_cols])

    # Perform one-hot encoding on all categorical features
    encoder = OneHotEncoder(sparse=False, drop='first')
    train_data_encoded = pd.DataFrame(encoder.fit_transform(train_data.drop(columns=['outcome variable'])), columns=encoder.get_feature_names_out(train_data.drop(columns=['outcome variable']).columns))
    test_data_encoded = pd.DataFrame(encoder.transform(test_data.drop(columns=['outcome variable'])), columns=encoder.get_feature_names_out(test_data.drop(columns=['outcome variable']).columns))

    # Ensure binary target variable
    train_data_encoded['outcome variable'] = train_data['outcome variable'].astype(int)
    test_data_encoded['outcome variable'] = test_data['outcome variable'].astype(int)

    return train_data_encoded, test_data_encoded

def train_model(model, train_data, target):
    # Train the model
    model.fit(train_data, target)
    return model

def evaluate_model(model, test_data, target):
    # Predict on the test data
    y_pred = model.predict(test_data)
    y_true = target

    # Calculate performance scores
    performance_scores = {
        'ACC': accuracy_score(y_true, y_pred),
        'F1': f1_score(y_true, y_pred)
    }

    return performance_scores

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    """
    Main function to execute the tabular classification pipeline.
    """

    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_data = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/05_patient_profile/train.csv')
    test_data = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/05_patient_profile/test.csv')

    # Verify the datasets
    print("Train data head:\n", train_data.head())
    print("Test data head:\n", test_data.head())

    # Check for expected columns
    print("Train data columns:\n", train_data.columns)
    print("Test data columns:\n", test_data.columns)

    # Step 2. Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
    # Here, the train_loader contains 70% of the `dataset`, the valid_loader contains 20% of the `dataset`, and the test_loader contains 10% of the `dataset`.
    train_data, valid_data = train_test_split(train_data, test_size=0.3, random_state=SEED)
    valid_data, test_data = train_test_split(valid_data, test_size=0.33, random_state=SEED)

    # Step 3. With the split dataset, run data preprocessing and feature engineering (if applicable) using the "preprocess_data" function you defined
    train_data_encoded, valid_data_encoded = preprocess_data(train_data, valid_data)
    test_data_encoded, _ = preprocess_data(test_data, test_data)

    # Step 4. Define required model. You may retrieve model from available hub or library along with pretrained weights (if any).
    # If pretrained or predefined model is not available, please create the model according to the given user's requirements below using PyTorch and relevant libraries.
    model = Pipeline([
        ('feature_selection', RFE(LogisticRegression(), n_features_to_select=5)),
        ('classification', LogisticRegression())
    ])

    # Step 5. train the retrieved/loaded model using the defined "train_model" function
    # on top of the model training, please run hyperparamter optimization based on the suggested hyperparamters and their values before proceeding to the evaluation step to ensure model's optimality
    model = train_model(model, train_data_encoded.drop(columns=['outcome variable']), train_data_encoded['outcome variable'])

    # Step 6. evaluate the trained model using the defined "evaluate_model" function
    model_performance = evaluate_model(model, test_data_encoded.drop(columns=['outcome variable']), test_data_encoded['outcome variable'])

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
