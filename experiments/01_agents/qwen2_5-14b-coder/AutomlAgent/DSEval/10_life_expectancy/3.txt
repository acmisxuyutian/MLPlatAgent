import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import LabelEncoder, StandardScaler
from sklearn.linear_model import LinearRegression
from sklearn.metrics import r2_score
import joblib

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(train_df, test_df):
    # Handle missing values
    imputer = SimpleImputer(strategy='mean')
    train_df['GDP'] = imputer.fit_transform(train_df[['GDP']])
    test_df['GDP'] = imputer.transform(test_df[['GDP']])

    # Feature construction: Categorize GDP into bins
    train_df['GDP_bins'] = pd.cut(train_df['GDP'], bins=[0, 1000, 10000, float('inf')], labels=['Low GDP', 'Medium GDP', 'High GDP'])
    test_df['GDP_bins'] = pd.cut(test_df['GDP'], bins=[0, 1000, 10000, float('inf')], labels=['Low GDP', 'Medium GDP', 'High GDP'])

    # Encode categorical features
    label_encoder = LabelEncoder()
    train_df['GDP_bins'] = label_encoder.fit_transform(train_df['GDP_bins'])
    test_df['GDP_bins'] = label_encoder.transform(test_df['GDP_bins'])

    # Normalization/Standardization
    scaler = StandardScaler()
    train_df[['GDP', 'GDP_bins']] = scaler.fit_transform(train_df[['GDP', 'GDP_bins']])
    test_df[['GDP', 'GDP_bins']] = scaler.transform(test_df[['GDP', 'GDP_bins']])

    return train_df, test_df

def train_model(train_df):
    # Prepare the features and target
    X_train = train_df[['GDP_bins']]
    y_train = train_df['life_expectancy']

    # Train the Linear Regression model
    linear_model = LinearRegression()
    linear_model.fit(X_train, y_train)

    return linear_model

def evaluate_model(model, test_df):
    # Prepare the test features
    X_test = test_df[['GDP_bins']]

    # Make predictions
    y_pred = model.predict(X_test)

    # Evaluate the model
    r2 = r2_score(test_df['life_expectancy'], y_pred)
    performance_scores = {
        'R-squared': r2
    }

    return performance_scores

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    """
    Main function to execute the tabular regression pipeline.
    """

    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/10_life_expectancy/train.csv'
    test_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/10_life_expectancy/test.csv'

    train_df = pd.read_csv(train_data_path)
    test_df = pd.read_csv(test_data_path)

    # Verify the dataset
    print(train_df.head())
    print(train_df.info())
    print(test_df.head())
    print(test_df.info())

    # Ensure the necessary columns are present
    required_columns = ['life_expectancy', 'GDP']
    if not all(column in train_df.columns for column in required_columns):
        raise ValueError(f"Training dataset is missing required columns: {required_columns}")
    if not all(column in test_df.columns for column in required_columns):
        raise ValueError(f"Test dataset is missing required columns: {required_columns}")

    # Step 2. Preprocess the data
    train_df, test_df = preprocess_data(train_df, test_df)

    # Step 3. Define and train the model
    model = train_model(train_df)

    # Step 4. Evaluate the trained model
    model_performance = evaluate_model(model, test_df)

    # Step 5. Save the trained model
    model_save_path = './agent_workspace/trained_models/linear_regression_model.pkl'
    os.makedirs(os.path.dirname(model_save_path), exist_ok=True)
    joblib.dump(model, model_save_path)

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
