######代码######

import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.metrics import r2_score
import joblib

# Set random seed for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data paths
train_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/10_life_expectancy/train.csv'
test_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/10_life_expectancy/test.csv'

# Load the dataset
train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# Check if 'life_expectancy' column exists in the datasets
if 'life_expectancy' not in train_df.columns or 'life_expectancy' not in test_df.columns:
    raise ValueError("The 'life_expectancy' column is missing from the dataset(s).")

# Preprocess the dataset
# Define GDP bins
bins = [0, 1000, 10000, float('inf')]
labels = ['Low GDP', 'Medium GDP', 'High GDP']
train_df['GDP_bins'] = pd.cut(train_df['GDP'], bins=bins, labels=labels, right=False)
test_df['GDP_bins'] = pd.cut(test_df['GDP'], bins=bins, labels=labels, right=False)

# Handle missing values
imputer = SimpleImputer(strategy='mean')
train_df['GDP'] = imputer.fit_transform(train_df[['GDP']])
test_df['GDP'] = imputer.transform(test_df[['GDP']])
train_df['life_expectancy'] = imputer.fit_transform(train_df[['life_expectancy']])
test_df['life_expectancy'] = imputer.transform(test_df[['life_expectancy']])

# Split the dataset into training, validation, and test sets
train_df, val_df = train_test_split(train_df, test_size=0.2, random_state=SEED)

# Define features and target variable for training
X_train = train_df[['GDP']]
y_train = train_df['life_expectancy']

# Initialize and fit the Linear Regression model
model = LinearRegression()
model.fit(X_train, y_train)

# Make predictions on the test set
X_test = test_df[['GDP']]
y_pred = model.predict(X_test)

# Evaluate the performance of the model using R-squared score
y_test = test_df['life_expectancy']
r2 = r2_score(y_test, y_pred)
print(f'R-squared score: {r2}')

# Save the trained model
model_path = './agent_workspace/trained_models/linear_regression_model.pkl'
joblib.dump(model, model_path)
