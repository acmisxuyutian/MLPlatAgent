import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_absolute_error, mean_squared_error, mean_squared_log_error

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(train_df, test_df):
    # Handle missing values
    numeric_cols = train_df.select_dtypes(include=['number']).columns
    train_df[numeric_cols] = train_df[numeric_cols].fillna(train_df[numeric_cols].mean())
    test_df[numeric_cols] = test_df[numeric_cols].fillna(test_df[numeric_cols].mean())
    
    # Remove duplicates
    train_df.drop_duplicates(inplace=True)
    test_df.drop_duplicates(inplace=True)
    
    # Feature construction: categorize GDP into bins using quantile binning
    train_df['GDP_bin'] = pd.qcut(train_df['GDP'], q=4, labels=False, duplicates='drop')
    test_df['GDP_bin'] = pd.qcut(test_df['GDP'], q=4, labels=False, duplicates='drop')
    
    return train_df, test_df

def train_model(train_df, target_col):
    # Define features and target
    X_train = train_df.drop(columns=[target_col])
    y_train = train_df[target_col]
    
    # Initialize and train the model
    model = LinearRegression()
    model.fit(X_train, y_train)
    
    return model

def evaluate_model(model, test_df, target_col):
    # Define features and target
    X_test = test_df.drop(columns=[target_col])
    y_test = test_df[target_col]
    
    # Predict
    y_pred = model.predict(X_test)
    
    # Evaluate
    performance_scores = {
        "MAE": mean_absolute_error(y_test, y_pred),
        "MSE": mean_squared_error(y_test, y_pred),
        "RMSLE": np.sqrt(mean_squared_log_error(y_test, y_pred))
    }
    
    return performance_scores

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1. Retrieve or load a dataset from user's local storage
    train_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/10_life_expectancy/train.csv')
    test_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/10_life_expectancy/test.csv')
    
    # Step 2. Preprocess the dataset
    train_df, test_df = preprocess_data(train_df, test_df)
    
    # Step 3. Define required model
    target_col = 'Life expectancy'  # Assuming 'Life expectancy' is the target column
    model = train_model(train_df, target_col)
    
    # Step 4. Evaluate the trained model
    model_performance = evaluate_model(model, test_df, target_col)
    
    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
