import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import mean_absolute_error
from sklearn.impute import SimpleImputer
import joblib

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data retrieval and preprocessing
def preprocess_data(train_df, test_df):
    # Separate the target variable 'age'
    y_train = train_df['age']
    y_test = test_df['age']
    train_df = train_df.drop('age', axis=1)
    test_df = test_df.drop('age', axis=1)
    
    # One-hot encoding
    train_df_encoded = pd.get_dummies(train_df, drop_first=True)
    test_df_encoded = pd.get_dummies(test_df, drop_first=True)
    
    # Align columns
    missing_cols = set(train_df_encoded.columns) - set(test_df_encoded.columns)
    for c in missing_cols:
        test_df_encoded[c] = 0
    
    missing_cols = set(test_df_encoded.columns) - set(train_df_encoded.columns)
    for c in missing_cols:
        train_df_encoded[c] = 0
    
    test_df_encoded = test_df_encoded[train_df_encoded.columns]
    
    # Handle missing values
    imputer = SimpleImputer(strategy='median')
    train_df_imputed = pd.DataFrame(imputer.fit_transform(train_df_encoded), columns=train_df_encoded.columns)
    test_df_imputed = pd.DataFrame(imputer.transform(test_df_encoded), columns=test_df_encoded.columns)
    
    return train_df_imputed, test_df_imputed, y_train, y_test, imputer

def train_model(X_train, y_train):
    # Initialize the Random Forest Regressor
    rf_regressor = RandomForestRegressor(n_estimators=100, max_depth=10, min_samples_split=2, random_state=SEED)
    
    # Train the model
    rf_regressor.fit(X_train, y_train)
    
    return rf_regressor

def evaluate_model(model, X_test, y_test):
    # Evaluate the model on the test dataset
    y_pred = model.predict(X_test)
    mae = mean_absolute_error(y_test, y_pred)
    
    performance_scores = {
        'MAE': mae
    }
    
    return performance_scores

def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/crab_age/train.csv'
    test_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/crab_age/test.csv'
    
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    
    # Step 2. Preprocess the data
    X_train, X_test, y_train, y_test, imputer = preprocess_data(train_df, test_df)
    
    # Step 3. Define and train the model
    model = train_model(X_train, y_train)
    
    # Step 4. Evaluate the model
    model_performance = evaluate_model(model, X_test, y_test)
    
    # Step 5. Save the trained model and preprocessing pipeline
    model_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/models/crab_age_rf_model.joblib'
    imputer_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/models/crab_age_imputer.joblib'
    
    joblib.dump(model, model_path)
    joblib.dump(imputer, imputer_path)
    
    # Step 6. Generate predictions on the test dataset using the deployed model
    loaded_rf_regressor = joblib.load(model_path)
    loaded_imputer = joblib.load(imputer_path)
    
    test_df_encoded = pd.get_dummies(test_df, drop_first=True)
    test_df_imputed = pd.DataFrame(loaded_imputer.transform(test_df_encoded), columns=test_df_encoded.columns)
    
    y_pred = loaded_rf_regressor.predict(test_df_imputed)
    
    # Step 7. Save the prediction results
    predictions_df = pd.DataFrame({'predicted_age': y_pred})
    predictions_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/predictions/crab_age_predictions.csv'
    predictions_df.to_csv(predictions_path, index=False)
    
    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
