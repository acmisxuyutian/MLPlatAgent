######代码######

import os
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import mean_absolute_error
from sklearn.preprocessing import OneHotEncoder
import numpy as np

# Constants
SEED = 42
np.random.seed(SEED)

# Data paths
TRAIN_DATASET_PATH = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/crab_age/train.csv'
TEST_DATASET_PATH = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/crab_age/test.csv'

# Target variable
TARGET_VARIABLE = 'age'

# Function to load data
def load_data(train_path, test_path):
    train_data = pd.read_csv(train_path)
    test_data = pd.read_csv(test_path)
    return train_data, test_data

# Function to preprocess data
def preprocess_data(train_data, test_data):
    # Identify categorical columns
    categorical_cols = train_data.select_dtypes(include=['object']).columns
    
    # One-hot encode categorical columns
    encoder = OneHotEncoder(handle_unknown='ignore')
    train_encoded = encoder.fit_transform(train_data[categorical_cols])
    test_encoded = encoder.transform(test_data[categorical_cols])
    
    # Create DataFrame from encoded data
    train_encoded_df = pd.DataFrame(train_encoded.toarray(), columns=encoder.get_feature_names_out(categorical_cols))
    test_encoded_df = pd.DataFrame(test_encoded.toarray(), columns=encoder.get_feature_names_out(categorical_cols))
    
    # Concatenate encoded data with original data
    train_data = pd.concat([train_data.drop(columns=categorical_cols), train_encoded_df], axis=1)
    test_data = pd.concat([test_data.drop(columns=categorical_cols), test_encoded_df], axis=1)
    
    return train_data, test_data

# Function to split data into features and target
def split_data(data, target):
    X = data.drop(columns=[target])
    y = data[target]
    return X, y

# Function to train model
def train_model(X_train, y_train):
    model = RandomForestRegressor(random_state=SEED)
    model.fit(X_train, y_train)
    return model

# Function to evaluate model
def evaluate_model(model, X_test, y_test):
    y_pred = model.predict(X_test)
    mae = mean_absolute_error(y_test, y_pred)
    performance_scores = {
        'MAE': mae
    }
    return performance_scores

# Function to save predictions
def save_predictions(predictions, filename):
    predictions_df = pd.DataFrame(predictions, columns=['predicted_age'])
    predictions_df.to_csv(filename, index=False)

# Function to save the trained model
def save_model(model, model_path):
    import joblib
    joblib.dump(model, model_path)

# Main function
def main():
    # Step 1: Load data
    train_data, test_data = load_data(TRAIN_DATASET_PATH, TEST_DATASET_PATH)
    
    # Step 2: Preprocess data
    train_data, test_data = preprocess_data(train_data, test_data)
    
    # Step 3: Split data into features and target
    X_train, y_train = split_data(train_data, TARGET_VARIABLE)
    X_test, y_test = split_data(test_data, TARGET_VARIABLE)
    
    # Step 4: Train model
    model = train_model(X_train, y_train)
    
    # Step 5: Evaluate model
    model_performance = evaluate_model(model, X_test, y_test)
    
    # Step 6: Save predictions
    predictions = model.predict(X_test)
    save_predictions(predictions, 'crab_age_predictions.csv')
    
    # Step 7: Save the trained model
    model_path = './agent_workspace/trained_models/crab_age_model.pkl'
    save_model(model, model_path)
    
    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
