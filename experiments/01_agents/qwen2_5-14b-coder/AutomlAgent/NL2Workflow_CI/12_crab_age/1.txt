######代码######

import os
import pandas as pd
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import mean_absolute_error
import joblib

# Define the source paths
train_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/crab_age/train.csv'
test_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/crab_age/test.csv'

# Ensure the data directory exists
os.makedirs('data', exist_ok=True)

# Load the datasets
train_df = pd.read_csv(train_path)
test_df = pd.read_csv(test_path)

# Separate the target variable
y_train = train_df['age']
y_test = test_df['age']

# Remove the target variable from the datasets
train_df = train_df.drop(columns=['age'])
test_df = test_df.drop(columns=['age'])

# Identify categorical columns
categorical_cols = train_df.select_dtypes(include=['object']).columns

# Apply one-hot encoding
train_df_encoded = pd.get_dummies(train_df, columns=categorical_cols)
test_df_encoded = pd.get_dummies(test_df, columns=categorical_cols)

# Ensure both datasets have the same columns
train_df_encoded, test_df_encoded = train_df_encoded.align(test_df_encoded, join='inner', axis=1)

# Add the target variable back to the datasets
train_df_encoded['age'] = y_train
test_df_encoded['age'] = y_test

# Define the target variable and features
X_train = train_df_encoded.drop(columns=['age'])
y_train = train_df_encoded['age']
X_test = test_df_encoded.drop(columns=['age'])
y_test = test_df_encoded['age']

# Initialize the Random Forest model with optimized hyperparameters
model = RandomForestRegressor(n_estimators=150, max_depth=12, min_samples_split=3, random_state=42)

# Train the model
model.fit(X_train, y_train)

# Make predictions on the test set
y_pred = model.predict(X_test)

# Evaluate the model using MAE
mae = mean_absolute_error(y_test, y_pred)
print(f'Mean Absolute Error: {mae}')

# Save the predictions to a CSV file
predictions_df = pd.DataFrame({'predicted_age': y_pred})
predictions_df.to_csv('predictions/crab_age_predictions.csv', index=False)

# Save the trained model
model_path = './agent_workspace/trained_models/crab_age_model.pkl'
os.makedirs(os.path.dirname(model_path), exist_ok=True)
joblib.dump(model, model_path)
print(f'Model saved to {model_path}')
