import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.metrics import mean_squared_error
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import MinMaxScaler, PolynomialFeatures
from scipy import stats
import torch
import torch.nn as nn
import torch.optim as optim

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(train_data, test_data):
    # Identify target variable and feature columns
    target_column = 'media_cost'
    feature_columns = [col for col in train_data.columns if col != target_column]

    # Impute missing values
    imputer = SimpleImputer(strategy='mean')
    train_data_imputed = pd.DataFrame(imputer.fit_transform(train_data[feature_columns]), columns=feature_columns)
    test_data_imputed = pd.DataFrame(imputer.transform(test_data[feature_columns]), columns=feature_columns)

    # Remove outliers using IQR
    Q1 = train_data_imputed.quantile(0.25)
    Q3 = train_data_imputed.quantile(0.75)
    IQR = Q3 - Q1
    train_data_cleaned = train_data_imputed[~((train_data_imputed < (Q1 - 1.5 * IQR)) | (train_data_imputed > (Q3 + 1.5 * IQR))).any(axis=1)]

    # Normalize numerical features
    scaler = MinMaxScaler()
    train_data_normalized = pd.DataFrame(scaler.fit_transform(train_data_cleaned), columns=train_data_cleaned.columns)
    test_data_normalized = pd.DataFrame(scaler.transform(test_data_imputed), columns=test_data_imputed.columns)

    # Create interaction terms
    interaction_terms = []
    for i in range(len(feature_columns)):
        for j in range(i + 1, len(feature_columns)):
            interaction_term = feature_columns[i] + '_' + feature_columns[j]
            train_data_normalized[interaction_term] = train_data_normalized[feature_columns[i]] * train_data_normalized[feature_columns[j]]
            test_data_normalized[interaction_term] = test_data_normalized[feature_columns[i]] * test_data_normalized[feature_columns[j]]
            interaction_terms.append(interaction_term)

    # Generate polynomial features
    poly = PolynomialFeatures(degree=2, include_bias=False)
    train_data_poly = poly.fit_transform(train_data_normalized)
    test_data_poly = poly.transform(test_data_normalized)

    return train_data_poly, test_data_poly, train_data[target_column].values, test_data[target_column].values

# Define the MLP model
class MLP(nn.Module):
    def __init__(self, input_size):
        super(MLP, self).__init__()
        self.fc1 = nn.Linear(input_size, 128)
        self.fc2 = nn.Linear(128, 64)
        self.fc3 = nn.Linear(64, 32)
        self.fc4 = nn.Linear(32, 1)

    def forward(self, x):
        x = torch.relu(self.fc1(x))
        x = torch.relu(self.fc2(x))
        x = torch.relu(self.fc3(x))
        x = self.fc4(x)
        return x

# Train the model
def train_model(model, train_data, train_labels, epochs=150, lr=0.0005):
    criterion = nn.MSELoss()
    optimizer = optim.Adam(model.parameters(), lr=lr)

    for epoch in range(epochs):
        model.train()
        optimizer.zero_grad()
        outputs = model(torch.tensor(train_data, dtype=torch.float32))
        loss = criterion(outputs, torch.tensor(train_labels.reshape(-1, 1), dtype=torch.float32))
        loss.backward()
        optimizer.step()

    return model

# Evaluate the model
def evaluate_model(model, test_data, test_labels):
    model.eval()
    with torch.no_grad():
        predictions = model(torch.tensor(test_data, dtype=torch.float32)).numpy()

    rmse = mean_squared_error(test_labels, predictions, squared=False)
    performance_scores = {
        'RMSE': rmse
    }
    return performance_scores

# Main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_data = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/media_campaign_cost/train.csv')
    test_data = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/media_campaign_cost/test.csv')

    # Step 2. Preprocess the data
    train_data_poly, test_data_poly, train_labels, test_labels = preprocess_data(train_data, test_data)

    # Step 3. Define the model
    model = MLP(train_data_poly.shape[1])

    # Step 4. Train the model
    model = train_model(model, train_data_poly, train_labels)

    # Step 5. Evaluate the model
    model_performance = evaluate_model(model, test_data_poly, test_labels)

    # Step 6. Save the model
    model_path = '/remote-home/cs_acmis_xyt/AutomlAgent/./agent_workspace/trained_models/media_campaign_cost_model.pth'
    torch.save(model.state_dict(), model_path)

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
