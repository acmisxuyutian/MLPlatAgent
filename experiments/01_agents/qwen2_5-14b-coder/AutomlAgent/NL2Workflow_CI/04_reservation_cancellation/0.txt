######代码######

import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import OneHotEncoder, StandardScaler
from sklearn.metrics import accuracy_score, f1_score, roc_auc_score
import lightgbm as lgb
from sklearn.model_selection import GridSearchCV
import joblib

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Step 1: Retrieve or load a dataset from hub (if available) or user's local storage (if given)
def load_datasets(train_path, test_path):
    train_data = pd.read_csv(train_path)
    test_data = pd.read_csv(test_path)
    
    return train_data, test_data

# Step 2: Preprocess the retrieved dataset(s)
def preprocess_data(train_data, test_data):
    # Identify categorical and numerical columns
    categorical_cols = train_data.select_dtypes(include=['object']).columns
    numerical_cols = train_data.select_dtypes(exclude=['object']).columns
    
    # Handle missing values
    imputer_num = SimpleImputer(strategy='median')
    imputer_cat = SimpleImputer(strategy='most_frequent')

    train_data[numerical_cols] = imputer_num.fit_transform(train_data[numerical_cols])
    test_data[numerical_cols] = imputer_num.transform(test_data[numerical_cols])

    train_data[categorical_cols] = imputer_cat.fit_transform(train_data[categorical_cols])
    test_data[categorical_cols] = imputer_cat.transform(test_data[categorical_cols])
    
    # Encode categorical variables
    encoder = OneHotEncoder(handle_unknown='ignore', sparse=False)
    train_encoded = encoder.fit_transform(train_data[categorical_cols])
    test_encoded = encoder.transform(test_data[categorical_cols])
    
    # Normalize numerical features
    scaler = StandardScaler()
    train_data[numerical_cols] = scaler.fit_transform(train_data[numerical_cols])
    test_data[numerical_cols] = scaler.transform(test_data[numerical_cols])
    
    # Combine encoded features with numerical features
    train_data = pd.concat([pd.DataFrame(train_encoded, columns=encoder.get_feature_names_out(categorical_cols)), train_data[numerical_cols]], axis=1)
    test_data = pd.concat([pd.DataFrame(test_encoded, columns=encoder.get_feature_names_out(categorical_cols)), test_data[numerical_cols]], axis=1)
    
    return train_data, test_data

# Step 3: Train the retrieved model
def train_model(train_data, valid_data, params):
    X_train = train_data.drop('cancellation', axis=1)
    y_train = train_data['cancellation']
    X_valid = valid_data.drop('cancellation', axis=1)
    y_valid = valid_data['cancellation']
    
    lgb_train = lgb.Dataset(X_train, label=y_train)
    lgb_valid = lgb.Dataset(X_valid, label=y_valid, reference=lgb_train)
    
    gbm = lgb.train(params,
                    lgb_train,
                    num_boost_round=100,
                    valid_sets=[lgb_train, lgb_valid],
                    early_stopping_rounds=10,
                    verbose_eval=False)
    
    return gbm

# Step 4: Evaluate the trained model
def evaluate_model(model, test_data):
    X_test = test_data.drop('cancellation', axis=1)
    y_test = test_data['cancellation']
    
    y_pred = model.predict(X_test, num_iteration=model.best_iteration)
    y_pred_binary = (y_pred > 0.5).astype(int)
    
    performance_scores = {
        'ACC': accuracy_score(y_test, y_pred_binary),
        'F1': f1_score(y_test, y_pred_binary),
        'AUC-ROC': roc_auc_score(y_test, y_pred)
    }
    
    return performance_scores

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Define dataset paths
    train_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/reservation_cancellation/train.csv'
    test_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/reservation_cancellation/test.csv'
    
    # Load datasets
    train_data, test_data = load_datasets(train_path, test_path)
    
    # Split train data into train and validation sets
    train_data, valid_data = train_test_split(train_data, test_size=0.2, random_state=SEED)
    
    # Preprocess data
    train_data, valid_data = preprocess_data(train_data, valid_data)
    test_data = preprocess_data(None, test_data)[1]  # Only preprocess test data
    
    # Define LightGBM parameters
    param_grid = {
        'num_leaves': [31],
        'learning_rate': [0.05],
        'n_estimators': [200],
        'max_depth': [7],
        'min_child_samples': [20]
    }
    
    # Train model
    lgbm = lgb.LGBMClassifier(objective='binary', metric='auc')
    grid_search = GridSearchCV(lgbm, param_grid, cv=3, scoring='roc_auc', n_jobs=-1)
    grid_search.fit(train_data.drop('cancellation', axis=1), train_data['cancellation'])
    best_params = grid_search.best_params_
    
    best_lgbm = lgb.LGBMClassifier(**best_params, objective='binary', metric='auc')
    best_lgbm.fit(train_data.drop('cancellation', axis=1), train_data['cancellation'])
    
    # Evaluate model
    model_performance = evaluate_model(best_lgbm, test_data)
    
    # Save the trained model
    model_save_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/models/lightgbm_cancellation_model.pkl'
    joblib.dump(best_lgbm, model_save_path)
    
    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
