import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import roc_auc_score
from imblearn.over_sampling import SMOTE
import lightgbm as lgb
import joblib
from sklearn.model_selection import GridSearchCV

# Set random seed for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(train_data, test_data):
    # Handle missing values
    missing_columns = train_data.columns[train_data.isnull().any()]
    imputer = SimpleImputer(strategy='mean')
    train_data[missing_columns] = imputer.fit_transform(train_data[missing_columns])
    test_data[missing_columns] = imputer.transform(test_data[missing_columns])

    # Encode categorical variables
    train_data = pd.get_dummies(train_data, drop_first=True)
    test_data = pd.get_dummies(test_data, drop_first=True)

    # Ensure both train and test data have the same columns
    test_data = test_data.reindex(columns=train_data.columns, fill_value=0)

    # Normalize/Standardize features
    numerical_columns = train_data.select_dtypes(include=[np.number]).columns
    scaler = StandardScaler()
    train_data[numerical_columns] = scaler.fit_transform(train_data[numerical_columns])
    test_data[numerical_columns] = scaler.transform(test_data[numerical_columns])

    return train_data, test_data

def train_model(train_data, valid_data, test_data):
    # Split the dataset into training and validation sets
    X_train, y_train = train_data.drop('target_column', axis=1), train_data['target_column']
    X_valid, y_valid = valid_data.drop('target_column', axis=1), valid_data['target_column']
    X_test, y_test = test_data.drop('target_column', axis=1), test_data['target_column']

    # Augment the dataset using SMOTE
    smote = SMOTE(random_state=SEED)
    X_resampled, y_resampled = smote.fit_resample(X_train, y_train)

    # Define hyperparameters
    param_grid = {
        'learning_rate': [0.01, 0.05, 0.1],
        'n_estimators': [50, 100, 200],
        'max_depth': [5, 10, 15]
    }

    # Train the model with hyperparameter tuning
    model = lgb.LGBMClassifier(random_state=SEED)
    grid_search = GridSearchCV(estimator=model, param_grid=param_grid, cv=3, scoring='roc_auc', n_jobs=-1)
    grid_search.fit(X_resampled, y_resampled)

    # Get the best model
    best_model = grid_search.best_estimator_

    return best_model, X_test, y_test

def evaluate_model(model, X_test, y_test):
    # Evaluate the model
    y_pred_proba = model.predict_proba(X_test)[:, 1]
    auc_roc = roc_auc_score(y_test, y_pred_proba)

    performance_scores = {
        'AUC-ROC': auc_roc
    }

    return performance_scores

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    """
    Main function to execute the tabular classification pipeline.
    """

    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_data = pd.read_csv(r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/reservation_cancellation/train.csv')
    test_data = pd.read_csv(r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/reservation_cancellation/test.csv')

    # Step 2. Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
    train_data, valid_data = train_test_split(train_data, test_size=0.2, random_state=SEED)

    # Step 3. With the split dataset, run data preprocessing and feature engineering (if applicable) using the "preprocess_data" function you defined
    train_data, valid_data = preprocess_data(train_data, valid_data)
    test_data, _ = preprocess_data(test_data, pd.DataFrame())

    # Step 4. Define required model. You may retrieve model from available hub or library along with pretrained weights (if any).
    # If pretrained or predefined model is not available, please create the model according to the given user's requirements below using PyTorch and relevant libraries.
    model, X_test, y_test = train_model(train_data, valid_data, test_data)

    # Step 6. evaluate the trained model using the defined "evaluate_model" function
    model_performance = evaluate_model(model, X_test, y_test)

    # Step 7. Save the model
    model_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/models/lightgbm_model.pkl'
    joblib.dump(model, model_path)

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
