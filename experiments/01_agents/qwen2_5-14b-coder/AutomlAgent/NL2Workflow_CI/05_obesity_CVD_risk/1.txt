import os, random, time, json
import numpy as np
import pandas as pd
from sklearn.preprocessing import OneHotEncoder, StandardScaler, PolynomialFeatures
from sklearn.feature_selection import SelectKBest, mutual_info_classif
from sklearn.metrics import accuracy_score, f1_score
from pytorch_tabnet.tab_model import TabNetClassifier
import torch

# Set the random seed for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(train_df, test_df):
    # Identify categorical columns
    categorical_cols = train_df.select_dtypes(include=['object', 'category']).columns

    # Initialize the OneHotEncoder
    encoder = OneHotEncoder(sparse_output=False, drop='first', handle_unknown='ignore')

    # Fit and transform the categorical features
    train_encoded = encoder.fit_transform(train_df[categorical_cols])
    test_encoded = encoder.transform(test_df[categorical_cols])

    # Create a DataFrame from the encoded features
    train_encoded_df = pd.DataFrame(train_encoded, columns=encoder.get_feature_names_out(categorical_cols))
    test_encoded_df = pd.DataFrame(test_encoded, columns=encoder.get_feature_names_out(categorical_cols))

    # Concatenate the encoded features with the original DataFrame
    train_df = pd.concat([train_df.drop(columns=categorical_cols), train_encoded_df], axis=1)
    test_df = pd.concat([test_df.drop(columns=categorical_cols), test_encoded_df], axis=1)

    # Create interaction terms and polynomial features
    poly = PolynomialFeatures(degree=2, interaction_only=True, include_bias=False)
    train_poly = poly.fit_transform(train_df.drop(columns=['obesity_risk_level']))
    test_poly = poly.transform(test_df.drop(columns=['obesity_risk_level']))

    # Feature selection using mutual information
    X_train = train_poly
    y_train = train_df['obesity_risk_level']
    X_test = test_poly
    y_test = test_df['obesity_risk_level']

    selector = SelectKBest(score_func=mutual_info_classif, k=10)
    X_train_selected = selector.fit_transform(X_train, y_train)
    X_test_selected = selector.transform(X_test)

    # Standardize the features
    scaler = StandardScaler()
    X_train_selected = scaler.fit_transform(X_train_selected)
    X_test_selected = scaler.transform(X_test_selected)

    return X_train_selected, y_train, X_test_selected, y_test

def train_model(X_train, y_train):
    # Initialize the TabNetClassifier
    model = TabNetClassifier(
        n_d=32,
        n_a=32,
        n_steps=4,
        gamma=2.0,
        lambda_sparse=0.01,
        optimizer_fn=torch.optim.Adam,
        optimizer_params=dict(lr=0.02),
        mask_type="entmax"
    )

    # Train the model
    model.fit(
        X_train=X_train,
        y_train=y_train,
        eval_set=[(X_train, y_train)],
        max_epochs=150,
        patience=50,
        batch_size=64,
        virtual_batch_size=128,
        num_workers=0,
        drop_last=False
    )

    return model

def evaluate_model(model, X_test, y_test):
    # Predict on the test set
    predictions = model.predict(X_test)

    # Calculate performance scores
    performance_scores = {
        'ACC': accuracy_score(y_test, predictions),
        'F1': f1_score(y_test, predictions, average='weighted')
    }

    return performance_scores

    """

    train_df = pd.read_csv(train_path, on_bad_lines='skip')
    test_df = pd.read_csv(test_path, on_bad_lines='skip')

    # Step 2. Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
    # Here, the train_loader contains 70% of the `dataset`, the valid_loader contains 20% of the `dataset`, and the test_loader contains 10% of the `dataset`.
    from sklearn.model_selection import train_test_split

    train_df, valid_df = train_test_split(train_df, test_size=0.3, random_state=SEED)
    valid_df, test_df = train_test_split(valid_df, test_size=0.333, random_state=SEED)

    # Step 3. With the split dataset, run data preprocessing and feature engineering (if applicable) using the "preprocess_data" function you defined
