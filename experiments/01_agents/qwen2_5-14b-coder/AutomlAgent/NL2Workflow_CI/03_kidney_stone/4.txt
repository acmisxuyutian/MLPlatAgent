# The following code is for "tabular classification" task.

import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
from sklearn.metrics import roc_auc_score
from sklearn.model_selection import cross_val_score
import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense, Dropout
from tensorflow.keras.optimizers import Adam
from tensorflow.keras.callbacks import EarlyStopping
from sklearn.metrics import roc_auc_score

SEED = 42
random.seed(SEED)
np.random.seed(SEED)
os.environ['PYTHONHASHSEED'] = str(SEED)
tf.random.set_seed(SEED)

# Data retrieval and preparation
def load_dataset():
    dataset_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/kidney_stone/kidney_stone.csv'
    dataset = pd.read_csv(dataset_path)
    return dataset

def split_dataset(dataset):
    train, temp = train_test_split(dataset, test_size=0.3, random_state=SEED)
    validation, test = train_test_split(temp, test_size=0.5, random_state=SEED)
    
    train.to_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/kidney_stone/train.csv', index=False)
    validation.to_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/kidney_stone/validation.csv', index=False)
    test.to_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/kidney_stone/test.csv', index=False)
    
    return train, validation, test

# Data preprocessing and feature engineering
def preprocess_data(train, validation, test):
    # Identify numerical and categorical columns
    numerical_cols = train.select_dtypes(include=['int64', 'float64']).columns
    categorical_cols = train.select_dtypes(include=['object']).columns
    
    # Impute missing values
    numerical_transformer = Pipeline(steps=[
        ('imputer', SimpleImputer(strategy='mean')),
        ('scaler', StandardScaler())
    ])
    
    categorical_transformer = Pipeline(steps=[
        ('imputer', SimpleImputer(strategy='most_frequent')),
        ('onehot', OneHotEncoder(handle_unknown='ignore'))
    ])
    
    preprocessor = ColumnTransformer(
        transformers=[
            ('num', numerical_transformer, numerical_cols),
            ('cat', categorical_transformer, categorical_cols)
        ])
    
    # Fit and transform the data
    X_train = preprocessor.fit_transform(train.drop('target', axis=1))
    y_train = train['target']
    
    X_validation = preprocessor.transform(validation.drop('target', axis=1))
    y_validation = validation['target']
    
    X_test = preprocessor.transform(test.drop('target', axis=1))
    y_test = test['target']
    
    return X_train, y_train, X_validation, y_validation, X_test, y_test

# Model training
def train_model(X_train, y_train, X_validation, y_validation):
    model = Sequential([
        Dense(128, activation='relu', input_shape=(X_train.shape[1],)),
        Dropout(0.3),
        Dense(64, activation='relu'),
        Dropout(0.3),
        Dense(1, activation='sigmoid')
    ])
    
    model.compile(optimizer=Adam(learning_rate=0.001), loss='binary_crossentropy', metrics=['accuracy'])
    
    early_stopping = EarlyStopping(monitor='val_loss', patience=10, restore_best_weights=True)
    
    model.fit(X_train, y_train, validation_data=(X_validation, y_validation), epochs=100, batch_size=32, callbacks=[early_stopping])
    
    return model

# Model evaluation
def evaluate_model(model, X_test, y_test):
    y_pred_proba = model.predict(X_test)
    auc_roc = roc_auc_score(y_test, y_pred_proba)
    
    performance_scores = {
        'AUC-ROC': auc_roc
    }
    
    return performance_scores

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    """
    Main function to execute the tabular classification pipeline.
    """
    
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    dataset = load_dataset()
    
    # Step 2. Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
    train, validation, test = split_dataset(dataset)
    
    # Step 3. With the split dataset, run data preprocessing and feature engineering (if applicable) using the "preprocess_data" function you defined
    X_train, y_train, X_validation, y_validation, X_test, y_test = preprocess_data(train, validation, test)
    
    # Step 4. Define required model. You may retrieve model from available hub or library along with pretrained weights (if any).
    # If pretrained or predefined model is not available, please create the model according to the given user's requirements below using PyTorch and relevant libraries.
    model = train_model(X_train, y_train, X_validation, y_validation)
    
    # Step 5. train the retrieved/loaded model using the defined "train_model" function
    # on top of the model training, please run hyperparamter optimization based on the suggested hyperparamters and their values before proceeding to the evaluation step to ensure model's optimality
    
    # Step 6. evaluate the trained model using the defined "evaluate_model" function
    model_performance = evaluate_model(model, X_test, y_test)
    
    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
