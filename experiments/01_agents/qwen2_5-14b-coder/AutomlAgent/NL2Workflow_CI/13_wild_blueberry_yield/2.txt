######代码######

import os
import random
import time
import json
import numpy as np
import pandas as pd
import torch
import torch.nn as nn
import torch.optim as optim
from sklearn.metrics import mean_absolute_error
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Load the dataset into a Pandas DataFrame
train_dataset = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/wild_blueberry_yield/train.csv')
test_dataset = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/wild_blueberry_yield/test.csv')

# Example: Create interaction terms
if 'temperature' in train_dataset.columns and 'rainfall' in train_dataset.columns:
    train_dataset['temperature_rainfall'] = train_dataset['temperature'] * train_dataset['rainfall']
if 'temperature' in test_dataset.columns and 'rainfall' in test_dataset.columns:
    test_dataset['temperature_rainfall'] = test_dataset['temperature'] * test_dataset['rainfall']

# Correlation matrix
corr_matrix = train_dataset.corr()
target_corr = corr_matrix['yield'].abs().sort_values(ascending=False)
relevant_features = target_corr[target_corr > 0.1].index.tolist()

train_dataset_selected = train_dataset[relevant_features]
test_dataset_selected = test_dataset[relevant_features[:-1]]  # Exclude the target variable for test set

# Split the dataset
train, temp = train_test_split(train_dataset_selected, test_size=0.3, random_state=42)
validation, test = train_test_split(temp, test_size=0.5, random_state=42)

class NeuralNetwork(nn.Module):
    def __init__(self, input_size):
        super(NeuralNetwork, self).__init__()
        self.fc1 = nn.Linear(input_size, 64)
        self.fc2 = nn.Linear(64, 32)
        self.fc3 = nn.Linear(32, 1)
        self.dropout = nn.Dropout(0.2)

    def forward(self, x):
        x = torch.relu(self.fc1(x))
        x = self.dropout(x)
        x = torch.relu(self.fc2(x))
        x = self.dropout(x)
        x = self.fc3(x)
        return x

input_size = len(train.columns) - 1  # Exclude the target variable
model = NeuralNetwork(input_size)
criterion = nn.MSELoss()
optimizer = optim.Adam(model.parameters(), lr=0.001)

# Standardize the data
scaler = StandardScaler()
train_features = scaler.fit_transform(train.drop('yield', axis=1))
train_labels = train['yield'].values
validation_features = scaler.transform(validation.drop('yield', axis=1))
validation_labels = validation['yield'].values

# Convert to PyTorch tensors
train_features = torch.tensor(train_features, dtype=torch.float32)
train_labels = torch.tensor(train_labels, dtype=torch.float32).view(-1, 1)
validation_features = torch.tensor(validation_features, dtype=torch.float32)
validation_labels = torch.tensor(validation_labels, dtype=torch.float32).view(-1, 1)

# Training loop
num_epochs = 50
for epoch in range(num_epochs):
    model.train()
    optimizer.zero_grad()
    outputs = model(train_features)
    loss = criterion(outputs, train_labels)
    loss.backward()
    optimizer.step()

    # Validation
    model.eval()
    with torch.no_grad():
        val_outputs = model(validation_features)
        val_loss = criterion(val_outputs, validation_labels)
    print(f'Epoch [{epoch+1}/{num_epochs}], Loss: {loss.item():.4f}, Val Loss: {val_loss.item():.4f}')

# Standardize the test data
test_features = scaler.transform(test_dataset_selected)

# Convert to PyTorch tensors
test_features = torch.tensor(test_features, dtype=torch.float32)
test_labels = torch.tensor(test_dataset['yield'].values, dtype=torch.float32).view(-1, 1)

# Evaluate the model
model.eval()
with torch.no_grad():
    test_outputs = model(test_features)
    test_mae = mean_absolute_error(test_labels.numpy(), test_outputs.numpy())
print(f'Test MAE: {test_mae:.4f}')

# Save the trained model
model_path = './agent_workspace/trained_models/wild_blueberry_yield_model.pth'
torch.save(model.state_dict(), model_path)
print(f'Model saved to {model_path}')
