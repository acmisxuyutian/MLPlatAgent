import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.metrics import mean_absolute_error
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import PolynomialFeatures, MinMaxScaler, StandardScaler
from sklearn.feature_selection import mutual_info_regression
from sklearn.ensemble import RandomForestRegressor
import torch
import torch.nn as nn
import torch.optim as optim

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(train_df, test_df):
    # Data Inspection
    print(train_df.head())
    print(train_df.dtypes)
    print(train_df.isnull().sum())
    
    # Feature Construction
    # Polynomial Features
    poly = PolynomialFeatures(degree=2, include_bias=False)
    poly_features_train = poly.fit_transform(train_df.drop(columns=['yield']))
    poly_features_test = poly.transform(test_df.drop(columns=['yield']))
    poly_df_train = pd.DataFrame(poly_features_train, columns=poly.get_feature_names_out(train_df.drop(columns=['yield']).columns))
    poly_df_test = pd.DataFrame(poly_features_test, columns=poly.get_feature_names_out(test_df.drop(columns=['yield']).columns))
    train_df = pd.concat([train_df, poly_df_train], axis=1)
    test_df = pd.concat([test_df, poly_df_test], axis=1)
    
    # Interaction Terms
    train_df['temp_precip_interaction'] = train_df['temperature'] * train_df['precipitation']
    test_df['temp_precip_interaction'] = test_df['temperature'] * test_df['precipitation']
    
    # Domain-Specific Transformations
    scaler = MinMaxScaler()
    train_df[['temperature', 'precipitation']] = scaler.fit_transform(train_df[['temperature', 'precipitation']])
    test_df[['temperature', 'precipitation']] = scaler.transform(test_df[['temperature', 'precipitation']])
    
    # Feature Selection
    # Correlation Analysis
    corr_matrix = train_df.corr().abs()
    upper = corr_matrix.where(np.triu(np.ones(corr_matrix.shape), k=1).astype(bool))
    to_drop = [column for column in upper.columns if any(upper[column] > 0.8)]
    train_df.drop(columns=to_drop, inplace=True)
    test_df.drop(columns=to_drop, inplace=True)
    
    # Mutual Information
    mi_scores = mutual_info_regression(train_df.drop(columns=['yield']), train_df['yield'])
    mi_scores = pd.Series(mi_scores, index=train_df.drop(columns=['yield']).columns)
    selected_features = mi_scores.sort_values(ascending=False).head(10).index
    train_df = train_df[selected_features.tolist() + ['yield']]
    test_df = test_df[selected_features.tolist() + ['yield']]
    
    # Tree-Based Feature Importance
    rf = RandomForestRegressor(n_estimators=100, random_state=42)
    rf.fit(train_df.drop(columns=['yield']), train_df['yield'])
    feature_importance = pd.Series(rf.feature_importances_, index=train_df.drop(columns=['yield']).columns)
    selected_features = feature_importance.sort_values(ascending=False).head(10).index
    train_df = train_df[selected_features.tolist() + ['yield']]
    test_df = test_df[selected_features.tolist() + ['yield']]
    
    # Data Cleaning
    train_df.fillna(train_df.mean(), inplace=True)
    test_df.fillna(train_df.mean(), inplace=True)
    train_df['temperature'] = train_df['temperature'].astype(float)
    test_df['temperature'] = test_df['temperature'].astype(float)
    train_df['precipitation'] = train_df['precipitation'].astype(float)
    test_df['precipitation'] = test_df['precipitation'].astype(float)
    
    return train_df, test_df

# Define the model architecture
class MLP(nn.Module):
    def __init__(self, input_size, hidden_size, output_size):
        super(MLP, self).__init__()
        self.fc1 = nn.Linear(input_size, hidden_size)
        self.fc2 = nn.Linear(hidden_size, hidden_size)
        self.fc3 = nn.Linear(hidden_size, output_size)
        self.relu = nn.ReLU()

    def forward(self, x):
        out = self.relu(self.fc1(x))
        out = self.relu(self.fc2(out))
        out = self.fc3(out)
        return out

# Train the model
def train_model(model, train_loader, criterion, optimizer, num_epochs):
    for epoch in range(num_epochs):
        model.train()
        optimizer.zero_grad()
        outputs = model(torch.tensor(train_loader['features'].values, dtype=torch.float32))
        loss = criterion(outputs, torch.tensor(train_loader['yield'].values, dtype=torch.float32).view(-1, 1))
        loss.backward()
        optimizer.step()
        if (epoch+1) % 10 == 0:
            print(f'Epoch [{epoch+1}/{num_epochs}], Loss: {loss.item():.4f}')
    return model

# Evaluate the model
def evaluate_model(model, test_loader):
    model.eval()
    with torch.no_grad():
        test_predictions = model(torch.tensor(test_loader['features'].values, dtype=torch.float32))
        test_mae = mean_absolute_error(test_loader['yield'].values, test_predictions.numpy())
    performance_scores = {
        'MAE': test_mae
    }
    return performance_scores

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    dataset_url = 'https://example.com/wild_blueberry_yield.csv'
    try:
        df = pd.read_csv(dataset_url)
    except Exception as e:
        print(f"Error loading dataset: {e}")
        return
    
    # Step 2. Create a train-valid-test split of the data
    train_df, test_df = train_test_split(df, test_size=0.3, random_state=SEED)
    train_df, valid_df = train_test_split(train_df, test_size=0.2, random_state=SEED)
    
    # Step 3. With the split dataset, run data preprocessing and feature engineering
    train_df, test_df = preprocess_data(train_df, test_df)
    
    # Step 4. Define required model
    input_size = train_df.shape[1] - 1
    hidden_size = 128
    output_size = 1
    model = MLP(input_size, hidden_size, output_size)
    
    # Step 5. Train the model
    criterion = nn.MSELoss()
    optimizer = optim.Adam(model.parameters(), lr=0.0005, weight_decay=1e-4)
    num_epochs = 100
    model = train_model(model, train_df, criterion, optimizer, num_epochs)
    
    # Step 6. Evaluate the model
    model_performance = evaluate_model(model, test_df)
    
    # Step 7. Save the model
    model_path = "./agent_workspace/trained_models/wild_blueberry_yield_model.pth"
    torch.save(model.state_dict(), model_path)
    
    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
