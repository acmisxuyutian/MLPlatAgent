# The following code is for "tabular classification" task.

import os, random, time, json
import numpy as np
import pandas as pd
from sklearn.metrics import accuracy_score, f1_score
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.model_selection import train_test_split
import xgboost as xgb
from sklearn.metrics import roc_auc_score
import joblib

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(train_df, test_df):
    # Normalize numerical features
    numerical_features = train_df.select_dtypes(include=['float64', 'int64']).columns
    scaler = StandardScaler()
    train_df[numerical_features] = scaler.fit_transform(train_df[numerical_features])
    test_df[numerical_features] = scaler.transform(test_df[numerical_features])

    # Encode categorical features
    categorical_features = train_df.select_dtypes(include=['object', 'category']).columns
    encoder = OneHotEncoder(handle_unknown='ignore')
    encoded_features_train = encoder.fit_transform(train_df[categorical_features])
    encoded_features_test = encoder.transform(test_df[categorical_features])

    encoded_df_train = pd.DataFrame(encoded_features_train, columns=encoder.get_feature_names_out(categorical_features))
    encoded_df_test = pd.DataFrame(encoded_features_test, columns=encoder.get_feature_names_out(categorical_features))

    train_df = pd.concat([train_df.drop(categorical_features, axis=1), encoded_df_train], axis=1)
    test_df = pd.concat([test_df.drop(categorical_features, axis=1), encoded_df_test], axis=1)

    # Create new features based on domain knowledge
    if 'signal1' in train_df.columns and 'signal2' in train_df.columns:
        train_df['new_feature'] = train_df['signal1'] * train_df['signal2']
        test_df['new_feature'] = test_df['signal1'] * test_df['signal2']

    return train_df, test_df, scaler, encoder

def train_model(train_loader, valid_loader):
    X_train, y_train = train_loader
    X_val, y_val = valid_loader

    model = xgb.XGBClassifier(
        objective='binary:logistic',
        eval_metric='auc',
        learning_rate=0.1,
        max_depth=6,
        min_child_weight=1,
        gamma=0.1,
        subsample=0.8,
        colsample_bytree=0.8,
        reg_alpha=0.01,
        reg_lambda=1,
        use_label_encoder=False,
        enable_categorical=True
    )

    model.fit(X_train, y_train, eval_set=[(X_val, y_val)], early_stopping_rounds=10, verbose=True)
    return model

def evaluate_model(model, test_loader):
    X_test, y_test = test_loader
    y_pred_proba = model.predict_proba(X_test)[:, 1]
    auc_roc = roc_auc_score(y_test, y_pred_proba)
    performance_scores = {
        'AUC-ROC': auc_roc
    }
    return performance_scores

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/smoker_status_Bio_Signals/train.csv')
    test_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/smoker_status_Bio_Signals/test.csv')

    # Step 2. Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
    X_train = train_df.drop('smoker_status', axis=1)
    y_train = train_df['smoker_status']

    X_train, X_val, y_train, y_val = train_test_split(X_train, y_train, test_size=0.2, random_state=SEED, stratify=y_train)

    train_loader = (X_train, y_train)
    valid_loader = (X_val, y_val)
    test_loader = (test_df.drop('smoker_status', axis=1), test_df['smoker_status'])

    # Step 3. With the split dataset, run data preprocessing and feature engineering (if applicable) using the "preprocess_data" function you defined
    train_df, test_df, scaler, encoder = preprocess_data(train_df, test_df)

    # Step 4. Define required model. You may retrieve model from available hub or library along with pretrained weights (if any).
    # If pretrained or predefined model is not available, please create the model according to the given user's requirements below using PyTorch and relevant libraries.
    model = train_model(train_loader, valid_loader)

    # Step 5. train the retrieved/loaded model using the defined "train_model" function
    # on top of the model training, please run hyperparamter optimization based on the suggested hyperparamters and their values before proceeding to the evaluation step to ensure model's optimality

    # Step 6. evaluate the trained model using the defined "evaluate_model" function
    model_performance = evaluate_model(model, test_loader)

    # Save the model and preprocessing pipeline
    joblib.dump(model, './agent_workspace/trained_models/smoker_status_XGboost_model.pkl')
    joblib.dump(scaler, './agent_workspace/trained_models/scaler.pkl')
    joblib.dump(encoder, './agent_workspace/trained_models/encoder.pkl')

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
