import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import roc_auc_score, confusion_matrix
import xgboost as xgb
import shap
import pickle

# Ensure reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(train_df, test_df):
    # Data Exploration
    print("Train Data Summary:")
    print(train_df.info())
    print(train_df.describe())
    print(train_df.isnull().sum())
    print(train_df.nunique())

    print("Test Data Summary:")
    print(test_df.info())
    print(test_df.describe())
    print(test_df.isnull().sum())
    print(test_df.nunique())

    # Handle Missing Values
    imputer = SimpleImputer(strategy='mean')
    train_df_imputed = pd.DataFrame(imputer.fit_transform(train_df), columns=train_df.columns)
    test_df_imputed = pd.DataFrame(imputer.transform(test_df), columns=test_df.columns)

    # Feature Scaling
    scaler = StandardScaler()
    train_df_scaled = pd.DataFrame(scaler.fit_transform(train_df_imputed), columns=train_df.columns)
    test_df_scaled = pd.DataFrame(scaler.transform(test_df_imputed), columns=test_df.columns)

    # Outlier Detection and Handling (using Z-score)
    z_scores = np.abs((train_df_scaled - train_df_scaled.mean()) / train_df_scaled.std())
    train_df_no_outliers = train_df_scaled[(z_scores < 3).all(axis=1)]

    # Feature Engineering - Correlation Analysis
    corr_matrix = train_df_no_outliers.corr().abs()
    upper = corr_matrix.where(np.triu(np.ones(corr_matrix.shape), k=1).astype(bool))
    to_drop = [column for column in upper.columns if any(upper[column] > 0.8)]
    train_df_no_outliers = train_df_no_outliers.drop(columns=to_drop)
    test_df_scaled = test_df_scaled.drop(columns=to_drop)

    # Feature Engineering - Feature Importance Analysis
    if 'smoker_status' in train_df_no_outliers.columns:
        X_train = train_df_no_outliers.drop(columns=['smoker_status'])
        y_train = train_df_no_outliers['smoker_status']
    else:
        raise KeyError("Column 'smoker_status' not found in the training data")

    if 'smoker_status' in test_df_scaled.columns:
        X_test = test_df_scaled.drop(columns=['smoker_status'])
        y_test = test_df_scaled['smoker_status']
    else:
        raise KeyError("Column 'smoker_status' not found in the test data")

    return X_train, y_train, X_test, y_test

def train_model(X_train, y_train):
    # Configure XGBoost model with DART algorithm
    model = xgb.XGBClassifier(
        objective='binary:logistic',
        booster='dart',
        learning_rate=0.1,
        max_depth=6,
        n_estimators=100,
        subsample=0.8,
        colsample_bytree=0.8,
        random_state=SEED
    )

    # Train the model
    model.fit(X_train, y_train)

    return model

def evaluate_model(model, X_test, y_test):
    # Predict probabilities
    y_pred_proba = model.predict_proba(X_test)[:, 1]

    # Calculate AUC-ROC score
    auc_roc = roc_auc_score(y_test, y_pred_proba)

    # Generate confusion matrix
    y_pred = model.predict(X_test)
    cm = confusion_matrix(y_test, y_pred)

    # Performance scores
    performance_scores = {
        'AUC-ROC': auc_roc,
        'Confusion Matrix': cm
    }

    return performance_scores

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    """
    Main function to execute the tabular classification pipeline.
    """

    # Step 1. Retrieve or load a dataset from user's local storage
    train_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/smoker_status_Bio_Signals/train.csv'
    test_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/smoker_status_Bio_Signals/test.csv'

    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)

    # Step 2. Preprocess the data
    X_train, y_train, X_test, y_test = preprocess_data(train_df, test_df)

    # Step 3. Define and train the model
    model = train_model(X_train, y_train)

    # Step 4. Evaluate the trained model
    model_performance = evaluate_model(model, X_test, y_test)

    # Step 5. Save the trained model
    model_save_path = './agent_workspace/trained_models/smoker_status_Bio_Signals_xgboost_model.pkl'
    with open(model_save_path, 'wb') as model_file:
        pickle.dump(model, model_file)

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
