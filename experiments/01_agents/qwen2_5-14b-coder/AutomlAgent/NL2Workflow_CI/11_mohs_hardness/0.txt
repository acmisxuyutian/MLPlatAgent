import os
import random
import time
import json
import numpy as np
import pandas as pd
import torch
import torch.nn as nn
import torch.optim as optim
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler, MinMaxScaler
from sklearn.metrics import mean_absolute_error, mean_squared_error

SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(train_df, test_df):
    # Check for missing values
    missing_values_train = train_df.isnull().sum()
    missing_values_test = test_df.isnull().sum()
    
    # Handle missing values
    numerical_features = train_df.select_dtypes(include=[np.number]).columns
    for feature in numerical_features:
        train_df[feature] = train_df[feature].fillna(train_df[feature].mean())
        test_df[feature] = test_df[feature].fillna(test_df[feature].mean())
    
    categorical_features = train_df.select_dtypes(include=[object]).columns
    for feature in categorical_features:
        train_df[feature] = train_df[feature].fillna(train_df[feature].mode()[0])
        test_df[feature] = test_df[feature].fillna(test_df[feature].mode()[0])
    
    # Detect and handle outliers
    for feature in numerical_features:
        Q1 = train_df[feature].quantile(0.25)
        Q3 = train_df[feature].quantile(0.75)
        IQR = Q3 - Q1
        lower_bound = Q1 - 1.5 * IQR
        upper_bound = Q3 + 1.5 * IQR
        train_df = train_df[(train_df[feature] >= lower_bound) & (train_df[feature] <= upper_bound)]
        test_df = test_df[(test_df[feature] >= lower_bound) & (test_df[feature] <= upper_bound)]
    
    # Feature engineering
    if 'feature1' in train_df.columns and 'feature2' in train_df.columns:
        train_df['interaction_feature'] = train_df['feature1'] * train_df['feature2']
        test_df['interaction_feature'] = test_df['feature1'] * test_df['feature2']
    
    from sklearn.preprocessing import PolynomialFeatures
    if 'feature1' in train_df.columns and 'feature2' in train_df.columns:
        poly = PolynomialFeatures(degree=2, include_bias=False)
        poly_features_train = poly.fit_transform(train_df[['feature1', 'feature2']])
        poly_features_test = poly.transform(test_df[['feature1', 'feature2']])
        train_df = pd.concat([train_df, pd.DataFrame(poly_features_train, columns=poly.get_feature_names_out(['feature1', 'feature2']))], axis=1)
        test_df = pd.concat([test_df, pd.DataFrame(poly_features_test, columns=poly.get_feature_names_out(['feature1', 'feature2']))], axis=1)
    
    # Min-Max Scaling
    if 'feature1' in train_df.columns and 'feature2' in train_df.columns:
        scaler = MinMaxScaler()
        train_df[['scaled_feature1', 'scaled_feature2']] = scaler.fit_transform(train_df[['feature1', 'feature2']])
        test_df[['scaled_feature1', 'scaled_feature2']] = scaler.transform(test_df[['feature1', 'feature2']])
    
    # Data augmentation
    if 'feature1' in train_df.columns:
        noise = np.random.normal(0, 0.1, train_df['feature1'].shape)
        train_df['noisy_feature1'] = train_df['feature1'] + noise
    
    return train_df, test_df

# Define the Neural Network Architecture
class NeuralNetwork(nn.Module):
    def __init__(self, input_size):
        super(NeuralNetwork, self).__init__()
        self.fc1 = nn.Linear(input_size, 64)
        self.fc2 = nn.Linear(64, 128)
        self.fc3 = nn.Linear(128, 64)
        self.fc4 = nn.Linear(64, 1)

    def forward(self, x):
        x = torch.tanh(self.fc1(x))
        x = torch.tanh(self.fc2(x))
        x = torch.tanh(self.fc3(x))
        x = self.fc4(x)
        return x

# Train the model
def train_model(model, train_loader, valid_loader, criterion, optimizer, num_epochs=100):
    for epoch in range(num_epochs):
        model.train()
        optimizer.zero_grad()
        outputs = model(train_loader['X'])
        loss = criterion(outputs, train_loader['y'])
        loss.backward()
        optimizer.step()

        if (epoch+1) % 10 == 0:
            model.eval()
            with torch.no_grad():
                val_outputs = model(valid_loader['X'])
                val_loss = criterion(val_outputs, valid_loader['y'])
                print(f'Epoch [{epoch+1}/{num_epochs}], Loss: {loss.item():.4f}, Val Loss: {val_loss.item():.4f}')
    return model

# Evaluate the model
def evaluate_model(model, test_loader, criterion):
    model.eval()
    with torch.no_grad():
        test_outputs = model(test_loader['X'])
        test_loss = criterion(test_outputs, test_loader['y'])
        mae = mean_absolute_error(test_loader['y'].numpy(), test_outputs.numpy())
        mse = mean_squared_error(test_loader['y'].numpy(), test_outputs.numpy())
        rmse = np.sqrt(mse)
        performance_scores = {
            "MAE": mae,
            "MSE": mse,
            "RMSE": rmse
        }
    return performance_scores

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/mohs_hardness/train.csv')
    test_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/mohs_hardness/test.csv')
    
    # Step 2. Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
    # Here, the train_loader contains 70% of the `dataset`, the valid_loader contains 20% of the `dataset`, and the test_loader contains 10% of the `dataset`.
    train_df, valid_df = train_test_split(train_df, test_size=0.3, random_state=SEED)
    valid_df, test_df = train_test_split(valid_df, test_size=0.33, random_state=SEED)
    
    # Step 3. With the split dataset, run data preprocessing and feature engineering (if applicable) using the "preprocess_data" function you defined
    train_df, valid_df = preprocess_data(train_df, valid_df)
    test_df, _ = preprocess_data(test_df, pd.DataFrame())
    
    # Step 4. Define required model. You may retrieve model from available hub or library along with pretrained weights (if any).
    # If pretrained or predefined model is not available, please create the model according to the given user's requirements below using PyTorch and relevant libraries.
    input_size = train_df.shape[1] - 1  # Exclude the target variable
    model = NeuralNetwork(input_size)
    
    # Step 5. train the retrieved/loaded model using the defined "train_model" function
    # on top of the model training, please run hyperparamter optimization based on the suggested hyperparamters and their values before proceeding to the evaluation step to ensure model's optimality
    criterion = nn.MSELoss()
    optimizer = optim.SGD(model.parameters(), lr=0.01, weight_decay=0.01)
    
    # Convert data to PyTorch tensors
    X_train_tensor = torch.tensor(train_df.drop('mohs_hardness', axis=1).values, dtype=torch.float32)
    y_train_tensor = torch.tensor(train_df['mohs_hardness'].values, dtype=torch.float32).view(-1, 1)
    X_val_tensor = torch.tensor(valid_df.drop('mohs_hardness', axis=1).values, dtype=torch.float32)
    y_val_tensor = torch.tensor(valid_df['mohs_hardness'].values, dtype=torch.float32).view(-1, 1)
    
    train_loader = {'X': X_train_tensor, 'y': y_train_tensor}
    valid_loader = {'X': X_val_tensor, 'y': y_val_tensor}
    
    model = train_model(model, train_loader, valid_loader, criterion, optimizer)
    
    # Step 6. evaluate the trained model using the defined "evaluate_model" function
    X_test_tensor = torch.tensor(test_df.drop('mohs_hardness', axis=1).values, dtype=torch.float32)
    y_test_tensor = torch.tensor(test_df['mohs_hardness'].values, dtype=torch.float32).view(-1, 1)
    
    test_loader = {'X': X_test_tensor, 'y': y_test_tensor}
    
    model_performance = evaluate_model(model, test_loader, criterion)
    
    # Step 7. Save the trained model
    torch.save(model.state_dict(), './agent_workspace/trained_models/mohs_hardness_model.pth')
    
    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
