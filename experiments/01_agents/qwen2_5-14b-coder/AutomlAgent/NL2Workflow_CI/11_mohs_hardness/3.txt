import os
import random
import time
import json
import numpy as np
import pandas as pd
import torch
import torch.nn as nn
import torch.optim as optim
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import MinMaxScaler
from sklearn.metrics import median_absolute_error

# Set random seed for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)

# Define the dataset paths
train_dataset_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/mohs_hardness/train.csv'
test_dataset_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/mohs_hardness/test.csv'

# Define the model save path
model_save_path = './agent_workspace/trained_models/mohs_hardness_model.pth'

# Data preprocessing and feature engineering
def preprocess_data(train_df, test_df):
    # Inspect dataset
    print(train_df.head())
    print(train_df.info())
    print(train_df.describe())
    
    # Handle missing values
    train_df.dropna(inplace=True)
    test_df.dropna(inplace=True)
    
    # Ensure data consistency
    train_df = train_df.astype(float)
    test_df = test_df.astype(float)
    
    # Normalize features
    scaler = MinMaxScaler()
    train_features = scaler.fit_transform(train_df.drop(columns=['mohs_hardness']))
    test_features = scaler.transform(test_df.drop(columns=['mohs_hardness']))
    
    # Split dataset into features and target
    train_labels = train_df['mohs_hardness'].values
    test_labels = test_df['mohs_hardness'].values
    
    return train_features, train_labels, test_features, test_labels

# Define the FFNN model architecture
class FFNN(nn.Module):
    def __init__(self, input_size):
        super(FFNN, self).__init__()
        self.fc1 = nn.Linear(input_size, 64)
        self.fc2 = nn.Linear(64, 128)
        self.fc3 = nn.Linear(128, 64)
        self.fc4 = nn.Linear(64, 1)
        self.dropout = nn.Dropout(0.2)
    
    def forward(self, x):
        x = torch.tanh(self.fc1(x))
        x = self.dropout(x)
        x = torch.tanh(self.fc2(x))
        x = self.dropout(x)
        x = torch.tanh(self.fc3(x))
        x = self.dropout(x)
        x = self.fc4(x)
        return x

# Train the model
def train_model(model, train_features, train_labels, val_features, val_labels, epochs=100, batch_size=32, learning_rate=0.01):
    criterion = nn.L1Loss()
    optimizer = optim.SGD(model.parameters(), lr=learning_rate)
    
    train_dataset = torch.utils.data.TensorDataset(torch.tensor(train_features, dtype=torch.float32), torch.tensor(train_labels, dtype=torch.float32))
    val_dataset = torch.utils.data.TensorDataset(torch.tensor(val_features, dtype=torch.float32), torch.tensor(val_labels, dtype=torch.float32))
    
    train_loader = torch.utils.data.DataLoader(train_dataset, batch_size=batch_size, shuffle=True)
    val_loader = torch.utils.data.DataLoader(val_dataset, batch_size=batch_size, shuffle=False)
    
    best_val_loss = float('inf')
    patience = 10
    early_stopping_counter = 0
    
    for epoch in range(epochs):
        model.train()
        running_loss = 0.0
        for inputs, labels in train_loader:
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs.squeeze(), labels)
            loss.backward()
            optimizer.step()
            running_loss += loss.item()
        
        avg_train_loss = running_loss / len(train_loader)
        
        model.eval()
        val_loss = 0.0
        with torch.no_grad():
            for inputs, labels in val_loader:
                outputs = model(inputs)
                loss = criterion(outputs.squeeze(), labels)
                val_loss += loss.item()
        
        avg_val_loss = val_loss / len(val_loader)
        
        print(f'Epoch {epoch+1}/{epochs}, Train Loss: {avg_train_loss:.4f}, Val Loss: {avg_val_loss:.4f}')
        
        if avg_val_loss < best_val_loss:
            best_val_loss = avg_val_loss
            early_stopping_counter = 0
            torch.save(model.state_dict(), model_save_path)
        else:
            early_stopping_counter += 1
            if early_stopping_counter >= patience:
                print('Early stopping triggered.')
                break
    
    return model

# Evaluate the model
def evaluate_model(model, test_features, test_labels):
    model.load_state_dict(torch.load(model_save_path))
    model.eval()
    
    test_dataset = torch.utils.data.TensorDataset(torch.tensor(test_features, dtype=torch.float32), torch.tensor(test_labels, dtype=torch.float32))
    test_loader = torch.utils.data.DataLoader(test_dataset, batch_size=32, shuffle=False)
    
    predictions = []
    true_labels = []
    
    with torch.no_grad():
        for inputs, labels in test_loader:
            outputs = model(inputs)
            predictions.extend(outputs.squeeze().numpy())
            true_labels.extend(labels.numpy())
    
    medae = median_absolute_error(true_labels, predictions)
    
    performance_scores = {
        'MEDAE': medae
    }
    
    return performance_scores

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_df = pd.read_csv(train_dataset_path)
    test_df = pd.read_csv(test_dataset_path)
    
    # Step 2. Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
    train_features, train_labels, test_features, test_labels = preprocess_data(train_df, test_df)
    
    # Split into train and validation sets
    train_features, val_features, train_labels, val_labels = train_test_split(train_features, train_labels, test_size=0.1, random_state=SEED)
    
    # Step 3. With the split dataset, run data preprocessing and feature engineering (if applicable) using the "preprocess_data" function you defined
    # Preprocessing is already done in the preprocess_data function
    
    # Step 4. Define required model. You may retrieve model from available hub or library along with pretrained weights (if any).
    # If pretrained or predefined model is not available, please create the model according to the given user's requirements below using PyTorch and relevant libraries.
    input_size = train_features.shape[1]
    model = FFNN(input_size)
    
    # Step 5. train the retrieved/loaded model using the defined "train_model" function
    # on top of the model training, please run hyperparamter optimization based on the suggested hyperparamters and their values before proceeding to the evaluation step to ensure model's optimality
    model = train_model(model, train_features, train_labels, val_features, val_labels)
    
    # Step 6. evaluate the trained model using the defined "evaluate_model" function
    model_performance = evaluate_model(model, test_features, test_labels)
    
    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
