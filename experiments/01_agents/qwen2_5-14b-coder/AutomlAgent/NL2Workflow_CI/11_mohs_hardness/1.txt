import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.impute import SimpleImputer
import torch
import torch.nn as nn
import torch.optim as optim

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(train_df, test_df):
    # Check if 'mohs_hardness' column exists
    if 'mohs_hardness' not in train_df.columns or 'mohs_hardness' not in test_df.columns:
        raise ValueError("The 'mohs_hardness' column is not found in the dataset.")

    # Handle missing values
    imputer = SimpleImputer(strategy='mean')
    train_df_imputed = pd.DataFrame(imputer.fit_transform(train_df), columns=train_df.columns)
    test_df_imputed = pd.DataFrame(imputer.transform(test_df), columns=test_df.columns)

    # Remove duplicate rows
    train_df_imputed.drop_duplicates(inplace=True)
    test_df_imputed.drop_duplicates(inplace=True)

    # Feature scaling
    scaler = StandardScaler()
    train_features = scaler.fit_transform(train_df_imputed.drop('mohs_hardness', axis=1))
    test_features = scaler.transform(test_df_imputed.drop('mohs_hardness', axis=1))

    train_labels = train_df_imputed['mohs_hardness'].values
    test_labels = test_df_imputed['mohs_hardness'].values

    # Data splitting
    X_train, X_val, y_train, y_val = train_test_split(train_features, train_labels, test_size=0.2, random_state=42)
    X_train, X_test, y_train, y_test = train_test_split(X_train, y_train, test_size=0.125, random_state=42)  # 10% test set

    return X_train, X_val, X_test, y_train, y_val, y_test

# Define the Neural Network Model
class MohsHardnessModel(nn.Module):
    def __init__(self, input_size):
        super(MohsHardnessModel, self).__init__()
        self.fc1 = nn.Linear(input_size, 64)
        self.fc2 = nn.Linear(64, 64)
        self.fc3 = nn.Linear(64, 1)
    
    def forward(self, x):
        x = torch.tanh(self.fc1(x))
        x = torch.tanh(self.fc2(x))
        x = self.fc3(x)
        return x

# Train the Model
def train_model(model, X_train, y_train, X_val, y_val, num_epochs=100, batch_size=64):
    optimizer = optim.SGD(model.parameters(), lr=0.001)
    criterion = nn.L1Loss()

    train_dataset = torch.utils.data.TensorDataset(torch.tensor(X_train, dtype=torch.float32), torch.tensor(y_train, dtype=torch.float32))
    train_loader = torch.utils.data.DataLoader(train_dataset, batch_size=batch_size, shuffle=True)

    for epoch in range(num_epochs):
        model.train()
        for inputs, labels in train_loader:
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, labels.view(-1, 1))
            loss.backward()
            optimizer.step()

    return model

# Evaluate the Model
def evaluate_model(model, X_test, y_test):
    model.eval()
    with torch.no_grad():
        test_outputs = model(torch.tensor(X_test, dtype=torch.float32))
        test_loss = nn.L1Loss()(test_outputs, torch.tensor(y_test, dtype=torch.float32).view(-1, 1))
    
    performance_scores = {
        "MAE": test_loss.item()
    }
    return performance_scores

# Save the trained model
def save_model(model, path):
    torch.save(model.state_dict(), path)

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1: Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_df = pd.read_csv(r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/mohs_hardness/train.csv')
    test_df = pd.read_csv(r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/mohs_hardness/test.csv')

    # Step 2: Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
    X_train, X_val, X_test, y_train, y_val, y_test = preprocess_data(train_df, test_df)

    # Step 3: With the split dataset, run data preprocessing and feature engineering (if applicable) using the "preprocess_data" function you defined
    # Preprocessing is already done in the preprocess_data function

    # Step 4: Define required model. You may retrieve model from available hub or library along with pretrained weights (if any).
    # If pretrained or predefined model is not available, please create the model according to the given user's requirements below using PyTorch and relevant libraries.
    input_size = X_train.shape[1]
    model = MohsHardnessModel(input_size)

    # Step 5: train the retrieved/loaded model using the defined "train_model" function
    model = train_model(model, X_train, y_train, X_val, y_val)

    # Step 6: evaluate the trained model using the defined "evaluate_model" function
    model_performance = evaluate_model(model, X_test, y_test)

    # Step 7: Save the trained model
    model_save_path = "./agent_workspace/trained_models/mohs_hardness_model.pth"
    save_model(model, model_save_path)

    return model_performance

if __name__ == "__main__":
