# Data preprocessing and feature engineering
def preprocess_data(df):
    # Identify numerical and categorical columns
    numerical_cols = df.select_dtypes(include=['int64', 'float64']).columns
    categorical_cols = df.select_dtypes(include=['object', 'category']).columns

    # Impute missing values
    numerical_imputer = SimpleImputer(strategy='mean')
    categorical_imputer = SimpleImputer(strategy='most_frequent')

    # One-Hot Encoding for categorical variables
    one_hot_encoder = OneHotEncoder(handle_unknown='ignore')

    # Label Encoding for ordinal categorical variables (if any)
    label_encoder = LabelEncoder()

    # Min-Max Scaling for numerical features
    min_max_scaler = MinMaxScaler()

    # Standardization for numerical features
    standard_scaler = StandardScaler()

    # Create a preprocessing pipeline
    preprocessor = ColumnTransformer(
        transformers=[
            ('num_imputer', numerical_imputer, numerical_cols),
            ('cat_imputer', categorical_imputer, categorical_cols),
            ('one_hot', one_hot_encoder, categorical_cols),
            ('min_max_scaler', min_max_scaler, numerical_cols),
            ('standard_scaler', standard_scaler, numerical_cols)
        ])

    # Apply preprocessing
    processed_data = preprocessor.fit_transform(df)

    return processed_data

def train_model(X_train, y_train):
    # Define the model
    model = xgb.XGBClassifier(
        n_estimators=100,
        max_depth=6,
        learning_rate=0.1,
        subsample=0.8,
        colsample_bytree=0.8
    )

    # Train the model
    model.fit(X_train, y_train)

    return model

def evaluate_model(model, X_test, y_test):    
    # Predict on the test set
    y_pred = model.predict(X_test)

    # Calculate performance scores
    performance_scores = {
        'ACC': accuracy_score(y_test, y_pred),
        'F1': f1_score(y_test, y_pred)
    }

    return performance_scores

    X_train = preprocess_data(train_df.drop('health_status', axis=1))
