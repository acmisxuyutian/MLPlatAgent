import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.metrics import mean_absolute_error, mean_squared_error
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler, OneHotEncoder, PolynomialFeatures
from sklearn.feature_selection import RFE
from sklearn.ensemble import RandomForestRegressor
import torch
import torch.nn as nn
import torch.optim as optim

SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(df):
    # Inspect the dataset
    print(df.isnull().sum())
    print(df.describe())
    print(df.head())

    # Handle missing values
    numerical_features = df.select_dtypes(include=[np.number]).columns
    categorical_features = df.select_dtypes(include=[object]).columns

    for feature in numerical_features:
        df[feature].fillna(df[feature].mean(), inplace=True)

    for feature in categorical_features:
        df[feature].fillna(df[feature].mode()[0], inplace=True)

    # Normalize or standardize numerical features
    scaler = StandardScaler()
    df[numerical_features] = scaler.fit_transform(df[numerical_features])

    # Encode categorical variables
    encoder = OneHotEncoder(sparse_output=False)
    encoded_features = encoder.fit_transform(df[categorical_features])
    df_encoded = pd.DataFrame(encoded_features, columns=encoder.get_feature_names_out(categorical_features))
    df = pd.concat([df, df_encoded], axis=1)

    # Create interaction terms
    poly = PolynomialFeatures(degree=2, interaction_only=True, include_bias=False)
    interaction_features = poly.fit_transform(df[numerical_features])
    df_interaction = pd.DataFrame(interaction_features, columns=poly.get_feature_names_out(numerical_features))
    df = pd.concat([df, df_interaction], axis=1)

    # Feature selection
    model = RandomForestRegressor()
    rfe = RFE(model, n_features_to_select=10)
    X_rfe = rfe.fit_transform(df.drop('yield', axis=1), df['yield'])
    selected_features = df.drop('yield', axis=1).columns[rfe.support_]
    df = df[['yield'] + list(selected_features)]

    return df

# Define the Feedforward Neural Network
class FeedforwardNeuralNetwork(nn.Module):
    def __init__(self, input_size, hidden_size, num_layers):
        super(FeedforwardNeuralNetwork, self).__init__()
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.num_layers = num_layers
        self.layers = nn.ModuleList([nn.Linear(self.input_size, self.hidden_size)])
        for _ in range(num_layers - 1):
            self.layers.append(nn.Linear(self.hidden_size, self.hidden_size))
        self.fc = nn.Linear(self.hidden_size, 1)

    def forward(self, x):
        for layer in self.layers:
            x = torch.relu(layer(x))
        x = self.fc(x)
        return x

# Train the model
def train_model(model, train_loader, valid_loader, learning_rate, batch_size, num_epochs, dropout_rate):
    criterion = nn.MSELoss()
    optimizer = optim.Adam(model.parameters(), lr=learning_rate)

    for epoch in range(num_epochs):
        model.train()
        for inputs, targets in train_loader:
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, targets)
            loss.backward()
            optimizer.step()

        model.eval()
        with torch.no_grad():
            valid_loss = 0
            for inputs, targets in valid_loader:
                outputs = model(inputs)
                valid_loss += criterion(outputs, targets).item()
            valid_loss /= len(valid_loader)

        print(f'Epoch [{epoch+1}/{num_epochs}], Validation Loss: {valid_loss:.4f}')

    return model

# Evaluate the model
def evaluate_model(model, test_loader):
    model.eval()
    y_true = []
    y_pred = []

    with torch.no_grad():
        for inputs, targets in test_loader:
            outputs = model(inputs)
            y_true.extend(targets.numpy())
            y_pred.extend(outputs.numpy())

    mae = mean_absolute_error(y_true, y_pred)
    mse = mean_squared_error(y_true, y_pred)
    rmse = np.sqrt(mse)

    performance_scores = {
        "MAE": mae,
        "MSE": mse,
        "RMSE": rmse
    }

    return performance_scores

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/wild_blueberry_yield/train.csv')

    # Step 2. Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
    train_df, temp_df = train_test_split(df, test_size=0.3, random_state=SEED)
    valid_df, test_df = train_test_split(temp_df, test_size=0.33, random_state=SEED)

    # Step 3. With the split dataset, run data preprocessing and feature engineering (if applicable) using the "preprocess_data" function you defined
    train_df = preprocess_data(train_df)
    valid_df = preprocess_data(valid_df)
    test_df = preprocess_data(test_df)

    # Step 4. Define required model. You may retrieve model from available hub or library along with pretrained weights (if any).
    # If pretrained or predefined model is not available, please create the model according to the given user's requirements below using PyTorch and relevant libraries.
    input_size = train_df.shape[1] - 1  # Exclude the target variable
    hidden_size = 64
    num_layers = 3
    model = FeedforwardNeuralNetwork(input_size, hidden_size, num_layers)

    # Step 5. train the retrieved/loaded model using the defined "train_model" function
    # on top of the model training, please run hyperparamter optimization based on the suggested hyperparamters and their values before proceeding to the evaluation step to ensure model's optimality

    # Convert data to PyTorch tensors
    X_train = torch.tensor(train_df.drop('yield', axis=1).values, dtype=torch.float32)
    y_train = torch.tensor(train_df['yield'].values, dtype=torch.float32).view(-1, 1)
    X_valid = torch.tensor(valid_df.drop('yield', axis=1).values, dtype=torch.float32)
    y_valid = torch.tensor(valid_df['yield'].values, dtype=torch.float32).view(-1, 1)
    X_test = torch.tensor(test_df.drop('yield', axis=1).values, dtype=torch.float32)
    y_test = torch.tensor(test_df['yield'].values, dtype=torch.float32).view(-1, 1)

    # Create data loaders
    train_loader = torch.utils.data.DataLoader(torch.utils.data.TensorDataset(X_train, y_train), batch_size=32, shuffle=True)
    valid_loader = torch.utils.data.DataLoader(torch.utils.data.TensorDataset(X_valid, y_valid), batch_size=32, shuffle=False)
    test_loader = torch.utils.data.DataLoader(torch.utils.data.TensorDataset(X_test, y_test), batch_size=32, shuffle=False)

    # Train the model
    model = train_model(model, train_loader, valid_loader, learning_rate=0.01, batch_size=32, num_epochs=100, dropout_rate=0.3)

    # Step 6. evaluate the trained model using the defined "evaluate_model" function
    model_performance = evaluate_model(model, test_loader)

    # Save the trained model
    os.makedirs('./agent_workspace/trained_models', exist_ok=True)
    torch.save(model.state_dict(), './agent_workspace/trained_models/wild_blueberry_yield_model.pth')

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
