import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import MinMaxScaler, OneHotEncoder
from sklearn.metrics import mean_absolute_error
from sklearn.ensemble import GradientBoostingRegressor
import joblib
from flask import Flask, request, jsonify

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data retrieval and loading
def load_dataset():
    # Load the dataset
    dataset = pd.read_csv(r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/mohs_hardness/train.csv')
    return dataset

# Data preprocessing
def preprocess_data(dataset):
    # Generate summary statistics
    summary_stats = dataset.describe()

    # Check for missing values and handle them
    missing_values = dataset.isnull().sum()
    dataset.fillna(dataset.mean(), inplace=True)

    # Calculate correlation matrix
    correlation_matrix = dataset.corr()

    # Separate target variable
    target = dataset['Mohs hardness']
    dataset = dataset.drop('Mohs hardness', axis=1)

    # Normalize numerical features
    numerical_features = dataset.select_dtypes(include=[np.number]).columns
    scaler = MinMaxScaler()
    dataset[numerical_features] = scaler.fit_transform(dataset[numerical_features])

    # One-Hot Encoding
    categorical_features = dataset.select_dtypes(include=['object']).columns
    dataset = pd.get_dummies(dataset, columns=categorical_features)

    # Create interaction terms
    interaction_terms = []
    for i in range(len(numerical_features)):
        for j in range(i+1, len(numerical_features)):
            interaction_term = dataset[numerical_features[i]] * dataset[numerical_features[j]]
            interaction_terms.append(interaction_term)
            dataset[f'interaction_{i}_{j}'] = interaction_term

    return dataset, target

# Data splitting
def split_data(dataset, target):
    # Split the dataset
    train_data, temp_data, train_target, temp_target = train_test_split(dataset, target, test_size=0.3, random_state=SEED)
    validation_data, test_data, validation_target, test_target = train_test_split(temp_data, temp_target, test_size=0.33, random_state=SEED)
    return train_data, validation_data, test_data, train_target, validation_target, test_target

# Model training
def train_model(train_data, train_target):
    # Define the model
    gbm = GradientBoostingRegressor(n_estimators=100, learning_rate=0.1, max_depth=3, random_state=SEED)

    # Train the model
    gbm.fit(train_data, train_target)
    return gbm

# Model evaluation
def evaluate_model(model, validation_data, validation_target, test_data, test_target):
    # Evaluate on validation set
    val_predictions = model.predict(validation_data)
    val_mae = mean_absolute_error(validation_target, val_predictions)

    # Evaluate on test set
    test_predictions = model.predict(test_data)
    test_mae = mean_absolute_error(test_target, test_predictions)

    performance_scores = {
        'Validation MAE': val_mae,
        'Test MAE': test_mae
    }
    return performance_scores

# Model deployment
def deploy_model(model):
    # Save the model
    model_path = './agent_workspace/trained_models/gbm_model.joblib'
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    joblib.dump(model, model_path)

    # Create Flask app
    app = Flask(__name__)

    @app.route('/predict', methods=['POST'])
    def predict():
        data = request.get_json(force=True)
        prediction = model.predict([data['features']])
        return jsonify({'prediction': prediction.tolist()})

    return app

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    dataset = load_dataset()

    # Step 2. Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
    train_data, validation_data, test_data, train_target, validation_target, test_target = split_data(dataset, dataset['Mohs hardness'])

    # Step 3. With the split dataset, run data preprocessing and feature engineering (if applicable) using the "preprocess_data" function you defined
    processed_train_data, train_target = preprocess_data(train_data)
    processed_validation_data, validation_target = preprocess_data(validation_data)
    processed_test_data, test_target = preprocess_data(test_data)

    # Step 4. Define required model. You may retrieve model from available hub or library along with pretrained weights (if any).
    # If pretrained or predefined model is not available, please create the model according to the given user's requirements below using PyTorch and relevant libraries.
    model = train_model(processed_train_data, train_target)

    # Step 5. train the retrieved/loaded model using the defined "train_model" function
    # Step 6. evaluate the trained model using the defined "evaluate_model" function
    model_performance = evaluate_model(model, processed_validation_data, validation_target, processed_test_data, test_target)

    # Step 7. Deploy the model
    app = deploy_model(model)

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
