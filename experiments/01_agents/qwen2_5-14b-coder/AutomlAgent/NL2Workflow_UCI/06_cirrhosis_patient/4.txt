# Data preprocessing and feature engineering
def preprocess_data(dataset):
    # Identify missing values
    missing_values = dataset.isnull().sum()
    print("Missing values:\n", missing_values)
    
    # Handle missing values with KNN Imputation
    imputer = KNNImputer(n_neighbors=5)
    dataset_imputed = pd.DataFrame(imputer.fit_transform(dataset), columns=dataset.columns)
    
    # Identify categorical features
    categorical_features = dataset.select_dtypes(include=['object', 'category']).columns
    
    # Apply One-Hot Encoding
    encoder = OneHotEncoder(sparse=False, drop='first')
    encoded_data = encoder.fit_transform(dataset_imputed[categorical_features])
    encoded_df = pd.DataFrame(encoded_data, columns=encoder.get_feature_names_out(categorical_features))
    dataset_encoded = pd.concat([dataset_imputed.drop(categorical_features, axis=1), encoded_df], axis=1)
    
    # Identify continuous features
    continuous_features = dataset.select_dtypes(include=['int64', 'float64']).columns
    
    # Normalize continuous features
    scaler = MinMaxScaler()
    dataset_normalized = dataset_encoded.copy()
    dataset_normalized[continuous_features] = scaler.fit_transform(dataset_encoded[continuous_features])
    
    return dataset_normalized
