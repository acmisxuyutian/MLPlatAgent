import os
import random
import time
import json
import numpy as np
import pandas as pd
import torch
import torch.nn as nn
import torch.optim as optim
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import MinMaxScaler, PolynomialFeatures
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_log_error
from torch.utils.data import DataLoader, TensorDataset

SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)

# Step 1: Retrieve and Load Dataset
train_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/08_santander_value/split_train.csv')
eval_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/08_santander_value/split_eval.csv')

# Step 2: Inspect Dataset
print(train_df.info())
print(train_df.describe())
print(train_df.isnull().sum())

print(eval_df.info())
print(eval_df.describe())
print(eval_df.isnull().sum())

# Step 3: Handle Missing Values
imputer = SimpleImputer(strategy='mean')
train_df_imputed = pd.DataFrame(imputer.fit_transform(train_df), columns=train_df.columns)
eval_df_imputed = pd.DataFrame(imputer.transform(eval_df), columns=eval_df.columns)

# Step 4: Feature Scaling
scaler = MinMaxScaler()
train_features = scaler.fit_transform(train_df_imputed.drop('target', axis=1))
eval_features = scaler.transform(eval_df_imputed.drop('target', axis=1))

# Step 5: Feature Engineering
poly = PolynomialFeatures(degree=2, include_bias=False)
train_features_poly = poly.fit_transform(train_features)
eval_features_poly = poly.transform(eval_features)

# Step 6: Data Splitting
train_features_poly, test_features_poly, train_target, test_target = train_test_split(
    train_features_poly, train_df_imputed['target'], test_size=0.2, random_state=SEED
)
eval_features_poly, test_features_poly, eval_target, test_target = train_test_split(
    test_features_poly, test_target, test_size=0.5, random_state=SEED
)

# Step 7: Define Model Architecture
class NeuralNetwork(nn.Module):
    def __init__(self, input_size):
        super(NeuralNetwork, self).__init__()
        self.fc1 = nn.Linear(input_size, 128)
        self.bn1 = nn.BatchNorm1d(128)
        self.dropout1 = nn.Dropout(0.2)
        self.fc2 = nn.Linear(128, 64)
        self.bn2 = nn.BatchNorm1d(64)
        self.dropout2 = nn.Dropout(0.2)
        self.fc3 = nn.Linear(64, 1)

    def forward(self, x):
        x = torch.relu(self.fc1(x))
        x = self.bn1(x)
        x = self.dropout1(x)
        x = torch.relu(self.fc2(x))
        x = self.bn2(x)
        x = self.dropout2(x)
        x = self.fc3(x)
        return x

input_size = train_features_poly.shape[1]
model = NeuralNetwork(input_size)

# Step 8: Compile Model
criterion = nn.MSELoss()
optimizer = optim.Adam(model.parameters(), lr=0.001)

# Step 9: Hyperparameter Tuning
class EarlyStopping:
    def __init__(self, patience=10, verbose=True):
        self.patience = patience
        self.verbose = verbose
        self.counter = 0
        self.best_score = None
        self.early_stop = False

    def __call__(self, val_loss):
        score = -val_loss

        if self.best_score is None:
            self.best_score = score
        elif score < self.best_score:
            self.counter += 1
            if self.verbose:
                print(f'EarlyStopping counter: {self.counter} out of {self.patience}')
            if self.counter >= self.patience:
                self.early_stop = True
        else:
            self.best_score = score
            self.counter = 0

# Convert data to PyTorch tensors
train_features_tensor = torch.tensor(train_features_poly, dtype=torch.float32)
train_target_tensor = torch.tensor(train_target.values, dtype=torch.float32).view(-1, 1)
eval_features_tensor = torch.tensor(eval_features_poly, dtype=torch.float32)
eval_target_tensor = torch.tensor(eval_target.values, dtype=torch.float32).view(-1, 1)
test_features_tensor = torch.tensor(test_features_poly, dtype=torch.float32)
test_target_tensor = torch.tensor(test_target.values, dtype=torch.float32).view(-1, 1)

# Create DataLoader
train_dataset = TensorDataset(train_features_tensor, train_target_tensor)
eval_dataset = TensorDataset(eval_features_tensor, eval_target_tensor)
test_dataset = TensorDataset(test_features_tensor, test_target_tensor)
train_loader = DataLoader(train_dataset, batch_size=32, shuffle=True)
eval_loader = DataLoader(eval_dataset, batch_size=32, shuffle=False)
test_loader = DataLoader(test_dataset, batch_size=32, shuffle=False)

scheduler = ReduceLROnPlateau(optimizer, mode='min', factor=0.2, patience=5, verbose=True)
early_stopping = EarlyStopping(patience=10, verbose=True)

# Step 10: Train Model
num_epochs = 100
for epoch in range(num_epochs):
    model.train()
    running_loss = 0.0
    for inputs, targets in train_loader:
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, torch.log1p(targets))
        loss.backward()
        optimizer.step()
        running_loss += loss.item()

    model.eval()
    eval_loss = 0.0
    with torch.no_grad():
        for inputs, targets in eval_loader:
            outputs = model(inputs)
            loss = criterion(outputs, torch.log1p(targets))
            eval_loss += loss.item()

    scheduler.step(eval_loss)
    print(f'Epoch {epoch+1}/{num_epochs}, Train Loss: {running_loss/len(train_loader)}, Eval Loss: {eval_loss/len(eval_loader)}')

    if early_stopping(eval_loss):
        print("Early stopping triggered")
        break

# Step 11: Evaluate Model
predictions = []
true_values = []
with torch.no_grad():
    for inputs, targets in test_loader:
        outputs = model(inputs)
        predictions.extend(outputs.numpy().flatten())
        true_values.extend(targets.numpy().flatten())

rmsle = np.sqrt(mean_squared_log_error(true_values, np.expm1(predictions)))
print(f'RMSLE: {rmsle}')

# Step 12: Save the Model
model_save_path = "./agent_workspace/trained_models/model.pth"
os.makedirs(os.path.dirname(model_save_path), exist_ok=True)
torch.save(model.state_dict(), model_save_path)
