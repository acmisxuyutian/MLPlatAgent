import os
import random
import time
import json
import numpy as np
import pandas as pd
import torch
import torch.nn as nn
import torch.optim as optim
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score, f1_score
from sklearn.preprocessing import MinMaxScaler, LabelEncoder
from sklearn.feature_selection import SelectKBest, f_classif

SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(train_data, eval_data):
    # Handle missing values
    train_data['Age'].fillna(train_data['Age'].median(), inplace=True)
    eval_data['Age'].fillna(eval_data['Age'].median(), inplace=True)

    train_data['Embarked'].fillna(train_data['Embarked'].mode()[0], inplace=True)
    eval_data['Embarked'].fillna(eval_data['Embarked'].mode()[0], inplace=True)

    # Encode categorical variables
    train_data = pd.get_dummies(train_data, columns=['Sex', 'Embarked'], drop_first=True)
    eval_data = pd.get_dummies(eval_data, columns=['Sex', 'Embarked'], drop_first=True)

    # Feature scaling
    scaler = MinMaxScaler()

    train_data_scaled = scaler.fit_transform(train_data.drop('Survived', axis=1))
    eval_data_scaled = scaler.transform(eval_data.drop('Survived', axis=1))

    # Feature selection
    selector = SelectKBest(score_func=f_classif, k=10)

    train_data_selected = selector.fit_transform(train_data_scaled, train_data['Survived'])
    eval_data_selected = selector.transform(eval_data_scaled)

    return train_data_selected, eval_data_selected, train_data['Survived'], eval_data['Survived']

def train_model(model, train_loader, criterion, optimizer, num_epochs=50):
    model.train()
    for epoch in range(num_epochs):
        for inputs, labels in train_loader:
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, labels)
            loss.backward()
            optimizer.step()
        print(f'Epoch [{epoch+1}/{num_epochs}], Loss: {loss.item():.4f}')
    return model

def evaluate_model(model, eval_features, eval_labels):
    model.eval()
    with torch.no_grad():
        outputs = model(eval_features)
        predictions = (outputs > 0.5).float()

    y_true = eval_labels.numpy()
    y_pred = predictions.numpy()

    performance_scores = {
        'ACC': accuracy_score(y_true, y_pred),
        'F1': f1_score(y_true, y_pred)
    }

    return performance_scores

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/04_titanic/split_train.csv'
    eval_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/04_titanic/split_eval.csv'

    train_data = pd.read_csv(train_data_path)
    eval_data = pd.read_csv(eval_data_path)

    # Step 2. Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
    # Here, the train_loader contains 70% of the `dataset`, the valid_loader contains 20% of the `dataset`, and the test_loader contains 10% of the `dataset`.
    train_data, valid_data = train_test_split(train_data, test_size=0.3, random_state=SEED)
    valid_data, test_data = train_test_split(valid_data, test_size=0.33, random_state=SEED)

    # Step 3. With the split dataset, run data preprocessing and feature engineering (if applicable) using the "preprocess_data" function you defined
    train_features, eval_features, train_labels, eval_labels = preprocess_data(train_data, eval_data)

    # Convert to PyTorch tensors
    train_features = torch.tensor(train_features, dtype=torch.float32)
    train_labels = torch.tensor(train_labels.values, dtype=torch.float32).view(-1, 1)
    eval_features = torch.tensor(eval_features, dtype=torch.float32)
    eval_labels = torch.tensor(eval_labels.values, dtype=torch.float32).view(-1, 1)

    # Create DataLoader
    train_dataset = torch.utils.data.TensorDataset(train_features, train_labels)
    train_loader = torch.utils.data.DataLoader(train_dataset, batch_size=32, shuffle=True)

    # Step 4. Define required model. You may retrieve model from available hub or library along with pretrained weights (if any).
    # If pretrained or predefined model is not available, please create the model according to the given user's requirements below using PyTorch and relevant libraries.
    input_size = train_features.shape[1]
    model = TitanicModel(input_size)

    # Step 5. train the retrieved/loaded model using the defined "train_model" function
    criterion = nn.BCELoss()
    optimizer = optim.Adam(model.parameters(), lr=0.001)

    model = train_model(model, train_loader, criterion, optimizer)

    # Step 6. evaluate the trained model using the defined "evaluate_model" function
    model_performance = evaluate_model(model, eval_features, eval_labels)

    return model_performance

# Define the neural network architecture
class TitanicModel(nn.Module):
    def __init__(self, input_size):
        super(TitanicModel, self).__init__()
        self.fc1 = nn.Linear(input_size, 128)
        self.fc2 = nn.Linear(128, 128)
        self.fc3 = nn.Linear(128, 1)
        self.dropout = nn.Dropout(0.3)

    def forward(self, x):
        x = torch.relu(self.fc1(x))
        x = self.dropout(x)
        x = torch.relu(self.fc2(x))
        x = self.dropout(x)
        x = torch.sigmoid(self.fc3(x))
        return x

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
