import os, random, time, json
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import OneHotEncoder, MinMaxScaler
from sklearn.feature_selection import mutual_info_classif
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, TensorDataset

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data retrieval and loading
def load_data():
    train_data = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/04_titanic/split_train.csv')
    eval_data = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/04_titanic/split_eval.csv')
    return train_data, eval_data

# Data preprocessing and feature engineering
def preprocess_data(train_data, eval_data):
    # Imputation
    numerical_imputer = SimpleImputer(strategy='mean')
    train_data[['Age', 'Fare']] = numerical_imputer.fit_transform(train_data[['Age', 'Fare']])
    eval_data[['Age', 'Fare']] = numerical_imputer.transform(eval_data[['Age', 'Fare']])

    categorical_imputer = SimpleImputer(strategy='most_frequent')
    train_data[['Embarked']] = categorical_imputer.fit_transform(train_data[['Embarked']])
    eval_data[['Embarked']] = categorical_imputer.transform(eval_data[['Embarked']])

    # Encoding
    one_hot_encoder = OneHotEncoder(sparse_output=False)
    encoded_train_features = one_hot_encoder.fit_transform(train_data[['Sex', 'Embarked']])
    encoded_eval_features = one_hot_encoder.transform(eval_data[['Sex', 'Embarked']])

    encoded_train_df = pd.DataFrame(encoded_train_features, columns=one_hot_encoder.get_feature_names_out(['Sex', 'Embarked']))
    encoded_eval_df = pd.DataFrame(encoded_eval_features, columns=one_hot_encoder.get_feature_names_out(['Sex', 'Embarked']))

    train_data = pd.concat([train_data, encoded_train_df], axis=1)
    eval_data = pd.concat([eval_data, encoded_eval_df], axis=1)

    train_data.drop(['Sex', 'Embarked'], axis=1, inplace=True)
    eval_data.drop(['Sex', 'Embarked'], axis=1, inplace=True)

    # Feature Selection
    X_train = train_data.drop('Survived', axis=1)
    y_train = train_data['Survived']
    X_eval = eval_data.drop('Survived', axis=1)
    y_eval = eval_data['Survived']

    # Compute correlation matrix for numerical features only
    numeric_features = X_train.select_dtypes(include=[np.number]).columns
    correlation_matrix = X_train[numeric_features].corr()
    mutual_info_scores = mutual_info_classif(X_train[numeric_features], y_train)

    # Select relevant features based on correlation and mutual information
    selected_features = correlation_matrix['Survived'].sort_values(ascending=False).index[1:]  # Exclude 'Survived'
    selected_features = list(set(selected_features).intersection(numeric_features))  # Ensure only numeric features are selected
    X_train_selected = X_train[selected_features]
    X_eval_selected = X_eval[selected_features]

    # Feature Scaling
    scaler = MinMaxScaler()
    X_train_scaled = scaler.fit_transform(X_train_selected)
    X_eval_scaled = scaler.transform(X_eval_selected)

    return X_train_scaled, y_train, X_eval_scaled, y_eval

# Model design and training
class TitanicSurvivalModel(nn.Module):
    def __init__(self, input_size):
        super(TitanicSurvivalModel, self).__init__()
        self.fc1 = nn.Linear(input_size, 64)
        self.fc2 = nn.Linear(64, 32)
        self.fc3 = nn.Linear(32, 1)
        self.relu = nn.ReLU()
        self.sigmoid = nn.Sigmoid()

    def forward(self, x):
        x = self.relu(self.fc1(x))
        x = self.relu(self.fc2(x))
        x = self.sigmoid(self.fc3(x))
        return x

def train_model(X_train_scaled, y_train):
    # Convert to PyTorch tensors
    X_train_tensor = torch.tensor(X_train_scaled, dtype=torch.float32)
    y_train_tensor = torch.tensor(y_train.values, dtype=torch.float32).view(-1, 1)

    # Create DataLoader
    train_dataset = TensorDataset(X_train_tensor, y_train_tensor)
    train_loader = DataLoader(train_dataset, batch_size=32, shuffle=True)

    # Define the neural network
    model = TitanicSurvivalModel(input_size=X_train_scaled.shape[1])
    criterion = nn.BCELoss()
    optimizer = optim.Adam(model.parameters(), lr=0.01)

    # Training loop
    num_epochs = 50
    best_val_loss = float('inf')
    patience = 5
    early_stopping_counter = 0

    for epoch in range(num_epochs):
        model.train()
        train_loss = 0.0
        for inputs, labels in train_loader:
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, labels)
            loss.backward()
            optimizer.step()
            train_loss += loss.item()

        train_loss /= len(train_loader)

        model.eval()
        val_loss = 0.0
        with torch.no_grad():
            for inputs, labels in train_loader:
                outputs = model(inputs)
                loss = criterion(outputs, labels)
                val_loss += loss.item()

        val_loss /= len(train_loader)

        print(f'Epoch {epoch+1}, Train Loss: {train_loss:.4f}, Val Loss: {val_loss:.4f}')

        if val_loss < best_val_loss:
            best_val_loss = val_loss
            early_stopping_counter = 0
            torch.save(model.state_dict(), './agent_workspace/trained_models/titanic_survival_model.pth')
        else:
            early_stopping_counter += 1
            if early_stopping_counter >= patience:
                print('Early stopping triggered.')
                break

    return model

# Model evaluation
def evaluate_model(model, X_eval_scaled, y_eval):
    model.load_state_dict(torch.load('./agent_workspace/trained_models/titanic_survival_model.pth'))
    model.eval()

    with torch.no_grad():
        eval_outputs = model(torch.tensor(X_eval_scaled, dtype=torch.float32))
        eval_predictions = (eval_outputs > 0.5).float().numpy().flatten()

    eval_accuracy = accuracy_score(y_eval, eval_predictions)
    eval_precision = precision_score(y_eval, eval_predictions)
    eval_recall = recall_score(y_eval, eval_predictions)
    eval_f1 = f1_score(y_eval, eval_predictions)

    performance_scores = {
        'Accuracy': eval_accuracy,
        'Precision': eval_precision,
        'Recall': eval_recall,
        'F1-Score': eval_f1
    }

    return performance_scores

    """

    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_data, eval_data = load_data()

    # Step 3. With the split dataset, run data preprocessing and feature engineering (if applicable) using the "preprocess_data" function you defined
    X_train_scaled, y_train, X_eval_scaled, y_eval = preprocess_data(train_data, eval_data)

    # Step 4. Define required model. You may retrieve model from available hub or library along with pretrained weights (if any).
    # If pretrained or predefined model is not available, please create the model according to the given user's requirements below using PyTorch and relevant libraries.
    model = None
