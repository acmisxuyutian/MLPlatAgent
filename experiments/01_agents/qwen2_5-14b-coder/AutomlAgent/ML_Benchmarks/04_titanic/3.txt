# The following code is for "tabular classification" task.

import os, random, time, json
import numpy as np
import pandas as pd
from sklearn.metrics import accuracy_score, f1_score
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import MinMaxScaler
from sklearn.impute import SimpleImputer
import torch
import torch.nn as nn
import torch.optim as optim

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(train_val_data, test_data):
    # One-hot encoding for categorical variables
    train_val_data = pd.get_dummies(train_val_data, columns=['Sex', 'Embarked'], drop_first=True)
    test_data = pd.get_dummies(test_data, columns=['Sex', 'Embarked'], drop_first=True)

    # Min-Max Scaling
    scaler = MinMaxScaler()
    numerical_features = ['Age', 'Fare']
    train_val_data[numerical_features] = scaler.fit_transform(train_val_data[numerical_features])
    test_data[numerical_features] = scaler.transform(test_data[numerical_features])

    # Impute missing values
    imputer = SimpleImputer(strategy='mean')
    train_val_data['Age'] = imputer.fit_transform(train_val_data[['Age']])
    test_data['Age'] = imputer.transform(test_data[['Age']])

    imputer = SimpleImputer(strategy='most_frequent')
    train_val_data['Embarked'] = imputer.fit_transform(train_val_data[['Embarked']])
    test_data['Embarked'] = imputer.transform(test_data[['Embarked']])

    # Feature Engineering
    train_val_data['FamilySize'] = train_val_data['SibSp'] + train_val_data['Parch'] + 1
    test_data['FamilySize'] = test_data['SibSp'] + test_data['Parch'] + 1

    train_val_data['Title'] = train_val_data['Name'].str.extract(' ([A-Za-z]+)\.', expand=False)
    test_data['Title'] = test_data['Name'].str.extract(' ([A-Za-z]+)\.', expand=False)

    train_val_data = pd.get_dummies(train_val_data, columns=['Title'], drop_first=True)
    test_data = pd.get_dummies(test_data, columns=['Title'], drop_first=True)

    # Splitting the dataset
    X = train_val_data.drop('Survived', axis=1)
    y = train_val_data['Survived']

    X_train, X_val, y_train, y_val = train_test_split(X, y, test_size=0.3, stratify=y, random_state=42)

    return X_train, X_val, y_train, y_val, test_data.drop('Survived', axis=1)

# Define the neural network model
class SimpleFFNN(nn.Module):
    def __init__(self, input_size):
        super(SimpleFFNN, self).__init__()
        self.fc1 = nn.Linear(input_size, 64)
        self.fc2 = nn.Linear(64, 32)
        self.fc3 = nn.Linear(32, 1)
        self.relu = nn.ReLU()
        self.sigmoid = nn.Sigmoid()

    def forward(self, x):
        x = self.relu(self.fc1(x))
        x = self.relu(self.fc2(x))
        x = self.sigmoid(self.fc3(x))
        return x

# Train the model
def train_model(model, train_loader, valid_loader, epochs=10, learning_rate=0.001):
    criterion = nn.BCELoss()
    optimizer = optim.Adam(model.parameters(), lr=learning_rate)

    for epoch in range(epochs):
        model.train()
        for inputs, labels in train_loader:
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, labels.view(-1, 1))
            loss.backward()
            optimizer.step()

        model.eval()
        with torch.no_grad():
            val_loss = 0
            for inputs, labels in valid_loader:
                outputs = model(inputs)
                loss = criterion(outputs, labels.view(-1, 1))
                val_loss += loss.item()

        print(f'Epoch {epoch+1}, Validation Loss: {val_loss/len(valid_loader)}')

    return model

# Evaluate the model
def evaluate_model(model, test_loader):
    model.eval()
    y_true = []
    y_pred = []

    with torch.no_grad():
        for inputs, labels in test_loader:
            outputs = model(inputs)
            predictions = (outputs > 0.5).float()
            y_true.extend(labels.numpy())
            y_pred.extend(predictions.numpy())

    performance_scores = {
        'ACC': accuracy_score(y_true, y_pred),
        'F1': f1_score(y_true, y_pred)
    }

    return performance_scores

# Save the trained model
def save_model(model, path):
    torch.save(model.state_dict(), path)

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_val_data = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/04_titanic/split_train.csv')
    test_data = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/04_titanic/split_eval.csv')

    # Step 2. Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
    X_train, X_val, y_train, y_val, X_test = preprocess_data(train_val_data, test_data)

    # Convert to PyTorch tensors
    X_train_tensor = torch.tensor(X_train.values, dtype=torch.float32)
    y_train_tensor = torch.tensor(y_train.values, dtype=torch.float32)
    X_val_tensor = torch.tensor(X_val.values, dtype=torch.float32)
    y_val_tensor = torch.tensor(y_val.values, dtype=torch.float32)
    X_test_tensor = torch.tensor(X_test.values, dtype=torch.float32)

    # Create data loaders
    train_loader = torch.utils.data.DataLoader(torch.utils.data.TensorDataset(X_train_tensor, y_train_tensor), batch_size=32, shuffle=True)
    valid_loader = torch.utils.data.DataLoader(torch.utils.data.TensorDataset(X_val_tensor, y_val_tensor), batch_size=32, shuffle=False)
    test_loader = torch.utils.data.DataLoader(torch.utils.data.TensorDataset(X_test_tensor, torch.tensor(test_data['Survived'].values, dtype=torch.float32)), batch_size=32, shuffle=False)

    # Step 3. With the split dataset, run data preprocessing and feature engineering (if applicable) using the "preprocess_data" function you defined
    # This is already done in the preprocess_data function

    # Step 4. Define required model. You may retrieve model from available hub or library along with pretrained weights (if any).
    # If pretrained or predefined model is not available, please create the model according to the given user's requirements below using PyTorch and relevant libraries.
    model = SimpleFFNN(input_size=X_train.shape[1])

    # Step 5. train the retrieved/loaded model using the defined "train_model" function
    # on top of the model training, please run hyperparamter optimization based on the suggested hyperparamters and their values before proceeding to the evaluation step to ensure model's optimality
    model = train_model(model, train_loader, valid_loader)

    # Step 6. evaluate the trained model using the defined "evaluate_model" function
    model_performance = evaluate_model(model, test_loader)

    # Save the trained model
    save_model(model, './agent_workspace/trained_models/titanic_model.pth')

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
