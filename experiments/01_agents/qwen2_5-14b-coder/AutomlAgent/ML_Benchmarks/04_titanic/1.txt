import os, random, time, json
import numpy as np
import pandas as pd
import torch
import torch.nn as nn
import torch.optim as optim
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.feature_selection import mutual_info_classif
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score, f1_score

SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(train_df, eval_df):
    # Handle missing values
    train_df['Age'].fillna(train_df['Age'].median(), inplace=True)
    eval_df['Age'].fillna(eval_df['Age'].median(), inplace=True)
    train_df['Embarked'].fillna(train_df['Embarked'].mode()[0], inplace=True)
    eval_df['Embarked'].fillna(eval_df['Embarked'].mode()[0], inplace=True)

    # Encode categorical variables
    train_df = pd.get_dummies(train_df, columns=['Sex', 'Embarked'])
    eval_df = pd.get_dummies(eval_df, columns=['Sex', 'Embarked'])

    # Feature selection
    if 'Survived' not in train_df.columns:
        raise ValueError("Survived column not found in the training data")

    # Calculate correlation matrix
    corr_matrix = train_df.corr(numeric_only=True)
    relevant_features_corr = corr_matrix['Survived'].abs().sort_values(ascending=False).index[1:]

    # Calculate mutual information
    X = train_df.drop('Survived', axis=1)
    y = train_df['Survived']
    mi = mutual_info_classif(X, y)
    mi_series = pd.Series(mi, index=X.columns).sort_values(ascending=False)
    relevant_features_mi = mi_series.index[:5]

    # Extract feature importance using RandomForestClassifier
    rf = RandomForestClassifier(random_state=42)
    rf.fit(X, y)
    feature_importance = pd.Series(rf.feature_importances_, index=X.columns).sort_values(ascending=False)
    relevant_features_rf = feature_importance.index[:5]

    # Combine relevant features
    relevant_features = set(relevant_features_corr).union(set(relevant_features_mi)).union(set(relevant_features_rf))

    # Feature scaling
    scaler = StandardScaler()
    X_train = scaler.fit_transform(train_df[relevant_features])
    X_eval = scaler.transform(eval_df[relevant_features])
    y_train = train_df['Survived']
    y_eval = eval_df['Survived']

    return X_train, X_eval, y_train, y_eval, relevant_features

# Define the model architecture
class TitanicModel(nn.Module):
    def __init__(self, input_size):
        super(TitanicModel, self).__init__()
        self.fc1 = nn.Linear(input_size, 64)
        self.fc2 = nn.Linear(64, 64)
        self.fc3 = nn.Linear(64, 1)
        self.relu = nn.ReLU()
        self.sigmoid = nn.Sigmoid()

    def forward(self, x):
        x = self.relu(self.fc1(x))
        x = self.relu(self.fc2(x))
        x = self.sigmoid(self.fc3(x))
        return x

# Train the model
def train_model(model, X_train, y_train, X_val, y_val, num_epochs=100, batch_size=32, learning_rate=0.001):
    criterion = nn.BCELoss()
    optimizer = optim.Adam(model.parameters(), lr=learning_rate)

    train_losses = []
    val_losses = []

    for epoch in range(num_epochs):
        model.train()
        for i in range(0, len(X_train), batch_size):
            inputs = torch.tensor(X_train[i:i+batch_size], dtype=torch.float32)
            labels = torch.tensor(y_train[i:i+batch_size], dtype=torch.float32).view(-1, 1)

            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, labels)
            loss.backward()
            optimizer.step()

        model.eval()
        with torch.no_grad():
            val_inputs = torch.tensor(X_val, dtype=torch.float32)
            val_labels = torch.tensor(y_val, dtype=torch.float32).view(-1, 1)
            val_outputs = model(val_inputs)
            val_loss = criterion(val_outputs, val_labels)

        train_losses.append(loss.item())
        val_losses.append(val_loss.item())

        print(f'Epoch [{epoch+1}/{num_epochs}], Loss: {loss.item():.4f}, Val Loss: {val_loss.item():.4f}')

    return model, train_losses, val_losses

# Evaluate the model
def evaluate_model(model, X_eval, y_eval):
    model.eval()
    with torch.no_grad():
        eval_inputs = torch.tensor(X_eval, dtype=torch.float32)
        eval_labels = torch.tensor(y_eval, dtype=torch.float32).view(-1, 1)
        eval_outputs = model(eval_inputs)
        eval_predictions = (eval_outputs > 0.5).float()
        accuracy = (eval_predictions == eval_labels).float().mean().item()

    performance_scores = {
        'ACC': accuracy
    }

    return performance_scores

# Save the trained model
def save_model(model, path):
    torch.save(model.state_dict(), path)

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    """
    Main function to execute the tabular classification pipeline.
    """

    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/04_titanic/split_train.csv'
    eval_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/04_titanic/split_eval.csv'

    train_df = pd.read_csv(train_data_path)
    eval_df = pd.read_csv(eval_data_path)

    # Step 2. Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
    # Here, the train_loader contains 70% of the `dataset`, the valid_loader contains 20% of the `dataset`, and the test_loader contains 10% of the `dataset`.
    X_train, X_eval, y_train, y_eval, relevant_features = preprocess_data(train_df, eval_df)
    X_train, X_val, y_train, y_val = train_test_split(X_train, y_train, test_size=0.15, random_state=SEED)

    # Step 3. With the split dataset, run data preprocessing and feature engineering (if applicable) using the "preprocess_data" function you defined
    input_size = len(relevant_features)

    # Step 4. Define required model. You may retrieve model from available hub or library along with pretrained weights (if any).
    # If pretrained or predefined model is not available, please create the model according to the given user's requirements below using PyTorch and relevant libraries.
    model = TitanicModel(input_size)

    # Step 5. train the retrieved/loaded model using the defined "train_model" function
    # on top of the model training, please run hyperparamter optimization based on the suggested hyperparamters and their values before proceeding to the evaluation step to ensure model's optimality
    model, train_losses, val_losses = train_model(model, X_train, y_train, X_val, y_val)

    # Step 6. evaluate the trained model using the defined "evaluate_model" function
    model_performance = evaluate_model(model, X_eval, y_eval)

    # Step 7. Save the trained model
    model_save_path = "./agent_workspace/trained_models/titanic_model.pth"
    save_model(model, model_save_path)

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
