######代码######

import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.metrics import accuracy_score, f1_score, roc_auc_score
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.decomposition import PCA
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, TensorDataset

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Load the train dataset
train_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/06_santander_customer/split_train.csv'
train_df = pd.read_csv(train_data_path)

# Load the eval dataset
eval_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/06_santander_customer/split_eval.csv'
eval_df = pd.read_csv(eval_data_path)

# Display basic information
print(train_df.info())
print(eval_df.info())

# Display the first few rows
print(train_df.head())
print(eval_df.head())

# Check for missing values
print(train_df.isnull().sum())
print(eval_df.isnull().sum())

# Analyze the target variable distribution
print(train_df['target'].value_counts(normalize=True))
print(eval_df['target'].value_counts(normalize=True))

# Handle missing values
train_df.fillna(train_df.mean(), inplace=True)
eval_df.fillna(eval_df.mean(), inplace=True)

# Separate features and target
X_train = train_df.drop('target', axis=1)
y_train = train_df['target']
X_eval = eval_df.drop('target', axis=1)
y_eval = eval_df['target']

# Identify categorical columns
categorical_cols = X_train.select_dtypes(include=['object']).columns

# One-hot encode categorical columns
encoder = OneHotEncoder(sparse_output=False)
X_train_encoded = encoder.fit_transform(X_train[categorical_cols])
X_eval_encoded = encoder.transform(X_eval[categorical_cols])

# Combine scaled numerical features and encoded categorical features
X_train_final = np.concatenate([X_train.drop(categorical_cols, axis=1).values, X_train_encoded], axis=1)
X_eval_final = np.concatenate([X_eval.drop(categorical_cols, axis=1).values, X_eval_encoded], axis=1)

# Scale the features
scaler = StandardScaler()
X_train_scaled = scaler.fit_transform(X_train_final)
X_eval_scaled = scaler.transform(X_eval_final)

# Check if 'feature1' and 'feature2' exist in the dataset
if 'feature1' in X_train.columns and 'feature2' in X_train.columns:
    train_df['interaction_feature'] = train_df['feature1'] * train_df['feature2']
    eval_df['interaction_feature'] = eval_df['feature1'] * eval_df['feature2']
else:
    print("Columns 'feature1' and/or 'feature2' do not exist in the dataset. Skipping interaction feature creation.")

# Separate features and target after creating new features
X_train = train_df.drop('target', axis=1)
y_train = train_df['target']
X_eval = eval_df.drop('target', axis=1)
y_eval = eval_df['target']

# Identify categorical columns
categorical_cols = X_train.select_dtypes(include=['object']).columns

# One-hot encode categorical columns
X_train_encoded = encoder.fit_transform(X_train[categorical_cols])
X_eval_encoded = encoder.transform(X_eval[categorical_cols])

# Combine scaled numerical features and encoded categorical features
X_train_final = np.concatenate([X_train.drop(categorical_cols, axis=1).values, X_train_encoded], axis=1)
X_eval_final = np.concatenate([X_eval.drop(categorical_cols, axis=1).values, X_eval_encoded], axis=1)

# Scale the features
X_train_scaled = scaler.fit_transform(X_train_final)
X_eval_scaled = scaler.transform(X_eval_final)

# Reduce dimensionality using PCA
pca = PCA(n_components=0.95)  # Retain 95% of variance
X_train_pca = pca.fit_transform(X_train_scaled)
X_eval_pca = pca.transform(X_eval_scaled)

# Define the FFNN model
class FFNN(nn.Module):
    def __init__(self, input_size, hidden_size, num_classes):
        super(FFNN, self).__init__()
        self.fc1 = nn.Linear(input_size, hidden_size)
        self.relu = nn.ReLU()
        self.fc2 = nn.Linear(hidden_size, num_classes)
        self.sigmoid = nn.Sigmoid()

    def forward(self, x):
        out = self.fc1(x)
        out = self.relu(out)
        out = self.fc2(out)
        out = self.sigmoid(out)
        return out

# Hyperparameters
input_size = X_train_pca.shape[1]
hidden_size = 64
num_classes = 1
num_epochs = 50
batch_size = 32
learning_rate = 0.001

# Convert data to PyTorch tensors
X_train_tensor = torch.tensor(X_train_pca, dtype=torch.float32)
y_train_tensor = torch.tensor(y_train.values, dtype=torch.float32).view(-1, 1)
X_eval_tensor = torch.tensor(X_eval_pca, dtype=torch.float32)
y_eval_tensor = torch.tensor(y_eval.values, dtype=torch.float32).view(-1, 1)

# Create DataLoader
train_dataset = TensorDataset(X_train_tensor, y_train_tensor)
train_loader = DataLoader(dataset=train_dataset, batch_size=batch_size, shuffle=True)

# Initialize the model, loss function, and optimizer
model = FFNN(input_size, hidden_size, num_classes)
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=learning_rate)

# Training loop
for epoch in range(num_epochs):
    for i, (inputs, labels) in enumerate(train_loader):
        # Forward pass
        outputs = model(inputs)
        loss = criterion(outputs, labels)

        # Backward and optimize
        optimizer.zero_grad()
        loss.backward()
        optimizer.step()

    if (epoch+1) % 5 == 0:
        print(f'Epoch [{epoch+1}/{num_epochs}], Loss: {loss.item():.4f}')

# Evaluation
with torch.no_grad():
    model.eval()
    outputs = model(X_eval_tensor)
    preds = outputs.numpy().flatten()

# Calculate AUC
auc = roc_auc_score(y_eval, preds)
print(f'AUC: {auc:.4f}')

# Save the trained model
model_save_path = './agent_workspace/trained_models/santander_customer_ffnn.pth'
torch.save(model.state_dict(), model_save_path)
print(f'Model saved to {model_save_path}')
