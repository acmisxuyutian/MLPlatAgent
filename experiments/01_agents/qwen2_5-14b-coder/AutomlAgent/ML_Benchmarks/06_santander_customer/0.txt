import os, random, time, json
import numpy as np
import pandas as pd
import torch
import torch.nn as nn
import torch.optim as optim
from sklearn.metrics import accuracy_score, f1_score, roc_auc_score
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import MinMaxScaler, OneHotEncoder
from sklearn.feature_selection import SelectKBest, f_classif
import torch.utils.data as data

SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(train_data, eval_data):
    # Check for missing values
    print(train_data.isnull().sum())
    print(eval_data.isnull().sum())

    # Impute missing values for numerical features
    numerical_features = train_data.select_dtypes(include=['float64', 'int64']).columns
    imputer = SimpleImputer(strategy='mean')
    train_data[numerical_features] = imputer.fit_transform(train_data[numerical_features])
    eval_data[numerical_features] = imputer.transform(eval_data[numerical_features])

    # Impute missing values for categorical features
    categorical_features = train_data.select_dtypes(include=['object']).columns
    imputer = SimpleImputer(strategy='most_frequent')
    train_data[categorical_features] = imputer.fit_transform(train_data[categorical_features])
    eval_data[categorical_features] = imputer.transform(eval_data[categorical_features])

    # Normalize numerical features
    scaler = MinMaxScaler()
    train_data[numerical_features] = scaler.fit_transform(train_data[numerical_features])
    eval_data[numerical_features] = scaler.transform(eval_data[numerical_features])

    # Encode categorical variables
    encoder = OneHotEncoder(sparse=False, handle_unknown='ignore')
    train_encoded = encoder.fit_transform(train_data[categorical_features])
    eval_encoded = encoder.transform(eval_data[categorical_features])

    # Create a DataFrame with the encoded features
    train_encoded_df = pd.DataFrame(train_encoded, columns=encoder.get_feature_names_out(categorical_features))
    eval_encoded_df = pd.DataFrame(eval_encoded, columns=encoder.get_feature_names_out(categorical_features))

    # Concatenate the encoded features with the numerical features
    train_data = pd.concat([train_data[numerical_features], train_encoded_df], axis=1)
    eval_data = pd.concat([eval_data[numerical_features], eval_encoded_df], axis=1)

    # Feature selection
    X_train = train_data.drop('target', axis=1)
    y_train = train_data['target']
    X_eval = eval_data.drop('target', axis=1)
    y_eval = eval_data['target']

    selector = SelectKBest(score_func=f_classif, k=10)
    X_train_selected = selector.fit_transform(X_train, y_train)
    X_eval_selected = selector.transform(X_eval)

    # Get the selected feature names
    selected_features = X_train.columns[selector.get_support()]

    # Feature transformation
    for feature in selected_features:
        if X_train[feature].skew() > 1:
            X_train[feature] = np.log1p(X_train[feature])
            X_eval[feature] = np.log1p(X_eval[feature])

    return X_train_selected, y_train, X_eval_selected, y_eval

# Define the Neural Network Model
class NeuralNetwork(nn.Module):
    def __init__(self, input_size):
        super(NeuralNetwork, self).__init__()
        self.fc1 = nn.Linear(input_size, 128)
        self.fc2 = nn.Linear(128, 64)
        self.fc3 = nn.Linear(64, 1)
        self.dropout = nn.Dropout(0.2)
        self.bn1 = nn.BatchNorm1d(128)
        self.bn2 = nn.BatchNorm1d(64)

    def forward(self, x):
        x = torch.relu(self.fc1(x))
        x = self.bn1(x)
        x = self.dropout(x)
        x = torch.relu(self.fc2(x))
        x = self.bn2(x)
        x = self.dropout(x)
        x = torch.sigmoid(self.fc3(x))
        return x

# Train the model
def train_model(model, train_loader, criterion, optimizer, num_epochs=10):
    for epoch in range(num_epochs):
        model.train()
        for inputs, targets in train_loader:
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, targets)
            loss.backward()
            optimizer.step()

        # Evaluation
        model.eval()
        with torch.no_grad():
            eval_loss = 0
            correct = 0
            total = 0
            for inputs, targets in eval_loader:
                outputs = model(inputs)
                loss = criterion(outputs, targets)
                eval_loss += loss.item()
                predicted = (outputs > 0.5).float()
                correct += (predicted == targets).sum().item()
                total += targets.size(0)

        print(f'Epoch [{epoch+1}/{num_epochs}], Loss: {eval_loss/len(eval_loader):.4f}, Accuracy: {100*correct/total:.2f}%')
    return model

# Evaluate the model
def evaluate_model(model, eval_loader):
    model.eval()
    with torch.no_grad():
        y_pred = []
        y_true = []
        for inputs, targets in eval_loader:
            outputs = model(inputs)
            y_pred.extend(outputs.numpy())
            y_true.extend(targets.numpy())

    auc = roc_auc_score(y_true, y_pred)
    print(f'AUC: {auc:.4f}')
    return {'AUC': auc}

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/06_santander_customer/split_train.csv'
    eval_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/06_santander_customer/split_eval.csv'

    train_data = pd.read_csv(train_data_path)
    eval_data = pd.read_csv(eval_data_path)

    # Step 2. Preprocess the data
    X_train_selected, y_train, X_eval_selected, y_eval = preprocess_data(train_data, eval_data)

    # Step 3. Convert data to PyTorch tensors
    X_train_tensor = torch.tensor(X_train_selected, dtype=torch.float32)
    y_train_tensor = torch.tensor(y_train.values, dtype=torch.float32).view(-1, 1)
    X_eval_tensor = torch.tensor(X_eval_selected, dtype=torch.float32)
    y_eval_tensor = torch.tensor(y_eval.values, dtype=torch.float32).view(-1, 1)

    # Step 4. Create data loaders
    train_dataset = data.TensorDataset(X_train_tensor, y_train_tensor)
    eval_dataset = data.TensorDataset(X_eval_tensor, y_eval_tensor)

    train_loader = data.DataLoader(train_dataset, batch_size=32, shuffle=True)
    eval_loader = data.DataLoader(eval_dataset, batch_size=32, shuffle=False)

    # Step 5. Define required model
    input_size = X_train_selected.shape[1]
    model = NeuralNetwork(input_size)

    # Step 6. Compile the model
    criterion = nn.BCELoss()
    optimizer = optim.Adam(model.parameters(), lr=0.001)

    # Step 7. Train the model
    model = train_model(model, train_loader, criterion, optimizer)

    # Step 8. Evaluate the model
    model_performance = evaluate_model(model, eval_loader)

    # Step 9. Save the trained model
    model_save_path = './agent_workspace/trained_models/santander_customer_model.pth'
    torch.save(model.state_dict(), model_save_path)
    print(f'Model saved to {model_save_path}')

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
