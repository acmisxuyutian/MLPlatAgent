# The following code is for "tabular classification" task.

import os, random, time, json
import numpy as np
import pandas as pd
import torch
from sklearn.model_selection import train_test_split
from sklearn.metrics import roc_auc_score
from pytorch_tabnet.tab_model import TabNetClassifier
from feature_engine.creation import PolynomialFeatures
from feature_engine.selection import SelectByTargetMean

SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)

# Data retrieval and preparation
def load_data():
    train_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/06_santander_customer/split_train.csv'
    eval_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/06_santander_customer/split_eval.csv'

    train_df = pd.read_csv(train_data_path)
    eval_df = pd.read_csv(eval_data_path)

    # Data inspection
    print(train_df.info())
    print(train_df.describe())
    print(train_df['target'].value_counts())

    # Data cleaning
    train_df.fillna(train_df.median(), inplace=True)
    eval_df.fillna(eval_df.median(), inplace=True)

    Q1 = train_df.quantile(0.25)
    Q3 = train_df.quantile(0.75)
    IQR = Q3 - Q1
    train_df = train_df[~((train_df < (Q1 - 1.5 * IQR)) | (train_df > (Q3 + 1.5 * IQR))).any(axis=1)]
    eval_df = eval_df[~((eval_df < (Q1 - 1.5 * IQR)) | (eval_df > (Q3 + 1.5 * IQR))).any(axis=1)]

    # Ensure target variable encoding
    train_df['target'] = train_df['target'].astype(int)
    eval_df['target'] = eval_df['target'].astype(int)

    return train_df, eval_df

# Feature engineering and selection
def preprocess_data(train_df, eval_df):
    # Automated feature engineering
    poly = PolynomialFeatures(degree=2, interaction_only=True, include_bias=False)
    train_poly = poly.fit_transform(train_df.drop(columns='target'))
    eval_poly = poly.transform(eval_df.drop(columns='target'))

    train_poly_df = pd.DataFrame(train_poly, columns=poly.get_feature_names_out(train_df.drop(columns='target').columns))
    eval_poly_df = pd.DataFrame(eval_poly, columns=poly.get_feature_names_out(eval_df.drop(columns='target').columns))

    # Feature selection
    selector = SelectByTargetMean(threshold=0.05)
    train_selected = selector.fit_transform(train_poly_df, train_df['target'])
    eval_selected = selector.transform(eval_poly_df)

    train_selected_df = pd.DataFrame(train_selected, columns=selector.get_support())
    eval_selected_df = pd.DataFrame(eval_selected, columns=selector.get_support())

    return train_selected_df, eval_selected_df

# Model training
def train_model(train_selected_df, train_df):
    # Define the model
    model = TabNetClassifier(
        n_d=64, n_a=64, n_steps=3,
        gamma=1.5, lambda_sparse=1e-3,
        optimizer_fn=torch.optim.Adam,
        optimizer_params=dict(lr=0.001),
        mask_type="entmax",
        scheduler_params={"step_size":50, "gamma":0.9},
        scheduler_fn=torch.optim.lr_scheduler.StepLR,
        seed=42,
        verbose=10
    )

    # Split the training data into training and validation sets
    X_train, X_val, y_train, y_val = train_test_split(train_selected_df, train_df['target'], test_size=0.2, random_state=42)

    # Train the model
    model.fit(
        X_train=X_train.values, y_train=y_train.values,
        eval_set=[(X_val.values, y_val.values)],
        max_epochs=50,
        patience=20,
        batch_size=1024,
        virtual_batch_size=128,
        num_workers=0,
        drop_last=False
    )

    return model

# Model evaluation
def evaluate_model(model, eval_selected_df, eval_df):
    # Predict probabilities on the evaluation set
    y_pred_proba = model.predict_proba(eval_selected_df.values)[:, 1]

    # Calculate AUC
    auc = roc_auc_score(eval_df['target'], y_pred_proba)
    performance_scores = {
        'AUC': auc
    }

    return performance_scores

# Save the trained model
def save_model(model, model_path):
    model.save_model(model_path)

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_df, eval_df = load_data()

    # Step 2. Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
    # Here, the train_loader contains 70% of the `dataset`, the valid_loader contains 20% of the `dataset`, and the test_loader contains 10% of the `dataset`.
    train_loader, valid_loader, test_loader = (
        None,
        None,
        None,
    )  # corresponding to 70%, 20%, 10% of `dataset`

    # Step 3. With the split dataset, run data preprocessing and feature engineering (if applicable) using the "preprocess_data" function you defined
    processed_train_df, processed_eval_df = preprocess_data(train_df, eval_df)

    # Step 4. Define required model. You may retrieve model from available hub or library along with pretrained weights (if any).
    # If pretrained or predefined model is not available, please create the model according to the given user's requirements below using PyTorch and relevant libraries.
    model = train_model(processed_train_df, train_df)

    # Step 5. train the retrieved/loaded model using the defined "train_model" function
    # Step 6. evaluate the trained model using the defined "evaluate_model" function
    model_performance = evaluate_model(model, processed_eval_df, eval_df)

    # Step 7. Save the trained model
    model_path = './agent_workspace/trained_models/tabnet_model'
    save_model(model, model_path)

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
