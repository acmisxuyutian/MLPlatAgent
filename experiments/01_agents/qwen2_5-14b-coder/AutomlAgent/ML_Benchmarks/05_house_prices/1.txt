import os

import time
import json
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer, KNNImputer
from sklearn.preprocessing import OneHotEncoder, MinMaxScaler
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, TensorDataset
from sklearn.model_selection import KFold
import optuna
import matplotlib.pyplot as plt
import seaborn as sns

# Set random seed for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)

# Data paths
TRAIN_DATA_PATH = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/05_house_prices/split_train.csv'
EVAL_DATA_PATH = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/05_house_prices/split_eval.csv'
MODEL_LOCATION = './agent_workspace/trained_models/house_prices_model.pth'

# Load dataset
def load_dataset(train_path, eval_path):
    train_df = pd.read_csv(train_path)
    eval_df = pd.read_csv(eval_path)
    return train_df, eval_df

# Preprocess data
def preprocess_data(train_df, eval_df):
    # Identify numerical and categorical columns
    numerical_cols = train_df.select_dtypes(include=[np.number]).columns
    categorical_cols = train_df.select_dtypes(include=['object']).columns

    # Handle missing values
    num_imputer = SimpleImputer(strategy='mean')
    cat_imputer = KNNImputer(n_neighbors=5)

    train_df[numerical_cols] = num_imputer.fit_transform(train_df[numerical_cols])
    eval_df[numerical_cols] = num_imputer.transform(eval_df[numerical_cols])

    # Convert categorical columns to numerical using OneHotEncoder
    encoder = OneHotEncoder(handle_unknown='ignore')
    train_encoded = encoder.fit_transform(train_df[categorical_cols])
    eval_encoded = encoder.transform(eval_df[categorical_cols])

    train_encoded_df = pd.DataFrame(train_encoded.toarray(), columns=encoder.get_feature_names_out(categorical_cols))
    eval_encoded_df = pd.DataFrame(eval_encoded.toarray(), columns=encoder.get_feature_names_out(categorical_cols))

    train_df = pd.concat([train_df[numerical_cols], train_encoded_df], axis=1)
    eval_df = pd.concat([eval_df[numerical_cols], eval_encoded_df], axis=1)

    # Normalize numerical features
    scaler = MinMaxScaler()
    train_df[numerical_cols] = scaler.fit_transform(train_df[numerical_cols])
    eval_df[numerical_cols] = scaler.transform(eval_df[numerical_cols])

    # Split features and target
    X_train = train_df.drop(columns=['SalePrice'])
    y_train = train_df['SalePrice']
    X_eval = eval_df.drop(columns=['SalePrice'])
    y_eval = eval_df['SalePrice']

    return X_train, y_train, X_eval, y_eval

# Define the neural network model
class HousePriceModel(nn.Module):
    def __init__(self, input_size, hidden_layers, hidden_units, activation):
        super(HousePriceModel, self).__init__()
        layers = []
        for i in range(hidden_layers):
            if i == 0:
                layers.append(nn.Linear(input_size, hidden_units))
            else:
                layers.append(nn.Linear(hidden_units, hidden_units))
            if activation == 'relu':
                layers.append(nn.ReLU())
            elif activation == 'tanh':
                layers.append(nn.Tanh())
        layers.append(nn.Linear(hidden_units, 1))
        self.model = nn.Sequential(*layers)

    def forward(self, x):
        return self.model(x)

# Train the model
def train_model(model, train_loader, criterion, optimizer, num_epochs):
    model.train()
    for epoch in range(num_epochs):
        for inputs, targets in train_loader:
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, targets)
            loss.backward()
            optimizer.step()
        if (epoch+1) % 10 == 0:
            print(f'Epoch [{epoch+1}/{num_epochs}], Loss: {loss.item():.4f}')
    return model

# Evaluate the model
def evaluate_model(model, eval_loader, criterion):
    model.eval()
    with torch.no_grad():
        total_loss = 0
        for inputs, targets in eval_loader:
            outputs = model(inputs)
            loss = criterion(outputs, targets)
            total_loss += loss.item()
    avg_loss = total_loss / len(eval_loader)
    return avg_loss

# Hyperparameter optimization using Optuna
def objective(trial):
    input_size = X_train.shape[1]
    hidden_layers = trial.suggest_int('hidden_layers', 1, 3)
    hidden_units = trial.suggest_int('hidden_units', 64, 256)
    activation = trial.suggest_categorical('activation', ['relu', 'tanh'])
    learning_rate = trial.suggest_float('learning_rate', 0.001, 0.1, log=True)
    batch_size = trial.suggest_categorical('batch_size', [32, 64, 128])
    l2_reg = trial.suggest_float('l2_reg', 0.001, 0.1, log=True)

    model = HousePriceModel(input_size, hidden_layers, hidden_units, activation)
    criterion = nn.MSELoss()
    optimizer = optim.Adam(model.parameters(), lr=learning_rate, weight_decay=l2_reg)

    train_loader = DataLoader(TensorDataset(torch.tensor(X_train.values, dtype=torch.float32), torch.tensor(y_train.values, dtype=torch.float32).view(-1, 1)), batch_size=batch_size, shuffle=True)
    eval_loader = DataLoader(TensorDataset(torch.tensor(X_eval.values, dtype=torch.float32), torch.tensor(y_eval.values, dtype=torch.float32).view(-1, 1)), batch_size=batch_size, shuffle=False)

    model = train_model(model, train_loader, criterion, optimizer, num_epochs=100)
    eval_loss = evaluate_model(model, eval_loader, criterion)
    return eval_loss

# Main function
def main():
    # Load dataset
    train_df, eval_df = load_dataset(TRAIN_DATA_PATH, EVAL_DATA_PATH)

    # Preprocess data
    X_train, y_train, X_eval, y_eval = preprocess_data(train_df, eval_df)

    # Hyperparameter optimization
    study = optuna.create_study(direction='minimize')
    study.optimize(objective, n_trials=10)

    # Get the best hyperparameters
    best_params = study.best_params
    print("Best hyperparameters:", best_params)

    # Train the final model with the best hyperparameters
    input_size = X_train.shape[1]
    model = HousePriceModel(input_size, best_params['hidden_layers'], best_params['hidden_units'], best_params['activation'])
    criterion = nn.MSELoss()
    optimizer = optim.Adam(model.parameters(), lr=best_params['learning_rate'], weight_decay=best_params['l2_reg'])
    train_loader = DataLoader(TensorDataset(torch.tensor(X_train.values, dtype=torch.float32), torch.tensor(y_train.values, dtype=torch.float32).view(-1, 1)), batch_size=best_params['batch_size'], shuffle=True)
    eval_loader = DataLoader(TensorDataset(torch.tensor(X_eval.values, dtype=torch.float32), torch.tensor(y_eval.values, dtype=torch.float32).view(-1, 1)), batch_size=best_params['batch_size'], shuffle=False)

    model = train_model(model, train_loader, criterion, optimizer, num_epochs=100)

    # Evaluate the model
    eval_loss = evaluate_model(model, eval_loader, criterion)
    print(f'Evaluation Loss: {eval_loss:.4f}')

    # Save the trained model
    torch.save(model.state_dict(), MODEL_LOCATION)

if __name__ == "__main__":
    main()
