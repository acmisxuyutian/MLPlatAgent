import os, random, time, json
import numpy as np
import pandas as pd
import torch
import torch.nn as nn
import torch.optim as optim
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import MinMaxScaler
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import mean_squared_error
import seaborn as sns
import matplotlib.pyplot as plt

# Set random seed for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(train_data, eval_data):
    # Display statistical summaries
    print(train_data.describe())
    
    # Identify categorical variables
    categorical_vars = train_data.select_dtypes(include=['object']).columns
    print("Categorical Variables:", categorical_vars)
    
    # One-hot encode categorical variables
    train_data_encoded = pd.get_dummies(train_data, columns=categorical_vars)
    eval_data_encoded = pd.get_dummies(eval_data, columns=categorical_vars)
    
    # Compute correlation matrix
    corr_matrix = train_data_encoded.corr()
    print("Correlation Matrix:\n", corr_matrix)
    
    # Feature importance using RandomForestRegressor
    rf = RandomForestRegressor()
    rf.fit(train_data_encoded.drop('SalePrice', axis=1), train_data_encoded['SalePrice'])
    feature_importances = pd.Series(rf.feature_importances_, index=train_data_encoded.drop('SalePrice', axis=1).columns)
    print("Feature Importances:\n", feature_importances.sort_values(ascending=False))
    
    # Select the top 10 features
    selected_features = feature_importances.sort_values(ascending=False).index[:10]
    train_data_selected = train_data_encoded[selected_features.tolist() + ['SalePrice']]
    eval_data_selected = eval_data_encoded[selected_features.tolist() + ['SalePrice']]
    
    # Fill missing values with the mean
    train_data_selected = train_data_selected.fillna(train_data_selected.mean())
    eval_data_selected = eval_data_selected.fillna(train_data_selected.mean())
    
    # Normalize features using Min-Max Scaling
    scaler = MinMaxScaler()
    train_data_scaled = scaler.fit_transform(train_data_selected.drop('SalePrice', axis=1))
    eval_data_scaled = scaler.transform(eval_data_selected.drop('SalePrice', axis=1))
    
    train_data_scaled_df = pd.DataFrame(train_data_scaled, columns=train_data_selected.drop('SalePrice', axis=1).columns)
    train_data_scaled_df['SalePrice'] = train_data_selected['SalePrice']
    
    eval_data_scaled_df = pd.DataFrame(eval_data_scaled, columns=eval_data_selected.drop('SalePrice', axis=1).columns)
    eval_data_scaled_df['SalePrice'] = eval_data_selected['SalePrice']
    
    return train_data_scaled_df, eval_data_scaled_df

# Define the neural network architecture
class HousePricePredictor(nn.Module):
    def __init__(self, input_size):
        super(HousePricePredictor, self).__init__()
        self.fc1 = nn.Linear(input_size, 64)
        self.fc2 = nn.Linear(64, 32)
        self.fc3 = nn.Linear(32, 16)
        self.fc4 = nn.Linear(16, 1)

    def forward(self, x):
        x = torch.relu(self.fc1(x))
        x = torch.relu(self.fc2(x))
        x = torch.relu(self.fc3(x))
        x = self.fc4(x)
        return x

# Train the model
def train_model(model, train_loader, val_loader, num_epochs=100, learning_rate=0.001):
    criterion = nn.MSELoss()
    optimizer = optim.Adam(model.parameters(), lr=learning_rate)
    
    for epoch in range(num_epochs):
        model.train()
        optimizer.zero_grad()
        outputs = model(train_loader)
        loss = criterion(outputs, val_loader)
        loss.backward()
        optimizer.step()
        
        if (epoch+1) % 10 == 0:
            print(f'Epoch [{epoch+1}/{num_epochs}], Loss: {loss.item():.4f}')
    
    return model

# Evaluate the model
def evaluate_model(model, test_loader):
    model.eval()
    with torch.no_grad():
        outputs = model(torch.tensor(test_loader.drop('SalePrice', axis=1).values, dtype=torch.float32))
        predicted = outputs.numpy()
        actual = test_loader['SalePrice'].values
    
    rmse = np.sqrt(np.mean((predicted - actual) ** 2))
    performance_scores = {
        'RMSE': rmse
    }
    
    return performance_scores

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/05_house_prices/split_train.csv'
    eval_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/05_house_prices/split_eval.csv'
    
    train_data = pd.read_csv(train_data_path)
    eval_data = pd.read_csv(eval_data_path)
    
    # Step 2. Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
    train_data, val_data = train_test_split(train_data, test_size=0.2, random_state=SEED)
    
    # Step 3. With the split dataset, run data preprocessing and feature engineering (if applicable) using the "preprocess_data" function you defined
    train_data_scaled, eval_data_scaled = preprocess_data(train_data, eval_data)
    
    # Step 4. Define required model. You may retrieve model from available hub or library along with pretrained weights (if any).
    # If pretrained or predefined model is not available, please create the model according to the given user's requirements below using PyTorch and relevant libraries.
    input_size = train_data_scaled.shape[1] - 1  # Exclude 'SalePrice'
    model = HousePricePredictor(input_size)
    
    # Step 5. train the retrieved/loaded model using the defined "train_model" function
    # on top of the model training, please run hyperparamter optimization based on the suggested hyperparamters and their values before proceeding to the evaluation step to ensure model's optimality
    model = train_model(model, torch.tensor(train_data_scaled.drop('SalePrice', axis=1).values, dtype=torch.float32), torch.tensor(train_data_scaled['SalePrice'].values, dtype=torch.float32).view(-1, 1))
    
    # Step 6. evaluate the trained model using the defined "evaluate_model" function
    model_performance = evaluate_model(model, eval_data_scaled)
    
    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
