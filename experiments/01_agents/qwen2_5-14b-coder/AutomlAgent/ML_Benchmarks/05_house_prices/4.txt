import os, random, time, json
import numpy as np
import pandas as pd
from sklearn.metrics import mean_squared_error
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.ensemble import RandomForestRegressor
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, TensorDataset

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(train_data, eval_data):
    # Identify missing values
    missing_values_train = train_data.isnull().sum()
    missing_values_eval = eval_data.isnull().sum()

    # Handle missing values for numerical features
    numerical_features = train_data.select_dtypes(include=[int, float]).columns
    train_data[numerical_features] = train_data[numerical_features].fillna(train_data[numerical_features].mean())
    eval_data[numerical_features] = eval_data[numerical_features].fillna(train_data[numerical_features].mean())

    # Handle missing values for categorical features
    categorical_features = train_data.select_dtypes(include=['object']).columns
    train_data = pd.get_dummies(train_data, columns=categorical_features)
    eval_data = pd.get_dummies(eval_data, columns=categorical_features)

    # Ensure both datasets have the same columns
    eval_data = eval_data.reindex(columns=train_data.columns, fill_value=0)

    # Standardize features
    scaler = StandardScaler()
    train_data[numerical_features] = scaler.fit_transform(train_data[numerical_features])
    eval_data[numerical_features] = scaler.transform(eval_data[numerical_features])

    # Feature selection using Random Forest
    X_train = train_data.drop('SalePrice', axis=1)
    y_train = train_data['SalePrice']

    model = RandomForestRegressor()
    model.fit(X_train, y_train)
    feature_importances = model.feature_importances_

    # Select top features (e.g., top 20 features)
    top_features = X_train.columns[feature_importances.argsort()[-20:]]
    X_train = X_train[top_features]
    eval_data = eval_data[top_features]

    # Create new features
    train_data['total_square_footage'] = train_data['GrLivArea'] + train_data['TotalBsmtSF']
    eval_data['total_square_footage'] = eval_data['GrLivArea'] + eval_data['TotalBsmtSF']
    train_data['age'] = pd.Timestamp.now().year - train_data['YearBuilt']
    eval_data['age'] = pd.Timestamp.now().year - eval_data['YearBuilt']

    return X_train, y_train, eval_data

# Define the Neural Network Architecture
class HousePricePredictor(nn.Module):
    def __init__(self, input_size):
        super(HousePricePredictor, self).__init__()
        self.fc1 = nn.Linear(input_size, 128)
        self.dropout1 = nn.Dropout(0.2)
        self.fc2 = nn.Linear(128, 64)
        self.dropout2 = nn.Dropout(0.2)
        self.fc3 = nn.Linear(64, 1)
    
    def forward(self, x):
        x = torch.relu(self.fc1(x))
        x = self.dropout1(x)
        x = torch.relu(self.fc2(x))
        x = self.dropout2(x)
        x = self.fc3(x)
        return x

def evaluate_model(model, eval_data):
    model.load_state_dict(torch.load('./agent_workspace/trained_models/best_model.pth'))

    # Evaluate on eval data
    model.eval()
    with torch.no_grad():
        eval_inputs = torch.tensor(eval_data.values, dtype=torch.float32)
        eval_outputs = model(eval_inputs)
        eval_preds = eval_outputs.numpy().flatten()

    # Calculate RMSE
    rmse = mean_squared_error(eval_data['SalePrice'], eval_preds, squared=False)
    performance_scores = {
        'RMSE': rmse
    }

    return performance_scores

def main():

    Main function to execute the tabular regression pipeline.

    train_data = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/05_house_prices/split_train.csv')

    X_train, y_train, eval_data = preprocess_data(train_data, eval_data)
    X_train, X_val, y_train, y_val = train_test_split(X_train, y_train, test_size=0.2, random_state=SEED)

    # Convert data to PyTorch tensors
    X_train_tensor = torch.tensor(X_train.values, dtype=torch.float32)
    y_train_tensor = torch.tensor(y_train.values, dtype=torch.float32).view(-1, 1)
    X_val_tensor = torch.tensor(X_val.values, dtype=torch.float32)
    y_val_tensor = torch.tensor(y_val.values, dtype=torch.float32).view(-1, 1)

    # Create DataLoader
    train_dataset = TensorDataset(X_train_tensor, y_train_tensor)
    train_loader = DataLoader(train_dataset, batch_size=32, shuffle=True)

    model = HousePricePredictor(input_size)

    num_epochs = 100
    early_stopping_patience = 10

    # Step 5. evaluate the trained model using the defined "evaluate_model" function
    model_performance = evaluate_model(model, eval_data)

    return model_performance
