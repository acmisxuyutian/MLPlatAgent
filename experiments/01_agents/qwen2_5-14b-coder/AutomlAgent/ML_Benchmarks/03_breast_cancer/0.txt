import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.datasets import load_breast_cancer
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
import torch
import torch.nn as nn
import torch.optim as optim
from sklearn.metrics import accuracy_score, f1_score

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(X, y):
    # Convert to DataFrame
    df = pd.DataFrame(X, columns=["feature_{}".format(i) for i in range(X.shape[1])])
    df['target'] = y

    # Standardize features
    scaler = StandardScaler()
    X_scaled = scaler.fit_transform(X)

    # Split the dataset
    X_train, X_val, y_train, y_val = train_test_split(X_scaled, y, test_size=0.2, random_state=SEED)

    # Convert to PyTorch tensors
    X_train_tensor = torch.tensor(X_train, dtype=torch.float32)
    y_train_tensor = torch.tensor(y_train, dtype=torch.float32).view(-1, 1)
    X_val_tensor = torch.tensor(X_val, dtype=torch.float32)
    y_val_tensor = torch.tensor(y_val, dtype=torch.float32).view(-1, 1)

    return X_train_tensor, y_train_tensor, X_val_tensor, y_val_tensor

# Define the model
class FFNN(nn.Module):
    def __init__(self, input_size):
        super(FFNN, self).__init__()
        self.fc1 = nn.Linear(input_size, 64)
        self.fc2 = nn.Linear(64, 64)
        self.fc3 = nn.Linear(64, 1)

    def forward(self, x):
        x = torch.relu(self.fc1(x))
        x = torch.relu(self.fc2(x))
        x = torch.sigmoid(self.fc3(x))
        return x

def train_model(model, X_train_tensor, y_train_tensor, num_epochs=100, learning_rate=0.001):
    criterion = nn.BCELoss()
    optimizer = optim.Adam(model.parameters(), lr=learning_rate)

    for epoch in range(num_epochs):
        model.train()
        optimizer.zero_grad()
        outputs = model(X_train_tensor)
        loss = criterion(outputs, y_train_tensor)
        loss.backward()
        optimizer.step()

        if (epoch+1) % 10 == 0:
            print(f'Epoch [{epoch+1}/{num_epochs}], Loss: {loss.item():.4f}')

    return model

def evaluate_model(model, X_val_tensor, y_val_tensor):
    model.eval()
    with torch.no_grad():
        outputs = model(X_val_tensor)
        predicted = (outputs > 0.5).float()
        accuracy = accuracy_score(y_val_tensor.numpy(), predicted.numpy())
        f1 = f1_score(y_val_tensor.numpy(), predicted.numpy())

    performance_scores = {
        'ACC': accuracy,
        'F1': f1
    }

    return performance_scores

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    """
    Main function to execute the tabular classification pipeline.
    """

    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    dataset = load_breast_cancer()
    X, y = dataset.data, dataset.target

    # Step 2. Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
    # Here, the train_loader contains 80% of the `dataset`, the valid_loader contains 20% of the `dataset`.
    X_train_tensor, y_train_tensor, X_val_tensor, y_val_tensor = preprocess_data(X, y)

    # Step 3. With the split dataset, run data preprocessing and feature engineering (if applicable) using the "preprocess_data" function you defined
    # processed_data = preprocess_data()

    # Step 4. Define required model. You may retrieve model from available hub or library along with pretrained weights (if any).
    # If pretrained or predefined model is not available, please create the model according to the given user's requirements below using PyTorch and relevant libraries.
    model = FFNN(input_size=X_train_tensor.shape[1])

    # Step 5. Train the retrieved/loaded model using the defined "train_model" function
    model = train_model(model, X_train_tensor, y_train_tensor)

    # Step 6. Evaluate the trained model using the defined "evaluate_model" function
    model_performance = evaluate_model(model, X_val_tensor, y_val_tensor)

    # Save the trained model
    os.makedirs("./agent_workspace/trained_models", exist_ok=True)
    torch.save(model.state_dict(), "./agent_workspace/trained_models/breast_cancer_model.pth")

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Validation Set:", model_performance)
