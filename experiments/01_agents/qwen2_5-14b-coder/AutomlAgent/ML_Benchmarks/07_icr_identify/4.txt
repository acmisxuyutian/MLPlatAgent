import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.metrics import accuracy_score, f1_score
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, PolynomialFeatures
from imblearn.over_sampling import SMOTE
import torch
import torch.nn as nn
import torch.optim as optim

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data preprocessing and feature engineering
def preprocess_data(train_data, eval_data):
    # Handle non-numeric columns
    for col in train_data.columns:
        if train_data[col].dtype == 'object':
            try:
                train_data[col] = pd.to_numeric(train_data[col], errors='coerce')
                eval_data[col] = pd.to_numeric(eval_data[col], errors='coerce')
            except ValueError:
                print(f"Column {col} cannot be converted to numeric and will be dropped.")
                train_data.drop(columns=[col], inplace=True)
                eval_data.drop(columns=[col], inplace=True)

    # Handle missing values
    imputer = SimpleImputer(strategy='mean')
    train_data_imputed = pd.DataFrame(imputer.fit_transform(train_data), columns=train_data.columns)
    eval_data_imputed = pd.DataFrame(imputer.transform(eval_data), columns=eval_data.columns)

    # Detect and handle outliers
    Q1 = train_data_imputed.quantile(0.25)
    Q3 = train_data_imputed.quantile(0.75)
    IQR = Q3 - Q1

    train_data_cleaned = train_data_imputed[~((train_data_imputed < (Q1 - 1.5 * IQR)) | (train_data_imputed > (Q3 + 1.5 * IQR))).any(axis=1)]
    eval_data_cleaned = eval_data_imputed[~((eval_data_imputed < (Q1 - 1.5 * IQR)) | (eval_data_imputed > (Q3 + 1.5 * IQR))).any(axis=1)]

    # Normalize and scale features
    scaler = StandardScaler()
    train_features = scaler.fit_transform(train_data_cleaned.drop('Class', axis=1))
    eval_features = scaler.transform(eval_data_cleaned.drop('Class', axis=1))

    # Check class distribution
    if len(train_data_cleaned['Class'].unique()) <= 1:
        raise ValueError("The target variable 'Class' needs to have more than 1 class.")

    # Apply SMOTE
    smote = SMOTE(random_state=42)
    train_features, train_labels = smote.fit_resample(train_features, train_data_cleaned['Class'])

    # Create interaction terms
    poly = PolynomialFeatures(degree=2, interaction_only=True, include_bias=False)
    train_features = poly.fit_transform(train_features)
    eval_features = poly.transform(eval_features)

    return train_features, train_labels, eval_features, eval_data_cleaned['Class']

# Define the FNN model
class FNN(nn.Module):
    def __init__(self, input_size, hidden_sizes, output_size):
        super(FNN, self).__init__()
        layers = []
        layers.append(nn.Linear(input_size, hidden_sizes[0]))
        layers.append(nn.ReLU())
        for i in range(1, len(hidden_sizes)):
            layers.append(nn.Linear(hidden_sizes[i-1], hidden_sizes[i]))
            layers.append(nn.ReLU())
        layers.append(nn.Linear(hidden_sizes[-1], output_size))
        self.model = nn.Sequential(*layers)

    def forward(self, x):
        return self.model(x)

# Train the model
def train_model(model, train_features, train_labels):
    criterion = nn.BCEWithLogitsLoss()
    optimizer = optim.Adam(model.parameters(), lr=0.001)

    train_features = torch.tensor(train_features, dtype=torch.float32)
    train_labels = torch.tensor(train_labels.values, dtype=torch.float32).unsqueeze(1)

    for epoch in range(100):
        model.train()
        optimizer.zero_grad()
        outputs = model(train_features)
        loss = criterion(outputs, train_labels)
        loss.backward()
        optimizer.step()
        if (epoch+1) % 10 == 0:
            print(f'Epoch [{epoch+1}/100], Loss: {loss.item():.4f}')
    return model

# Evaluate the model
def evaluate_model(model, eval_features, eval_labels):
    eval_features = torch.tensor(eval_features, dtype=torch.float32)
    eval_labels = torch.tensor(eval_labels.values, dtype=torch.float32).unsqueeze(1)

    model.eval()
    with torch.no_grad():
        outputs = model(eval_features)
        predicted = torch.round(torch.sigmoid(outputs))
        f1_score_val = f1_score(eval_labels, predicted, average='binary')
        print(f'F1 Score: {f1_score_val:.4f}')
    return {'F1': f1_score_val}

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_data = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/07_icr_identify/split_train.csv')
    eval_data = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/07_icr_identify/split_eval.csv')

    # Step 2. Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
    train_features, train_labels, eval_features, eval_labels = preprocess_data(train_data, eval_data)

    # Step 3. With the split dataset, run data preprocessing and feature engineering (if applicable) using the "preprocess_data" function you defined
    # Step 4. Define required model. You may retrieve model from available hub or library along with pretrained weights (if any).
    # If pretrained or predefined model is not available, please create the model according to the given user's requirements below using PyTorch and relevant libraries.
    input_size = train_features.shape[1]
    hidden_sizes = [64, 32, 16]
    output_size = 1
    model = FNN(input_size, hidden_sizes, output_size)

    # Step 5. train the retrieved/loaded model using the defined "train_model" function
    model = train_model(model, train_features, train_labels)

    # Step 6. evaluate the trained model using the defined "evaluate_model" function
    model_performance = evaluate_model(model, eval_features, eval_labels)

    # Save the trained model
    os.makedirs('./agent_workspace/trained_models', exist_ok=True)
    torch.save(model.state_dict(), './agent_workspace/trained_models/fnn_model.pth')

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
