import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.datasets import load_iris

iris = load_iris()
X = iris.data
y = iris.target
feature_names = iris.feature_names
target_names = iris.target_names

iris_df = pd.DataFrame(data=X, columns=feature_names)
iris_df['species'] = y
iris_df['species'] = iris_df['species'].map({0: target_names[0], 1: target_names[1], 2: target_names[2]})

print(iris_df.describe())

print(iris_df.head())

plt.figure(figsize=(10, 6))
for species in target_names:
    subset = iris_df[iris_df['species'] == species]
    plt.scatter(subset['sepal length (cm)'], subset['sepal width (cm)'], label=species)

plt.xlabel('Sepal Length (cm)')
plt.ylabel('Sepal Width (cm)')
plt.title('Iris Dataset: Sepal Length vs Sepal Width')
plt.legend()
plt.grid(True)
plt.show()

plt.savefig('iris_sepal_plot.png')
