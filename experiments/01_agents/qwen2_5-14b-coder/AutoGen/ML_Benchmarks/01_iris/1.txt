import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.datasets import load_iris

# Load the Iris dataset
iris = load_iris()
X = iris.data
y = iris.target
feature_names = iris.feature_names
target_names = iris.target_names

# Create a DataFrame for easier analysis
iris_df = pd.DataFrame(data=X, columns=feature_names)
iris_df['species'] = y
iris_df['species'] = iris_df['species'].map({0: target_names[0], 1: target_names[1], 2: target_names[2]})

# Display basic statistics
print(iris_df.describe())

# Display the first few rows of the DataFrame
print(iris_df.head())

# Create a pairplot to visualize the relationships between features
sns.pairplot(iris_df, hue='species', markers=["o", "s", "D"])
plt.suptitle('Pairplot of Iris Dataset', y=1.02)
plt.show()

# Create a boxplot to visualize the distribution of each feature
plt.figure(figsize=(12, 8))
sns.boxplot(x='species', y='sepal length (cm)', data=iris_df)
plt.title('Boxplot of Sepal Length by Species')
plt.show()

# Create a violin plot to visualize the distribution of each feature
plt.figure(figsize=(12, 8))
sns.violinplot(x='species', y='sepal length (cm)', data=iris_df)
plt.title('Violin Plot of Sepal Length by Species')
plt.show()

# Create a heatmap to visualize the correlation between features
plt.figure(figsize=(10, 8))
sns.heatmap(iris_df.drop('species', axis=1).corr(), annot=True, cmap='coolwarm')
plt.title('Correlation Heatmap of Iris Features')
plt.show()
