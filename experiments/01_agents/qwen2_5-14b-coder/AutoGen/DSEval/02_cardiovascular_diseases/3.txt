import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import roc_auc_score
from sklearn.preprocessing import OneHotEncoder

# Load the datasets
train_path = 'D:\\LLM-Driven_AI-Studio\\MLAgent\\data\\benchmark/DSEval/datasets/02_cardiovascular_diseases/train.csv'
test_path = 'D:\\LLM-Driven_AI-Studio\\MLAgent\\data\\benchmark/DSEval/datasets/02_cardiovascular_diseases/test.csv'

train_data = pd.read_csv(train_path)
test_data = pd.read_csv(test_path)

# Drop rows with missing values and duplicates
train_data = train_data.dropna().drop_duplicates()
test_data = test_data.dropna().drop_duplicates()

# Define a function to categorize BMI
def categorize_bmi(bmi):
    if bmi < 18.5:
        return 'Underweight'
    elif 18.5 <= bmi < 25:
        return 'Normal weight'
    elif 25 <= bmi < 30:
        return 'Overweight'
    else:
        return 'Obesity'

# Apply the function to create the BMI_Category feature
train_data['BMI_Category'] = train_data['BMI'].apply(categorize_bmi).astype('category')
test_data['BMI_Category'] = test_data['BMI'].apply(categorize_bmi).astype('category')

# One-hot encode categorical variables
categorical_cols = ['General_Health', 'Checkup', 'Exercise', 'Heart_Disease', 'Skin_Cancer', 'Other_Cancer', 'Depression', 'Diabetes', 'Arthritis', 'Sex', 'Age_Category', 'Smoking_History']
encoder = OneHotEncoder(sparse=False, drop='first')

# Fit and transform the encoder on the training data
X_train_encoded = encoder.fit_transform(train_data[categorical_cols])
X_test_encoded = encoder.transform(test_data[categorical_cols])

# Create DataFrames from the encoded data
X_train_encoded_df = pd.DataFrame(X_train_encoded, columns=encoder.get_feature_names_out(categorical_cols))
X_test_encoded_df = pd.DataFrame(X_test_encoded, columns=encoder.get_feature_names_out(categorical_cols))

# Concatenate the encoded data with the numeric data
X_train = pd.concat([train_data.drop(columns=['Heart_Disease'] + categorical_cols), X_train_encoded_df], axis=1)
X_test = pd.concat([test_data.drop(columns=['Heart_Disease'] + categorical_cols), X_test_encoded_df], axis=1)

# Define target variable
y_train = train_data['Heart_Disease']
y_test = test_data['Heart_Disease']

# Initialize and fit the logistic regression model
model = LogisticRegression(max_iter=1000)
model.fit(X_train, y_train)

# Make predictions
y_pred_proba = model.predict_proba(X_test)[:, 1]

# Compute AUC
auc = roc_auc_score(y_test, y_pred_proba)
print(f'Area Under ROC Curve: {auc}')
