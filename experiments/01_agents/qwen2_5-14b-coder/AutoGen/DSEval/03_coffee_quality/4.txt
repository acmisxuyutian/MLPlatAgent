# Load the training and test data
train_data_path = 'D:\\LLM-Driven_AI-Studio\\MLAgent\\data\\benchmark/DSEval/datasets/03_coffee_quality/train.csv'
test_data_path = 'D:\\LLM-Driven_AI-Studio\\MLAgent\\data\\benchmark/DSEval/datasets/03_coffee_quality/test.csv'

train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# Drop duplicated rows from the training and test datasets
train_df = train_df.drop_duplicates()
test_df = test_df.drop_duplicates()

# List of columns to drop
columns_to_drop = [
    "ID", "ICO Number", "Owner", "Region", "Certification Contact", "Certification Address", 
    "Farm Name", "Lot Number", "Mill", "ICO Number", "Producer", "Company", "Expiration", 
    "Harvest Year", "Unnamed: 0", "Number of Bags", "Bag Weight", "In-Country Partner", 
    "Country of Origin", "Processing Method", "Grading Date", "Variety", "Status", "Defects", 
    "Uniformity", "Clean Cup", "Sweetness", "Certification Body"
]

# Drop the specified columns from the training and test datasets
train_df = train_df.drop(columns=columns_to_drop)
test_df = test_df.drop(columns=columns_to_drop)

# One-hot encode the "Color" column for both training and test datasets
train_df = pd.get_dummies(train_df, columns=['Color'], drop_first=True)
test_df = pd.get_dummies(test_df, columns=['Color'], drop_first=True)

# Separate features and target variable for the training set
X_train = train_df.drop(columns=['Total Cup Points'])
y_train = train_df['Total Cup Points']

# Separate features for the test set
X_test = test_df.drop(columns=['Total Cup Points'])

from sklearn.ensemble import RandomForestRegressor

# Initialize the RandomForestRegressor
regressor = RandomForestRegressor(n_estimators=100, random_state=42)

# Train the model
regressor.fit(X_train, y_train)

# Make predictions on the test set
y_pred = regressor.predict(X_test)

from sklearn.metrics import mean_squared_error

# Calculate the mean-squared error
mse = mean_squared_error(test_df['Total Cup Points'], y_pred)
print(f"Mean Squared Error: {mse}")
