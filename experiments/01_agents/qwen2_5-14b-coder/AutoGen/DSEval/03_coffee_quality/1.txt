# 加载训练数据
train_data = pd.read_csv('D:\\LLM-Driven_AI-Studio\\MLAgent\\data\\benchmark/DSEval/datasets/03_coffee_quality/train.csv')

# 加载测试数据
test_data = pd.read_csv('D:\\LLM-Driven_AI-Studio\\MLAgent\\data\\benchmark/DSEval/datasets/03_coffee_quality/test.csv')

# 删除训练数据中的重复行

columns_to_drop = [

    "Farm Name", "Lot Number", "Mill", "ICO Number", "Producer", "Company", "Expiration", 
    "Harvest Year", "Unnamed: 0", "Number of Bags", "Bag Weight", "In-Country Partner",

]

train_data.drop(columns=columns_to_drop, inplace=True)

test_data.drop(columns=columns_to_drop, inplace=True)

# 对训练数据中的 'Color' 列进行 one-hot 编码
train_data = pd.get_dummies(train_data, columns=['Color'], drop_first=True)

from sklearn.ensemble import RandomForestRegressor
from sklearn.model_selection import train_test_split

# 分离特征和目标变量
X_train = train_data.drop('Total Cup Points', axis=1)
y_train = train_data['Total Cup Points']

# 初始化随机森林回归器
regressor = RandomForestRegressor(n_estimators=100, random_state=42)

# 训练模型
regressor.fit(X_train, y_train)

# 分离测试集的特征
X_test = test_data.drop('Total Cup Points', axis=1)

# 进行预测
predictions = regressor.predict(X_test)

from sklearn.metrics import mean_squared_error

# 计算均方误差
mse = mean_squared_error(test_data['Total Cup Points'], predictions)

print(f"Mean Squared Error: {mse}")
