import pandas as pd

# 加载训练数据
train_data = pd.read_csv('D:\\LLM-Driven_AI-Studio\\MLAgent\\data\\benchmark/DSEval/datasets/10_life_expectancy/train.csv')

# 加载测试数据
test_data = pd.read_csv('D:\\LLM-Driven_AI-Studio\\MLAgent\\data\\benchmark/DSEval/datasets/10_life_expectancy/test.csv')

# 定义 GDP 分类函数
def categorize_gdp(gdp):
    if gdp <= 1000:
        return 'Low GDP'
    elif 1000 < gdp <= 10000:
        return 'Medium GDP'
    else:
        return 'High GDP'

# 在训练数据和测试数据上应用函数
train_data['GDP_bins'] = train_data['GDP'].apply(categorize_gdp)
test_data['GDP_bins'] = test_data['GDP'].apply(categorize_gdp)

from sklearn.linear_model import LinearRegression
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import OneHotEncoder
import numpy as np

# 选择特征和目标变量
X_train = train_data[['GDP', 'GDP_bins']]
y_train = train_data['Life_Expectancy']

# 对分类特征进行 one-hot 编码
encoder = OneHotEncoder(sparse=False)
X_train_encoded = encoder.fit_transform(X_train[['GDP_bins']])

# 合并数值特征和编码后的分类特征
X_train_final = np.hstack((X_train[['GDP']].values, X_train_encoded))

# 创建并训练模型
model = LinearRegression()
model.fit(X_train_final, y_train)

# 选择测试数据的特征
X_test = test_data[['GDP', 'GDP_bins']]

# 对测试数据的分类特征进行 one-hot 编码
X_test_encoded = encoder.transform(X_test[['GDP_bins']])

# 合并数值特征和编码后的分类特征
X_test_final = np.hstack((X_test[['GDP']].values, X_test_encoded))

# 进行预测
y_pred = model.predict(X_test_final)

# 计算 R-squared score
from sklearn.metrics import r2_score
r2 = r2_score(test_data['Life_Expectancy'], y_pred)
print(f'R-squared score: {r2}')
