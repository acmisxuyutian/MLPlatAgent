import pandas as pd

# 读取训练数据
train_data = pd.read_csv('D:\\LLM-Driven_AI-Studio\\MLAgent\\data\\benchmark/DSEval/datasets/05_patient_profile/train.csv')

# 读取测试数据
test_data = pd.read_csv('D:\\LLM-Driven_AI-Studio\\MLAgent\\data\\benchmark/DSEval/datasets/05_patient_profile/test.csv')

# 选择需要 one-hot 编码的列
categorical_features = train_data.columns.difference(['Disease', 'outcome variable'])

# 对训练数据进行 one-hot 编码
train_encoded = pd.get_dummies(train_data, columns=categorical_features, drop_first=True)

# 对测试数据进行 one-hot 编码
test_encoded = pd.get_dummies(test_data, columns=categorical_features, drop_first=True)

from sklearn.linear_model import LogisticRegression
from sklearn.feature_selection import RFE

# 定义目标变量和特征
X_train = train_encoded.drop(columns=['outcome variable', 'Disease'])
y_train = train_encoded['outcome variable']

# 初始化逻辑回归模型
log_reg = LogisticRegression(max_iter=10000)

# 使用 RFE 选择最重要的 5 个特征
rfe = RFE(log_reg, n_features_to_select=5)
rfe.fit(X_train, y_train)

# 获取选中的特征
selected_features = X_train.columns[rfe.support_]
print("Selected features:", selected_features)

# 模型1：使用所有特征
X_train_all = train_encoded.drop(columns=['outcome variable', 'Disease'])
X_test_all = test_encoded.drop(columns=['outcome variable', 'Disease'])

log_reg_all = LogisticRegression(max_iter=10000)
log_reg_all.fit(X_train_all, y_train)

# 模型2：使用选中的 5 个特征
X_train_selected = train_encoded[selected_features]
X_test_selected = test_encoded[selected_features]

log_reg_selected = LogisticRegression(max_iter=10000)
log_reg_selected.fit(X_train_selected, y_train)

from sklearn.metrics import accuracy_score, classification_report

# 评估模型1
y_pred_all = log_reg_all.predict(X_test_all)
accuracy_all = accuracy_score(test_encoded['outcome variable'], y_pred_all)
print("Model1 Accuracy:", accuracy_all)
print("Model1 Classification Report:\n", classification_report(test_encoded['outcome variable'], y_pred_all))

# 评估模型2
y_pred_selected = log_reg_selected.predict(X_test_selected)
accuracy_selected = accuracy_score(test_encoded['outcome variable'], y_pred_selected)
print("Model2 Accuracy:", accuracy_selected)
print("Model2 Classification Report:\n", classification_report(test_encoded['outcome variable'], y_pred_selected))
