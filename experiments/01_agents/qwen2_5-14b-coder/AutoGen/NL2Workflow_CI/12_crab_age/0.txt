import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import mean_absolute_error
from sklearn.preprocessing import OneHotEncoder
import numpy as np

# 1. 加载数据集
train_path = r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/crab_age/train.csv'
test_path = r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/crab_age/test.csv'

train_data = pd.read_csv(train_path)
test_data = pd.read_csv(test_path)

# 2. 数据预处理
# 假设训练集和测试集的特征列相同
features = train_data.columns[:-1]  # 假设最后一列是目标变量
target = train_data.columns[-1]

# 进行独热编码
encoder = OneHotEncoder(sparse=False, drop='first')
train_encoded = encoder.fit_transform(train_data[features])
test_encoded = encoder.transform(test_data[features])

# 将编码后的特征重新组合成DataFrame
train_encoded_df = pd.DataFrame(train_encoded, columns=encoder.get_feature_names_out(features))
test_encoded_df = pd.DataFrame(test_encoded, columns=encoder.get_feature_names_out(features))

# 3. 建立模型
X_train = train_encoded_df
y_train = train_data[target]

model = RandomForestRegressor(n_estimators=100, random_state=42)
model.fit(X_train, y_train)

# 4. 模型预测
X_test = test_encoded_df
y_pred = model.predict(X_test)

# 5. 保存预测结果
output_path = r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/crab_age/predictions.csv'
predictions_df = pd.DataFrame(y_pred, columns=[target])
predictions_df.to_csv(output_path, index=False)

# 6. 评估模型
y_test = test_data[target]
mae = mean_absolute_error(y_test, y_pred)
print(f'MAE on test set: {mae}')
