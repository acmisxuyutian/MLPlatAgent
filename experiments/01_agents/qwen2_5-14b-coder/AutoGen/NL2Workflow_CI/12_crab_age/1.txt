import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import mean_absolute_error
from sklearn.preprocessing import OneHotEncoder
import numpy as np

# 1. 加载数据集
train_path = r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/crab_age/train.csv'
test_path = r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/crab_age/test.csv'

train_df = pd.read_csv(train_path)
test_df = pd.read_csv(test_path)

# 2. 数据预处理
# 假设训练集和测试集的特征列相同
features = train_df.columns[:-1]  # 假设最后一列是目标变量

encoder = OneHotEncoder(sparse=False, drop='first')
train_encoded = encoder.fit_transform(train_df[features])
test_encoded = encoder.transform(test_df[features])

# 3. 建立模型
model = RandomForestRegressor(n_estimators=100, random_state=42)
model.fit(train_encoded, train_df[target])

# 4. 模型预测
predictions = model.predict(test_encoded)

# 5. 保存预测结果
predictions_df = pd.DataFrame(predictions, columns=[target])
predictions_df.to_csv('crab_age_predictions.csv', index=False)

# 6. 评估模型
# 假设测试集有目标变量
test_df[target] = predictions
mae = mean_absolute_error(test_df[target], predictions)

print(f'MAE on test set: {mae}')
