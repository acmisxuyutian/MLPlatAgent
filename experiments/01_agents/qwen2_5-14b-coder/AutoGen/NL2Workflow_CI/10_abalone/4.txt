import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import OneHotEncoder
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import mean_squared_log_error

# 1. 加载数据
train_path = r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/abalone/train.csv'
test_path = r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/abalone/test.csv'

train_df = pd.read_csv(train_path)
test_df = pd.read_csv(test_path)

# 2. 数据预处理
# 独热编码 'Sex' 列
encoder = OneHotEncoder(sparse=False)
sex_encoded = encoder.fit_transform(train_df[['Sex']])
sex_encoded_test = encoder.transform(test_df[['Sex']])

# 将编码后的数据添加回 DataFrame
train_df_encoded = pd.concat([train_df.drop('Sex', axis=1), pd.DataFrame(sex_encoded, columns=encoder.get_feature_names_out(['Sex']))], axis=1)
test_df_encoded = pd.concat([test_df.drop('Sex', axis=1), pd.DataFrame(sex_encoded_test, columns=encoder.get_feature_names_out(['Sex']))], axis=1)

# 3. 数据划分
X_train = train_df_encoded.drop('Rings', axis=1)
y_train = train_df_encoded['Rings']

# 4. 模型构建
model = RandomForestRegressor(n_estimators=100, random_state=42)

# 5. 模型训练
model.fit(X_train, y_train)

# 6. 模型评估
X_test = test_df_encoded.drop('Rings', axis=1)
y_test = test_df_encoded['Rings']

# 预测
y_pred = model.predict(X_test)

# 计算 RMLSE
rmlse = np.sqrt(mean_squared_log_error(y_test, y_pred))

print(f"RMLSE: {rmlse}")
