import pandas as pd

# 读取训练集和测试集
train_data = pd.read_csv(r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/monster/train.csv')
test_data = pd.read_csv(r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/monster/test.csv')

# 查看数据的前几行
print(train_data.head())
print(test_data.head())

# 检查数据的基本信息
print(train_data.info())
print(test_data.info())

# 检查缺失值
print(train_data.isnull().sum())
print(test_data.isnull().sum())

# 处理缺失值（假设用均值填充）
train_data.fillna(train_data.mean(), inplace=True)
test_data.fillna(test_data.mean(), inplace=True)

# 检查异常值（假设用四分位数范围进行处理）
Q1 = train_data.quantile(0.25)
Q3 = train_data.quantile(0.75)
IQR = Q3 - Q1

# 移除异常值
train_data = train_data[~((train_data < (Q1 - 1.5 * IQR)) | (train_data > (Q3 + 1.5 * IQR))).any(axis=1)]

# 特征选择和编码
# 假设最后一列是目标变量，其余是特征
X_train = train_data.iloc[:, :-1]
y_train = train_data.iloc[:, -1]
X_test = test_data.iloc[:, :-1]
y_test = test_data.iloc[:, -1]

# 对类别变量进行编码
from sklearn.preprocessing import LabelEncoder

label_encoder = LabelEncoder()
y_train = label_encoder.fit_transform(y_train)
y_test = label_encoder.transform(y_test)

from sklearn.ensemble import RandomForestClassifier

# 初始化模型
model = RandomForestClassifier(n_estimators=100, random_state=42)

# 训练模型
model.fit(X_train, y_train)

from sklearn.metrics import accuracy_score

# 在测试集上进行预测
y_pred = model.predict(X_test)

# 计算准确性
accuracy = accuracy_score(y_test, y_pred)
print(f"模型在测试集上的准确性: {accuracy:.4f}")

# 保存预测结果到文件
import numpy as np

np.savetxt(r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/monster/predictions.csv', y_pred, delimiter=',', fmt='%d')

print(f"模型在测试集上的准确性: {accuracy:.4f}")
