import pandas as pd

# 读取训练集和测试集
train_data = pd.read_csv(r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/horse_health_outcomes/train.csv')
test_data = pd.read_csv(r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/horse_health_outcomes/test.csv')

# 查看数据的前几行
print(train_data.head())
print(test_data.head())

# 检查缺失值
print(train_data.isnull().sum())
print(test_data.isnull().sum())

# 处理缺失值（这里假设用均值填充数值型缺失值，用众数填充分类型缺失值）
for col in train_data.columns:
    if train_data[col].dtype == 'object':
        train_data[col].fillna(train_data[col].mode()[0], inplace=True)
    else:
        train_data[col].fillna(train_data[col].mean(), inplace=True)

for col in test_data.columns:
    if test_data[col].dtype == 'object':
        test_data[col].fillna(test_data[col].mode()[0], inplace=True)
    else:
        test_data[col].fillna(test_data[col].mean(), inplace=True)

# 编码分类变量
from sklearn.preprocessing import LabelEncoder

label_encoders = {}
for col in train_data.select_dtypes(include=['object']).columns:
    le = LabelEncoder()
    train_data[col] = le.fit_transform(train_data[col])
    test_data[col] = le.transform(test_data[col])
    label_encoders[col] = le

# 假设目标变量是 'outcome'
X_train = train_data.drop('outcome', axis=1)
y_train = train_data['outcome']
X_test = test_data.drop('outcome', axis=1)
y_test = test_data['outcome']

from sklearn.ensemble import RandomForestClassifier

# 初始化模型
model = RandomForestClassifier(random_state=42)

# 训练模型
model.fit(X_train, y_train)

# 进行预测
y_pred = model.predict(X_test)

from sklearn.metrics import f1_score

# 计算F1分数
f1 = f1_score(y_test, y_pred, average='weighted')
print(f'F1 Score: {f1}')

# 保存预测结果到CSV文件
test_data['predicted_outcome'] = y_pred
test_data.to_csv(r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/horse_health_outcomes/predictions.csv', index=False)
