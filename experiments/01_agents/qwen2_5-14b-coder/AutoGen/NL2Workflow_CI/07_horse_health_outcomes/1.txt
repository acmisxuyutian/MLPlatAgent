import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler, LabelEncoder
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import f1_score
import joblib

# 1. 加载数据
train_path = r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/horse_health_outcomes/train.csv'
test_path = r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/horse_health_outcomes/test.csv'

train_data = pd.read_csv(train_path)
test_data = pd.read_csv(test_path)

# 2. 数据预处理
# 假设目标变量是 'outcome'
X_train = train_data.drop('outcome', axis=1)
y_train = train_data['outcome']

# 处理缺失值
X_train.fillna(X_train.mean(), inplace=True)
test_data.fillna(test_data.mean(), inplace=True)

# 编码分类变量
label_encoders = {}
for column in X_train.select_dtypes(include=['object']).columns:
    le = LabelEncoder()
    X_train[column] = le.fit_transform(X_train[column])
    test_data[column] = le.transform(test_data[column])
    label_encoders[column] = le

# 3. 特征选择
# 这里我们假设所有特征都是有用的，如果有特征选择的需求，可以使用特征选择方法

# 4. 模型训练
scaler = StandardScaler()
X_train_scaled = scaler.fit_transform(X_train)

model = RandomForestClassifier(random_state=42)
model.fit(X_train_scaled, y_train)

# 5. 模型预测
X_test_scaled = scaler.transform(test_data)
y_pred = model.predict(X_test_scaled)

# 6. 评估模型
f1 = f1_score(test_data['outcome'], y_pred, average='weighted')
print(f'F1 Score: {f1}')

# 7. 保存预测结果
test_data['predicted_outcome'] = y_pred
test_data[['id', 'predicted_outcome']].to_csv('predicted_outcomes.csv', index=False)

# 保存模型和预处理工具
joblib.dump(model, 'horse_health_model.pkl')
joblib.dump(scaler, 'scaler.pkl')
joblib.dump(label_encoders, 'label_encoders.pkl')

print('模型训练和预测完成，预测结果已保存到 predicted_outcomes.csv')
