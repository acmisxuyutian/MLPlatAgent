######代码######

import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.metrics import f1_score, accuracy_score, precision_score, recall_score, roc_auc_score
from imblearn.over_sampling import SMOTE
from pytorch_tabnet.tab_model import TabNetClassifier
import optuna
import torch

SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)

# Data Retrieval
def load_data():
    train_data = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/07_icr_identify/split_train.csv')
    eval_data = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/07_icr_identify/split_eval.csv')
    return train_data, eval_data

# Data Preprocessing
def preprocess_data(train_data, eval_data):
    # Data Analysis
    print(train_data.describe())
    print(train_data.isnull().sum())
    print(train_data['Class'].value_counts())

    # Identify numerical and categorical features
    numerical_features = train_data.select_dtypes(include=['int64', 'float64']).columns.tolist()
    categorical_features = train_data.select_dtypes(include=['object', 'category']).columns.tolist()

    # Handling Missing Values
    imputer_num = SimpleImputer(strategy='mean')
    imputer_cat = SimpleImputer(strategy='most_frequent')

    train_data[numerical_features] = imputer_num.fit_transform(train_data[numerical_features])
    train_data[categorical_features] = imputer_cat.fit_transform(train_data[categorical_features])

    eval_data[numerical_features] = imputer_num.transform(eval_data[numerical_features])
    eval_data[categorical_features] = imputer_cat.transform(eval_data[categorical_features])

    # Feature Scaling
    scaler = StandardScaler()
    train_data[numerical_features] = scaler.fit_transform(train_data[numerical_features])
    eval_data[numerical_features] = scaler.transform(eval_data[numerical_features])

    # Encoding Categorical Variables
    encoder = OneHotEncoder(handle_unknown='ignore', sparse_output=False)
    encoded_train = encoder.fit_transform(train_data[categorical_features])
    encoded_eval = encoder.transform(eval_data[categorical_features])

    encoded_train_df = pd.DataFrame(encoded_train, columns=encoder.get_feature_names_out(categorical_features))
    encoded_eval_df = pd.DataFrame(encoded_eval, columns=encoder.get_feature_names_out(categorical_features))

    train_data = pd.concat([train_data.drop(columns=categorical_features), encoded_train_df], axis=1)
    eval_data = pd.concat([eval_data.drop(columns=categorical_features), encoded_eval_df], axis=1)

    # Splitting the Data
    X_train = train_data.drop(columns=['Class'])
    y_train = train_data['Class']

    X_eval = eval_data.drop(columns=['Class'])
    y_eval = eval_data['Class']

    return X_train, y_train, X_eval, y_eval

# Data Augmentation
def augment_data(X_train, y_train):
    # Ensure y_train is a 1D array and in integer format
    y_train = y_train.astype(int).values.ravel()
    
    smote = SMOTE()
    X_train_resampled, y_train_resampled = smote.fit_resample(X_train, y_train)
    return X_train_resampled, y_train_resampled

# Model Training and Optimization
def train_model(X_train, y_train, X_eval, y_eval):
    model = TabNetClassifier()

    def objective(trial):
        params = {
            'n_d': trial.suggest_int('n_d', 8, 64),
            'n_a': trial.suggest_int('n_a', 8, 64),
            'n_steps': trial.suggest_int('n_steps', 3, 10),
            'gamma': trial.suggest_float('gamma', 1.0, 2.0),
            'lambda_sparse': trial.suggest_float('lambda_sparse', 1e-6, 1e-3, log=True),
            'batch_size': trial.suggest_categorical('batch_size', [16, 32, 64, 128]),
            'virtual_batch_size': trial.suggest_categorical('virtual_batch_size', [4, 8, 16, 32]),
            'max_epochs': 100,
            'patience': 10,
            'device_name': 'cuda' if torch.cuda.is_available() else 'cpu'
        }

        model.fit(
            X_train=X_train, y_train=y_train,
            eval_set=[(X_eval, y_eval)],
            eval_metric=['f1'],
            max_epochs=params['max_epochs'],
            patience=params['patience'],
            batch_size=params['batch_size'],
            virtual_batch_size=params['virtual_batch_size'],
            num_workers=0,
            drop_last=False,
            scheduler_params={'step_size': 50, 'gamma': 0.9},
            scheduler_fn=torch.optim.lr_scheduler.StepLR,
            loss_fn=torch.nn.CrossEntropyLoss()
        )

        eval_pred = model.predict(X_eval)
        f1 = f1_score(y_eval, eval_pred, average='weighted')
        return f1

    study = optuna.create_study(direction='maximize')
    study.optimize(objective, n_trials=50)

    best_params = study.best_params
    print(f"Best hyperparameters: {best_params}")

    model = TabNetClassifier(**best_params)
    model.fit(
        X_train=X_train, y_train=y_train,
        eval_set=[(X_eval, y_eval)],
        eval_metric=['f1'],
        max_epochs=best_params['max_epochs'],
        patience=best_params['patience'],
        batch_size=best_params['batch_size'],
        virtual_batch_size=best_params['virtual_batch_size'],
        num_workers=0,
        drop_last=False,
        scheduler_params={'step_size': 50, 'gamma': 0.9},
        scheduler_fn=torch.optim.lr_scheduler.StepLR,
        loss_fn=torch.nn.CrossEntropyLoss()
    )

    return model

# Model Evaluation
def evaluate_model(model, X_eval, y_eval):
    eval_pred = model.predict(X_eval)
    f1 = f1_score(y_eval, eval_pred, average='weighted')
    accuracy = accuracy_score(y_eval, eval_pred)
    precision = precision_score(y_eval, eval_pred, average='weighted')
    recall = recall_score(y_eval, eval_pred, average='weighted')
    roc_auc = roc_auc_score(y_eval, model.predict_proba(X_eval), multi_class='ovr')

    performance_scores = {
        'F1': f1,
        'Accuracy': accuracy,
        'Precision': precision,
        'Recall': recall,
        'ROC-AUC': roc_auc
    }

    return performance_scores

# Main function to orchestrate the pipeline
def main():
    # Step 1. Retrieve or load a dataset
    train_data, eval_data = load_data()

    # Step 2. Preprocess the data
    X_train, y_train, X_eval, y_eval = preprocess_data(train_data, eval_data)

    # Step 3. Augment the data
    X_train_resampled, y_train_resampled = augment_data(X_train, y_train)

    # Step 4. Train the model
    model = train_model(X_train_resampled, y_train_resampled, X_eval, y_eval)

    # Step 5. Evaluate the model
    model_performance = evaluate_model(model, X_eval, y_eval)

    # Save the trained model
    os.makedirs('./agent_workspace/trained_models', exist_ok=True)
    model.save_model('./agent_workspace/trained_models/tabnet_model.pkl')

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
