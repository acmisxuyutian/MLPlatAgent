import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.metrics import f1_score
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.feature_selection import SelectKBest, mutual_info_classif
from pytorch_tabnet.tab_model import TabNetClassifier
import optuna
import joblib

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data Retrieval and Collection
def load_data():
    train_data = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/07_icr_identify/split_train.csv')
    eval_data = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/07_icr_identify/split_eval.csv')
    return train_data, eval_data

# Data Preprocessing
def preprocess_data(train_data, eval_data):
    # Check for missing values
    print("Missing values in training data:")
    print(train_data.isnull().sum())
    print("Missing values in evaluation data:")
    print(eval_data.isnull().sum())

    # Identify outliers
    numerical_features = train_data.select_dtypes(include=['number']).columns
    for feature in numerical_features:
        q1 = train_data[feature].quantile(0.25)
        q3 = train_data[feature].quantile(0.75)
        iqr = q3 - q1
        lower_bound = q1 - 1.5 * iqr
        upper_bound = q3 + 1.5 * iqr
        outliers = train_data[(train_data[feature] < lower_bound) | (train_data[feature] > upper_bound)]
        print(f"Outliers in {feature}: {len(outliers)}")

    # Impute missing values
    numerical_features = train_data.select_dtypes(include=['number']).columns
    categorical_features = train_data.select_dtypes(include=['object']).columns

    # Impute numerical features with median
    train_data[numerical_features] = train_data[numerical_features].fillna(train_data[numerical_features].median())
    eval_data[numerical_features] = eval_data[numerical_features].fillna(train_data[numerical_features].median())

    # Impute categorical features with mode
    train_data[categorical_features] = train_data[categorical_features].fillna(train_data[categorical_features].mode().iloc[0])
    eval_data[categorical_features] = eval_data[categorical_features].fillna(train_data[categorical_features].mode().iloc[0])

    # Feature Scaling
    scaler = StandardScaler()
    train_data_scaled = scaler.fit_transform(train_data[numerical_features])
    eval_data_scaled = scaler.transform(eval_data[numerical_features])

    train_data_scaled = pd.DataFrame(train_data_scaled, columns=numerical_features)
    eval_data_scaled = pd.DataFrame(eval_data_scaled, columns=numerical_features)

    # Encoding Categorical Variables
    encoder = OneHotEncoder(handle_unknown='ignore')
    train_data_encoded = encoder.fit_transform(train_data[categorical_features])
    eval_data_encoded = encoder.transform(eval_data[categorical_features])

    train_data_encoded = pd.DataFrame(train_data_encoded.toarray(), columns=encoder.get_feature_names_out(categorical_features))
    eval_data_encoded = pd.DataFrame(eval_data_encoded.toarray(), columns=encoder.get_feature_names_out(categorical_features))

    # Combine numerical and categorical features
    train_data_combined = pd.concat([train_data_scaled, train_data_encoded], axis=1)
    eval_data_combined = pd.concat([eval_data_scaled, eval_data_encoded], axis=1)

    # Feature Selection
    selector = SelectKBest(mutual_info_classif, k=10)
    train_data_selected = selector.fit_transform(train_data_combined, train_data['Class'])
    eval_data_selected = selector.transform(eval_data_combined)

    train_data_selected = pd.DataFrame(train_data_selected, columns=train_data_combined.columns[selector.get_support()])
    eval_data_selected = pd.DataFrame(eval_data_selected, columns=eval_data_combined.columns[selector.get_support()])

    return train_data_selected, eval_data_selected, train_data['Class'], eval_data['Class'], scaler, encoder, selector

# Model Training and Hyperparameter Tuning
def train_model(train_data_selected, train_labels, eval_data_selected, eval_labels):
    def objective(trial):
        n_d = trial.suggest_int('n_d', 32, 128)
        n_a = trial.suggest_int('n_a', 32, 128)
        n_steps = trial.suggest_int('n_steps', 3, 10)
        gamma = trial.suggest_float('gamma', 1.0, 2.0)
        n_independent = trial.suggest_int('n_independent', 1, 5)
        n_shared = trial.suggest_int('n_shared', 1, 5)
        lambda_sparse = trial.suggest_float('lambda_sparse', 0.0001, 0.1)
        optimizer = trial.suggest_categorical('optimizer', ['Adam', 'SGD'])
        learning_rate = trial.suggest_float('learning_rate', 0.001, 0.1)

        model = TabNetClassifier(
            n_d=n_d, n_a=n_a, n_steps=n_steps, gamma=gamma,
            n_independent=n_independent, n_shared=n_shared,
            lambda_sparse=lambda_sparse, optimizer_fn=optimizer,
            optimizer_params={'lr': learning_rate}
        )

        model.fit(
            X_train=train_data_selected.values, y_train=train_labels.values,
            eval_set=[(eval_data_selected.values, eval_labels.values)],
            eval_metric=['accuracy']
        )

        # Evaluate the model on the evaluation set to get the F1 score
        eval_predictions = model.predict(eval_data_selected.values)
        f1 = f1_score(eval_labels, eval_predictions, average='weighted')
        return f1

    study = optuna.create_study(direction='maximize')
    study.optimize(objective, n_trials=50)

    best_params = study.best_params
    print(f"Best hyperparameters: {best_params}")

    final_model = TabNetClassifier(
        n_d=best_params['n_d'], n_a=best_params['n_a'], n_steps=best_params['n_steps'], gamma=best_params['gamma'],
        n_independent=best_params['n_independent'], n_shared=best_params['n_shared'],
        lambda_sparse=best_params['lambda_sparse'], optimizer_fn=best_params['optimizer'],
        optimizer_params={'lr': best_params['learning_rate']}
    )

    final_model.fit(
        X_train=train_data_selected.values, y_train=train_labels.values,
        eval_set=[(eval_data_selected.values, eval_labels.values)],
        eval_metric=['accuracy']
    )

    return final_model

# Model Evaluation
def evaluate_model(model, eval_data_selected, eval_labels):
    eval_predictions = model.predict(eval_data_selected.values)
    f1 = f1_score(eval_labels, eval_predictions, average='weighted')
    print(f"F1 Score on evaluation data: {f1}")
    return f1

# Model Saving and Deployment
def save_model(model, scaler, encoder, selector):
    model_path = './agent_workspace/trained_models/tabnet_model.pkl'
    preprocessing_path = './agent_workspace/trained_models/preprocessing.pkl'

    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    os.makedirs(os.path.dirname(preprocessing_path), exist_ok=True)

    joblib.dump(model, model_path)
    preprocessing_steps = {
        'scaler': scaler,
        'encoder': encoder,
        'selector': selector
    }
    joblib.dump(preprocessing_steps, preprocessing_path)

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_data, eval_data = load_data()

    # Step 2. Preprocess the data
    train_data_selected, eval_data_selected, train_labels, eval_labels, scaler, encoder, selector = preprocess_data(train_data, eval_data)

    # Step 3. Train the model
    model = train_model(train_data_selected, train_labels, eval_data_selected, eval_labels)

    # Step 4. Evaluate the model
    model_performance = evaluate_model(model, eval_data_selected, eval_labels)

    # Step 5. Save the model and preprocessing steps
    save_model(model, scaler, encoder, selector)

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Evaluation Set:", model_performance)
