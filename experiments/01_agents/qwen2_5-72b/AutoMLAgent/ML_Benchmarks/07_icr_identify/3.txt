import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, FunctionTransformer
from sklearn.feature_selection import SelectKBest, mutual_info_classif
from sklearn.metrics import f1_score, precision_score, recall_score, roc_auc_score
from torch.utils.data import DataLoader, TensorDataset
import torch
import torch.nn as nn
import torch.optim as optim
from itertools import product

# Set the random seed for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)

# Data retrieval and loading
def load_data(train_data_path, eval_data_path):
    train_df = pd.read_csv(train_data_path)
    eval_df = pd.read_csv(eval_data_path)
    print("Training Data:")
    print(train_df.head())
    print(train_df.shape)
    print("Evaluation Data:")
    print(eval_df.head())
    print(eval_df.shape)
    return train_df, eval_df

# Data preprocessing
def preprocess_data(train_df, eval_df):
    # Statistical summary
    print("Statistical Summary:")
    print(train_df.describe())

    # Handle missing values
    imputer_num = SimpleImputer(strategy='mean')
    imputer_cat = SimpleImputer(strategy='most_frequent')

    train_df_numerical = train_df.select_dtypes(include=[np.number])
    train_df_categorical = train_df.select_dtypes(exclude=[np.number])

    train_df_numerical = pd.DataFrame(imputer_num.fit_transform(train_df_numerical), columns=train_df_numerical.columns)
    train_df_categorical = pd.DataFrame(imputer_cat.fit_transform(train_df_categorical), columns=train_df_categorical.columns)

    train_df = pd.concat([train_df_numerical, train_df_categorical], axis=1)

    # Feature scaling
    scaler = StandardScaler()
    train_df_numerical = pd.DataFrame(scaler.fit_transform(train_df_numerical), columns=train_df_numerical.columns)

    # Categorical encoding
    train_df_categorical = pd.get_dummies(train_df_categorical)

    train_df = pd.concat([train_df_numerical, train_df_categorical], axis=1)

    # Ensure 'Class' is an integer for classification
    train_df['Class'] = train_df['Class'].astype(int)

    # Feature selection
    selector = SelectKBest(score_func=mutual_info_classif, k=30)
    train_df_selected = selector.fit_transform(train_df.drop('Class', axis=1), train_df['Class'])

    # Feature transformation
    log_transformer = FunctionTransformer(np.log1p, validate=False)
    train_df_selected = log_transformer.transform(train_df_selected)

    # Repeat the same steps for the evaluation dataset
    eval_df_numerical = eval_df.select_dtypes(include=[np.number])
    eval_df_categorical = eval_df.select_dtypes(exclude=[np.number])

    eval_df_numerical = pd.DataFrame(imputer_num.transform(eval_df_numerical), columns=eval_df_numerical.columns)
    eval_df_categorical = pd.DataFrame(imputer_cat.transform(eval_df_categorical), columns=eval_df_categorical.columns)

    eval_df = pd.concat([eval_df_numerical, eval_df_categorical], axis=1)

    eval_df_numerical = pd.DataFrame(scaler.transform(eval_df_numerical), columns=eval_df_numerical.columns)
    eval_df_categorical = pd.get_dummies(eval_df_categorical)

    # Ensure the same columns in the evaluation dataset as in the training dataset
    missing_cols = set(train_df_categorical.columns) - set(eval_df_categorical.columns)
    for col in missing_cols:
        eval_df_categorical[col] = 0

    eval_df_categorical = eval_df_categorical[train_df_categorical.columns]

    eval_df = pd.concat([eval_df_numerical, eval_df_categorical], axis=1)

    eval_df['Class'] = eval_df['Class'].astype(int)

    eval_df_selected = selector.transform(eval_df.drop('Class', axis=1))
    eval_df_selected = log_transformer.transform(eval_df_selected)

    return train_df_selected, train_df['Class'], eval_df_selected, eval_df['Class']

# Define the FNN model
class FNN(nn.Module):
    def __init__(self, input_size, hidden_sizes, output_size, dropout_rate):
        super(FNN, self).__init__()
        layers = []
        for i in range(len(hidden_sizes)):
            if i == 0:
                layers.append(nn.Linear(input_size, hidden_sizes[i]))
            else:
                layers.append(nn.Linear(hidden_sizes[i-1], hidden_sizes[i]))
            layers.append(nn.ReLU())
            layers.append(nn.Dropout(dropout_rate))
        layers.append(nn.Linear(hidden_sizes[-1], output_size))
        layers.append(nn.Softmax(dim=1))
        self.model = nn.Sequential(*layers)

    def forward(self, x):
        return self.model(x)

# Hyperparameter tuning function
def train_fnn(learning_rate, batch_size, num_layers, num_neurons, dropout_rate, train_df_selected, train_labels, eval_df_selected, eval_labels):
    input_size = train_df_selected.shape[1]
    output_size = len(np.unique(train_labels))
    hidden_sizes = [int(num_neurons)] * int(num_layers)
    
    model = FNN(input_size, hidden_sizes, output_size, dropout_rate)
    criterion = nn.CrossEntropyLoss()
    optimizer = optim.Adam(model.parameters(), lr=learning_rate)
    
    train_dataset = TensorDataset(torch.tensor(train_df_selected, dtype=torch.float32), torch.tensor(train_labels, dtype=torch.long))
    train_loader = DataLoader(train_dataset, batch_size=int(batch_size), shuffle=True)
    
    best_val_f1 = 0
    patience = 5
    epochs_without_improvement = 0

    for epoch in range(100):
        model.train()
        for inputs, targets in train_loader:
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, targets)
            loss.backward()
            optimizer.step()
        
        model.eval()
        with torch.no_grad():
            eval_inputs = torch.tensor(eval_df_selected, dtype=torch.float32)
            eval_targets = torch.tensor(eval_labels, dtype=torch.long)
            eval_outputs = model(eval_inputs)
            _, predicted = torch.max(eval_outputs, 1)
            f1 = f1_score(eval_targets, predicted, average='weighted')
        
        if f1 > best_val_f1:
            best_val_f1 = f1
            epochs_without_improvement = 0
        else:
            epochs_without_improvement += 1
            if epochs_without_improvement >= patience:
                break

    return best_val_f1, model

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/07_icr_identify/split_train.csv'
    eval_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/07_icr_identify/split_eval.csv'
    train_df, eval_df = load_data(train_data_path, eval_data_path)

    # Step 2. Preprocess the data
    train_df_selected, train_labels, eval_df_selected, eval_labels = preprocess_data(train_df, eval_df)

    # Step 3. Train the model with hyperparameter tuning
    hyperparameters = {
        'learning_rate': [0.001, 0.01, 0.1],
        'batch_size': [32, 64, 128],
        'num_layers': [2, 3, 4],
        'num_neurons': [32, 64, 128],
        'dropout_rate': [0.2, 0.3, 0.4]
    }

    best_f1 = 0
    best_model = None
    best_params = None

    for params in product(*hyperparameters.values()):
        params = dict(zip(hyperparameters.keys(), params))
        f1, model = train_fnn(**params, train_df_selected=train_df_selected, train_labels=train_labels, eval_df_selected=eval_df_selected, eval_labels=eval_labels)
        if f1 > best_f1:
            best_f1 = f1
            best_model = model
            best_params = params

    print("Best Hyperparameters:", best_params)

    # Evaluate the final model
    with torch.no_grad():
        eval_inputs = torch.tensor(eval_df_selected, dtype=torch.float32)
        eval_targets = torch.tensor(eval_labels, dtype=torch.long)
        eval_outputs = best_model(eval_inputs)
        _, predicted = torch.max(eval_outputs, 1)
        f1 = f1_score(eval_targets, predicted, average='weighted')
        precision = precision_score(eval_targets, predicted, average='weighted')
        recall = recall_score(eval_targets, predicted, average='weighted')
        auc_roc = roc_auc_score(eval_targets, predicted, multi_class='ovr')

    print("F1 Score:", f1)
    print("Precision:", precision)
    print("Recall:", recall)
    print("AUC-ROC:", auc_roc)

    # Save the trained model
    model_save_path = "./agent_workspace/trained_models/fnn_model.pth"
    os.makedirs(os.path.dirname(model_save_path), exist_ok=True)
    torch.save(best_model.state_dict(), model_save_path)
    print(f"Model saved to {model_save_path}")

    # Return model performance
    model_performance = {
        'F1 Score': f1,
        'Precision': precision,
        'Recall': recall,
        'AUC-ROC': auc_roc
    }

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
