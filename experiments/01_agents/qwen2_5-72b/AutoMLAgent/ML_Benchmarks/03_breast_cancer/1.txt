######代码######

import os
import random
import numpy as np
import pandas as pd
from sklearn.datasets import load_breast_cancer
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score
import joblib

# Set the random seed for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Define the path for saving the model
MODEL_PATH = "./agent_workspace/trained_models/random_forest_model.pkl"

# Step 1: Data Retrieval and Loading
def load_dataset():
    data = load_breast_cancer()
    df = pd.DataFrame(data.data, columns=data.feature_names)
    df['target'] = data.target
    return df

# Step 2: Data Preprocessing
def preprocess_data(df):
    # Handling missing values
    if df.isnull().sum().sum() > 0:
        df.fillna(df.mean(), inplace=True)
    
    # Feature scaling
    scaler = StandardScaler()
    df[df.columns[:-1]] = scaler.fit_transform(df[df.columns[:-1]])
    
    return df

# Step 3: Data Splitting
def split_data(df):
    X = df[df.columns[:-1]]
    y = df['target']
    X_train, X_val, y_train, y_val = train_test_split(X, y, test_size=0.2, stratify=y, random_state=SEED)
    return X_train, X_val, y_train, y_val

# Step 4: Model Training
def train_model(X_train, y_train):
    rf_model = RandomForestClassifier(
        n_estimators=100,
        max_depth=20,
        min_samples_split=5,
        min_samples_leaf=2,
        random_state=SEED
    )
    rf_model.fit(X_train, y_train)
    return rf_model

# Step 5: Model Evaluation
def evaluate_model(model, X_val, y_val):
    y_val_pred = model.predict(X_val)
    val_accuracy = accuracy_score(y_val, y_val_pred)
    return val_accuracy

# Step 6: Model Saving
def save_model(model, model_path):
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    joblib.dump(model, model_path)

# Step 7: Data Analysis (Optional)
def data_analysis(df, model):
    summary_stats = df.describe()
    print("Summary Statistics:\n", summary_stats)
    
    class_distribution = df['target'].value_counts()
    print("Class Distribution:\n", class_distribution)
    
    correlation_matrix = df.corr()
    print("Correlation Matrix:\n", correlation_matrix)
    
    feature_importances = pd.DataFrame({'feature': df.columns[:-1], 'importance': model.feature_importances_})
    feature_importances = feature_importances.sort_values(by='importance', ascending=False)
    print("Feature Importances:\n", feature_importances)

# The main function to orchestrate the entire pipeline
def main():
    # Step 1: Load the dataset
    df = load_dataset()
    
    # Step 2: Preprocess the data
    df = preprocess_data(df)
    
    # Step 3: Split the data
    X_train, X_val, y_train, y_val = split_data(df)
    
    # Step 4: Train the model
    model = train_model(X_train, y_train)
    
    # Step 5: Evaluate the model
    val_accuracy = evaluate_model(model, X_val, y_val)
    print(f'Validation Accuracy: {val_accuracy:.4f}')
    
    # Step 6: Save the model
    save_model(model, MODEL_PATH)
    
    # Step 7: Perform data analysis (optional)
    data_analysis(df, model)
    
    return val_accuracy

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Validation Set:", model_performance)
