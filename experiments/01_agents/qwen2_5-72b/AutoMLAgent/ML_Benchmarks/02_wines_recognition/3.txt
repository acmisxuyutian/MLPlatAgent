######代码######

import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from xgboost import XGBClassifier
from sklearn.metrics import accuracy_score, f1_score
import joblib

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Define the path for saving the trained model
MODEL_PATH = "./agent_workspace/trained_models/xgboost_wine_quality_model.pkl"

def load_dataset():
    """
    Load the dataset from the UCI Machine Learning Repository.
    """
    url = "https://archive.ics.uci.edu/ml/machine-learning-databases/wine/wine.data"
    data = pd.read_csv(url, header=None)
    return data

def initial_exploration(data):
    """
    Perform initial data exploration.
    """
    summary_stats = data.describe()
    class_distribution = data[0].value_counts()
    print("Summary Statistics:\n", summary_stats)
    print("Class Distribution:\n", class_distribution)

def preprocess_data(data):
    """
    Preprocess the dataset: handle missing values, scale features.
    """
    # Check for missing values
    missing_values = data.isnull().sum()
    print("Missing Values:\n", missing_values)

    # Impute missing values with the mean
    data.fillna(data.mean(), inplace=True)

    # Scale the features
    scaler = StandardScaler()
    features = data.iloc[:, 1:]
    target = data.iloc[:, 0]

    # Adjust target to start from 0
    target = target - 1

    scaled_features = scaler.fit_transform(features)
    scaled_data = pd.DataFrame(scaled_features, columns=features.columns)

    return scaled_data, target

def split_data(scaled_data, target):
    """
    Split the dataset into training (80%) and testing (20%) sets.
    """
    X_train, X_test, y_train, y_test = train_test_split(scaled_data, target, test_size=0.2, random_state=SEED)
    return X_train, X_test, y_train, y_test

def train_model(X_train, y_train):
    """
    Train an XGBoost model on the training set.
    """
    xgb_model = XGBClassifier(n_estimators=100, max_depth=6, learning_rate=0.1, subsample=0.8, random_state=SEED)
    xgb_model.fit(X_train, y_train)
    return xgb_model

def evaluate_model(model, X_test, y_test):
    """
    Evaluate the model's performance on the test set.
    """
    y_pred = model.predict(X_test)
    accuracy = accuracy_score(y_test, y_pred)
    f1 = f1_score(y_test, y_pred, average='weighted')
    performance_scores = {
        'Accuracy': accuracy,
        'F1': f1
    }
    return performance_scores

def save_model(model, model_path):
    """
    Save the trained model to a file.
    """
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    joblib.dump(model, model_path)

def load_model(model_path):
    """
    Load the saved model for inference.
    """
    return joblib.load(model_path)

def main():
    """
    Main function to execute the tabular classification pipeline.
    """
    # Step 1: Load the dataset
    data = load_dataset()
    initial_exploration(data)

    # Step 2: Preprocess the data
    scaled_data, target = preprocess_data(data)

    # Step 3: Split the data
    X_train, X_test, y_train, y_test = split_data(scaled_data, target)

    # Step 4: Train the model
    model = train_model(X_train, y_train)

    # Step 5: Evaluate the model
    performance_scores = evaluate_model(model, X_test, y_test)
    print("Model Performance on Test Set:", performance_scores)

    # Step 6: Save the model
    save_model(model, MODEL_PATH)

    return performance_scores

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
