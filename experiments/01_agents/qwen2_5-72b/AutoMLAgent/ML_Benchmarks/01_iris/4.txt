######代码######

import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.datasets import load_iris
from sklearn.preprocessing import StandardScaler, LabelEncoder
from sklearn.model_selection import train_test_split, GridSearchCV
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score, classification_report, confusion_matrix
import seaborn as sns
import matplotlib.pyplot as plt
import joblib

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

DATASET_PATH = "_experiments/datasets"  # path for saving and loading dataset(s) (or the user's uploaded dataset) for preprocessing, training, hyperparamter tuning, and evaluation

# Data retrieval and loading
def load_iris_dataset():
    iris = load_iris()
    data = pd.DataFrame(iris.data, columns=iris.feature_names)
    data['species'] = iris.target
    return data, iris

# Data preprocessing
def preprocess_data(data, iris):
    # Check for missing values
    print("Missing values:\n", data.isnull().sum())

    # Standardize the features
    scaler = StandardScaler()
    data[iris.feature_names] = scaler.fit_transform(data[iris.feature_names])

    # Encode the target variable
    encoder = LabelEncoder()
    data['species'] = encoder.fit_transform(data['species'])

    return data

# Data analysis and visualization
def visualize_data(data):
    sns.pairplot(data, hue='species')
    plt.savefig('iris_pairplot.png')
    plt.show()

# Model training and hyperparameter tuning
def train_model(X_train, y_train):
    param_grid = {
        'n_estimators': [10, 50, 100, 200],
        'max_depth': [None, 10, 20, 30],
        'min_samples_split': [2, 5, 10]
    }

    rf = RandomForestClassifier()
    grid_search = GridSearchCV(rf, param_grid, cv=5, scoring='accuracy')
    grid_search.fit(X_train, y_train)

    best_rf = grid_search.best_estimator_
    return best_rf

# Model evaluation
def evaluate_model(model, X_test, y_test):
    y_pred = model.predict(X_test)
    accuracy = accuracy_score(y_test, y_pred)
    report = classification_report(y_test, y_pred)
    cm = confusion_matrix(y_test, y_pred)

    performance_scores = {
        'Accuracy': accuracy,
        'Classification Report': report,
        'Confusion Matrix': cm
    }

    return performance_scores

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    """
    Main function to execute the tabular classification pipeline.
    """

    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    data, iris = load_iris_dataset()

    # Step 2. Create a train-valid-test split of the data
    X = data[iris.feature_names]
    y = data['species']
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, random_state=42)

    # Step 3. With the split dataset, run data preprocessing and feature engineering (if applicable)
    data = preprocess_data(data, iris)

    # Step 4. Define required model
    model = train_model(X_train, y_train)

    # Step 5. Save the trained model
    model_path = "./agent_workspace/trained_models/best_random_forest_model.pkl"
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    joblib.dump(model, model_path)

    # Step 6. Evaluate the trained model
    model_performance = evaluate_model(model, X_test, y_test)

    # Step 7. Save evaluation results
    with open('./agent_workspace/model_evaluation.txt', 'w') as f:
        f.write(f'Accuracy: {model_performance["Accuracy"]}\n')
        f.write(f'Classification Report:\n{model_performance["Classification Report"]}\n')
        f.write(f'Confusion Matrix:\n{model_performance["Confusion Matrix"]}\n')

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
