import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, LabelEncoder
from scipy.stats import zscore
from imblearn.over_sampling import SMOTE
from sklearn.decomposition import PCA
from sklearn.model_selection import train_test_split
from sklearn.metrics import roc_auc_score
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, TensorDataset
import optuna

SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)

# Data retrieval
def load_data(train_path, eval_path):
    train_df = pd.read_csv(train_path)
    eval_df = pd.read_csv(eval_path)
    return train_df, eval_df

# Data preprocessing
def preprocess_data(train_df, eval_df):
    # Handle missing values
    imputer_num = SimpleImputer(strategy='mean')
    imputer_cat = SimpleImputer(strategy='most_frequent')

    # Identify numerical and categorical features
    numerical_features = train_df.select_dtypes(include=[np.number]).columns.tolist()
    categorical_features = train_df.select_dtypes(include=[object]).columns.tolist()

    # Impute numerical and categorical features
    if numerical_features:
        train_df.loc[:, numerical_features] = imputer_num.fit_transform(train_df[numerical_features])
        eval_df.loc[:, numerical_features] = imputer_num.transform(eval_df[numerical_features])

    if categorical_features:
        train_df.loc[:, categorical_features] = imputer_cat.fit_transform(train_df[categorical_features])
        eval_df.loc[:, categorical_features] = imputer_cat.transform(eval_df[categorical_features])

    # Remove outliers
    if numerical_features:
        z_scores = np.abs(zscore(train_df[numerical_features]))
        train_df = train_df[(z_scores < 3).all(axis=1)]

    # Remove duplicates
    train_df.drop_duplicates(inplace=True)
    eval_df.drop_duplicates(inplace=True)

    # Standardize numerical features
    if numerical_features:
        scaler = StandardScaler()
        train_df.loc[:, numerical_features] = scaler.fit_transform(train_df[numerical_features])
        eval_df.loc[:, numerical_features] = scaler.transform(eval_df[numerical_features])

    # Encode categorical features
    if categorical_features:
        for feature in categorical_features:
            le = LabelEncoder()
            train_df.loc[:, feature] = le.fit_transform(train_df[feature])
            eval_df.loc[:, feature] = le.transform(eval_df[feature])

    return train_df, eval_df

# Data augmentation
def augment_data(train_df):
    smote = SMOTE()
    X_train = train_df.drop(columns=['target'])
    y_train = train_df['target']
    
    # Ensure the target is categorical
    if y_train.dtype == 'float64':
        y_train = y_train.astype(int)
    
    X_train, y_train = smote.fit_resample(X_train, y_train)
    train_df = pd.concat([X_train, y_train], axis=1)
    return train_df

# Feature engineering
def feature_engineering(train_df, eval_df):
    numerical_features = train_df.select_dtypes(include=[np.number]).columns.tolist()

    # Example: Create interaction features
    interaction_features = []
    for i in range(len(numerical_features)):
        for j in range(i + 1, len(numerical_features)):
            interaction_feature = f'{numerical_features[i]}_x_{numerical_features[j]}'
            train_df[interaction_feature] = train_df[numerical_features[i]] * train_df[numerical_features[j]]
            eval_df[interaction_feature] = eval_df[numerical_features[i]] * eval_df[numerical_features[j]]
            interaction_features.append(interaction_feature)

    # Dimensionality reduction
    pca = PCA(n_components=0.95)
    X_train = train_df.drop(columns=['target'])
    X_eval = eval_df.drop(columns=['target'])
    X_train_pca = pca.fit_transform(X_train)
    X_eval_pca = pca.transform(X_eval)

    train_df = pd.concat([pd.DataFrame(X_train_pca), train_df['target'].reset_index(drop=True)], axis=1)
    eval_df = pd.concat([pd.DataFrame(X_eval_pca), eval_df['target'].reset_index(drop=True)], axis=1)

    return train_df, eval_df

# Model definition
class DNN(nn.Module):
    def __init__(self, input_size, hidden_layers, hidden_size, dropout_rate):
        super(DNN, self).__init__()
        layers = []
        layers.append(nn.Linear(input_size, hidden_size))
        layers.append(nn.ReLU())
        layers.append(nn.Dropout(dropout_rate))

        for _ in range(hidden_layers - 1):
            layers.append(nn.Linear(hidden_size, hidden_size))
            layers.append(nn.ReLU())
            layers.append(nn.Dropout(dropout_rate))

        layers.append(nn.Linear(hidden_size, 1))
        layers.append(nn.Sigmoid())

        self.model = nn.Sequential(*layers)

    def forward(self, x):
        return self.model(x)

# Early stopping
class EarlyStopping:
    def __init__(self, patience=7, verbose=False, delta=0, path='checkpoint.pt'):
        self.patience = patience
        self.verbose = verbose
        self.counter = 0
        self.best_score = None
        self.early_stop = False
        self.val_loss_min = np.Inf
        self.delta = delta
        self.path = path

    def __call__(self, val_loss, model):
        score = -val_loss
        if self.best_score is None:
            self.best_score = score
            self.save_checkpoint(val_loss, model)
        elif score < self.best_score + self.delta:
            self.counter += 1
            print(f'EarlyStopping counter: {self.counter} out of {self.patience}')
            if self.counter >= self.patience:
                self.early_stop = True
        else:
            self.best_score = score
            self.save_checkpoint(val_loss, model)
            self.counter = 0

    def save_checkpoint(self, val_loss, model):
        if self.verbose:
            print(f'Validation loss decreased ({self.val_loss_min:.6f} --> {val_loss:.6f}).  Saving model ...')
        torch.save(model.state_dict(), self.path)
        self.val_loss_min = val_loss

# Model training
def train_model(model, train_loader, valid_loader, num_epochs, learning_rate, patience):
    criterion = nn.BCELoss()
    optimizer = optim.Adam(model.parameters(), lr=learning_rate)
    early_stopping = EarlyStopping(patience=patience, verbose=True)

    for epoch in range(num_epochs):
        model.train()
        running_loss = 0.0
        for inputs, labels in train_loader:
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, labels)
            loss.backward()
            optimizer.step()
            running_loss += loss.item()

        model.eval()
        valid_loss = 0.0
        with torch.no_grad():
            for inputs, labels in valid_loader:
                outputs = model(inputs)
                loss = criterion(outputs, labels)
                valid_loss += loss.item()

        print(f'Epoch {epoch + 1}/{num_epochs}, Training Loss: {running_loss / len(train_loader)}, Validation Loss: {valid_loss / len(valid_loader)}')

        early_stopping(valid_loss, model)
        if early_stopping.early_stop:
            print("Early stopping")
            break

    model.load_state_dict(torch.load('checkpoint.pt'))
    return model

# Model evaluation
def evaluate_model(model, test_loader):
    model.eval()
    y_true, y_pred = [], []
    with torch.no_grad():
        for inputs, labels in test_loader:
            outputs = model(inputs)
            y_true.extend(labels.cpu().numpy())
            y_pred.extend(outputs.cpu().numpy())

    y_true = np.array(y_true)
    y_pred = np.array(y_pred)
    auc_roc = roc_auc_score(y_true, y_pred)
    performance_scores = {
        'AUC-ROC': auc_roc
    }
    return performance_scores

# Objective function for hyperparameter tuning
def objective(trial, train_df, X_train, X_valid, y_train, y_valid, X_test, y_test):
    input_size = train_df.shape[1] - 1
    hidden_layers = trial.suggest_int('hidden_layers', 1, 5)
    hidden_size = trial.suggest_int('hidden_size', 32, 256)
    dropout_rate = trial.suggest_float('dropout_rate', 0.1, 0.5)
    learning_rate = trial.suggest_float('learning_rate', 0.001, 0.1)
    batch_size = trial.suggest_int('batch_size', 32, 128)

    model = DNN(input_size, hidden_layers, hidden_size, dropout_rate)
    train_loader = DataLoader(TensorDataset(torch.tensor(X_train.values, dtype=torch.float32), torch.tensor(y_train.values, dtype=torch.float32)), batch_size=batch_size, shuffle=True)
    valid_loader = DataLoader(TensorDataset(torch.tensor(X_valid.values, dtype=torch.float32), torch.tensor(y_valid.values, dtype=torch.float32)), batch_size=batch_size, shuffle=False)

    model = train_model(model, train_loader, valid_loader, num_epochs=100, learning_rate=learning_rate, patience=10)
    test_loader = DataLoader(TensorDataset(torch.tensor(X_test.values, dtype=torch.float32), torch.tensor(y_test.values, dtype=torch.float32)), batch_size=batch_size, shuffle=False)
    performance_scores = evaluate_model(model, test_loader)
    return performance_scores['AUC-ROC']

# Main function
def main():
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/06_santander_customer/split_train.csv'
    eval_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/06_santander_customer/split_eval.csv'

    # Load data
    train_df, eval_df = load_data(train_path, eval_path)

    # Preprocess data
    train_df, eval_df = preprocess_data(train_df, eval_df)

    # Augment data
    train_df = augment_data(train_df)

    # Feature engineering
    train_df, eval_df = feature_engineering(train_df, eval_df)

    # Split data
    X_train, X_test, y_train, y_test = train_test_split(train_df.drop(columns=['target']), train_df['target'], test_size=0.1, random_state=42)
    X_train, X_valid, y_train, y_valid = train_test_split(X_train, y_train, test_size=0.2222, random_state=42)  # 70% train, 20% valid, 10% test

    # Hyperparameter tuning
    study = optuna.create_study(direction='maximize')
    study.optimize(lambda trial: objective(trial, train_df, X_train, X_valid, y_train, y_valid, X_test, y_test), n_trials=50)

    # Train the final model with the best hyperparameters
    best_params = study.best_params
    model = DNN(train_df.shape[1] - 1, best_params['hidden_layers'], best_params['hidden_size'], best_params['dropout_rate'])
    train_loader = DataLoader(TensorDataset(torch.tensor(X_train.values, dtype=torch.float32), torch.tensor(y_train.values, dtype=torch.float32)), batch_size=best_params['batch_size'], shuffle=True)
    valid_loader = DataLoader(TensorDataset(torch.tensor(X_valid.values, dtype=torch.float32), torch.tensor(y_valid.values, dtype=torch.float32)), batch_size=best_params['batch_size'], shuffle=False)
    model = train_model(model, train_loader, valid_loader, num_epochs=100, learning_rate=best_params['learning_rate'], patience=10)

    # Evaluate the final model
    test_loader = DataLoader(TensorDataset(torch.tensor(X_test.values, dtype=torch.float32), torch.tensor(y_test.values, dtype=torch.float32)), batch_size=best_params['batch_size'], shuffle=False)
    model_performance = evaluate_model(model, test_loader)

    # Save the model
    model_save_path = './agent_workspace/trained_models/santander_customer_dnn.pth'
    os.makedirs(os.path.dirname(model_save_path), exist_ok=True)
    torch.save(model.state_dict(), model_save_path)

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
