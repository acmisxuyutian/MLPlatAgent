######代码######

import os
import random
import numpy as np
import pandas as pd
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, TensorDataset
from sklearn.impute import KNNImputer
from sklearn.preprocessing import StandardScaler
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import roc_auc_score
import optuna

# Set random seeds for reproducibility
random.seed(42)
np.random.seed(42)
torch.manual_seed(42)
torch.cuda.manual_seed(42)
torch.backends.cudnn.deterministic = True
torch.backends.cudnn.benchmark = False

# Data Retrieval
train_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/06_santander_customer/split_train.csv'
eval_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/06_santander_customer/split_eval.csv'

train_df = pd.read_csv(train_data_path)
eval_df = pd.read_csv(eval_data_path)

# Data Overview
print("Training Data Shape:", train_df.shape)
print("Evaluation Data Shape:", eval_df.shape)
print("Target Variable:", train_df['target'].unique())

# Handle Missing Values
imputer = KNNImputer(n_neighbors=5)
train_df = pd.DataFrame(imputer.fit_transform(train_df), columns=train_df.columns)
eval_df = pd.DataFrame(imputer.transform(eval_df), columns=eval_df.columns)

# Feature Scaling
scaler = StandardScaler()
train_df[train_df.columns.difference(['target'])] = scaler.fit_transform(train_df[train_df.columns.difference(['target'])])
eval_df[eval_df.columns.difference(['target'])] = scaler.transform(eval_df[eval_df.columns.difference(['target'])])

# Feature Selection
X_train = train_df.drop(columns=['target'])
y_train = train_df['target']

model = RandomForestClassifier()
model.fit(X_train, y_train)
feature_importances = model.feature_importances_
important_features = X_train.columns[feature_importances > 0.01]

X_train = X_train[important_features]
X_eval = eval_df[important_features]

# Model Architecture
class MLP(nn.Module):
    def __init__(self, input_size, hidden_layers, output_size):
        super(MLP, self).__init__()
        layers = []
        layers.append(nn.Linear(input_size, hidden_layers[0]))
        layers.append(nn.ReLU())
        for i in range(1, len(hidden_layers)):
            layers.append(nn.Linear(hidden_layers[i-1], hidden_layers[i]))
            layers.append(nn.ReLU())
        layers.append(nn.Linear(hidden_layers[-1], output_size))
        layers.append(nn.Sigmoid())
        self.model = nn.Sequential(*layers)

    def forward(self, x):
        return self.model(x)

input_size = X_train.shape[1]
hidden_layers = [128, 64, 32]
output_size = 1
model = MLP(input_size, hidden_layers, output_size)

# Training and Validation Split
X_train, X_val, y_train, y_val = train_test_split(X_train, y_train, test_size=0.2, random_state=42)

# Data Loaders
train_dataset = TensorDataset(torch.tensor(X_train.values, dtype=torch.float32), torch.tensor(y_train.values, dtype=torch.float32))
val_dataset = TensorDataset(torch.tensor(X_val.values, dtype=torch.float32), torch.tensor(y_val.values, dtype=torch.float32))
eval_dataset = TensorDataset(torch.tensor(X_eval.values, dtype=torch.float32), torch.tensor(eval_df['target'].values, dtype=torch.float32))

train_loader = DataLoader(train_dataset, batch_size=64, shuffle=True)
val_loader = DataLoader(val_dataset, batch_size=64, shuffle=False)
eval_loader = DataLoader(eval_dataset, batch_size=64, shuffle=False)

# Training Loop
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=0.001)

num_epochs = 30
best_auc = 0.0
for epoch in range(num_epochs):
    model.train()
    for inputs, labels in train_loader:
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels.unsqueeze(1))
        loss.backward()
        optimizer.step()

    model.eval()
    with torch.no_grad():
        val_predictions = []
        val_labels = []
        for inputs, labels in val_loader:
            outputs = model(inputs)
            val_predictions.extend(outputs.squeeze().numpy())
            val_labels.extend(labels.numpy())

        val_auc = roc_auc_score(val_labels, val_predictions)
        if val_auc > best_auc:
            best_auc = val_auc
            torch.save(model.state_dict(), 'best_model.pth')

    print(f'Epoch [{epoch+1}/{num_epochs}], Loss: {loss.item():.4f}, Validation AUC: {val_auc:.4f}')

# Model Evaluation
model.load_state_dict(torch.load('best_model.pth'))
model.eval()

with torch.no_grad():
    eval_predictions = []
    eval_labels = []
    for inputs, labels in eval_loader:
        outputs = model(inputs)
        eval_predictions.extend(outputs.squeeze().numpy())
        eval_labels.extend(labels.numpy())

eval_auc = roc_auc_score(eval_labels, eval_predictions)
print(f'Evaluation AUC: {eval_auc:.4f}')

# Hyperparameter Tuning
def objective(trial):
    hidden_layers = [trial.suggest_int('hidden_layer_1', 64, 256), trial.suggest_int('hidden_layer_2', 32, 128), trial.suggest_int('hidden_layer_3', 16, 64)]
    learning_rate = trial.suggest_float('learning_rate', 0.0001, 0.1, log=True)
    batch_size = trial.suggest_categorical('batch_size', [32, 64, 128])
    num_epochs = trial.suggest_int('num_epochs', 10, 30)

    model = MLP(input_size, hidden_layers, output_size)
    criterion = nn.BCELoss()
    optimizer = optim.Adam(model.parameters(), lr=learning_rate)

    train_loader = DataLoader(train_dataset, batch_size=batch_size, shuffle=True)
    val_loader = DataLoader(val_dataset, batch_size=batch_size, shuffle=False)

    best_auc = 0.0
    for epoch in range(num_epochs):
        model.train()
        for inputs, labels in train_loader:
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, labels.unsqueeze(1))
            loss.backward()
            optimizer.step()

        model.eval()
        with torch.no_grad():
            val_predictions = []
            val_labels = []
            for inputs, labels in val_loader:
                outputs = model(inputs)
                val_predictions.extend(outputs.squeeze().numpy())
                val_labels.extend(labels.numpy())

            val_auc = roc_auc_score(val_labels, val_predictions)
            if val_auc > best_auc:
                best_auc = val_auc

    return best_auc

study = optuna.create_study(direction='maximize')
study.optimize(objective, n_trials=50)

print('Best hyperparameters:', study.best_params)
print('Best AUC:', study.best_value)

# Final Model Training and Evaluation
best_params = study.best_params
model = MLP(input_size, [best_params['hidden_layer_1'], best_params['hidden_layer_2'], best_params['hidden_layer_3']], output_size)
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=best_params['learning_rate'])

train_loader = DataLoader(train_dataset, batch_size=best_params['batch_size'], shuffle=True)
val_loader = DataLoader(val_dataset, batch_size=best_params['batch_size'], shuffle=False)
eval_loader = DataLoader(eval_dataset, batch_size=best_params['batch_size'], shuffle=False)

best_auc_path = './agent_workspace/trained_models/best_model.pth'
best_auc = 0.0
for epoch in range(best_params['num_epochs']):
    model.train()
    for inputs, labels in train_loader:
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels.unsqueeze(1))
        loss.backward()
        optimizer.step()

    model.eval()
    with torch.no_grad():
        val_predictions = []
        val_labels = []
        for inputs, labels in val_loader:
            outputs = model(inputs)
            val_predictions.extend(outputs.squeeze().numpy())
            val_labels.extend(labels.numpy())

        val_auc = roc_auc_score(val_labels, val_predictions)
        if val_auc > best_auc:
            best_auc = val_auc
            torch.save(model.state_dict(), best_auc_path)

model.load_state_dict(torch.load(best_auc_path))
model.eval()

with torch.no_grad():
    eval_predictions = []
    eval_labels = []
    for inputs, labels in eval_loader:
        outputs = model(inputs)
        eval_predictions.extend(outputs.squeeze().numpy())
        eval_labels.extend(labels.numpy())

eval_auc = roc_auc_score(eval_labels, eval_predictions)
print(f'Final Evaluation AUC: {eval_auc:.4f}')

# Model Deployment
os.makedirs('./agent_workspace/trained_models', exist_ok=True)
torch.save(model.state_dict(), './agent_workspace/trained_models/final_model.pth')
import joblib
joblib.dump(scaler, './agent_workspace/trained_models/scaler.pkl')
joblib.dump(important_features, './agent_workspace/trained_models/important_features.pkl')
