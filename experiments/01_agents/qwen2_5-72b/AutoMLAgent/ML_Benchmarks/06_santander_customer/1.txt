import os
import random
import time
import json
import numpy as np
import pandas as pd
import torch
import torch.nn as nn
import torch.optim as optim
import torch.utils.data as data
from sklearn.metrics import accuracy_score, f1_score, roc_auc_score
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler
from sklearn.feature_selection import SelectKBest, mutual_info_classif
from imblearn.over_sampling import SMOTE

# Set random seeds for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)
torch.cuda.manual_seed(SEED)
torch.backends.cudnn.deterministic = True
torch.backends.cudnn.benchmark = False

# Data Retrieval
def load_data(train_path, eval_path):
    train_df = pd.read_csv(train_path)
    eval_df = pd.read_csv(eval_path)
    return train_df, eval_df

# Data Preprocessing
def preprocess_data(train_df, eval_df):
    # Handling Missing Values
    imputer = SimpleImputer(strategy='mean')
    train_df = pd.DataFrame(imputer.fit_transform(train_df), columns=train_df.columns)
    eval_df = pd.DataFrame(imputer.transform(eval_df), columns=eval_df.columns)

    # Feature Scaling
    scaler = StandardScaler()
    train_df[train_df.columns] = scaler.fit_transform(train_df[train_df.columns])
    eval_df[eval_df.columns] = scaler.transform(eval_df[eval_df.columns])

    # Categorical Encoding
    train_df = pd.get_dummies(train_df)
    eval_df = pd.get_dummies(eval_df)

    # Feature Engineering
    numerical_features = train_df.select_dtypes(include=[np.number]).columns
    new_features = []
    for i in range(len(numerical_features)):
        for j in range(i + 1, len(numerical_features)):
            new_feature = numerical_features[i] + '_' + numerical_features[j]
            train_df[new_feature] = train_df[numerical_features[i]] * train_df[numerical_features[j]]
            eval_df[new_feature] = eval_df[numerical_features[i]] * eval_df[numerical_features[j]]
            new_features.append(new_feature)

    # Feature Selection
    X_train = train_df.drop(columns=['target'])
    y_train = train_df['target']
    X_eval = eval_df.drop(columns=['target'])
    y_eval = eval_df['target']

    selector = SelectKBest(mutual_info_classif, k=10)
    X_train_selected = selector.fit_transform(X_train, y_train)
    X_eval_selected = selector.transform(X_eval)

    return X_train_selected, y_train, X_eval_selected, y_eval

# Data Augmentation
def augment_data(X_train, y_train):
    smote = SMOTE()
    X_train_augmented, y_train_augmented = smote.fit_resample(X_train, y_train)
    return X_train_augmented, y_train_augmented

# Model Definition
class MLP(nn.Module):
    def __init__(self, input_size, hidden_size, output_size, dropout_rate):
        super(MLP, self).__init__()
        self.fc1 = nn.Linear(input_size, hidden_size)
        self.relu = nn.ReLU()
        self.fc2 = nn.Linear(hidden_size, hidden_size // 2)
        self.fc3 = nn.Linear(hidden_size // 2, output_size)
        self.dropout = nn.Dropout(dropout_rate)
        self.sigmoid = nn.Sigmoid()

    def forward(self, x):
        x = self.fc1(x)
        x = self.relu(x)
        x = self.dropout(x)
        x = self.fc2(x)
        x = self.relu(x)
        x = self.dropout(x)
        x = self.fc3(x)
        x = self.sigmoid(x)
        return x

# Training the Model
def train_model(model, train_loader, criterion, optimizer, num_epochs=100, batch_size=64):
    for epoch in range(num_epochs):
        model.train()
        for inputs, labels in train_loader:
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, labels.unsqueeze(1))
            loss.backward()
            optimizer.step()
        print(f'Epoch [{epoch+1}/{num_epochs}], Loss: {loss.item():.4f}')
    return model

# Evaluation
def evaluate_model(model, eval_loader, y_eval):
    model.eval()
    y_pred = []
    with torch.no_grad():
        for inputs, _ in eval_loader:
            outputs = model(inputs)
            y_pred.extend(outputs.squeeze().numpy())
    y_pred = np.array(y_pred)
    auc = roc_auc_score(y_eval, y_pred)
    performance_scores = {
        'AUC': auc
    }
    return performance_scores

# Main function to orchestrate the pipeline
def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/06_santander_customer/split_train.csv'
    eval_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/06_santander_customer/split_eval.csv'
    train_df, eval_df = load_data(train_path, eval_path)

    # Step 2. Preprocess the data
    X_train, y_train, X_eval, y_eval = preprocess_data(train_df, eval_df)

    # Step 3. Data Augmentation
    X_train, y_train = augment_data(X_train, y_train)

    # Step 4. Define the model
    input_size = X_train.shape[1]
    hidden_size = 64
    output_size = 1
    dropout_rate = 0.2
    model = MLP(input_size, hidden_size, output_size, dropout_rate)

    # Step 5. Train the model
    criterion = nn.BCELoss()
    optimizer = optim.Adam(model.parameters(), lr=0.001)
    train_dataset = data.TensorDataset(torch.tensor(X_train, dtype=torch.float32), torch.tensor(y_train, dtype=torch.float32))
    train_loader = data.DataLoader(train_dataset, batch_size=64, shuffle=True)
    model = train_model(model, train_loader, criterion, optimizer)

    # Step 6. Evaluate the model
    eval_dataset = data.TensorDataset(torch.tensor(X_eval, dtype=torch.float32), torch.tensor(y_eval, dtype=torch.float32))
    eval_loader = data.DataLoader(eval_dataset, batch_size=64, shuffle=False)
    model_performance = evaluate_model(model, eval_loader, y_eval)

    # Step 7. Save the model
    model_save_path = './agent_workspace/trained_models/santander_customer_mlp.pth'
    os.makedirs(os.path.dirname(model_save_path), exist_ok=True)
    torch.save(model.state_dict(), model_save_path)

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
