import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import OneHotEncoder, StandardScaler, PolynomialFeatures
from sklearn.model_selection import train_test_split, KFold
from sklearn.metrics import mean_squared_log_error, mean_squared_error
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, TensorDataset
from sklearn.model_selection import RandomizedSearchCV

# Set random seeds for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)
if torch.cuda.is_available():
    torch.cuda.manual_seed(SEED)
    torch.cuda.manual_seed_all(SEED)
torch.backends.cudnn.deterministic = True
torch.backends.cudnn.benchmark = False

# Data Retrieval and Initial Setup
def load_data(train_path, eval_path):
    train_data = pd.read_csv(train_path)
    eval_data = pd.read_csv(eval_path)
    return train_data, eval_data

# Data Preprocessing
def preprocess_data(train_data, eval_data):
    # Identify numerical and categorical features
    numerical_features = train_data.select_dtypes(include=['number']).columns.tolist()
    categorical_features = train_data.select_dtypes(include=['object']).columns.tolist()

    # Impute missing values
    numerical_imputer = SimpleImputer(strategy='median')
    categorical_imputer = SimpleImputer(strategy='most_frequent')

    train_data[numerical_features] = numerical_imputer.fit_transform(train_data[numerical_features])
    train_data[categorical_features] = categorical_imputer.fit_transform(train_data[categorical_features])

    eval_data[numerical_features] = numerical_imputer.transform(eval_data[numerical_features])
    eval_data[categorical_features] = categorical_imputer.transform(eval_data[categorical_features])

    # Encode categorical variables
    encoder = OneHotEncoder(sparse_output=False, handle_unknown='ignore')
    encoded_train = encoder.fit_transform(train_data[categorical_features])
    encoded_eval = encoder.transform(eval_data[categorical_features])

    train_data = pd.concat([train_data[numerical_features], pd.DataFrame(encoded_train, columns=encoder.get_feature_names_out(categorical_features))], axis=1)
    eval_data = pd.concat([eval_data[numerical_features], pd.DataFrame(encoded_eval, columns=encoder.get_feature_names_out(categorical_features))], axis=1)

    # Feature scaling
    scaler = StandardScaler()
    train_data = pd.DataFrame(scaler.fit_transform(train_data), columns=train_data.columns.astype(str))
    eval_data = pd.DataFrame(scaler.transform(eval_data), columns=eval_data.columns.astype(str))

    return train_data, eval_data

# Feature Engineering
def feature_engineering(train_data, eval_data):
    # Generate polynomial features
    numerical_features = train_data.select_dtypes(include=['number']).columns.tolist()
    poly = PolynomialFeatures(degree=2, interaction_only=True, include_bias=False)
    train_data_poly = poly.fit_transform(train_data[numerical_features])
    eval_data_poly = poly.transform(eval_data[numerical_features])

    train_data = pd.concat([train_data, pd.DataFrame(train_data_poly, columns=poly.get_feature_names_out(numerical_features))], axis=1)
    eval_data = pd.concat([eval_data, pd.DataFrame(eval_data_poly, columns=poly.get_feature_names_out(numerical_features))], axis=1)

    return train_data, eval_data

# Data Augmentation
def data_augmentation(train_data, eval_data):
    # No specific data augmentation is required for this task
    return train_data, eval_data

# Extract and Understand the Underlying Characteristics of the Dataset
def explore_data(train_data):
    # Generate summary statistics
    summary_stats = train_data.describe()

    # Compute the correlation matrix
    correlation_matrix = train_data.corr()

    # Identify and handle outliers using the IQR method
    Q1 = train_data.quantile(0.25)
    Q3 = train_data.quantile(0.75)
    IQR = Q3 - Q1
    train_data = train_data[~((train_data < (Q1 - 1.5 * IQR)) | (train_data > (Q3 + 1.5 * IQR))).any(axis=1)]

    return summary_stats, correlation_matrix, train_data

# Model Selection and Training
class DNN(nn.Module):
    def __init__(self, input_dim, hidden_layers, dropout_rate):
        super(DNN, self).__init__()
        layers = []
        for i in range(len(hidden_layers)):
            if i == 0:
                layers.append(nn.Linear(input_dim, hidden_layers[i]))
            else:
                layers.append(nn.Linear(hidden_layers[i-1], hidden_layers[i]))
            layers.append(nn.ReLU())
            layers.append(nn.Dropout(dropout_rate))
        layers.append(nn.Linear(hidden_layers[-1], 1))
        self.model = nn.Sequential(*layers)

    def forward(self, x):
        return self.model(x)

def train_model(model, train_loader, valid_loader, criterion, optimizer, num_epochs=100, patience=10):
    model.train()
    best_valid_loss = float('inf')
    epochs_without_improvement = 0
    for epoch in range(num_epochs):
        train_loss = 0.0
        for batch in train_loader:
            inputs, targets = batch
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, targets)
            loss.backward()
            optimizer.step()
            train_loss += loss.item()

        train_loss /= len(train_loader)
        valid_loss = evaluate_model(model, valid_loader, criterion)

        print(f'Epoch {epoch+1}/{num_epochs}, Train Loss: {train_loss:.4f}, Valid Loss: {valid_loss:.4f}')

        if valid_loss < best_valid_loss:
            best_valid_loss = valid_loss
            epochs_without_improvement = 0
        else:
            epochs_without_improvement += 1
            if epochs_without_improvement >= patience:
                print(f'Early stopping after {patience} epochs without improvement')
                break

    return model

def evaluate_model(model, data_loader, criterion):
    model.eval()
    total_loss = 0.0
    with torch.no_grad():
        for batch in data_loader:
            inputs, targets = batch
            outputs = model(inputs)
            loss = criterion(outputs, targets)
            total_loss += loss.item()
    return total_loss / len(data_loader)

def hyperparameter_optimization(train_loader, valid_loader, input_dim):
    param_distributions = {
        'hidden_layers': [3, 4, 5],
        'neurons_per_layer': [128, 256, 512],
        'learning_rate': [0.001, 0.01, 0.1],
        'batch_size': [32, 64, 128],
        'dropout_rate': [0.1, 0.2, 0.3]
    }

    def objective(params):
        hidden_layers = [params['neurons_per_layer']] * params['hidden_layers']
        model = DNN(input_dim, hidden_layers, params['dropout_rate']).to(device)
        criterion = nn.MSELoss()
        optimizer = optim.Adam(model.parameters(), lr=params['learning_rate'])
        model = train_model(model, train_loader, valid_loader, criterion, optimizer, num_epochs=10, patience=3)
        valid_loss = evaluate_model(model, valid_loader, criterion)
        return -valid_loss

    search = RandomizedSearchCV(objective, param_distributions, n_iter=10, cv=3, random_state=SEED)
    search.fit(None, None)
    best_params = search.best_params_
    return best_params

# Main function to orchestrate the pipeline
def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/05_house_prices/split_train.csv'
    eval_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/05_house_prices/split_eval.csv'
    train_data, eval_data = load_data(train_path, eval_path)

    # Step 2. Create a train-valid-test split of the data
    train_data, valid_data = train_test_split(train_data, test_size=0.2, random_state=SEED)
    train_data, eval_data = preprocess_data(train_data, eval_data)
    valid_data, _ = preprocess_data(valid_data, valid_data)

    # Step 3. Run data preprocessing and feature engineering
    train_data, eval_data = feature_engineering(train_data, eval_data)
    valid_data, _ = feature_engineering(valid_data, valid_data)

    # Step 4. Explore the dataset
    summary_stats, correlation_matrix, train_data = explore_data(train_data)

    # Step 5. Prepare the data for training
    X_train = train_data.drop(columns=['SalePrice']).values
    y_train = np.log1p(train_data['SalePrice'].values)
    X_valid = valid_data.drop(columns=['SalePrice']).values
    y_valid = np.log1p(valid_data['SalePrice'].values)
    X_eval = eval_data.drop(columns=['SalePrice']).values
    y_eval = np.log1p(eval_data['SalePrice'].values)

    train_dataset = TensorDataset(torch.tensor(X_train, dtype=torch.float32), torch.tensor(y_train, dtype=torch.float32).view(-1, 1))
    valid_dataset = TensorDataset(torch.tensor(X_valid, dtype=torch.float32), torch.tensor(y_valid, dtype=torch.float32).view(-1, 1))
    eval_dataset = TensorDataset(torch.tensor(X_eval, dtype=torch.float32), torch.tensor(y_eval, dtype=torch.float32).view(-1, 1))

    train_loader = DataLoader(train_dataset, batch_size=64, shuffle=True)
    valid_loader = DataLoader(valid_dataset, batch_size=64, shuffle=False)
    eval_loader = DataLoader(eval_dataset, batch_size=64, shuffle=False)

    # Step 6. Define required model and hyperparameter optimization
    input_dim = X_train.shape[1]
    best_params = hyperparameter_optimization(train_loader, valid_loader, input_dim)
    hidden_layers = [best_params['neurons_per_layer']] * best_params['hidden_layers']
    model = DNN(input_dim, hidden_layers, best_params['dropout_rate']).to(device)
    criterion = nn.MSELoss()
    optimizer = optim.Adam(model.parameters(), lr=best_params['learning_rate'])

    # Step 7. Train the model
    model = train_model(model, train_loader, valid_loader, criterion, optimizer)

    # Step 8. Evaluate the model
    eval_loss = evaluate_model(model, eval_loader, criterion)
    rmse = np.sqrt(eval_loss)
    model_performance = {
        "RMSE": rmse
    }

    # Step 9. Save the model
    model_save_path = "./agent_workspace/trained_models/house_price_model.pth"
    os.makedirs(os.path.dirname(model_save_path), exist_ok=True)
    torch.save(model.state_dict(), model_save_path)

    return model_performance

if __name__ == "__main__":
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
