import os
import random
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import OneHotEncoder, StandardScaler
from sklearn.feature_selection import RFE
from sklearn.ensemble import RandomForestClassifier
from xgboost import XGBClassifier
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score
import joblib

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data Retrieval
def load_data(train_path, eval_path):
    train_df = pd.read_csv(train_path)
    eval_df = pd.read_csv(eval_path)
    return train_df, eval_df

# Data Preprocessing
def preprocess_data(train_df, eval_df):
    # Impute missing values in numerical features
    numerical_imputer = SimpleImputer(strategy='median')
    train_df[['Age', 'Fare']] = numerical_imputer.fit_transform(train_df[['Age', 'Fare']])
    eval_df[['Age', 'Fare']] = numerical_imputer.transform(eval_df[['Age', 'Fare']])

    # Impute missing values in categorical features
    categorical_imputer = SimpleImputer(strategy='most_frequent')
    train_df[['Embarked']] = categorical_imputer.fit_transform(train_df[['Embarked']])
    eval_df[['Embarked']] = categorical_imputer.transform(eval_df[['Embarked']])

    # One-hot encode categorical features
    encoder = OneHotEncoder(handle_unknown='ignore')
    categorical_features = ['Sex', 'Embarked']
    train_encoded = encoder.fit_transform(train_df[categorical_features])
    eval_encoded = encoder.transform(eval_df[categorical_features])

    train_df = train_df.drop(columns=categorical_features)
    eval_df = eval_df.drop(columns=categorical_features)

    train_df = pd.concat([train_df, pd.DataFrame(train_encoded.toarray(), columns=encoder.get_feature_names_out(categorical_features))], axis=1)
    eval_df = pd.concat([eval_df, pd.DataFrame(eval_encoded.toarray(), columns=encoder.get_feature_names_out(categorical_features))], axis=1)

    # Standardize numerical features
    scaler = StandardScaler()
    numerical_features = ['Age', 'Fare']
    train_df[numerical_features] = scaler.fit_transform(train_df[numerical_features])
    eval_df[numerical_features] = scaler.transform(eval_df[numerical_features])

    return train_df, eval_df

# Feature Engineering
def feature_engineering(train_df, eval_df):
    # Create 'FamilySize' feature
    train_df['FamilySize'] = train_df['SibSp'] + train_df['Parch'] + 1
    eval_df['FamilySize'] = eval_df['SibSp'] + eval_df['Parch'] + 1

    # Extract 'Title' from 'Name'
    train_df['Title'] = train_df['Name'].str.extract(' ([A-Za-z]+)\.')
    eval_df['Title'] = eval_df['Name'].str.extract(' ([A-Za-z]+)\.')

    # One-hot encode 'Title'
    title_encoder = OneHotEncoder(handle_unknown='ignore')
    train_title_encoded = title_encoder.fit_transform(train_df[['Title']])
    eval_title_encoded = title_encoder.transform(eval_df[['Title']])

    train_df = train_df.drop(columns=['Title'])
    eval_df = eval_df.drop(columns=['Title'])

    train_df = pd.concat([train_df, pd.DataFrame(train_title_encoded.toarray(), columns=title_encoder.get_feature_names_out(['Title']))], axis=1)
    eval_df = pd.concat([eval_df, pd.DataFrame(eval_title_encoded.toarray(), columns=title_encoder.get_feature_names_out(['Title']))], axis=1)

    # Feature Selection
    model = RandomForestClassifier()
    rfe = RFE(estimator=model, n_features_to_select=10)
    rfe.fit(train_df.drop(columns=['Survived']).select_dtypes(include=[np.number]), train_df['Survived'])

    selected_features = train_df.select_dtypes(include=[np.number]).columns[rfe.support_]
    train_df = train_df[selected_features.insert(0, 'Survived')]
    eval_df = eval_df[selected_features.insert(0, 'Survived')]

    return train_df, eval_df

# Model Training
def train_model(train_df):
    model = XGBClassifier(
        n_estimators=100,
        learning_rate=0.1,
        max_depth=5,
        subsample=0.7,
        colsample_bytree=0.7
    )
    model.fit(train_df.drop(columns=['Survived']), train_df['Survived'])
    return model

# Model Evaluation
def evaluate_model(model, eval_df):
    eval_predictions = model.predict(eval_df.drop(columns=['Survived']))

    accuracy = accuracy_score(eval_df['Survived'], eval_predictions)
    precision = precision_score(eval_df['Survived'], eval_predictions)
    recall = recall_score(eval_df['Survived'], eval_predictions)
    f1 = f1_score(eval_df['Survived'], eval_predictions)

    performance_scores = {
        'Accuracy': accuracy,
        'Precision': precision,
        'Recall': recall,
        'F1-Score': f1
    }
    return performance_scores

# Main Function to Orchestrate the Pipeline
def main():
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/04_titanic/split_train.csv'
    eval_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/04_titanic/split_eval.csv'

    # Step 1: Load the datasets
    train_df, eval_df = load_data(train_path, eval_path)

    # Step 2: Preprocess the datasets
    train_df, eval_df = preprocess_data(train_df, eval_df)

    # Step 3: Feature engineering
    train_df, eval_df = feature_engineering(train_df, eval_df)

    # Step 4: Train the model
    model = train_model(train_df)

    # Step 5: Evaluate the model
    model_performance = evaluate_model(model, eval_df)

    # Step 6: Save the model
    model_path = './agent_workspace/trained_models/gradient_boosting_model.pkl'
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    joblib.dump(model, model_path)

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Evaluation Set:", model_performance)
