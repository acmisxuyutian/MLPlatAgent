######代码######

import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import OneHotEncoder, StandardScaler, OrdinalEncoder
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
from imblearn.over_sampling import SMOTE
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score, f1_score
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, TensorDataset

SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)

# Define paths
train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/04_titanic/split_train.csv'
eval_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/04_titanic/split_eval.csv'

# Load data
def load_data(train_path, eval_path):
    train_df = pd.read_csv(train_path)
    eval_df = pd.read_csv(eval_path)
    return train_df, eval_df

# Data preprocessing and feature engineering
def preprocess_data(train_df, eval_df):
    # Define numerical and categorical features
    numerical_features = ['Age', 'Fare']
    categorical_features = ['Sex', 'Embarked']

    # Impute missing values
    imputer_num = SimpleImputer(strategy='mean')
    imputer_cat = SimpleImputer(strategy='most_frequent')

    # Encode categorical variables
    encoder = OneHotEncoder(handle_unknown='ignore')

    # Scale numerical features
    scaler = StandardScaler()

    # Create preprocessing pipelines
    preprocessor = ColumnTransformer(
        transformers=[
            ('num', Pipeline(steps=[('imputer', imputer_num), ('scaler', scaler)]), numerical_features),
            ('cat', Pipeline(steps=[('imputer', imputer_cat), ('encoder', encoder)]), categorical_features)
        ])

    # Apply preprocessing to the training and evaluation datasets
    X_train = train_df.drop(columns=['Survived'])
    y_train = train_df['Survived']
    X_eval = eval_df.drop(columns=['Survived'])
    y_eval = eval_df['Survived']

    X_train_processed = preprocessor.fit_transform(X_train)
    X_eval_processed = preprocessor.transform(X_eval)

    return X_train_processed, y_train, X_eval_processed, y_eval, preprocessor

# Apply SMOTE
def apply_smote(X_train, y_train):
    smote = SMOTE()
    X_train_resampled, y_train_resampled = smote.fit_resample(X_train, y_train)
    return X_train_resampled, y_train_resampled

# Define the neural network architecture
class MLP(nn.Module):
    def __init__(self, input_size, hidden_size, output_size):
        super(MLP, self).__init__()
        self.fc1 = nn.Linear(input_size, hidden_size)
        self.relu = nn.ReLU()
        self.fc2 = nn.Linear(hidden_size, output_size)
        self.sigmoid = nn.Sigmoid()

    def forward(self, x):
        x = self.fc1(x)
        x = self.relu(x)
        x = self.fc2(x)
        x = self.sigmoid(x)
        return x

# Train the model
def train_model(X_train, y_train, input_size, hidden_size, output_size, num_epochs=30, batch_size=64, lr=0.01):
    # Convert data to PyTorch tensors
    X_train_tensor = torch.tensor(X_train, dtype=torch.float32)
    y_train_tensor = torch.tensor(y_train, dtype=torch.float32).unsqueeze(1)

    # Create data loaders
    train_dataset = TensorDataset(X_train_tensor, y_train_tensor)
    train_loader = DataLoader(train_dataset, batch_size=batch_size, shuffle=True)

    # Initialize the model, loss function, and optimizer
    model = MLP(input_size, hidden_size, output_size)
    criterion = nn.BCELoss()
    optimizer = optim.Adam(model.parameters(), lr=lr)

    # Train the model
    for epoch in range(num_epochs):
        for inputs, labels in train_loader:
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, labels)
            loss.backward()
            optimizer.step()
        print(f'Epoch [{epoch+1}/{num_epochs}], Loss: {loss.item():.4f}')

    return model

# Evaluate the model
def evaluate_model(model, X_eval, y_eval, batch_size=64):
    # Convert data to PyTorch tensors
    X_eval_tensor = torch.tensor(X_eval, dtype=torch.float32)
    y_eval_tensor = torch.tensor(y_eval, dtype=torch.float32).unsqueeze(1)

    # Create data loaders
    eval_dataset = TensorDataset(X_eval_tensor, y_eval_tensor)
    eval_loader = DataLoader(eval_dataset, batch_size=batch_size, shuffle=False)

    # Evaluate the model
    model.eval()
    y_true = []
    y_pred = []
    with torch.no_grad():
        for inputs, labels in eval_loader:
            outputs = model(inputs)
            predictions = (outputs > 0.5).float()
            y_true.extend(labels.numpy())
            y_pred.extend(predictions.numpy())

    y_true = np.array(y_true)
    y_pred = np.array(y_pred)

    performance_scores = {
        'ACC': accuracy_score(y_true, y_pred),
        'F1': f1_score(y_true, y_pred)
    }

    return performance_scores

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, and model evaluation
def main():
    # Step 1. Load the dataset
    train_df, eval_df = load_data(train_path, eval_path)

    # Step 2. Data preprocessing and feature engineering
    X_train_processed, y_train, X_eval_processed, y_eval, preprocessor = preprocess_data(train_df, eval_df)

    # Step 3. Data augmentation
    X_train_resampled, y_train_resampled = apply_smote(X_train_processed, y_train)

    # Step 4. Define the model
    input_size = X_train_resampled.shape[1]
    hidden_size = 128
    output_size = 1
    model = MLP(input_size, hidden_size, output_size)

    # Step 5. Train the model
    model = train_model(X_train_resampled, y_train_resampled, input_size, hidden_size, output_size)

    # Step 6. Evaluate the model
    model_performance = evaluate_model(model, X_eval_processed, y_eval)

    # Step 7. Save the trained model
    model_save_path = "./agent_workspace/trained_models/titanic_mlp_model.pth"
    os.makedirs(os.path.dirname(model_save_path), exist_ok=True)
    torch.save(model.state_dict(), model_save_path)

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
