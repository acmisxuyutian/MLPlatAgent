######代码######

import os
import random
import time
import json
import numpy as np
import pandas as pd
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, TensorDataset
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import OneHotEncoder, StandardScaler
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score

SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)

# Data Retrieval and Loading
def load_data(train_path, eval_path):
    train_df = pd.read_csv(train_path)
    eval_df = pd.read_csv(eval_path)
    return train_df, eval_df

# Data Preprocessing
def preprocess_data(train_df, eval_df):
    numerical_features = ['Age', 'Fare']
    categorical_features = ['Pclass', 'Sex', 'Embarked']

    numerical_transformer = Pipeline(steps=[
        ('imputer', SimpleImputer(strategy='median')),
        ('scaler', StandardScaler())
    ])

    categorical_transformer = Pipeline(steps=[
        ('imputer', SimpleImputer(strategy='most_frequent')),
        ('onehot', OneHotEncoder(handle_unknown='ignore'))
    ])

    preprocessor = ColumnTransformer(
        transformers=[
            ('num', numerical_transformer, numerical_features),
            ('cat', categorical_transformer, categorical_features)
        ])

    X_train = train_df.drop(columns=['Survived'])
    y_train = train_df['Survived']
    X_eval = eval_df.drop(columns=['Survived'])
    y_eval = eval_df['Survived']

    X_train_processed = preprocessor.fit_transform(X_train)
    X_eval_processed = preprocessor.transform(X_eval)

    return X_train_processed, y_train, X_eval_processed, y_eval, preprocessor

# Exploratory Data Analysis (EDA)
def exploratory_data_analysis(train_df):
    print("Survival Distribution:")
    print(train_df['Survived'].value_counts(normalize=True))

    print("\nCorrelation Matrix:")
    print(train_df.select_dtypes(include=[np.number]).corr())

    print("\nNumerical Feature Summary:")
    print(train_df[['Age', 'Fare']].describe())

    print("\nCategorical Feature Counts:")
    for feature in ['Pclass', 'Sex', 'Embarked']:
        print(f"\n{feature} Counts:")
        print(train_df[feature].value_counts())

    print("\nMissing Values:")
    print(train_df.isnull().sum())

# Define the FNN model
class FNN(nn.Module):
    def __init__(self, input_size, hidden_size, output_size, dropout_rate):
        super(FNN, self).__init__()
        self.fc1 = nn.Linear(input_size, hidden_size)
        self.fc2 = nn.Linear(hidden_size, hidden_size)
        self.fc3 = nn.Linear(hidden_size, hidden_size)
        self.fc4 = nn.Linear(hidden_size, output_size)
        self.relu = nn.ReLU()
        self.dropout = nn.Dropout(dropout_rate)
        self.sigmoid = nn.Sigmoid()

    def forward(self, x):
        x = self.relu(self.fc1(x))
        x = self.dropout(x)
        x = self.relu(self.fc2(x))
        x = self.dropout(x)
        x = self.relu(self.fc3(x))
        x = self.dropout(x)
        x = self.sigmoid(self.fc4(x))
        return x

# Train the model
def train_model(model, train_loader, eval_loader, criterion, optimizer, num_epochs):
    for epoch in range(num_epochs):
        model.train()
        for batch_X, batch_y in train_loader:
            optimizer.zero_grad()
            outputs = model(batch_X)
            loss = criterion(outputs, batch_y)
            loss.backward()
            optimizer.step()

        # Evaluation
        model.eval()
        with torch.no_grad():
            eval_outputs = model(eval_loader.dataset.tensors[0])
            eval_loss = criterion(eval_outputs, eval_loader.dataset.tensors[1])
            eval_predictions = (eval_outputs > 0.5).float()
            accuracy = (eval_predictions == eval_loader.dataset.tensors[1]).float().mean().item()

        print(f'Epoch [{epoch+1}/{num_epochs}], Loss: {loss.item():.4f}, Eval Loss: {eval_loss.item():.4f}, Accuracy: {accuracy:.4f}')

# Evaluate the model
def evaluate_model(model, eval_loader):
    model.eval()
    y_true = []
    y_pred = []
    with torch.no_grad():
        for batch_X, batch_y in eval_loader:
            outputs = model(batch_X)
            predictions = (outputs > 0.5).float()
            y_true.extend(batch_y.numpy())
            y_pred.extend(predictions.numpy())

    y_true = np.array(y_true)
    y_pred = np.array(y_pred)

    performance_scores = {
        'Accuracy': accuracy_score(y_true, y_pred),
        'Precision': precision_score(y_true, y_pred),
        'Recall': recall_score(y_true, y_pred),
        'F1 Score': f1_score(y_true, y_pred)
    }

    return performance_scores

# Main function to orchestrate the pipeline
def main():
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/04_titanic/split_train.csv'
    eval_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/04_titanic/split_eval.csv'

    # Load data
    train_df, eval_df = load_data(train_path, eval_path)

    # Exploratory Data Analysis
    exploratory_data_analysis(train_df)

    # Preprocess data
    X_train_processed, y_train, X_eval_processed, y_eval, preprocessor = preprocess_data(train_df, eval_df)

    # Convert processed data to tensors
    X_train_tensor = torch.tensor(X_train_processed, dtype=torch.float32)
    y_train_tensor = torch.tensor(y_train.values, dtype=torch.float32).unsqueeze(1)
    X_eval_tensor = torch.tensor(X_eval_processed, dtype=torch.float32)
    y_eval_tensor = torch.tensor(y_eval.values, dtype=torch.float32).unsqueeze(1)

    # Create data loaders
    train_dataset = TensorDataset(X_train_tensor, y_train_tensor)
    train_loader = DataLoader(train_dataset, batch_size=32, shuffle=True)

    eval_dataset = TensorDataset(X_eval_tensor, y_eval_tensor)
    eval_loader = DataLoader(eval_dataset, batch_size=32, shuffle=False)

    # Initialize the model, loss function, and optimizer
    input_size = X_train_processed.shape[1]
    hidden_size = 100
    output_size = 1
    dropout_rate = 0.2
    learning_rate = 0.001
    l2_reg = 0.0001

    model = FNN(input_size, hidden_size, output_size, dropout_rate)
    criterion = nn.BCELoss()
    optimizer = optim.Adam(model.parameters(), lr=learning_rate, weight_decay=l2_reg)

    # Train the model
    num_epochs = 100
    train_model(model, train_loader, eval_loader, criterion, optimizer, num_epochs)

    # Evaluate the model
    model_performance = evaluate_model(model, eval_loader)
    print("Model Performance on Evaluation Set:", model_performance)

    # Save the trained model
    model_save_path = "./agent_workspace/trained_models/fnn_model.pth"
    os.makedirs(os.path.dirname(model_save_path), exist_ok=True)
    torch.save(model.state_dict(), model_save_path)

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
