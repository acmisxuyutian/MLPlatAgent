######代码######

import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import mean_squared_log_error
import torch
import torch.nn as nn
import torch.optim as optim

# Set the random seed for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)

# Data Retrieval and Initial Setup
def load_data():
    train_data = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/08_santander_value/split_train.csv')
    eval_data = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/ml_benchmark/08_santander_value/split_eval.csv')
    return train_data, eval_data

# Data Preprocessing
def preprocess_data(train_data, eval_data):
    # Handle Missing Values
    numerical_features = train_data.select_dtypes(include=['number']).columns
    categorical_features = train_data.select_dtypes(include=['object']).columns

    if not numerical_features.empty:
        numerical_imputer = SimpleImputer(strategy='median')
        train_data[numerical_features] = numerical_imputer.fit_transform(train_data[numerical_features])
        eval_data[numerical_features] = numerical_imputer.transform(eval_data[numerical_features])

    if not categorical_features.empty:
        categorical_imputer = SimpleImputer(strategy='most_frequent')
        train_data[categorical_features] = categorical_imputer.fit_transform(train_data[categorical_features])
        eval_data[categorical_features] = categorical_imputer.transform(eval_data[categorical_features])

    # Handle Outliers
    def handle_outliers(df, columns):
        for col in columns:
            Q1 = df[col].quantile(0.25)
            Q3 = df[col].quantile(0.75)
            IQR = Q3 - Q1
            lower_bound = Q1 - 1.5 * IQR
            upper_bound = Q3 + 1.5 * IQR
            df[col] = df[col].clip(lower_bound, upper_bound)
        return df

    if not numerical_features.empty:
        train_data = handle_outliers(train_data, numerical_features)
        eval_data = handle_outliers(eval_data, numerical_features)

    # Encode Categorical Variables
    if not categorical_features.empty:
        train_data = pd.get_dummies(train_data, columns=categorical_features)
        eval_data = pd.get_dummies(eval_data, columns=categorical_features)

    # Standardize Numerical Features
    if not numerical_features.empty:
        scaler = StandardScaler()
        train_data[numerical_features] = scaler.fit_transform(train_data[numerical_features])
        eval_data[numerical_features] = scaler.transform(eval_data[numerical_features])

    # Split Data
    X_train = train_data.drop(columns=['target'])
    y_train = train_data['target']

    X_eval = eval_data.drop(columns=['target'])
    y_eval = eval_data['target']

    return X_train, y_train, X_eval, y_eval

# Model Development and Training
class MLP(nn.Module):
    def __init__(self, input_size, hidden_sizes, output_size):
        super(MLP, self).__init__()
        layers = []
        layers.append(nn.Linear(input_size, hidden_sizes[0]))
        layers.append(nn.ReLU())
        for i in range(1, len(hidden_sizes)):
            layers.append(nn.Linear(hidden_sizes[i-1], hidden_sizes[i]))
            layers.append(nn.ReLU())
        layers.append(nn.Linear(hidden_sizes[-1], output_size))
        self.model = nn.Sequential(*layers)

    def forward(self, x):
        return self.model(x)

def train_model(model, X_train, y_train):
    criterion = nn.MSELoss()
    optimizer = optim.Adam(model.parameters(), lr=0.001)

    X_train_tensor = torch.tensor(X_train.values, dtype=torch.float32)
    y_train_tensor = torch.tensor(y_train.values, dtype=torch.float32).view(-1, 1)

    num_epochs = 100
    batch_size = 64
    for epoch in range(num_epochs):
        model.train()
        for i in range(0, len(X_train_tensor), batch_size):
            X_batch = X_train_tensor[i:i+batch_size]
            y_batch = y_train_tensor[i:i+batch_size]

            optimizer.zero_grad()
            outputs = model(X_batch)
            loss = criterion(outputs, y_batch)
            loss.backward()
            optimizer.step()

        if (epoch + 1) % 10 == 0:
            print(f'Epoch [{epoch+1}/{num_epochs}], Loss: {loss.item():.4f}')

    return model

def evaluate_model(model, X_eval, y_eval):
    model.eval()
    with torch.no_grad():
        X_eval_tensor = torch.tensor(X_eval.values, dtype=torch.float32)
        y_eval_tensor = torch.tensor(y_eval.values, dtype=torch.float32).view(-1, 1)

        y_pred = model(X_eval_tensor)
        y_pred = y_pred.numpy().flatten()
        y_eval = y_eval_tensor.numpy().flatten()

        rmsle = np.sqrt(mean_squared_log_error(y_eval, y_pred))
        print(f'RMSLE on evaluation set: {rmsle:.4f}')
        return {'RMSLE': rmsle}

def save_model(model, model_save_path):
    torch.save(model.state_dict(), model_save_path)

# Main Function
def main():
    # Step 1: Load the dataset
    train_data, eval_data = load_data()

    # Step 2: Preprocess the data
    X_train, y_train, X_eval, y_eval = preprocess_data(train_data, eval_data)

    # Step 3: Define the MLP model
    input_size = X_train.shape[1]
    hidden_sizes = [128, 64, 32]
    output_size = 1
    model = MLP(input_size, hidden_sizes, output_size)

    # Step 4: Train the model
    model = train_model(model, X_train, y_train)

    # Step 5: Evaluate the model
    model_performance = evaluate_model(model, X_eval, y_eval)

    # Step 6: Save the model
    model_save_path = './agent_workspace/trained_models/santander_value_mlp_model.pth'
    os.makedirs(os.path.dirname(model_save_path), exist_ok=True)
    save_model(model, model_save_path)

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Evaluation Set:", model_performance)
