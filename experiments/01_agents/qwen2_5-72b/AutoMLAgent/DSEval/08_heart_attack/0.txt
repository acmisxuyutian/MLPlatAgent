import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.preprocessing import LabelEncoder, StandardScaler
from sklearn.impute import SimpleImputer
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import GridSearchCV
from sklearn.metrics import roc_auc_score
import joblib

# Set the random seed for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data Retrieval and Collection
def load_datasets():
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/08_heart_attack/train.csv'
    test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/08_heart_attack/test.csv'
    
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    
    return train_df, test_df

# Data Preprocessing
def preprocess_data(train_df, test_df):
    # Convert categorical data
    label_encoder = LabelEncoder()
    train_df['Diet'] = label_encoder.fit_transform(train_df['Diet'])
    test_df['Diet'] = label_encoder.transform(test_df['Diet'])

    # Handle missing values
    imputer = SimpleImputer(strategy='mean')
    train_df_numerical = train_df.select_dtypes(include=[np.number])
    test_df_numerical = test_df.select_dtypes(include=[np.number])
    train_df_numerical = pd.DataFrame(imputer.fit_transform(train_df_numerical), columns=train_df_numerical.columns)
    test_df_numerical = pd.DataFrame(imputer.transform(test_df_numerical), columns=test_df_numerical.columns)

    # Feature scaling
    scaler = StandardScaler()
    train_df_scaled = pd.DataFrame(scaler.fit_transform(train_df_numerical), columns=train_df_numerical.columns)
    test_df_scaled = pd.DataFrame(scaler.transform(test_df_numerical), columns=test_df_numerical.columns)
    
    return train_df_scaled, test_df_scaled

# Model Initialization and Training
def train_model(train_df_scaled):
    # Ensure the target variable is binary
    if not np.issubdtype(train_df_scaled['Heart Attack Risk'].dtype, np.integer):
        raise ValueError("Target variable 'Heart Attack Risk' must be an integer (binary classification).")

    # Initialize the model
    rf = RandomForestClassifier(random_state=SEED)

    # Define the parameter grid
    param_grid = {
        'n_estimators': [50, 100, 200],
        'max_depth': [None, 10, 20, 30],
        'min_samples_split': [2, 5, 10],
        'min_samples_leaf': [1, 2, 4]
    }

    # Perform grid search with cross-validation
    grid_search = GridSearchCV(rf, param_grid, cv=5, scoring='roc_auc', n_jobs=-1, error_score='raise')
    grid_search.fit(train_df_scaled.drop('Heart Attack Risk', axis=1), train_df_scaled['Heart Attack Risk'])

    # Get the best model
    best_rf = grid_search.best_estimator_

    # Save the best model
    model_path = "./agent_workspace/trained_models/best_rf_model.pkl"
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    joblib.dump(best_rf, model_path)
    
    return best_rf

# Model Evaluation
def evaluate_model(best_rf, test_df_scaled):
    # Make predictions on the test set
    test_predictions = best_rf.predict_proba(test_df_scaled.drop('Heart Attack Risk', axis=1))[:, 1]

    # Evaluate the model
    roc_auc = roc_auc_score(test_df_scaled['Heart Attack Risk'], test_predictions)

    # Return the ROC AUC score
    return (roc_auc,)

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    """
    Main function to execute the tabular classification pipeline.
    """
    # Step 1. Retrieve or load the datasets
    train_df, test_df = load_datasets()

    # Step 2. Preprocess the data
    train_df_scaled, test_df_scaled = preprocess_data(train_df, test_df)

    # Step 3. Train the model
    best_rf = train_model(train_df_scaled)

    # Step 4. Evaluate the model
    model_performance = evaluate_model(best_rf, test_df_scaled)

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
