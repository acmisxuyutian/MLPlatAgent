import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import RandomizedSearchCV
from sklearn.metrics import roc_auc_score
import joblib

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data Retrieval and Loading
def load_data(train_path, test_path):
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    return train_df, test_df

# Data Preprocessing
def preprocess_data(train_df, test_df):
    # Convert 'Diet' column
    def convert_diet_column(df):
        diet_mapping = {'Healthy': 1, 'Average': 2, 'Unhealthy': 3}
        df['Diet'] = df['Diet'].map(diet_mapping)
        return df

    train_df = convert_diet_column(train_df)
    test_df = convert_diet_column(test_df)

    # Handle missing values
    numerical_imputer = SimpleImputer(strategy='mean')
    categorical_imputer = SimpleImputer(strategy='most_frequent')

    numerical_columns = train_df.select_dtypes(include=['float64', 'int64']).columns
    categorical_columns = train_df.select_dtypes(include=['object']).columns

    train_df[numerical_columns] = numerical_imputer.fit_transform(train_df[numerical_columns])
    train_df[categorical_columns] = categorical_imputer.fit_transform(train_df[categorical_columns])

    test_df[numerical_columns] = numerical_imputer.transform(test_df[numerical_columns])
    test_df[categorical_columns] = categorical_imputer.transform(test_df[categorical_columns])

    # Standardize numerical features
    scaler = StandardScaler()
    train_df[numerical_columns] = scaler.fit_transform(train_df[numerical_columns])
    test_df[numerical_columns] = scaler.transform(test_df[numerical_columns])

    # Encode categorical variables
    train_df = pd.get_dummies(train_df, drop_first=True)
    test_df = pd.get_dummies(test_df, drop_first=True)

    # Ensure feature names match
    train_columns = train_df.drop(columns=['Heart Attack Risk']).columns
    test_df = test_df[train_columns]

    return train_df, test_df

# Model Training and Hyperparameter Tuning
def train_model(train_df):
    model = RandomForestClassifier(random_state=42)

    param_dist = {
        'n_estimators': [50, 100, 200],
        'max_depth': [None, 10, 20, 30],
        'min_samples_split': [2, 5, 10],
        'min_samples_leaf': [1, 2, 4],
        'max_features': ['sqrt', 'log2']
    }

    random_search = RandomizedSearchCV(model, param_distributions=param_dist, n_iter=10, scoring='roc_auc', cv=5, n_jobs=-1, random_state=42)
    random_search.fit(train_df.drop(columns=['Heart Attack Risk']), train_df['Heart Attack Risk'])

    best_model = random_search.best_estimator_
    return best_model

# Model Evaluation
def evaluate_model(model, test_df):
    y_pred = model.predict_proba(test_df.drop(columns=['Heart Attack Risk']))[:, 1]
    roc_auc = roc_auc_score(test_df['Heart Attack Risk'], y_pred)
    return roc_auc

# Main function to orchestrate the pipeline
def main():
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/08_heart_attack/train.csv'
    test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/08_heart_attack/test.csv'

    # Step 1: Load the datasets
    train_df, test_df = load_data(train_path, test_path)

    # Step 2: Preprocess the datasets
    train_df, test_df = preprocess_data(train_df, test_df)

    # Step 3: Train the model
    model = train_model(train_df)

    # Step 4: Evaluate the model
    roc_auc = evaluate_model(model, test_df)
    print(f'ROC AUC Score: {roc_auc}')

    # Step 5: Save the model and evaluation results
    model_path = './agent_workspace/trained_models/heart_attack_risk_model.pkl'
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    joblib.dump(model, model_path)

    results_path = './agent_workspace/trained_models/heart_attack_risk_evaluation.txt'
    with open(results_path, 'w') as f:
        f.write(f'ROC AUC Score: {roc_auc}\n')

    return roc_auc

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
