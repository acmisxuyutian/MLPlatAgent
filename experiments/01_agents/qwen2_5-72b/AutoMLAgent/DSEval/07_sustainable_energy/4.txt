import os
import pandas as pd
import numpy as np
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error
import joblib

# Data paths
train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/07_sustainable_energy/train.csv'
test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/07_sustainable_energy/test.csv'
model_path = './agent_workspace/trained_models/linear_regression_model.pkl'

# Step 1: Load the datasets
def load_data(train_path, test_path):
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    
    # Check the first few rows and column names
    print("Training Data:")
    print(train_df.head())
    print(train_df.columns)
    print("Test Data:")
    print(test_df.head())
    print(test_df.columns)
    
    return train_df, test_df

# Step 2: Preprocess the datasets
def preprocess_data(train_df, test_df):
    # Exclude 'Entity' and 'Year' columns
    train_features = train_df.drop(columns=['Entity', 'Renewable energy share in the total final energy consumption (%)'])
    train_target = train_df['Renewable energy share in the total final energy consumption (%)']
    test_features = test_df.drop(columns=['Entity'])

    # Handle missing values
    train_features = train_features.apply(pd.to_numeric, errors='coerce')
    test_features = test_features.apply(pd.to_numeric, errors='coerce')

    train_features.fillna(train_features.mean(), inplace=True)
    test_features.fillna(test_features.mean(), inplace=True)

    # Ensure the feature names match
    test_features = test_features[train_features.columns]

    # Normalize/Standardize features
    scaler = StandardScaler()
    train_features = scaler.fit_transform(train_features)
    test_features = scaler.transform(test_features)

    # Ensure no NaN values in the target variable
    train_target = train_target.fillna(train_target.mean())

    return train_features, train_target, test_features

# Step 3: Train the model
def train_model(train_features, train_target):
    model = LinearRegression()
    model.fit(train_features, train_target)
    return model

# Step 4: Make predictions
def make_predictions(model, test_features):
    test_predictions = model.predict(test_features)
    return test_predictions

# Step 5: Evaluate the model
def evaluate_model(test_df, test_predictions):
    test_target = test_df['Renewable energy share in the total final energy consumption (%)']
    test_rmse = np.sqrt(mean_squared_error(test_target, test_predictions))
    return test_rmse

# Step 6: Save the model
def save_model(model, model_path):
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    joblib.dump(model, model_path)
    print(f"Model saved to {model_path}")

# Step 7: Load the model
def load_model(model_path):
    loaded_model = joblib.load(model_path)
    return loaded_model

# Main function to orchestrate the entire pipeline
def main():
    # Load the datasets
    train_df, test_df = load_data(train_path, test_path)

    # Preprocess the datasets
    train_features, train_target, test_features = preprocess_data(train_df, test_df)

    # Train the model
    model = train_model(train_features, train_target)

    # Make predictions
    test_predictions = make_predictions(model, test_features)

    # Evaluate the model
    test_rmse = evaluate_model(test_df, test_predictions)
    print(f'RMSE on the test set: {test_rmse}')

    # Save the model
    save_model(model, model_path)

    return test_rmse

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
