######代码######

import os
import random
import numpy as np
import pandas as pd
from sklearn.linear_model import LinearRegression
from sklearn.impute import SimpleImputer
from sklearn.metrics import mean_squared_error
from scipy import stats
from joblib import dump, load

# Set random seed for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Define the paths for the datasets and model
train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/07_sustainable_energy/train.csv'
test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/07_sustainable_energy/test.csv'
model_path = './agent_workspace/trained_models/linear_regression_model.joblib'

# Ensure the directory for saving the model exists
os.makedirs(os.path.dirname(model_path), exist_ok=True)

# Step 1: Load Datasets
def load_datasets(train_path, test_path):
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    return train_df, test_df

# Step 2: Preprocess Data
def preprocess_data(train_df, test_df):
    # Drop unnecessary columns
    columns_to_drop = ['Entity', 'Year']
    for column in columns_to_drop:
        if column in train_df.columns:
            train_df = train_df.drop(columns=[column])
        if column in test_df.columns:
            test_df = test_df.drop(columns=[column])

    # Separate features and target variable
    target_column = 'Renewable energy share in the total final energy consumption (%)'
    X_train = train_df.drop(columns=[target_column])
    y_train = train_df[target_column]
    X_test = test_df.drop(columns=[target_column])
    y_test = test_df[target_column]

    # Convert non-numeric data to numeric (handle commas in numbers)
    X_train = X_train.apply(lambda x: x.str.replace(',', '.').astype(float) if x.dtype == 'object' else x)
    X_test = X_test.apply(lambda x: x.str.replace(',', '.').astype(float) if x.dtype == 'object' else x)

    # Handle missing values in features
    imputer = SimpleImputer(strategy='mean')
    X_train = imputer.fit_transform(X_train)
    X_test = imputer.transform(X_test)

    # Convert back to DataFrame
    X_train = pd.DataFrame(X_train, columns=train_df.drop(columns=[target_column]).columns)
    X_test = pd.DataFrame(X_test, columns=test_df.drop(columns=[target_column]).columns)

    # Handle missing values in the target variable
    y_train = y_train.dropna()
    y_test = y_test.dropna()

    # Ensure the indices align after dropping NaNs
    X_train = X_train.loc[y_train.index]
    X_test = X_test.loc[y_test.index]

    # Identify and handle outliers
    z_scores = stats.zscore(X_train)
    outliers = (z_scores > 3).any(axis=1)
    X_train = X_train[~outliers]
    y_train = y_train[~outliers]

    return X_train, y_train, X_test, y_test

# Step 3: Train the Model
def train_model(X_train, y_train):
    model = LinearRegression()
    model.fit(X_train, y_train)
    return model

# Step 4: Evaluate the Model
def evaluate_model(model, X_test, y_test):
    y_pred = model.predict(X_test)
    rmse = np.sqrt(mean_squared_error(y_test, y_pred))
    performance_scores = {
        'RMSE': rmse
    }
    return performance_scores

# Step 5: Save the Model
def save_model(model, model_path):
    dump(model, model_path)

# Step 6: Load and Use the Model (for future reference)
def load_model(model_path):
    model = load(model_path)
    return model

# Main function to orchestrate the entire pipeline
def main():
    # Load datasets
    train_df, test_df = load_datasets(train_path, test_path)

    # Preprocess data
    X_train, y_train, X_test, y_test = preprocess_data(train_df, test_df)

    # Train the model
    model = train_model(X_train, y_train)

    # Evaluate the model
    performance_scores = evaluate_model(model, X_test, y_test)
    print("Model Performance on Test Set:", performance_scores)

    # Save the model
    save_model(model, model_path)

    return performance_scores

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
