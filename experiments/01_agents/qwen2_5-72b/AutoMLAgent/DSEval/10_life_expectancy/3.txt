import os
import pandas as pd
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LinearRegression
from sklearn.metrics import r2_score
import joblib

# Define paths
train_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/10_life_expectancy/train.csv'
test_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/10_life_expectancy/test.csv'
model_path = './agent_workspace/trained_models/linear_regression_model.pkl'
scaler_path = './agent_workspace/trained_models/standard_scaler.pkl'

# Ensure directories exist
os.makedirs(os.path.dirname(model_path), exist_ok=True)
os.makedirs(os.path.dirname(scaler_path), exist_ok=True)

# Load data
def load_data(train_path, test_path):
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    return train_df, test_df

# Preprocess data
def preprocess_data(train_df, test_df):
    # GDP Binning
    bins = [0, 1000, 10000, float('inf')]
    labels = ['Low GDP', 'Medium GDP', 'High GDP']
    train_df['GDP_bin'] = pd.cut(train_df['GDP'], bins=bins, labels=labels, include_lowest=True)
    test_df['GDP_bin'] = pd.cut(test_df['GDP'], bins=bins, labels=labels, include_lowest=True)
    
    # Handling Missing Values
    train_df.fillna(train_df.select_dtypes(include=['number']).mean(), inplace=True)
    test_df.fillna(test_df.select_dtypes(include=['number']).mean(), inplace=True)
    
    # Feature Scaling
    scaler = StandardScaler()
    continuous_features = ['GDP']  # Ensure 'Life expectancy' is not included in continuous features
    train_df[continuous_features] = scaler.fit_transform(train_df[continuous_features])
    test_df[continuous_features] = scaler.transform(test_df[continuous_features])
    
    return train_df, test_df, scaler

# Prepare data for training and testing
def prepare_data(train_df, test_df):
    # Ensure 'Life expectancy' and 'GDP_bin' columns exist
    if 'Life expectancy' not in train_df.columns or 'GDP_bin' not in train_df.columns:
        raise KeyError("Required columns 'Life expectancy' or 'GDP_bin' not found in the training dataset.")
    if 'Life expectancy' not in test_df.columns or 'GDP_bin' not in test_df.columns:
        raise KeyError("Required columns 'Life expectancy' or 'GDP_bin' not found in the test dataset.")
    
    X_train = train_df.drop(columns=['Life expectancy', 'GDP_bin'])
    y_train = train_df['Life expectancy']
    
    X_test = test_df.drop(columns=['Life expectancy', 'GDP_bin'])
    y_test = test_df['Life expectancy']
    
    return X_train, y_train, X_test, y_test

# Train the model
def train_model(X_train, y_train):
    model = LinearRegression()
    model.fit(X_train, y_train)
    return model

# Evaluate the model
def evaluate_model(model, X_test, y_test):
    y_pred = model.predict(X_test)
    r2 = r2_score(y_test, y_pred)
    print(f'R-squared score: {r2}')
    return r2

# Main function to orchestrate the pipeline
def main():
    # Load data
    train_df, test_df = load_data(train_data_path, test_data_path)
    
    # Preprocess data
    train_df, test_df, scaler = preprocess_data(train_df, test_df)
    
    # Prepare data
    X_train, y_train, X_test, y_test = prepare_data(train_df, test_df)
    
    # Train model
    model = train_model(X_train, y_train)
    
    # Evaluate model
    r2 = evaluate_model(model, X_test, y_test)
    
    # Save model and scaler
    joblib.dump(model, model_path)
    joblib.dump(scaler, scaler_path)
    
    return r2

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
