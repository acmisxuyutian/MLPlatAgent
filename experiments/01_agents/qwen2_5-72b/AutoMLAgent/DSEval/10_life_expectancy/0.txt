import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LinearRegression
from sklearn.metrics import r2_score, mean_absolute_error, mean_squared_error
import joblib
import logging

# Set the random seed for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Define the paths for the datasets
train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/10_life_expectancy/train.csv'
test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/10_life_expectancy/test.csv'

# Define the paths for saving the model and scaler
model_path = './agent_workspace/trained_models/life_expectancy_model.pkl'
scaler_path = './agent_workspace/trained_models/scaler.pkl'

# Ensure the directory for saving the model exists
os.makedirs(os.path.dirname(model_path), exist_ok=True)
os.makedirs(os.path.dirname(scaler_path), exist_ok=True)

# Data Retrieval and Loading
def load_data(train_path, test_path):
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    return train_df, test_df

# Data Preprocessing
def preprocess_data(train_df, test_df):
    # Handle missing values
    numerical_features = train_df.select_dtypes(include=['float64', 'int64']).columns
    numerical_imputer = SimpleImputer(strategy='mean')
    train_df[numerical_features] = numerical_imputer.fit_transform(train_df[numerical_features])
    test_df[numerical_features] = numerical_imputer.transform(test_df[numerical_features])

    categorical_features = train_df.select_dtypes(include=['object']).columns
    if not categorical_features.empty:
        categorical_imputer = SimpleImputer(strategy='most_frequent')
        train_df[categorical_features] = categorical_imputer.fit_transform(train_df[categorical_features])
        test_df[categorical_features] = categorical_imputer.transform(test_df[categorical_features])

    # Feature Engineering
    def categorize_gdp(gdp):
        if gdp <= 1000:
            return 'Low GDP'
        elif 1000 < gdp <= 10000:
            return 'Medium GDP'
        else:
            return 'High GDP'

    train_df['GDP_bins'] = train_df['GDP'].apply(categorize_gdp)
    test_df['GDP_bins'] = test_df['GDP'].apply(categorize_gdp)

    # Feature Scaling
    numerical_features = train_df.select_dtypes(include=['float64', 'int64']).columns
    scaler = StandardScaler()
    train_df[numerical_features] = scaler.fit_transform(train_df[numerical_features])
    test_df[numerical_features] = scaler.transform(test_df[numerical_features])

    # Encode Categorical Variables
    train_df = pd.get_dummies(train_df, columns=['GDP_bins'])
    test_df = pd.get_dummies(test_df, columns=['GDP_bins'])

    # Ensure column consistency between train and test datasets
    if not set(train_df.columns) == set(test_df.columns):
        # Align columns
        test_df = test_df.reindex(columns=train_df.columns, fill_value=0)

    return train_df, test_df, scaler

# Model Training
def train_model(X_train, y_train):
    model = LinearRegression()
    model.fit(X_train, y_train)
    return model

# Model Evaluation
def evaluate_model(model, X_test, y_test):
    y_pred = model.predict(X_test)
    performance_scores = {
        'R2': r2_score(y_test, y_pred),
        'MAE': mean_absolute_error(y_test, y_pred),
        'MSE': mean_squared_error(y_test, y_pred),
        'RMSE': mean_squared_error(y_test, y_pred, squared=False)
    }
    return performance_scores

# Main function to orchestrate the pipeline
def main():
    # Step 1: Load the datasets
    train_df, test_df = load_data(train_path, test_path)

    # Step 2: Preprocess the datasets
    train_df, test_df, scaler = preprocess_data(train_df, test_df)

    # Step 3: Split the data into features and target
    X_train = train_df.drop(columns=['Life expectancy'])
    y_train = train_df['Life expectancy']
    X_test = test_df.drop(columns=['Life expectancy'])
    y_test = test_df['Life expectancy']

    # Step 4: Train the model
    model = train_model(X_train, y_train)

    # Step 5: Evaluate the model
    performance_scores = evaluate_model(model, X_test, y_test)

    # Step 6: Save the model and scaler
    joblib.dump(model, model_path)
    joblib.dump(scaler, scaler_path)

    # Step 7: Set up logging
    logging.basicConfig(filename='./agent_workspace/logs/model_performance.log', level=logging.INFO)
    logging.info(f'Model Performance: {performance_scores}')

    return performance_scores

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
