import os
import random
import time
import json
import numpy as np
import pandas as pd
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, TensorDataset
from sklearn.metrics import accuracy_score, f1_score, roc_auc_score
from sklearn.model_selection import train_test_split

# Set random seed for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)

# Data Retrieval and Loading
def load_datasets(train_path, test_path):
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    assert train_df is not None, "Training dataset is not loaded correctly."
    assert test_df is not None, "Test dataset is not loaded correctly."
    return train_df, test_df

# Data Preprocessing
def preprocess_data(train_df, test_df):
    # Handle Missing Values
    numerical_features = train_df.select_dtypes(include=['float64', 'int64']).columns
    for feature in numerical_features:
        train_df[feature] = train_df[feature].fillna(train_df[feature].mean())
        test_df[feature] = test_df[feature].fillna(train_df[feature].mean())

    categorical_features = train_df.select_dtypes(include=['object']).columns
    for feature in categorical_features:
        train_df[feature] = train_df[feature].fillna(train_df[feature].mode()[0])
        test_df[feature] = test_df[feature].fillna(train_df[feature].mode()[0])

    # Remove Duplicates
    train_df.drop_duplicates(inplace=True)
    test_df.drop_duplicates(inplace=True)

    # Create 'BMI_Category' Feature
    def create_bmi_category(row):
        bmi = row['BMI']
        if bmi < 18.5:
            return 'Underweight'
        elif 18.5 <= bmi <= 24.9:
            return 'Normal weight'
        elif 25 <= bmi <= 29.9:
            return 'Overweight'
        else:
            return 'Obesity'

    train_df['BMI_Category'] = train_df.apply(create_bmi_category, axis=1)
    test_df['BMI_Category'] = test_df.apply(create_bmi_category, axis=1)

    # Convert Categorical Features to Numerical Using One-Hot Encoding
    train_df = pd.get_dummies(train_df, drop_first=True)
    test_df = pd.get_dummies(test_df, drop_first=True)

    return train_df, test_df

# Define the Neural Network Model
class FeedforwardNN(nn.Module):
    def __init__(self, input_size, hidden_size, output_size, dropout_rate):
        super(FeedforwardNN, self).__init__()
        self.fc1 = nn.Linear(input_size, hidden_size)
        self.relu = nn.ReLU()
        self.fc2 = nn.Linear(hidden_size, hidden_size)
        self.fc3 = nn.Linear(hidden_size, output_size)
        self.dropout = nn.Dropout(dropout_rate)

    def forward(self, x):
        x = self.fc1(x)
        x = self.relu(x)
        x = self.dropout(x)
        x = self.fc2(x)
        x = self.relu(x)
        x = self.dropout(x)
        x = self.fc3(x)
        return x

# Prepare the Data for Training
def prepare_data(train_df, valid_df, test_df):
    X_train = train_df.drop(columns=['Heart_Disease']).values
    y_train = train_df['Heart_Disease'].values
    X_valid = valid_df.drop(columns=['Heart_Disease']).values
    y_valid = valid_df['Heart_Disease'].values
    X_test = test_df.drop(columns=['Heart_Disease']).values
    y_test = test_df['Heart_Disease'].values

    # Ensure all data is of type float32
    X_train = X_train.astype(np.float32)
    y_train = y_train.astype(np.float32)
    X_valid = X_valid.astype(np.float32)
    y_valid = y_valid.astype(np.float32)
    X_test = X_test.astype(np.float32)
    y_test = y_test.astype(np.float32)

    X_train = torch.tensor(X_train, dtype=torch.float32)
    y_train = torch.tensor(y_train, dtype=torch.float32).view(-1, 1)
    X_valid = torch.tensor(X_valid, dtype=torch.float32)
    y_valid = torch.tensor(y_valid, dtype=torch.float32).view(-1, 1)
    X_test = torch.tensor(X_test, dtype=torch.float32)
    y_test = torch.tensor(y_test, dtype=torch.float32).view(-1, 1)

    train_dataset = TensorDataset(X_train, y_train)
    valid_dataset = TensorDataset(X_valid, y_valid)
    test_dataset = TensorDataset(X_test, y_test)

    train_loader = DataLoader(train_dataset, batch_size=64, shuffle=True)
    valid_loader = DataLoader(valid_dataset, batch_size=64, shuffle=False)
    test_loader = DataLoader(test_dataset, batch_size=64, shuffle=False)

    return train_loader, valid_loader, test_loader

# Train the Model
def train_model(model, train_loader, valid_loader, criterion, optimizer, num_epochs=100):
    for epoch in range(num_epochs):
        model.train()
        for inputs, targets in train_loader:
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, targets)
            loss.backward()
            optimizer.step()

        if (epoch + 1) % 10 == 0:
            print(f'Epoch [{epoch+1}/{num_epochs}], Loss: {loss.item():.4f}')

        # Validation
        model.eval()
        with torch.no_grad():
            y_pred = []
            y_true = []
            for inputs, targets in valid_loader:
                outputs = model(inputs)
                y_pred.extend(outputs.sigmoid().numpy())
                y_true.extend(targets.numpy())

            y_pred = np.array(y_pred)
            y_true = np.array(y_true)
            valid_auc_roc = roc_auc_score(y_true, y_pred)
            print(f'Epoch [{epoch+1}/{num_epochs}], Validation AUC-ROC: {valid_auc_roc:.4f}')

# Evaluate the Model
def evaluate_model(model, test_loader):
    model.eval()
    with torch.no_grad():
        y_pred = []
        y_true = []
        for inputs, targets in test_loader:
            outputs = model(inputs)
            y_pred.extend(outputs.sigmoid().numpy())
            y_true.extend(targets.numpy())

    y_pred = np.array(y_pred)
    y_true = np.array(y_true)

    auc_roc = roc_auc_score(y_true, y_pred)
    performance_scores = {
        'AUC-ROC': auc_roc,
        'Accuracy': accuracy_score(y_true, (y_pred > 0.5).astype(int)),
        'F1-Score': f1_score(y_true, (y_pred > 0.5).astype(int))
    }
    return performance_scores

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/02_cardiovascular_diseases/train.csv'
    test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/02_cardiovascular_diseases/test.csv'
    train_df, test_df = load_datasets(train_path, test_path)

    # Step 2. Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
    train_df, valid_df = train_test_split(train_df, test_size=0.2, random_state=SEED)

    # Step 3. With the split dataset, run data preprocessing and feature engineering (if applicable) using the "preprocess_data" function you defined
    train_df, valid_df = preprocess_data(train_df, valid_df)
    test_df, _ = preprocess_data(test_df, test_df)  # Preprocess test data separately

    # Step 4. Prepare the data for training
    train_loader, valid_loader, test_loader = prepare_data(train_df, valid_df, test_df)

    # Step 5. Define required model. You may retrieve model from available hub or library along with pretrained weights (if any).
    input_size = len(train_df.columns) - 1  # Exclude the target variable
    hidden_size = 128
    output_size = 1
    dropout_rate = 0.3

    model = FeedforwardNN(input_size, hidden_size, output_size, dropout_rate)

    # Step 6. Train the retrieved/loaded model using the defined "train_model" function
    criterion = nn.BCEWithLogitsLoss()
    optimizer = optim.Adam(model.parameters(), lr=0.001)
    train_model(model, train_loader, valid_loader, criterion, optimizer)

    # Step 7. Evaluate the trained model using the defined "evaluate_model" function
    model_performance = evaluate_model(model, test_loader)

    # Save the trained model
    model_save_path = './agent_workspace/trained_models/cardiovascular_diseases_model.pth'
    os.makedirs(os.path.dirname(model_save_path), exist_ok=True)
    torch.save(model.state_dict(), model_save_path)

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
