######代码######

import os
import pandas as pd
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error
import joblib
import numpy as np

# Set the random seed for reproducibility
SEED = 42
np.random.seed(SEED)

# Step 1: Data Retrieval and Loading
def load_data(train_path, test_path):
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    return train_df, test_df

# Step 2: Data Preprocessing
def preprocess_data(train_df, test_df):
    # Handling Missing Values
    # Identify numerical and categorical columns
    numerical_cols = train_df.select_dtypes(include=[np.number]).columns
    categorical_cols = train_df.select_dtypes(include=[object]).columns

    # Impute missing values for numerical columns
    train_df[numerical_cols] = train_df[numerical_cols].fillna(train_df[numerical_cols].median())
    test_df[numerical_cols] = test_df[numerical_cols].fillna(test_df[numerical_cols].median())

    # Impute missing values for categorical columns
    train_df[categorical_cols] = train_df[categorical_cols].fillna(train_df[categorical_cols].mode().iloc[0])
    test_df[categorical_cols] = test_df[categorical_cols].fillna(test_df[categorical_cols].mode().iloc[0])

    # Outlier Handling (IQR Method)
    Q1 = train_df[numerical_cols].quantile(0.25)
    Q3 = train_df[numerical_cols].quantile(0.75)
    IQR = Q3 - Q1
    train_df = train_df[~((train_df[numerical_cols] < (Q1 - 1.5 * IQR)) | (train_df[numerical_cols] > (Q3 + 1.5 * IQR))).any(axis=1)]

    # Normalization/Standardization
    scaler = StandardScaler()
    train_df[numerical_cols] = scaler.fit_transform(train_df[numerical_cols])
    test_df[numerical_cols] = scaler.transform(test_df[numerical_cols])

    # Feature Encoding (One-hot Encoding)
    train_df = pd.get_dummies(train_df)
    test_df = pd.get_dummies(test_df)

    # Ensure the same columns in both datasets
    test_df = test_df.reindex(columns=train_df.columns, fill_value=0)

    # Separate features and target variable
    X_train = train_df.drop(columns=['price'])
    y_train = train_df['price']
    X_test = test_df.drop(columns=['price'])
    y_test = test_df['price']

    return X_train, y_train, X_test, y_test

# Step 3: Model Training
def train_model(X_train, y_train):
    model = LinearRegression()
    model.fit(X_train, y_train)
    return model

# Step 4: Model Evaluation
def evaluate_model(model, X_test, y_test):
    y_pred = model.predict(X_test)
    rmse = np.sqrt(mean_squared_error(y_test, y_pred))
    return rmse

# The main function to orchestrate the data loading, data preprocessing, model training, and model evaluation
def main():
    # Define the paths to the datasets
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/09_housing_price/train.csv'
    test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/09_housing_price/test.csv'

    # Step 1: Load the datasets
    train_df, test_df = load_data(train_path, test_path)

    # Step 2: Preprocess the data
    X_train, y_train, X_test, y_test = preprocess_data(train_df, test_df)

    # Step 3: Train the model
    model = train_model(X_train, y_train)

    # Save the model
    model_path = './agent_workspace/trained_models/linear_regression_model.pkl'
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    joblib.dump(model, model_path)

    # Step 4: Evaluate the model
    rmse = evaluate_model(model, X_test, y_test)
    print(f'Root Mean Squared Error (RMSE): {rmse}')

    return rmse

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
