######代码######

import os
import random
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.compose import ColumnTransformer
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error
import joblib
import logging

# Set the random seed for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data retrieval and loading
def load_data():
    train_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/09_housing_price/train.csv')
    test_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/09_housing_price/test.csv')
    return train_df, test_df

# Data preprocessing
def preprocess_data(train_df, test_df):
    # Check for missing values
    print(train_df.isnull().sum())
    print(test_df.isnull().sum())

    # Identify numerical and categorical features
    numerical_features = train_df.select_dtypes(include=['int64', 'float64']).columns.tolist()
    categorical_features = train_df.select_dtypes(include=['object']).columns.tolist()

    # Ensure 'price' is in the list of columns
    if 'price' not in train_df.columns:
        raise ValueError("The 'price' column is missing in the training dataset.")
    if 'price' not in test_df.columns:
        raise ValueError("The 'price' column is missing in the test dataset.")

    # Impute missing values
    imputer = SimpleImputer(strategy='median')
    train_df[numerical_features] = imputer.fit_transform(train_df[numerical_features])
    test_df[numerical_features] = imputer.transform(test_df[numerical_features])

    # Feature scaling
    scaler = StandardScaler()
    train_df[numerical_features] = scaler.fit_transform(train_df[numerical_features])
    test_df[numerical_features] = scaler.transform(test_df[numerical_features])

    # Encoding categorical variables
    preprocessor = ColumnTransformer(
        transformers=[
            ('num', StandardScaler(), numerical_features),
            ('cat', OneHotEncoder(drop='first'), categorical_features)
        ]
    )

    # Apply preprocessing
    X_train = preprocessor.fit_transform(train_df.drop(columns=['price']))
    y_train = train_df['price']
    X_test = preprocessor.transform(test_df.drop(columns=['price']))
    y_test = test_df['price']

    return X_train, y_train, X_test, y_test, preprocessor

# Model training
def train_model(X_train, y_train):
    model = LinearRegression()
    model.fit(X_train, y_train)
    return model

# Model evaluation
def evaluate_model(model, X_test, y_test):
    y_pred = model.predict(X_test)
    rmse = mean_squared_error(y_test, y_pred, squared=False)
    return rmse

# Main function to orchestrate the entire pipeline
def main():
    """
    Main function to execute the tabular regression pipeline.
    """
    # Step 1: Load the datasets
    train_df, test_df = load_data()

    # Step 2: Preprocess the data
    X_train, y_train, X_test, y_test, preprocessor = preprocess_data(train_df, test_df)

    # Step 3: Train the model
    model = train_model(X_train, y_train)

    # Step 4: Evaluate the model
    rmse = evaluate_model(model, X_test, y_test)

    # Step 5: Save the model and preprocessing pipeline
    os.makedirs('./agent_workspace/trained_models', exist_ok=True)
    joblib.dump(model, './agent_workspace/trained_models/linear_regression_model.pkl')
    joblib.dump(preprocessor, './agent_workspace/trained_models/preprocessor.pkl')

    # Logging
    os.makedirs('./agent_workspace/logs', exist_ok=True)
    logging.basicConfig(filename='./agent_workspace/logs/model_training.log', level=logging.INFO)
    logging.info('Model training started')
    logging.info('Model training completed')
    logging.info(f'RMSE: {rmse}')

    return rmse

if __name__ == "__main__":
    rmse = main()
    print(f"Model Performance on Test Set (RMSE): {rmse}")
