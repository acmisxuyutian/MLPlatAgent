######代码######

import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.metrics import accuracy_score, precision_score, recall_score, confusion_matrix
from sklearn.preprocessing import StandardScaler
from sklearn.feature_selection import RFE
from sklearn.linear_model import LogisticRegression
from sklearn.model_selection import cross_val_score
import pickle

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data retrieval
def load_data(train_path, test_path):
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    return train_df, test_df

# Data preprocessing
def preprocess_data(train_df, test_df):
    # One-Hot Encoding
    categorical_features = [col for col in train_df.columns if train_df[col].dtype == 'object' and col != 'Disease']
    train_df = pd.get_dummies(train_df, columns=categorical_features, drop_first=True)
    test_df = pd.get_dummies(test_df, columns=categorical_features, drop_first=True)

    # Standardization
    numerical_features = [col for col in train_df.columns if train_df[col].dtype != 'object' and col != 'Disease']
    scaler = StandardScaler()
    train_df[numerical_features] = scaler.fit_transform(train_df[numerical_features])
    test_df[numerical_features] = scaler.transform(test_df[numerical_features])

    return train_df, test_df

# Feature selection
def select_features(X_train, y_train, n_features_to_select=5):
    model = LogisticRegression(solver='liblinear')
    rfe = RFE(model, n_features_to_select=n_features_to_select)
    rfe.fit(X_train, y_train)
    top_features = X_train.columns[rfe.support_]
    return top_features

# Model training
def train_model(X_train, y_train, C, solver):
    model = LogisticRegression(C=C, solver=solver, multi_class='auto')
    model.fit(X_train, y_train)
    return model

# Model evaluation
def evaluate_model(model, X_test, y_test):
    y_pred = model.predict(X_test)
    accuracy = accuracy_score(y_test, y_pred)
    precision = precision_score(y_test, y_pred, average='weighted', zero_division=0)
    recall = recall_score(y_test, y_pred, average='weighted', zero_division=0)
    confusion_matrix_result = confusion_matrix(y_test, y_pred)
    return accuracy, precision, recall, confusion_matrix_result

# Cross-validation
def cross_validate_model(model, X_train, y_train, cv=5):
    cv_scores = cross_val_score(model, X_train, y_train, cv=cv, scoring='accuracy')
    return cv_scores

# Main function to orchestrate the pipeline
def main():
    # Data retrieval
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/05_patient_profile/train.csv'
    test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/05_patient_profile/test.csv'
    train_df, test_df = load_data(train_path, test_path)

    # Data preprocessing
    train_df, test_df = preprocess_data(train_df, test_df)

    # Feature extraction
    X_train = train_df.drop(columns=['Disease'])
    y_train = train_df['Disease']
    X_test = test_df.drop(columns=['Disease'])
    y_test = test_df['Disease']

    # Model 1: Logistic Regression with All One-Hot Encoded Features
    model1 = train_model(X_train, y_train, C=0.1, solver='liblinear')
    accuracy1, precision1, recall1, confusion_matrix1 = evaluate_model(model1, X_test, y_test)
    cv_scores1 = cross_validate_model(model1, X_train, y_train)

    # Model 2: Logistic Regression with Top 5 Most Important Features
    top_features = select_features(X_train, y_train, n_features_to_select=5)
    X_train_selected = X_train[top_features]
    X_test_selected = X_test[top_features]
    model2 = train_model(X_train_selected, y_train, C=1, solver='saga')
    accuracy2, precision2, recall2, confusion_matrix2 = evaluate_model(model2, X_test_selected, y_test)
    cv_scores2 = cross_validate_model(model2, X_train_selected, y_train)

    # Model performance
    model_performance = {
        'Model 1': {
            'Accuracy': accuracy1,
            'Precision': precision1,
            'Recall': recall1,
            'Confusion Matrix': confusion_matrix1.tolist(),
            'Cross-Validation Scores': cv_scores1.tolist()
        },
        'Model 2': {
            'Accuracy': accuracy2,
            'Precision': precision2,
            'Recall': recall2,
            'Confusion Matrix': confusion_matrix2.tolist(),
            'Cross-Validation Scores': cv_scores2.tolist()
        }
    }

    # Save models
    model1_path = './agent_workspace/trained_models/model1.pkl'
    model2_path = './agent_workspace/trained_models/model2.pkl'
    os.makedirs(os.path.dirname(model1_path), exist_ok=True)
    os.makedirs(os.path.dirname(model2_path), exist_ok=True)
    with open(model1_path, 'wb') as f:
        pickle.dump(model1, f)
    with open(model2_path, 'wb') as f:
        pickle.dump(model2, f)

    # Save feature selection report
    with open('feature_selection_report.txt', 'w') as f:
        f.write(f"Feature Selection Method: RFE\n")
        f.write(f"Top 5 Features: {top_features}\n")
        f.write(f"Rationale: RFE iteratively removes the least important features and re-trains the model to select the most relevant features.\n")

    # Save model interpretation report
    feature_importance1 = model1.coef_[0]
    feature_importance2 = model2.coef_[0]
    with open('model_interpretation_report.txt', 'w') as f:
        f.write("Model 1 - Feature Importance:\n")
        for feature, importance in zip(X_train.columns, feature_importance1):
            f.write(f"{feature}: {importance}\n")

        f.write("\nModel 2 - Feature Importance:\n")
        for feature, importance in zip(top_features, feature_importance2):
            f.write(f"{feature}: {importance}\n")

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
