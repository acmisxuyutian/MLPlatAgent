######代码######

import os
import random
import numpy as np
import pandas as pd
from sklearn.preprocessing import OneHotEncoder, StandardScaler
from sklearn.impute import SimpleImputer
from sklearn.linear_model import LogisticRegression
from sklearn.feature_selection import RFE
from sklearn.metrics import accuracy_score, precision_score, recall_score
import joblib

# Set the random seed for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data Retrieval and Collection
def load_datasets():
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/05_patient_profile/train.csv'
    test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/05_patient_profile/test.csv'
    
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    
    return train_df, test_df

# Data Preprocessing
def preprocess_data(train_df, test_df):
    # Identify categorical features
    categorical_features = [col for col in train_df.columns if train_df[col].dtype == 'object' and col != 'Disease']
    
    # Apply one-hot encoding
    encoder = OneHotEncoder(handle_unknown='ignore')
    train_encoded = encoder.fit_transform(train_df[categorical_features])
    test_encoded = encoder.transform(test_df[categorical_features])
    
    # Create new DataFrames with encoded features
    train_encoded_df = pd.DataFrame(train_encoded.toarray(), columns=encoder.get_feature_names_out(categorical_features))
    test_encoded_df = pd.DataFrame(test_encoded.toarray(), columns=encoder.get_feature_names_out(categorical_features))
    
    # Concatenate encoded features with the original datasets
    train_df = pd.concat([train_df.drop(columns=categorical_features), train_encoded_df], axis=1)
    test_df = pd.concat([test_df.drop(columns=categorical_features), test_encoded_df], axis=1)
    
    # Handle missing values
    imputer = SimpleImputer(strategy='most_frequent')
    train_df = pd.DataFrame(imputer.fit_transform(train_df), columns=train_df.columns)
    test_df = pd.DataFrame(imputer.transform(test_df), columns=test_df.columns)
    
    # Standardize numerical features
    scaler = StandardScaler()
    numerical_features = [col for col in train_df.columns if col != 'Disease']
    train_df[numerical_features] = scaler.fit_transform(train_df[numerical_features])
    test_df[numerical_features] = scaler.transform(test_df[numerical_features])
    
    return train_df, test_df

# Feature Selection
def select_features(train_df):
    # Train a logistic regression model
    model = LogisticRegression(penalty='l2', solver='liblinear')
    rfe = RFE(estimator=model, n_features_to_select=5, step=1)
    rfe.fit(train_df.drop(columns=['Disease']), train_df['Disease'])
    
    # Select the top 5 features
    top_features = train_df.drop(columns=['Disease']).columns[rfe.support_]
    train_df_rfe = train_df[top_features]
    
    return train_df_rfe, top_features

# Model Training and Evaluation
def train_and_evaluate_model(train_df, test_df, top_features=None):
    # Train Model 1: Using all features
    model1 = LogisticRegression(C=1, penalty='l2', solver='liblinear')
    model1.fit(train_df.drop(columns=['Disease']), train_df['Disease'])
    
    # Train Model 2: Using top 5 features
    if top_features is not None:
        model2 = LogisticRegression(C=0.1, penalty='l2', solver='liblinear')
        model2.fit(train_df[top_features], train_df['Disease'])
    else:
        model2 = None
    
    # Evaluate Model 1
    y_pred1 = model1.predict(test_df.drop(columns=['Disease']))
    accuracy1 = accuracy_score(test_df['Disease'], y_pred1)
    precision1 = precision_score(test_df['Disease'], y_pred1, average='weighted')
    recall1 = recall_score(test_df['Disease'], y_pred1, average='weighted')
    
    # Evaluate Model 2
    if model2 is not None:
        y_pred2 = model2.predict(test_df[top_features])
        accuracy2 = accuracy_score(test_df['Disease'], y_pred2)
        precision2 = precision_score(test_df['Disease'], y_pred2, average='weighted')
        recall2 = recall_score(test_df['Disease'], y_pred2, average='weighted')
    else:
        accuracy2, precision2, recall2 = None, None, None
    
    # Print the performance metrics
    print(f"Model 1 - Accuracy: {accuracy1}, Precision: {precision1}, Recall: {recall1}")
    if model2 is not None:
        print(f"Model 2 - Accuracy: {accuracy2}, Precision: {precision2}, Recall: {recall2}")
    
    # Return the performance metrics
    model_performance = {
        'Model 1': {'Accuracy': accuracy1, 'Precision': precision1, 'Recall': recall1},
        'Model 2': {'Accuracy': accuracy2, 'Precision': precision2, 'Recall': recall2} if model2 is not None else None
    }
    
    return model1, model2, model_performance

# Model Saving
def save_models(model1, model2, model_dir):
    os.makedirs(model_dir, exist_ok=True)
    joblib.dump(model1, os.path.join(model_dir, 'logistic_regression_model1.pkl'))
    if model2 is not None:
        joblib.dump(model2, os.path.join(model_dir, 'logistic_regression_model2.pkl'))

# Main function to orchestrate the pipeline
def main():
    # Load datasets
    train_df, test_df = load_datasets()
    
    # Preprocess data
    train_df, test_df = preprocess_data(train_df, test_df)
    
    # Select features
    train_df_rfe, top_features = select_features(train_df)
    
    # Train and evaluate models
    model1, model2, model_performance = train_and_evaluate_model(train_df, test_df, top_features)
    
    # Save models
    model_dir = './agent_workspace/trained_models'
    save_models(model1, model2, model_dir)
    
    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
