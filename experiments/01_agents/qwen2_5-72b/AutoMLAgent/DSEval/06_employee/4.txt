import os
import pandas as pd
import numpy as np
from sklearn.preprocessing import OneHotEncoder, StandardScaler
from sklearn.impute import SimpleImputer
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import roc_auc_score, confusion_matrix
import joblib

SEED = 42
np.random.seed(SEED)

# Data Retrieval
def load_data(train_path, test_path):
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    return train_df, test_df

# Data Preprocessing
def preprocess_data(train_df, test_df):
    # Feature Construction
    train_df['YearsInCompany'] = 2023 - train_df['JoiningYear']
    test_df['YearsInCompany'] = 2023 - test_df['JoiningYear']

    # Identify categorical and numerical columns
    categorical_features = train_df.select_dtypes(include=['object', 'category']).columns
    numerical_features = train_df.select_dtypes(include=['number']).columns

    # One-Hot Encoding
    encoder = OneHotEncoder(handle_unknown='ignore', sparse_output=False)
    train_encoded = encoder.fit_transform(train_df[categorical_features])
    test_encoded = encoder.transform(test_df[categorical_features])

    train_df_encoded = pd.DataFrame(train_encoded, columns=encoder.get_feature_names_out(categorical_features))
    test_df_encoded = pd.DataFrame(test_encoded, columns=encoder.get_feature_names_out(categorical_features))

    train_df = pd.concat([train_df, train_df_encoded], axis=1).drop(columns=categorical_features)
    test_df = pd.concat([test_df, test_df_encoded], axis=1).drop(columns=categorical_features)

    # Handle Missing Values
    imputer = SimpleImputer(strategy='mean')
    train_df[numerical_features] = imputer.fit_transform(train_df[numerical_features])
    test_df[numerical_features] = imputer.transform(test_df[numerical_features])

    # Feature Scaling
    scaler = StandardScaler()
    train_df[numerical_features] = scaler.fit_transform(train_df[numerical_features])
    test_df[numerical_features] = scaler.transform(test_df[numerical_features])

    return train_df, test_df, encoder, imputer, scaler

# Model Training
def train_model(train_df):
    model = RandomForestClassifier(n_estimators=100, max_depth=10, min_samples_split=2, min_samples_leaf=1, random_state=42)
    X_train = train_df.drop(columns=['LeaveOrNot'])
    y_train = train_df['LeaveOrNot']
    model.fit(X_train, y_train)
    return model

# Model Evaluation
def evaluate_model(model, test_df):
    X_test = test_df.drop(columns=['LeaveOrNot'])
    y_test = test_df['LeaveOrNot']
    y_pred = model.predict(X_test)
    y_pred_proba = model.predict_proba(X_test)[:, 1]

    auc_roc = roc_auc_score(y_test, y_pred_proba)
    cm = confusion_matrix(y_test, y_pred)

    performance_scores = {
        'AUC-ROC Score': auc_roc,
        'Confusion Matrix': cm
    }

    return performance_scores

# Model Saving
def save_model(model, model_path, preprocessing_steps, preprocessing_path):
    joblib.dump(model, model_path)
    joblib.dump(preprocessing_steps, preprocessing_path)

# Main Function to Orchestrate the Pipeline
def main():
    # Data Retrieval
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/06_employee/train.csv'
    test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/06_employee/test.csv'
    train_df, test_df = load_data(train_path, test_path)

    # Data Preprocessing
    train_df, test_df, encoder, imputer, scaler = preprocess_data(train_df, test_df)

    # Model Training
    model = train_model(train_df)

    # Model Evaluation
    performance_scores = evaluate_model(model, test_df)

    # Model Saving
    model_path = './agent_workspace/trained_models/employee_turnover_random_forest.pkl'
    preprocessing_path = './agent_workspace/trained_models/employee_turnover_preprocessing.pkl'
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    os.makedirs(os.path.dirname(preprocessing_path), exist_ok=True)
    preprocessing_steps = {
        'encoder': encoder,
        'imputer': imputer,
        'scaler': scaler
    }
    save_model(model, model_path, preprocessing_steps, preprocessing_path)

    return performance_scores

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
