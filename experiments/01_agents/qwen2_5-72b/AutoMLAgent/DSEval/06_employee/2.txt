######代码######

import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split, RandomizedSearchCV
from sklearn.ensemble import RandomForestClassifier
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import roc_auc_score, confusion_matrix
import joblib

# Set the random seed for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Paths to the datasets
train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/06_employee/train.csv'
test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/06_employee/test.csv'

# Load the datasets
train_df = pd.read_csv(train_path)
test_df = pd.read_csv(test_path)

# Feature Construction
train_df['YearsInCompany'] = 2023 - train_df['JoiningYear']
test_df['YearsInCompany'] = 2023 - test_df['JoiningYear']

# Handling Missing Data
train_df.fillna(train_df.mean(numeric_only=True), inplace=True)
train_df.fillna(train_df.mode().iloc[0], inplace=True)

test_df.fillna(test_df.mean(numeric_only=True), inplace=True)
test_df.fillna(test_df.mode().iloc[0], inplace=True)

# One-Hot Encoding
categorical_columns = train_df.select_dtypes(include=['object']).columns
train_df = pd.get_dummies(train_df, columns=categorical_columns)
test_df = pd.get_dummies(test_df, columns=categorical_columns)

# Ensure that the test set has the same columns as the training set
missing_cols = set(train_df.columns) - set(test_df.columns)
for c in missing_cols:
    test_df[c] = 0

# Ensure that the order of columns is the same
test_df = test_df[train_df.columns]

# Feature Scaling
numerical_columns = train_df.select_dtypes(include=['int64', 'float64']).columns.drop('LeaveOrNot')
scaler = StandardScaler()
train_df[numerical_columns] = scaler.fit_transform(train_df[numerical_columns])
test_df[numerical_columns] = scaler.transform(test_df[numerical_columns])

# Data Splitting
X_train, X_val, y_train, y_val = train_test_split(
    train_df.drop(columns=['LeaveOrNot']),
    train_df['LeaveOrNot'],
    test_size=0.2,
    stratify=train_df['LeaveOrNot'],
    random_state=42
)

# Initialize the Model
rf = RandomForestClassifier(random_state=42)

# Define Hyperparameter Search Space
param_grid = {
    'n_estimators': [50, 100, 200],
    'max_depth': [None, 10, 20, 30],
    'min_samples_split': [2, 5, 10],
    'min_samples_leaf': [1, 2, 4]
}

# Perform Hyperparameter Tuning
random_search = RandomizedSearchCV(
    estimator=rf,
    param_distributions=param_grid,
    n_iter=50,
    cv=5,
    verbose=2,
    random_state=42,
    n_jobs=-1
)

# Fit the model
random_search.fit(X_train, y_train)

# Select the Best Model
best_rf = random_search.best_estimator_

# Make Predictions
y_val_pred = best_rf.predict(X_val)
y_val_pred_proba = best_rf.predict_proba(X_val)[:, 1]

y_test_pred = best_rf.predict(test_df.drop(columns=['LeaveOrNot']))
y_test_pred_proba = best_rf.predict_proba(test_df.drop(columns=['LeaveOrNot']))[:, 1]

# Compute AUC-ROC
val_auc_roc = roc_auc_score(y_val, y_val_pred_proba)
test_auc_roc = roc_auc_score(test_df['LeaveOrNot'], y_test_pred_proba)

# Generate Confusion Matrix
val_conf_matrix = confusion_matrix(y_val, y_val_pred)
test_conf_matrix = confusion_matrix(test_df['LeaveOrNot'], y_test_pred)

# Save the Model
model_path = './agent_workspace/trained_models/employee_turnover_random_forest.pkl'
os.makedirs(os.path.dirname(model_path), exist_ok=True)
joblib.dump(best_rf, model_path)

# Save Evaluation Results
results = {
    'val_auc_roc': val_auc_roc,
    'test_auc_roc': test_auc_roc,
    'val_conf_matrix': val_conf_matrix.tolist(),
    'test_conf_matrix': test_conf_matrix.tolist()
}

results_path = './agent_workspace/trained_models/employee_turnover_results.json'
os.makedirs(os.path.dirname(results_path), exist_ok=True)
with open(results_path, 'w') as f:
    json.dump(results, f, indent=4)

# Print the model performance on the test set
print("Model Performance on Test Set:")
print(f"AUC-ROC: {test_auc_roc}")
print(f"Confusion Matrix:\n{test_conf_matrix}")
