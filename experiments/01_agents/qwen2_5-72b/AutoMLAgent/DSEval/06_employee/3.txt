import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.preprocessing import OneHotEncoder
from sklearn.model_selection import train_test_split, RandomizedSearchCV
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import roc_auc_score, confusion_matrix
from imblearn.over_sampling import SMOTE
import joblib

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Step 1. Data Retrieval
def load_data():
    train_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/06_employee/train.csv')
    test_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/06_employee/test.csv')
    return train_df, test_df

# Step 2. Data Preprocessing
def preprocess_data(train_df, test_df):
    # Feature Construction
    train_df['YearsInCompany'] = 2023 - train_df['JoiningYear']
    test_df['YearsInCompany'] = 2023 - test_df['JoiningYear']
    
    # Handling Categorical Variables
    encoder = OneHotEncoder(drop='first')
    categorical_features = train_df.select_dtypes(include=['object', 'category']).columns
    train_encoded = pd.DataFrame(encoder.fit_transform(train_df[categorical_features]), columns=encoder.get_feature_names_out(categorical_features))
    test_encoded = pd.DataFrame(encoder.transform(test_df[categorical_features]), columns=encoder.get_feature_names_out(categorical_features))
    train_df = pd.concat([train_df.drop(columns=categorical_features), train_encoded], axis=1)
    test_df = pd.concat([test_df.drop(columns=categorical_features), test_encoded], axis=1)
    
    # Ensure Consistency
    train_df, test_df = train_df.align(test_df, join='inner', axis=1)
    
    return train_df, test_df, encoder

# Step 3. Data Splitting
def split_data(train_df):
    X_train, X_val, y_train, y_val = train_test_split(
        train_df.drop(columns=['LeaveOrNot']),
        train_df['LeaveOrNot'],
        test_size=0.2,
        stratify=train_df['LeaveOrNot'],
        random_state=42
    )
    return X_train, X_val, y_train, y_val

# Step 4. Data Augmentation
def augment_data(X_train, y_train):
    smote = SMOTE(random_state=42)
    X_train_resampled, y_train_resampled = smote.fit_resample(X_train, y_train)
    return X_train_resampled, y_train_resampled

# Step 5. Data Exploration
def explore_data(train_df):
    summary_stats = train_df.describe()
    class_distribution = train_df['LeaveOrNot'].value_counts(normalize=True)
    correlation_matrix = train_df.corr()
    return summary_stats, class_distribution, correlation_matrix

# Step 6. Model Training
def train_model(X_train_resampled, y_train_resampled):
    model = RandomForestClassifier(random_state=42)
    param_grid = {
        'n_estimators': [100, 200, 300],
        'max_depth': [10, 20, 30],
        'min_samples_split': [2, 5, 10],
        'min_samples_leaf': [1, 2, 4],
        'max_features': ['auto', 'sqrt']
    }
    random_search = RandomizedSearchCV(model, param_grid, n_iter=10, cv=5, scoring='roc_auc', random_state=42)
    random_search.fit(X_train_resampled, y_train_resampled)
    best_model = random_search.best_estimator_
    return best_model

# Step 7. Model Evaluation
def evaluate_model(model, X_val, y_val, test_df):
    y_val_pred = model.predict(X_val)
    y_val_prob = model.predict_proba(X_val)[:, 1]
    y_test_pred = model.predict(test_df.drop(columns=['LeaveOrNot']))
    y_test_prob = model.predict_proba(test_df.drop(columns=['LeaveOrNot']))[:, 1]
    
    roc_auc_val = roc_auc_score(y_val, y_val_prob)
    roc_auc_test = roc_auc_score(test_df['LeaveOrNot'], y_test_prob)
    conf_matrix_val = confusion_matrix(y_val, y_val_pred)
    conf_matrix_test = confusion_matrix(test_df['LeaveOrNot'], y_test_pred)
    
    feature_importances = model.feature_importances_
    feature_importance_df = pd.DataFrame({'Feature': X_val.columns, 'Importance': feature_importances})
    feature_importance_df = feature_importance_df.sort_values(by='Importance', ascending=False)
    
    performance_scores = {
        'ROC_AUC_Val': roc_auc_val,
        'ROC_AUC_Test': roc_auc_test,
        'Confusion_Matrix_Val': conf_matrix_val,
        'Confusion_Matrix_Test': conf_matrix_test,
        'Feature_Importance': feature_importance_df
    }
    
    return performance_scores

# Step 8. Model Saving and Deployment
def save_model(model, encoder):
    model_path = './agent_workspace/trained_models/employee_turnover_random_forest.pkl'
    joblib.dump(model, model_path)
    
    preprocessing_path = './agent_workspace/trained_models/employee_turnover_preprocessing.pkl'
    joblib.dump(encoder, preprocessing_path)

# Main function to orchestrate the pipeline
def main():
    # Step 1. Retrieve or load the dataset
    train_df, test_df = load_data()
    
    # Step 2. Preprocess the data
    train_df, test_df, encoder = preprocess_data(train_df, test_df)
    
    # Step 3. Split the data
    X_train, X_val, y_train, y_val = split_data(train_df)
    
    # Step 4. Augment the data
    X_train_resampled, y_train_resampled = augment_data(X_train, y_train)
    
    # Step 5. Explore the data
    summary_stats, class_distribution, correlation_matrix = explore_data(train_df)
    
    # Step 6. Train the model
    model = train_model(X_train_resampled, y_train_resampled)
    
    # Step 7. Evaluate the model
    performance_scores = evaluate_model(model, X_val, y_val, test_df)
    
    # Step 8. Save the model and preprocessing steps
    save_model(model, encoder)
    
    return performance_scores

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
