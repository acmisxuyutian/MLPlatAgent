import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score, f1_score, roc_auc_score
from sklearn.utils.class_weight import compute_class_weight
import xgboost as xgb
import joblib

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data Retrieval and Loading
def load_data(train_path, test_path):
    # Load the training dataset
    train_df = pd.read_csv(train_path)

    # Load the test dataset
    test_df = pd.read_csv(test_path)

    # Check the first few rows and data types
    print("Training Data:")
    print(train_df.head())
    print(train_df.dtypes)

    print("Test Data:")
    print(test_df.head())
    print(test_df.dtypes)

    return train_df, test_df

# Data Preprocessing
def preprocess_data(train_df, test_df):
    # Drop irrelevant columns
    columns_to_drop = ['CustomerId', 'Complain', 'Surname']
    train_df = train_df.drop(columns=columns_to_drop)
    test_df = test_df.drop(columns=columns_to_drop)

    # One-hot encoding for categorical columns
    train_df = pd.get_dummies(train_df, columns=['Geography', 'Card Type', 'Gender'])
    test_df = pd.get_dummies(test_df, columns=['Geography', 'Card Type', 'Gender'])

    # Handle missing values
    print("Missing values in training data:")
    print(train_df.isnull().sum())
    print("Missing values in test data:")
    print(test_df.isnull().sum())

    # Impute missing values for numerical columns
    train_df.fillna(train_df.select_dtypes(include=[np.number]).mean(), inplace=True)
    test_df.fillna(test_df.select_dtypes(include=[np.number]).mean(), inplace=True)

    # Impute missing values for categorical columns
    for col in train_df.select_dtypes(include=['object']).columns:
        train_df[col].fillna(train_df[col].mode()[0], inplace=True)
        test_df[col].fillna(test_df[col].mode()[0], inplace=True)

    return train_df, test_df

# Data Augmentation
def compute_class_weights(y_train):
    class_weights = compute_class_weight('balanced', classes=np.unique(y_train), y=y_train)
    scale_pos_weight = class_weights[1] / class_weights[0]
    return scale_pos_weight

# Model Training
def train_model(X_train, y_train, scale_pos_weight):
    # Initialize the XGBoost model
    xgb_model = xgb.XGBClassifier(
        learning_rate=0.1,
        max_depth=6,
        min_child_weight=1,
        gamma=0,
        subsample=0.8,
        colsample_bytree=0.8,
        scale_pos_weight=scale_pos_weight,
        n_estimators=100,
        random_state=42
    )

    # Train the model
    xgb_model.fit(X_train, y_train)

    return xgb_model

# Model Evaluation
def evaluate_model(model, X_test, y_test):
    # Make predictions
    y_pred_proba = model.predict_proba(X_test)[:, 1]

    # Compute AUC-ROC
    roc_auc = roc_auc_score(y_test, y_pred_proba)
    print(f'ROC AUC: {roc_auc}')

    # Compute Accuracy and F1 Score
    y_pred = model.predict(X_test)
    accuracy = accuracy_score(y_test, y_pred)
    f1 = f1_score(y_test, y_pred)

    performance_scores = {
        'ROC AUC': roc_auc,
        'Accuracy': accuracy,
        'F1 Score': f1
    }

    return performance_scores

# Model Saving
def save_model(model, model_path):
    joblib.dump(model, model_path)

# Main function to orchestrate the entire pipeline
def main():
    # Step 1: Load the dataset
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/01_bank_customer_churn/train.csv'
    test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/01_bank_customer_churn/test.csv'
    train_df, test_df = load_data(train_path, test_path)

    # Step 2: Preprocess the data
    train_df, test_df = preprocess_data(train_df, test_df)

    # Step 3: Split the data into features and target
    X_train = train_df.drop(columns=['Exited'])
    y_train = train_df['Exited']
    X_test = test_df.drop(columns=['Exited'])
    y_test = test_df['Exited']

    # Step 4: Compute class weights
    scale_pos_weight = compute_class_weights(y_train)

    # Step 5: Train the model
    model = train_model(X_train, y_train, scale_pos_weight)

    # Step 6: Evaluate the model
    performance_scores = evaluate_model(model, X_test, y_test)
    print("Model Performance on Test Set:", performance_scores)

    # Step 7: Save the model
    model_path = './agent_workspace/trained_models/xgboost_churn_model.pkl'
    save_model(model, model_path)

    return performance_scores

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
