import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import OneHotEncoder
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
from sklearn.model_selection import train_test_split, cross_val_score
from sklearn.metrics import roc_auc_score
import xgboost as xgb
import optuna
import joblib

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Step 1: Load the datasets
def load_data(train_data_path, test_data_path):
    train_df = pd.read_csv(train_data_path)
    test_df = pd.read_csv(test_data_path)
    return train_df, test_df

# Step 2: Preprocess the data
def preprocess_data(train_df, test_df):
    # Drop irrelevant features
    train_df = train_df.drop(columns=['CustomerId', 'Complain', 'Surname'])
    test_df = test_df.drop(columns=['CustomerId', 'Complain', 'Surname'])

    # Define categorical and numerical columns
    categorical_features = ['Geography', 'Card Type']
    numerical_features = train_df.select_dtypes(include=['int64', 'float64']).columns.tolist()

    # Ensure 'Exited' is in the training data
    if 'Exited' not in train_df.columns:
        raise ValueError("The 'Exited' column is missing in the training dataset.")

    # Define the preprocessing pipeline
    preprocessor = ColumnTransformer(
        transformers=[
            ('cat', OneHotEncoder(), categorical_features),
            ('num', SimpleImputer(strategy='mean'), numerical_features)
        ]
    )

    # Apply the preprocessing pipeline
    X_train = preprocessor.fit_transform(train_df.drop(columns=['Exited']))
    y_train = train_df['Exited']
    X_test = preprocessor.transform(test_df.drop(columns=['Exited'], errors='ignore'))

    return X_train, y_train, X_test, preprocessor

# Step 3: Train the model
def train_model(X_train, y_train):
    def objective(trial):
        params = {
            'max_depth': trial.suggest_int('max_depth', 3, 10),
            'learning_rate': trial.suggest_loguniform('learning_rate', 0.01, 0.3),
            'n_estimators': trial.suggest_int('n_estimators', 100, 500),
            'subsample': trial.suggest_uniform('subsample', 0.5, 1.0),
            'colsample_bytree': trial.suggest_uniform('colsample_bytree', 0.5, 1.0)
        }
        model = xgb.XGBClassifier(**params)
        scores = cross_val_score(model, X_train, y_train, cv=5, scoring='roc_auc')
        return scores.mean()

    # Perform Bayesian optimization
    study = optuna.create_study(direction='maximize')
    study.optimize(objective, n_trials=100)

    # Get the best hyperparameters
    best_params = study.best_params

    # Train the final model with the best hyperparameters
    final_model = xgb.XGBClassifier(**best_params)
    final_model.fit(X_train, y_train)

    return final_model

# Step 4: Evaluate the model
def evaluate_model(model, X_val, y_val):
    val_auc = roc_auc_score(y_val, model.predict_proba(X_val)[:, 1])
    return val_auc

# The main function to orchestrate the entire pipeline
def main():
    # Step 1: Load the datasets
    train_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/01_bank_customer_churn/train.csv'
    test_data_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/DSEval/datasets/01_bank_customer_churn/test.csv'
    train_df, test_df = load_data(train_data_path, test_data_path)

    # Step 2: Preprocess the data
    X_train, y_train, X_test, preprocessor = preprocess_data(train_df, test_df)

    # Step 3: Train the model
    model = train_model(X_train, y_train)

    # Step 4: Evaluate the model
    # Note: The test dataset does not have the target variable 'Exited', so we cannot evaluate it directly.
    # For the purpose of this example, we will split the training data into training and validation sets.
    X_train_split, X_val_split, y_train_split, y_val_split = train_test_split(X_train, y_train, test_size=0.2, random_state=42)
    val_auc = evaluate_model(model, X_val_split, y_val_split)
    print(f'Validation AUC-ROC: {val_auc}')

    # Step 5: Save the model and preprocessing pipeline
    model_path = './agent_workspace/trained_models/xgboost_churn_model.pkl'
    preprocessor_path = './agent_workspace/trained_models/preprocessor.pkl'
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    os.makedirs(os.path.dirname(preprocessor_path), exist_ok=True)
    joblib.dump(model, model_path)
    joblib.dump(preprocessor, preprocessor_path)

    # Return the validation AUC-ROC score
    return val_auc

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Validation Set:", model_performance)
