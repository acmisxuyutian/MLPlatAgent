import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import OneHotEncoder, StandardScaler
from imblearn.over_sampling import SMOTE
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import RandomizedSearchCV
from scipy.stats import randint
from sklearn.metrics import confusion_matrix, roc_auc_score
import pickle

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data Retrieval and Loading
def load_data(train_path, test_path):
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    return train_df, test_df

# Data Preprocessing
def preprocess_data(train_df, test_df):
    # Handle Missing Values
    numerical_columns = train_df.select_dtypes(include=[np.number]).columns.tolist()
    categorical_columns = train_df.select_dypes(include=[object]).columns.tolist()

    numerical_imputer = SimpleImputer(strategy='mean')
    train_df[numerical_columns] = numerical_imputer.fit_transform(train_df[numerical_columns])
    test_df[numerical_columns] = numerical_imputer.transform(test_df[numerical_columns])

    categorical_imputer = SimpleImputer(strategy='most_frequent')
    train_df[categorical_columns] = categorical_imputer.fit_transform(train_df[categorical_columns])
    test_df[categorical_columns] = categorical_imputer.transform(test_df[categorical_columns])

    # Encode Categorical Variables
    encoder = OneHotEncoder(handle_unknown='ignore', sparse=False)
    train_encoded = encoder.fit_transform(train_df[categorical_columns])
    test_encoded = encoder.transform(test_df[categorical_columns])

    train_df = pd.concat([train_df.drop(categorical_columns, axis=1), pd.DataFrame(train_encoded, columns=encoder.get_feature_names_out(categorical_columns))], axis=1)
    test_df = pd.concat([test_df.drop(categorical_columns, axis=1), pd.DataFrame(test_encoded, columns=encoder.get_feature_names_out(categorical_columns))], axis=1)

    # Feature Scaling
    scaler = StandardScaler()
    train_df[numerical_columns] = scaler.fit_transform(train_df[numerical_columns])
    test_df[numerical_columns] = scaler.transform(test_df[numerical_columns])

    # Data Augmentation
    X_train = train_df.drop('churn', axis=1)
    y_train = train_df['churn']

    smote = SMOTE()
    X_train_resampled, y_train_resampled = smote.fit_resample(X_train, y_train)
    train_df_resampled = pd.concat([pd.DataFrame(X_train_resampled, columns=X_train.columns), pd.Series(y_train_resampled, name='churn')], axis=1)

    return train_df_resampled, test_df

# Model Training and Hyperparameter Tuning
def train_model(train_df_resampled):
    X_train = train_df_resampled.drop('churn', axis=1)
    y_train = train_df_resampled['churn']

    rf = RandomForestClassifier(random_state=42)

    param_distributions = {
        'n_estimators': randint(100, 300),
        'max_depth': [None, 10, 20, 30],
        'min_samples_split': [2, 5, 10],
        'min_samples_leaf': [1, 2, 4]
    }

    random_search = RandomizedSearchCV(rf, param_distributions, n_iter=100, cv=5, scoring='roc_auc', random_state=42)
    random_search.fit(X_train, y_train)
    best_rf = random_search.best_estimator_

    return best_rf

# Model Evaluation
def evaluate_model(model, test_df):
    X_test = test_df.drop('churn', axis=1)
    y_test = test_df['churn']

    y_pred = model.predict(X_test)
    y_pred_proba = model.predict_proba(X_test)[:, 1]

    cm = confusion_matrix(y_test, y_pred)
    auc_roc = roc_auc_score(y_test, y_pred_proba)

    performance_scores = {
        'Confusion Matrix': cm,
        'AUC-ROC Score': auc_roc
    }

    return performance_scores, y_pred, y_pred_proba

# Save Model
def save_model(model, model_path):
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    with open(model_path, 'wb') as file:
        pickle.dump(model, file)

# Save Predictions
def save_predictions(test_df, y_pred, y_pred_proba, predictions_path):
    predictions_df = pd.DataFrame({'CustomerID': test_df['CustomerID'], 'Churn': y_pred, 'Churn_Probability': y_pred_proba})
    predictions_df.to_csv(predictions_path, index=False)

# Main Function to Orchestrate the Pipeline
def main():
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/bank_customer_churn/train.csv'
    test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/bank_customer_churn/test.csv'
    model_path = './agent_workspace/trained_models/random_forest_model.pkl'
    predictions_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/bank_customer_churn/predictions.csv'

    # Load data
    train_df, test_df = load_data(train_path, test_path)

    # Preprocess data
    train_df_resampled, test_df = preprocess_data(train_df, test_df)

    # Train model
    model = train_model(train_df_resampled)

    # Evaluate model
    performance_scores, y_pred, y_pred_proba = evaluate_model(model, test_df)

    # Save model
    save_model(model, model_path)

    # Save predictions
    save_predictions(test_df, y_pred, y_pred_proba, predictions_path)

    return performance_scores

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:")
    print("Confusion Matrix:")
    print(model_performance['Confusion Matrix'])
    print("AUC-ROC Score:", model_performance['AUC-ROC Score'])
