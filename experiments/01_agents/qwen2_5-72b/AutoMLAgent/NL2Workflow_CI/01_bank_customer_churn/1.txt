import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import roc_auc_score, confusion_matrix, classification_report
from sklearn.model_selection import train_test_split
from imblearn.over_sampling import SMOTE
import joblib

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data Retrieval and Loading
def load_datasets(train_path, test_path):
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    return train_df, test_df

# Data Preprocessing
def preprocess_data(train_df, test_df):
    # Handle Missing Values
    numerical_features = train_df.select_dtypes(include=['float64', 'int64']).columns
    numerical_imputer = SimpleImputer(strategy='mean')
    train_df[numerical_features] = numerical_imputer.fit_transform(train_df[numerical_features])
    test_df[numerical_features] = numerical_imputer.transform(test_df[numerical_features])

    categorical_features = train_df.select_dtypes(include=['object']).columns
    categorical_imputer = SimpleImputer(strategy='most_frequent')
    train_df[categorical_features] = categorical_imputer.fit_transform(train_df[categorical_features])
    test_df[categorical_features] = categorical_imputer.transform(test_df[categorical_features])

    # Encode Categorical Variables
    train_df = pd.get_dummies(train_df, columns=categorical_features, drop_first=True)
    test_df = pd.get_dummies(test_df, columns=categorical_features, drop_first=True)

    # Ensure Consistent Columns
    train_columns = set(train_df.columns)
    test_columns = set(test_df.columns)
    common_columns = list(train_columns.intersection(test_columns))

    for col in train_columns - test_columns:
        test_df[col] = 0

    for col in test_columns - train_columns:
        train_df[col] = 0

    train_df = train_df[common_columns]
    test_df = test_df[common_columns]

    # Scale Numerical Features
    scaler = StandardScaler()
    train_df[numerical_features] = scaler.fit_transform(train_df[numerical_features])
    test_df[numerical_features] = scaler.transform(test_df[numerical_features])

    return train_df, test_df

# Data Augmentation (Optional)
def apply_smote(X_train, y_train):
    smote = SMOTE()
    X_train_resampled, y_train_resampled = smote.fit_resample(X_train, y_train)
    return X_train_resampled, y_train_resampled

# Model Training
def train_model(X_train, y_train):
    rf_model = RandomForestClassifier(
        n_estimators=200,
        max_depth=20,
        min_samples_split=5,
        min_samples_leaf=2,
        random_state=42
    )
    rf_model.fit(X_train, y_train)
    return rf_model

# Model Evaluation
def evaluate_model(model, X_test, y_test):
    y_pred = model.predict(X_test)
    y_pred_proba = model.predict_proba(X_test)[:, 1]

    auc_roc = roc_auc_score(y_test, y_pred_proba)
    conf_matrix = confusion_matrix(y_test, y_pred)
    class_report = classification_report(y_test, y_pred)

    performance_scores = {
        'AUC-ROC': auc_roc,
        'Confusion Matrix': conf_matrix,
        'Classification Report': class_report
    }

    return performance_scores

# Save Model and Results
def save_model_and_results(model, predictions_df, model_path, predictions_path):
    joblib.dump(model, model_path)
    predictions_df.to_csv(predictions_path, index=False)

# Main function to orchestrate the pipeline
def main():
    # Load datasets
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/bank_customer_churn/train.csv'
    test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/bank_customer_churn/test.csv'
    train_df, test_df = load_datasets(train_path, test_path)

    # Preprocess data
    train_df, test_df = preprocess_data(train_df, test_df)

    # Separate features and target variable
    X_train = train_df.drop(columns=['churn'], errors='ignore')
    y_train = train_df['churn']
    X_test = test_df.drop(columns=['churn'], errors='ignore')
    y_test = test_df['churn']

    # Apply SMOTE (optional)
    X_train_resampled, y_train_resampled = apply_smote(X_train, y_train)

    # Train the model
    model = train_model(X_train_resampled, y_train_resampled)

    # Evaluate the model
    performance_scores = evaluate_model(model, X_test, y_test)

    # Save the model and results
    model_path = './agent_workspace/trained_models/random_forest_churn_model.pkl'
    predictions_path = './agent_workspace/results/predictions.csv'
    predictions_df = pd.DataFrame({
        'actual': y_test,
        'predicted': model.predict(X_test),
        'predicted_proba': model.predict_proba(X_test)[:, 1]
    })
    save_model_and_results(model, predictions_df, model_path, predictions_path)

    return performance_scores

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
