import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.metrics import f1_score
from sklearn.preprocessing import MinMaxScaler, OneHotEncoder
from imblearn.over_sampling import SMOTE
from pytorch_tabnet.tab_model import TabNetClassifier
import optuna

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data Retrieval and Loading
def load_data(train_path, test_path):
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    return train_df, test_df

# Data Preprocessing
def preprocess_data(train_df, test_df):
    # Handle Missing Values
    numerical_features = train_df.select_dtypes(include=['float64', 'int64']).columns
    categorical_features = train_df.select_dtypes(include=['object']).columns

    train_df[numerical_features] = train_df[numerical_features].fillna(train_df[numerical_features].mean())
    train_df[categorical_features] = train_df[categorical_features].fillna(train_df[categorical_features].mode().iloc[0])

    test_df[numerical_features] = test_df[numerical_features].fillna(test_df[numerical_features].mean())
    test_df[categorical_features] = test_df[categorical_features].fillna(test_df[categorical_features].mode().iloc[0])

    # Normalize Numerical Features
    scaler = MinMaxScaler()
    train_df[numerical_features] = scaler.fit_transform(train_df[numerical_features])
    test_df[numerical_features] = scaler.transform(test_df[numerical_features])

    # Encode Categorical Variables
    encoder = OneHotEncoder(handle_unknown='ignore', sparse=False)
    encoded_train = encoder.fit_transform(train_df[categorical_features])
    encoded_test = encoder.transform(test_df[categorical_features])

    train_df = train_df.drop(columns=categorical_features)
    test_df = test_df.drop(columns=categorical_features)

    train_df = pd.concat([train_df, pd.DataFrame(encoded_train, columns=encoder.get_feature_names_out(categorical_features))], axis=1)
    test_df = pd.concat([test_df, pd.DataFrame(encoded_test, columns=encoder.get_feature_names_out(categorical_features))], axis=1)

    return train_df, test_df

# Data Augmentation
def augment_data(train_df):
    # Synthetic Data Generation using SMOTE
    X_train = train_df.drop(columns=['health_status'])
    y_train = train_df['health_status']

    smote = SMOTE()
    X_resampled, y_resampled = smote.fit_resample(X_train, y_train)

    train_df = pd.concat([pd.DataFrame(X_resampled, columns=X_train.columns), pd.Series(y_resampled, name='health_status')], axis=1)

    # Feature Perturbation
    numerical_features = train_df.select_dtypes(include=['float64', 'int64']).columns
    train_df[numerical_features] += np.random.normal(0, 0.1, train_df[numerical_features].shape)

    return train_df

# Model Training and Hyperparameter Tuning
def train_model(X_train, y_train, X_test, y_test):
    def objective(trial):
        params = {
            'n_d': trial.suggest_int('n_d', 8, 64),
            'n_a': trial.suggest_int('n_a', 8, 64),
            'gamma': trial.suggest_float('gamma', 1.0, 2.0),
            'n_steps': trial.suggest_int('n_steps', 3, 10),
            'learning_rate': trial.suggest_float('learning_rate', 0.001, 0.1),
            'batch_size': trial.suggest_categorical('batch_size', [32, 64, 128]),
            'max_epochs': trial.suggest_int('max_epochs', 50, 100),
        }

        model = TabNetClassifier(**params)
        model.fit(
            X_train=X_train.values, y_train=y_train.values,
            eval_set=[(X_train.values, y_train.values), (X_test.values, y_test.values)],
            eval_name=['train', 'valid'],
            eval_metric=['f1'],
            max_epochs=params['max_epochs'],
            patience=10,
            batch_size=params['batch_size'],
            virtual_batch_size=128,
            num_workers=0,
            drop_last=False
        )

        preds = model.predict(X_test.values)
        f1 = f1_score(y_test, preds, average='weighted')

        return f1

    study = optuna.create_study(direction='maximize')
    study.optimize(objective, n_trials=100)

    best_params = study.best_params
    best_f1 = study.best_value

    return best_params, best_f1

# Model Evaluation and Prediction
def evaluate_model(best_params, X_train, y_train, X_test, y_test, test_df):
    best_model = TabNetClassifier(**best_params)
    best_model.fit(
        X_train=X_train.values, y_train=y_train.values,
        eval_set=[(X_train.values, y_train.values), (X_test.values, y_test.values)],
        eval_name=['train', 'valid'],
        eval_metric=['f1'],
        max_epochs=best_params['max_epochs'],
        patience=10,
        batch_size=best_params['batch_size'],
        virtual_batch_size=128,
        num_workers=0,
        drop_last=False
    )

    test_preds = best_model.predict(test_df.drop(columns=['health_status']).values)

    # Save the predictions to a CSV file
    test_df['predicted_health_status'] = test_preds
    test_df[['id', 'predicted_health_status']].to_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/horse_health_outcomes/predictions.csv', index=False)

    # Calculate the F1 score on the test set
    test_f1 = f1_score(test_df['health_status'], test_preds, average='weighted')
    print(f'Test F1 Score: {test_f1}')

    # Save the trained model
    model_path = os.path.join('./agent_workspace/trained_models', 'tabnet_model.pkl')
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    best_model.save_model(model_path)

    return test_f1

# Main Function to Orchestrate the Entire Pipeline
def main():
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/horse_health_outcomes/train.csv'
    test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/horse_health_outcomes/test.csv'

    # Load data
    train_df, test_df = load_data(train_path, test_path)

    # Preprocess data
    train_df, test_df = preprocess_data(train_df, test_df)

    # Augment data
    train_df = augment_data(train_df)

    # Split data into features and target
    X_train = train_df.drop(columns=['health_status'])
    y_train = train_df['health_status']
    X_test = test_df.drop(columns=['health_status'])
    y_test = test_df['health_status']

    # Train model and perform hyperparameter tuning
    best_params, best_f1 = train_model(X_train, y_train, X_test, y_test)

    # Evaluate model and save predictions
    test_f1 = evaluate_model(best_params, X_train, y_train, X_test, y_test, test_df)

    return test_f1

if __name__ == "__main__":
    test_f1 = main()
    print(f"Final Test F1 Score: {test_f1}")
