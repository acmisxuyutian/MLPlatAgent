######代码######

import os
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.impute import SimpleImputer
from imblearn.over_sampling import SMOTE
from xgboost import XGBClassifier
from sklearn.metrics import roc_auc_score
import logging

# Set up logging
logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')

# Set random seed for reproducibility
SEED = 42
np.random.seed(SEED)

# Data Retrieval and Loading
def load_data(train_path, test_path):
    logging.info("Loading training and test datasets...")
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    
    logging.info("Training dataset shape: %s", train_df.shape)
    logging.info("Test dataset shape: %s", test_df.shape)
    
    return train_df, test_df

# Data Preprocessing
def preprocess_data(train_df, test_df):
    logging.info("Preprocessing data...")
    
    # Separate features and target
    X_train = train_df.drop(columns=['target'])
    y_train = train_df['target']
    X_test = test_df.drop(columns=['target'])
    y_test = test_df['target']
    
    # Feature scaling
    scaler = StandardScaler()
    X_train_scaled = scaler.fit_transform(X_train)
    X_test_scaled = scaler.transform(X_test)
    
    # Handling missing values
    imputer = SimpleImputer(strategy='mean')
    X_train_imputed = imputer.fit_transform(X_train_scaled)
    X_test_imputed = imputer.transform(X_test_scaled)
    
    # Data splitting
    X_train_final, X_val, y_train_final, y_val = train_test_split(X_train_imputed, y_train, test_size=0.2, random_state=SEED)
    
    logging.info("Data preprocessing completed.")
    
    return X_train_final, y_train_final, X_val, y_val, X_test_imputed, y_test

# Data Augmentation
def augment_data(X_train, y_train):
    logging.info("Performing data augmentation...")
    
    smote = SMOTE(random_state=SEED)
    X_train_augmented, y_train_augmented = smote.fit_resample(X_train, y_train)
    
    logging.info("Data augmentation completed.")
    
    return X_train_augmented, y_train_augmented

# Model Training
def train_model(X_train, y_train):
    logging.info("Training the XGBoost model...")
    
    model = XGBClassifier(
        learning_rate=0.1,
        booster='dart',
        max_depth=7,
        n_estimators=200,
        subsample=0.8,
        colsample_bytree=0.8,
        reg_alpha=0.1,
        reg_lambda=0.5,
        random_state=SEED
    )
    
    model.fit(X_train, y_train)
    
    logging.info("Model training completed.")
    
    return model

# Model Evaluation
def evaluate_model(model, X_val, y_val):
    logging.info("Evaluating the model...")
    
    y_val_pred_proba = model.predict_proba(X_val)[:, 1]
    auc_roc = roc_auc_score(y_val, y_val_pred_proba)
    
    logging.info("AUC-ROC on validation set: %f", auc_roc)
    
    return auc_roc

# Final Model Evaluation on Test Set
def final_evaluation(model, X_test, y_test):
    logging.info("Evaluating the final model on the test set...")
    
    y_test_pred_proba = model.predict_proba(X_test)[:, 1]
    auc_roc = roc_auc_score(y_test, y_test_pred_proba)
    
    logging.info("AUC-ROC on test set: %f", auc_roc)
    
    return auc_roc

# Main function to orchestrate the pipeline
def main():
    # Define dataset paths
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/smoker_status_Bio_Signals/train.csv'
    test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/smoker_status_Bio_Signals/test.csv'
    
    # Load data
    train_df, test_df = load_data(train_path, test_path)
    
    # Preprocess data
    X_train, y_train, X_val, y_val, X_test, y_test = preprocess_data(train_df, test_df)
    
    # Augment data
    X_train_augmented, y_train_augmented = augment_data(X_train, y_train)
    
    # Train model
    model = train_model(X_train_augmented, y_train_augmented)
    
    # Evaluate model on validation set
    val_auc_roc = evaluate_model(model, X_val, y_val)
    
    # Final evaluation on test set
    test_auc_roc = final_evaluation(model, X_test, y_test)
    
    # Save the trained model
    model_save_path = "./agent_workspace/trained_models/xgboost_smoker_status_model.json"
    os.makedirs(os.path.dirname(model_save_path), exist_ok=True)
    model.save_model(model_save_path)
    logging.info("Model saved to %s", model_save_path)
    
    return test_auc_roc

if __name__ == "__main__":
    test_auc_roc = main()
    print("Final AUC-ROC on Test Set:", test_auc_roc)
