import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.preprocessing import MinMaxScaler
from sklearn.model_selection import train_test_split, GridSearchCV, StratifiedKFold
from sklearn.metrics import roc_auc_score
from imblearn.over_sampling import SMOTE
from xgboost import XGBClassifier
import joblib

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data Retrieval and Loading
def load_data(train_path, test_path):
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    return train_df, test_df

# Data Preprocessing
def preprocess_data(train_df, test_df):
    # Handle missing values
    train_df.fillna(train_df.median(), inplace=True)
    test_df.fillna(test_df.median(), inplace=True)

    # Feature scaling
    scaler = MinMaxScaler()
    train_df_scaled = pd.DataFrame(scaler.fit_transform(train_df), columns=train_df.columns)
    test_df_scaled = pd.DataFrame(scaler.transform(test_df), columns=test_df.columns)

    # Feature encoding
    train_df_encoded = pd.get_dummies(train_df_scaled, drop_first=True)
    test_df_encoded = pd.get_dummies(test_df_scaled, drop_first=True)

    return train_df_encoded, test_df_encoded

# Data Augmentation
def apply_data_augmentation(X_train, y_train):
    smote = SMOTE(random_state=42)
    X_train_resampled, y_train_resampled = smote.fit_resample(X_train, y_train)
    return X_train_resampled, y_train_resampled

# Model Initialization and Configuration
def initialize_model():
    model = XGBClassifier(
        booster='dart',
        learning_rate=0.1,
        max_depth=5,
        subsample=0.9,
        colsample_bytree=0.9,
        reg_lambda=1.0,
        reg_alpha=0.1,
        objective='binary:logistic',
        random_state=42
    )
    return model

# Hyperparameter Tuning
def tune_hyperparameters(model, X_train, y_train):
    param_grid = {
        'learning_rate': [0.01, 0.05, 0.1, 0.2],
        'max_depth': [3, 5, 7, 9],
        'subsample': [0.5, 0.7, 0.9, 1.0],
        'colsample_bytree': [0.5, 0.7, 0.9, 1.0],
        'reg_lambda': [0.1, 1.0, 5.0, 10.0],
        'reg_alpha': [0.1, 1.0, 5.0, 10.0]
    }

    kfold = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
    grid_search = GridSearchCV(model, param_grid, scoring='roc_auc', cv=kfold, n_jobs=-1)
    grid_search.fit(X_train, y_train)

    best_params = grid_search.best_params_
    best_model = grid_search.best_estimator_

    return best_model, best_params

# Model Training
def train_model(model, X_train, y_train):
    model.fit(X_train, y_train, eval_metric='auc', early_stopping_rounds=10, verbose=True)
    return model

# Model Evaluation
def evaluate_model(model, X_test, y_test):
    y_pred_proba = model.predict_proba(X_test)[:, 1]
    auc_roc = roc_auc_score(y_test, y_pred_proba)
    return auc_roc

# Main function to orchestrate the entire pipeline
def main():
    # Step 1. Retrieve or load a dataset from the specified paths
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/smoker_status_Bio_Signals/train.csv'
    test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/smoker_status_Bio_Signals/test.csv'
    train_df, test_df = load_data(train_path, test_path)

    # Step 2. Preprocess the data
    train_df_encoded, test_df_encoded = preprocess_data(train_df, test_df)

    # Step 3. Split the data into features and target
    target_column = 'smoker_status'
    if target_column not in train_df_encoded.columns:
        raise ValueError(f"Target column '{target_column}' not found in the training dataset.")
    if target_column not in test_df_encoded.columns:
        raise ValueError(f"Target column '{target_column}' not found in the test dataset.")

    X_train = train_df_encoded.drop(columns=[target_column])
    y_train = train_df_encoded[target_column]
    X_test = test_df_encoded.drop(columns=[target_column])
    y_test = test_df_encoded[target_column]

    # Step 4. Apply data augmentation if necessary
    X_train_resampled, y_train_resampled = apply_data_augmentation(X_train, y_train)

    # Step 5. Initialize the model
    model = initialize_model()

    # Step 6. Tune hyperparameters
    best_model, best_params = tune_hyperparameters(model, X_train_resampled, y_train_resampled)

    # Step 7. Train the model
    best_model = train_model(best_model, X_train_resampled, y_train_resampled)

    # Step 8. Evaluate the model
    auc_roc = evaluate_model(best_model, X_test, y_test)
    print(f"AUC-ROC on Test Set: {auc_roc}")

    # Step 9. Save the trained model
    model_save_path = './agent_workspace/trained_models/xgboost_model.pkl'
    os.makedirs(os.path.dirname(model_save_path), exist_ok=True)
    joblib.dump(best_model, model_save_path)

    return auc_roc

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
