######代码######

import os
import pandas as pd
import numpy as np
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
from sklearn.model_selection import cross_val_score
from xgboost import XGBRegressor
from sklearn.metrics import mean_squared_log_error
import joblib

SEED = 42
np.random.seed(SEED)

# Step 1: Load data
def load_data():
    train_df = pd.read_csv(r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/media_campaign_cost/train.csv')
    test_df = pd.read_csv(r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/media_campaign_cost/test.csv')
    return train_df, test_df

# Step 2: Preprocess data
def preprocess_data(train_df, test_df):
    numerical_features = train_df.select_dtypes(include=['number']).columns
    categorical_features = train_df.select_dtypes(include=['object']).columns

    # Impute missing values
    numerical_imputer = SimpleImputer(strategy='median')
    categorical_imputer = SimpleImputer(strategy='most_frequent')

    if not numerical_features.empty:
        train_df[numerical_features] = numerical_imputer.fit_transform(train_df[numerical_features])
        test_df[numerical_features] = numerical_imputer.transform(test_df[numerical_features])

    if not categorical_features.empty:
        train_df[categorical_features] = categorical_imputer.fit_transform(train_df[categorical_features])
        test_df[categorical_features] = categorical_imputer.transform(test_df[categorical_features])

    # One-hot encoding for categorical features
    train_df = pd.get_dummies(train_df, columns=categorical_features)
    test_df = pd.get_dummies(test_df, columns=categorical_features)

    # Ensure the test set has the same columns as the training set
    missing_cols = set(train_df.columns) - set(test_df.columns)
    for col in missing_cols:
        test_df[col] = 0

    # Ensure the training set has the same columns as the test set
    missing_cols = set(test_df.columns) - set(train_df.columns)
    for col in missing_cols:
        train_df[col] = 0

    # Reorder the columns to match
    test_df = test_df[train_df.columns]

    # Feature scaling
    scaler = StandardScaler()
    train_df[numerical_features] = scaler.fit_transform(train_df[numerical_features])
    test_df[numerical_features] = scaler.transform(test_df[numerical_features])

    # Separate features and target variable
    X_train = train_df.drop(columns=['media_campaign_cost'], errors='ignore')
    y_train = train_df['media_campaign_cost'] if 'media_campaign_cost' in train_df.columns else None

    X_test = test_df.drop(columns=['media_campaign_cost'], errors='ignore')
    y_test = test_df['media_campaign_cost'] if 'media_campaign_cost' in test_df.columns else None

    return X_train, y_train, X_test, y_test

# Step 3: Train model
def train_model(X_train, y_train):
    if y_train is not None:
        xgb_model = XGBRegressor(
            n_estimators=200,
            max_depth=5,
            learning_rate=0.1,
            subsample=0.8,
            colsample_bytree=0.8,
            random_state=42
        )
        cv_scores = cross_val_score(xgb_model, X_train, y_train, cv=5, scoring='neg_mean_squared_log_error')
        print(f'Cross-Validation RMLSE: {(-cv_scores).mean()}')
        xgb_model.fit(X_train, y_train)
        return xgb_model
    else:
        print("Training dataset does not have the target variable, skipping model training.")
        return None

# Step 4: Evaluate model
def evaluate_model(xgb_model, X_test, y_test):
    if xgb_model is not None and y_test is not None:
        y_pred = xgb_model.predict(X_test)
        rmlse = mean_squared_log_error(y_test, y_pred, squared=False)
        print(f'Test RMLSE: {rmlse}')
        return rmlse, y_pred
    else:
        if xgb_model is None:
            print("Model is not trained, skipping evaluation.")
        if y_test is None:
            print("Test dataset does not have the target variable, skipping evaluation.")
        return None, xgb_model.predict(X_test) if xgb_model is not None else None

# Step 5: Save results
def save_results(y_pred, xgb_model):
    if y_pred is not None:
        predictions_df = pd.DataFrame({'media_campaign_cost': y_pred})
        predictions_df.to_csv(r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/media_campaign_cost/predictions.csv', index=False)

    if xgb_model is not None:
        model_save_path = r'./agent_workspace/trained_models/xgb_media_campaign_cost_model.pkl'
        os.makedirs(os.path.dirname(model_save_path), exist_ok=True)
        joblib.dump(xgb_model, model_save_path)

# Main function to orchestrate the pipeline
def main():
    # Step 1: Load data
    train_df, test_df = load_data()

    # Step 2: Preprocess data
    X_train, y_train, X_test, y_test = preprocess_data(train_df, test_df)

    # Step 3: Train model
    xgb_model = train_model(X_train, y_train)

    # Step 4: Evaluate model
    if y_test is not None:
        rmlse, y_pred = evaluate_model(xgb_model, X_test, y_test)
    else:
        rmlse, y_pred = None, xgb_model.predict(X_test) if xgb_model is not None else None

    # Step 5: Save results
    save_results(y_pred, xgb_model)

    return rmlse

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set (RMLSE):", model_performance)
