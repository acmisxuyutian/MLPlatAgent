import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.feature_selection import mutual_info_regression
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_log_error
import lightgbm as lgb
import optuna

# Set the random seed for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data Retrieval and Collection
def load_data():
    train_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/media_campaign_cost/train.csv'
    test_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/media_campaign_cost/test.csv'
    
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    
    return train_df, test_df

# Data Inspection
def inspect_data(train_df):
    print("Train Data Head:")
    print(train_df.head())
    print("Train Data Info:")
    print(train_df.info())
    print("Train Data Descriptive Statistics:")
    print(train_df.describe())
    print("Train Data Missing Values:")
    print(train_df.isnull().sum())

# Data Preprocessing
def preprocess_data(train_df, test_df):
    # Handling Missing Values
    numerical_features = train_df.select_dtypes(include=['number']).columns
    categorical_features = train_df.select_dtypes(include=['object']).columns

    if not categorical_features.empty:
        categorical_imputer = SimpleImputer(strategy='most_frequent')
        train_df[categorical_features] = categorical_imputer.fit_transform(train_df[categorical_features])
        test_df[categorical_features] = categorical_imputer.transform(test_df[categorical_features])

    if not numerical_features.empty:
        numerical_imputer = SimpleImputer(strategy='median')
        train_df[numerical_features] = numerical_imputer.fit_transform(train_df[numerical_features])
        test_df[numerical_features] = numerical_imputer.transform(test_df[numerical_features])

    # Feature Encoding
    if not categorical_features.empty:
        train_df = pd.get_dummies(train_df, columns=categorical_features, drop_first=True)
        test_df = pd.get_dummies(test_df, columns=categorical_features, drop_first=True)

    # Feature Scaling
    if not numerical_features.empty:
        scaler = StandardScaler()
        train_df[numerical_features] = scaler.fit_transform(train_df[numerical_features])
        test_df[numerical_features] = scaler.transform(test_df[numerical_features])

    # Feature Engineering
    if 'date' in train_df.columns:
        train_df['day_of_week'] = pd.to_datetime(train_df['date']).dt.dayofweek
        train_df['month'] = pd.to_datetime(train_df['date']).dt.month
        train_df['time_of_day'] = pd.to_datetime(train_df['date']).dt.hour

        test_df['day_of_week'] = pd.to_datetime(test_df['date']).dt.dayofweek
        test_df['month'] = pd.to_datetime(test_df['date']).dt.month
        test_df['time_of_day'] = pd.to_datetime(test_df['date']).dt.hour

    if 'feature_1' in train_df.columns and 'feature_2' in train_df.columns:
        train_df['interaction_feature_1'] = train_df['feature_1'] * train_df['feature_2']
        test_df['interaction_feature_1'] = test_df['feature_1'] * test_df['feature_2']

    # Feature Selection
    target_column = 'media_campaign_cost'
    if target_column not in train_df.columns:
        raise ValueError(f"The target column '{target_column}' is not found in the training dataset.")

    X_train = train_df.drop(columns=[target_column])
    y_train = train_df[target_column]

    if not X_train.empty:
        mutual_info = mutual_info_regression(X_train, y_train)
        feature_importance = pd.Series(mutual_info, index=X_train.columns)
        selected_features = feature_importance.nlargest(10).index  # Select top 10 features

        X_train = X_train[selected_features]
        X_test = test_df[selected_features]

    return X_train, y_train, X_test

# Model Training and Optimization
def train_model(X_train, y_train):
    # Data Splitting
    X_train_split, X_val, y_train_split, y_val = train_test_split(X_train, y_train, test_size=0.2, random_state=42)

    # Model Training
    params = {
        'objective': 'regression',
        'metric': 'rmse',
        'n_estimators': 250,
        'max_depth': 6,
        'learning_rate': 0.05
    }

    model = lgb.LGBMRegressor(**params)
    model.fit(X_train_split, y_train_split, eval_set=[(X_val, y_val)], early_stopping_rounds=10, verbose=10)

    # Hyperparameter Tuning
    def objective(trial):
        params = {
            'objective': 'regression',
            'metric': 'rmse',
            'n_estimators': trial.suggest_int('n_estimators', 100, 300),
            'max_depth': trial.suggest_int('max_depth', 3, 7),
            'learning_rate': trial.suggest_loguniform('learning_rate', 0.01, 0.2)
        }

        model = lgb.LGBMRegressor(**params)
        model.fit(X_train_split, y_train_split, eval_set=[(X_val, y_val)], early_stopping_rounds=10, verbose=0)
        y_pred = model.predict(X_val)
        rmlse = np.sqrt(mean_squared_log_error(y_val, y_pred))
        return rmlse

    study = optuna.create_study(direction='minimize')
    study.optimize(objective, n_trials=50)

    best_params = study.best_params
    best_model = lgb.LGBMRegressor(**best_params)
    best_model.fit(X_train, y_train)

    return best_model

# Model Evaluation and Prediction
def evaluate_model(model, X_test, y_test):
    y_pred = model.predict(X_test)
    rmlse = np.sqrt(mean_squared_log_error(y_test, y_pred))
    print(f'RMLSE on test set: {rmlse}')
    return rmlse

def save_predictions_and_model(model, y_pred, rmlse):
    predictions = pd.DataFrame({'media_campaign_cost': y_pred})
    predictions.to_csv(r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/media_campaign_cost/predictions.csv', index=False)

    with open(r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/media_campaign_cost/rmlse_score.txt', 'w') as f:
        f.write(f'RMLSE: {rmlse}')

    model_path = './agent_workspace/trained_models/lightgbm_model.pkl'
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    model.booster_.save_model(model_path)

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_df, test_df = load_data()
    inspect_data(train_df)

    # Step 2. Preprocess the data
    X_train, y_train, X_test = preprocess_data(train_df, test_df)

    # Step 3. Train the model
    model = train_model(X_train, y_train)

    # Step 4. Evaluate the model
    y_test = test_df['media_campaign_cost']
    rmlse = evaluate_model(model, X_test, y_test)

    # Step 5. Save predictions and the trained model
    y_pred = model.predict(X_test)
    save_predictions_and_model(model, y_pred, rmlse)

    return rmlse

if __name__ == "__main__":
    rmlse = main()
    print(f"Model Performance on Test Set: RMLSE = {rmlse}")
