import os
import pandas as pd
import numpy as np
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
from sklearn.model_selection import KFold, GridSearchCV
from catboost import CatBoostRegressor
from sklearn.metrics import mean_squared_log_error
import joblib

# Set random seed for reproducibility
SEED = 42
np.random.seed(SEED)

# Data Retrieval and Loading
def load_data():
    train_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/media_campaign_cost/train.csv')
    test_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/media_campaign_cost/test.csv')
    return train_df, test_df

# Data Preprocessing
def preprocess_data(train_df, test_df):
    # Handling Missing Values
    numerical_features = train_df.select_dtypes(include=['number']).columns.tolist()
    categorical_features = train_df.select_dypes(include=['object']).columns.tolist()

    if not numerical_features:
        numerical_imputer = None
    else:
        numerical_imputer = SimpleImputer(strategy='median')

    if not categorical_features:
        categorical_imputer = None
    else:
        categorical_imputer = SimpleImputer(strategy='most_frequent')

    if numerical_imputer is not None:
        train_df[numerical_features] = numerical_imputer.fit_transform(train_df[numerical_features])
        test_df[numerical_features] = numerical_imputer.transform(test_df[numerical_features])

    if categorical_imputer is not None:
        train_df[categorical_features] = categorical_imputer.fit_transform(train_df[categorical_features])
        test_df[categorical_features] = categorical_imputer.transform(test_df[categorical_features])

    # Categorical Feature Encoding
    preprocessor = ColumnTransformer(
        transformers=[
            ('num', StandardScaler(), numerical_features) if numerical_features else ('num', 'passthrough', []),
            ('cat', OneHotEncoder(handle_unknown='ignore'), categorical_features) if categorical_features else ('cat', 'passthrough', [])
        ]
    )

    # Ensure the target variable is correctly identified
    if 'media_campaign_cost' in train_df.columns:
        X_train = train_df.drop(columns=['media_campaign_cost'])
        y_train = train_df['media_campaign_cost']
    else:
        raise KeyError("The target variable 'media_campaign_cost' is not found in the training dataset.")

    X_test = test_df.drop(columns=['media_campaign_cost'], errors='ignore')
    y_test = test_df['media_campaign_cost'] if 'media_campaign_cost' in test_df.columns else None

    X_train = preprocessor.fit_transform(X_train)
    X_test = preprocessor.transform(X_test)

    return X_train, y_train, X_test, y_test, preprocessor

# Model Training and Hyperparameter Tuning
def train_model(X_train, y_train):
    model = CatBoostRegressor(random_state=SEED, verbose=0)
    param_grid = {
        'learning_rate': [0.01, 0.05, 0.1],
        'depth': [4, 6, 8],
        'n_estimators': [100, 500, 1000],
        'l2_leaf_reg': [1, 3, 5]
    }

    kf = KFold(n_splits=5, shuffle=True, random_state=SEED)
    grid_search = GridSearchCV(model, param_grid, cv=kf, scoring='neg_mean_squared_log_error', n_jobs=-1)
    grid_search.fit(X_train, y_train)

    best_model = grid_search.best_estimator_
    return best_model

# Model Evaluation and Prediction
def evaluate_model(model, X_test, y_test):
    if y_test is not None:
        y_pred = model.predict(X_test)
        rmlse = mean_squared_log_error(y_test, y_pred, squared=False)
        print(f"RMLSE: {rmlse}")
    else:
        print("Test data not available for evaluation.")
        y_pred = model.predict(X_test)
        rmlse = None

    return rmlse, y_pred

# Saving Model and Predictions
def save_model_and_predictions(model, y_pred, preprocessor):
    model_path = './agent_workspace/trained_models/catboost_model.pkl'
    preprocessor_path = './agent_workspace/trained_models/preprocessor.pkl'
    predictions_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/media_campaign_cost/predictions.csv'

    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    os.makedirs(os.path.dirname(preprocessor_path), exist_ok=True)

    joblib.dump(model, model_path)
    joblib.dump(preprocessor, preprocessor_path)

    predictions_df = pd.DataFrame({'media_campaign_cost': y_pred})
    predictions_df.to_csv(predictions_path, index=False)

# Main Function to Orchestrate the Pipeline
def main():
    # Step 1: Load the data
    train_df, test_df = load_data()

    # Step 2: Preprocess the data
    X_train, y_train, X_test, y_test, preprocessor = preprocess_data(train_df, test_df)

    # Step 3: Train the model
    best_model = train_model(X_train, y_train)

    # Step 4: Evaluate the model
    rmlse, y_pred = evaluate_model(best_model, X_test, y_test)

    # Step 5: Save the model and predictions
    save_model_and_predictions(best_model, y_pred, preprocessor)

if __name__ == "__main__":
    main()
