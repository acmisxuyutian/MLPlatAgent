import os
import random
import time
import json
import numpy as np
import pandas as pd
from scipy import stats
from sklearn.model_selection import KFold, cross_val_score, GridSearchCV
from sklearn.metrics import mean_squared_log_error
from sklearn.preprocessing import StandardScaler, LabelEncoder
from xgboost import XGBRegressor

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data retrieval and initial inspection
def load_data():
    train_df = pd.read_csv(r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/media_campaign_cost/train.csv')
    test_df = pd.read_csv(r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/media_campaign_cost/test.csv')
    
    # Initial inspection
    print("Training dataset shape:", train_df.shape)
    print("Test dataset shape:", test_df.shape)
    print("Training dataset data types:\n", train_df.dtypes)
    print("Test dataset data types:\n", test_df.dtypes)
    print("Training dataset summary statistics:\n", train_df.describe())
    print("Test dataset summary statistics:\n", test_df.describe())
    print("Training dataset missing values:\n", train_df.isnull().sum())
    print("Test dataset missing values:\n", test_df.isnull().sum())
    
    # Outliers detection
    z_scores = np.abs(stats.zscore(train_df.select_dtypes(include=[np.number])))
    train_df = train_df[(z_scores < 3).all(axis=1)]
    
    # Inconsistencies check
    if 'media_campaign_cost' in train_df.columns:
        train_df['media_campaign_cost'] = train_df['media_campaign_cost'].apply(lambda x: max(x, 1))
    else:
        raise KeyError("The target variable 'media_campaign_cost' is not found in the training dataset.")
    
    return train_df, test_df

# Data cleaning
def clean_data(train_df, test_df):
    # Handling missing values
    train_df.fillna(train_df.mean(), inplace=True)
    test_df.fillna(test_df.mean(), inplace=True)
    
    # Remove columns with a high percentage of missing values
    train_df.dropna(thresh=len(train_df) * 0.5, axis=1, inplace=True)
    test_df.dropna(thresh=len(test_df) * 0.5, axis=1, inplace=True)
    
    # Handling outliers
    z_scores = np.abs(stats.zscore(train_df.select_dtypes(include=[np.number])))
    train_df = train_df[(z_scores < 3).all(axis=1)]
    
    # Consistency check
    if 'media_campaign_cost' in train_df.columns:
        train_df['media_campaign_cost'] = train_df['media_campaign_cost'].apply(lambda x: max(x, 1))
    else:
        raise KeyError("The target variable 'media_campaign_cost' is not found in the training dataset.")
    
    return train_df, test_df

# Feature engineering
def feature_engineering(train_df, test_df):
    # Interaction terms
    if 'feature1' in train_df.columns and 'feature2' in train_df.columns:
        train_df['interaction_term'] = train_df['feature1'] * train_df['feature2']
        test_df['interaction_term'] = test_df['feature1'] * test_df['feature2']
    
    # Categorical variables
    categorical_features = [col for col in train_df.columns if train_df[col].dtype == 'object']
    train_df = pd.get_dummies(train_df, columns=categorical_features)
    test_df = pd.get_dummies(test_df, columns=categorical_features)
    
    # Feature scaling
    scaler = StandardScaler()
    numerical_features = train_df.select_dtypes(include=[np.number]).columns
    train_df[numerical_features] = scaler.fit_transform(train_df[numerical_features])
    test_df[numerical_features] = scaler.transform(test_df[numerical_features])
    
    # Feature selection
    corr_matrix = train_df.corr().abs()
    upper = corr_matrix.where(np.triu(np.ones(corr_matrix.shape), k=1).astype(bool))
    to_drop = [column for column in upper.columns if any(upper[column] > 0.8)]
    train_df.drop(columns=to_drop, inplace=True)
    test_df.drop(columns=to_drop, inplace=True)
    
    return train_df, test_df

# Model selection and hyperparameter tuning
def select_and_tune_model(train_df):
    kf = KFold(n_splits=5, shuffle=True, random_state=42)
    model = XGBRegressor()
    scores = cross_val_score(model, train_df.drop(columns=['media_campaign_cost']), train_df['media_campaign_cost'], cv=kf, scoring='neg_mean_squared_log_error')
    print(f'Cross-Validation RMLSE: {np.sqrt(-scores.mean())}')
    
    param_grid = {
        'n_estimators': [100, 200, 300],
        'max_depth': [3, 4, 5, 6],
        'learning_rate': [0.01, 0.1, 0.2],
        'subsample': [0.6, 0.8, 1.0],
        'colsample_bytree': [0.6, 0.8, 1.0]
    }
    
    grid_search = GridSearchCV(estimator=XGBRegressor(), param_grid=param_grid, cv=kf, scoring='neg_mean_squared_log_error', n_jobs=-1)
    grid_search.fit(train_df.drop(columns=['media_campaign_cost']), train_df['media_campaign_cost'])
    
    best_params = grid_search.best_params_
    print(f'Best Hyperparameters: {best_params}')
    
    return best_params

# Model training and evaluation
def train_and_evaluate_model(train_df, test_df, best_params):
    best_model = XGBRegressor(**best_params)
    best_model.fit(train_df.drop(columns=['media_campaign_cost']), train_df['media_campaign_cost'])
    
    y_pred = best_model.predict(test_df.drop(columns=['media_campaign_cost']))
    rmlse = np.sqrt(mean_squared_log_error(test_df['media_campaign_cost'], y_pred))
    print(f'Test RMLSE: {rmlse}')
    
    return best_model, rmlse

# Final predictions and reporting
def save_predictions_and_report(test_df, y_pred, rmlse, best_params, best_model):
    test_df['predicted_media_campaign_cost'] = y_pred
    test_df[['predicted_media_campaign_cost']].to_csv(r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/media_campaign_cost/predictions.csv', index=False)
    
    with open(r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/media_campaign_cost/report.txt', 'w') as f:
        f.write(f'Test RMLSE: {rmlse}\n')
        f.write(f'Best Hyperparameters: {best_params}\n')
        f.write(f'Feature Engineering Steps: \n')
        f.write(f'1. Interaction Terms: feature1 * feature2\n')
        f.write(f'2. One-Hot Encoding: categorical_features\n')
        f.write(f'3. Standard Scaling: numerical_features\n')
        f.write(f'4. Feature Selection: mutual_info_regression\n')
    
    model_path = r'./agent_workspace/trained_models/best_model.pkl'
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    best_model.save_model(model_path)

# Main function to orchestrate the entire pipeline
def main():
    # Step 1: Load data
    train_df, test_df = load_data()
    
    # Step 2: Clean data
    train_df, test_df = clean_data(train_df, test_df)
    
    # Step 3: Feature engineering
    train_df, test_df = feature_engineering(train_df, test_df)
    
    # Step 4: Model selection and hyperparameter tuning
    best_params = select_and_tune_model(train_df)
    
    # Step 5: Train and evaluate model
    best_model, rmlse = train_and_evaluate_model(train_df, test_df, best_params)
    
    # Step 6: Save predictions and report
    save_predictions_and_report(test_df, best_model.predict(test_df.drop(columns=['media_campaign_cost'])), rmlse, best_params, best_model)
    
    return rmlse

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
