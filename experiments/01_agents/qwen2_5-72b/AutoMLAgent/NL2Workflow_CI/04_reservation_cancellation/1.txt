import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
from imblearn.over_sampling import SMOTE
from lightgbm import LGBMClassifier
from sklearn.model_selection import GridSearchCV
from sklearn.metrics import roc_auc_score, confusion_matrix, classification_report
from joblib import dump

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Step 1. Load datasets
def load_datasets(train_path, test_path):
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    return train_df, test_df

# Step 2. Preprocess data
def preprocess_data(train_df, test_df):
    # EDA
    print(train_df.describe())
    print(train_df.isnull().sum())

    # Ensure 'is_canceled' is present in the training dataset
    if 'is_canceled' not in train_df.columns:
        raise KeyError("The column 'is_canceled' is not present in the training dataset.")

    # Handle missing values
    numerical_features = train_df.select_dtypes(include=[np.number]).columns.tolist()
    categorical_features = train_df.select_dtypes(include=[object]).columns.tolist()

    numerical_transformer = SimpleImputer(strategy='mean')
    categorical_transformer = SimpleImputer(strategy='most_frequent')

    preprocessor = ColumnTransformer(
        transformers=[
            ('num', numerical_transformer, numerical_features),
            ('cat', categorical_transformer, categorical_features)
        ]
    )

    train_df = pd.DataFrame(preprocessor.fit_transform(train_df), columns=numerical_features + categorical_features)
    test_df = pd.DataFrame(preprocessor.transform(test_df), columns=numerical_features + categorical_features)

    # Feature scaling
    scaler = StandardScaler()
    train_df[numerical_features] = scaler.fit_transform(train_df[numerical_features])
    test_df[numerical_features] = scaler.transform(test_df[numerical_features])

    # Feature engineering
    if 'checkin_date' in train_df.columns and 'booking_date' in train_df.columns:
        train_df['booking_lead_time'] = (pd.to_datetime(train_df['checkin_date']) - pd.to_datetime(train_df['booking_date'])).dt.days
    if 'checkin_date' in test_df.columns and 'booking_date' in test_df.columns:
        test_df['booking_lead_time'] = (pd.to_datetime(test_df['checkin_date']) - pd.to_datetime(test_df['booking_date'])).dt.days

    # One-hot encoding
    encoder = OneHotEncoder(handle_unknown='ignore')
    encoded_train = encoder.fit_transform(train_df[categorical_features])
    encoded_test = encoder.transform(test_df[categorical_features])

    encoded_train_df = pd.DataFrame(encoded_train.toarray(), columns=encoder.get_feature_names_out(categorical_features))
    encoded_test_df = pd.DataFrame(encoded_test.toarray(), columns=encoder.get_feature_names_out(categorical_features))

    train_df = pd.concat([train_df, encoded_train_df], axis=1)
    test_df = pd.concat([test_df, encoded_test_df], axis=1)

    train_df.drop(categorical_features, axis=1, inplace=True)
    test_df.drop(categorical_features, axis=1, inplace=True)

    # Handle class imbalance
    X_train = train_df.drop('is_canceled', axis=1)
    y_train = train_df['is_canceled']

    smote = SMOTE()
    X_train_resampled, y_train_resampled = smote.fit_resample(X_train, y_train)

    return X_train_resampled, y_train_resampled, test_df

# Step 3. Train model
def train_model(X_train_resampled, y_train_resampled):
    param_grid = {
        'learning_rate': [0.01, 0.1, 0.2],
        'num_leaves': [31, 50, 100],
        'max_depth': [-1, 5, 10],
        'min_child_samples': [20, 50, 100],
        'subsample': [0.5, 0.7, 1.0]
    }

    model = LGBMClassifier(class_weight='balanced')
    grid_search = GridSearchCV(model, param_grid, cv=5, scoring='roc_auc')
    grid_search.fit(X_train_resampled, y_train_resampled)
    best_params = grid_search.best_params_

    best_model = LGBMClassifier(**best_params, class_weight='balanced')
    best_model.fit(X_train_resampled, y_train_resampled)

    return best_model

# Step 4. Evaluate model
def evaluate_model(best_model, X_test, y_test):
    y_pred_proba = best_model.predict_proba(X_test)[:, 1]
    auc_roc_score = roc_auc_score(y_test, y_pred_proba)
    print(f"AUC-ROC Score: {auc_roc_score}")

    y_pred = (y_pred_proba > 0.5).astype(int)
    confusion_matrix_result = confusion_matrix(y_test, y_pred)
    classification_report_result = classification_report(y_test, y_pred)

    print("Confusion Matrix:")
    print(confusion_matrix_result)
    print("Classification Report:")
    print(classification_report_result)

    performance_scores = {
        'AUC-ROC': auc_roc_score,
        'Confusion Matrix': confusion_matrix_result,
        'Classification Report': classification_report_result
    }

    return performance_scores

# Step 5. Save prediction results and model
def save_predictions_and_model(best_model, y_pred_proba, test_df):
    predictions_df = pd.DataFrame({'id': test_df['id'], 'is_canceled': y_pred_proba})
    predictions_df.to_csv('/remote-home/cs_acmis_xyt/AutomlAgent/agent_workspace/trained_models/predictions.csv', index=False)

    model_save_path = '/remote-home/cs_acmis_xyt/AutomlAgent/agent_workspace/trained_models/lightgbm_model.pkl'
    os.makedirs(os.path.dirname(model_save_path), exist_ok=True)
    dump(best_model, model_save_path)

# Main function to orchestrate the pipeline
def main():
    # Step 1. Load datasets
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/reservation_cancellation/train.csv'
    test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/reservation_cancellation/test.csv'

    train_df, test_df = load_datasets(train_path, test_path)

    # Step 2. Preprocess data
    X_train_resampled, y_train_resampled, test_df = preprocess_data(train_df, test_df)

    # Step 3. Train model
    best_model = train_model(X_train_resampled, y_train_resampled)

    # Step 4. Evaluate model
    if 'is_canceled' in test_df.columns:
        X_test = test_df.drop('is_canceled', axis=1, errors='ignore')
        y_test = test_df['is_canceled']
    else:
        raise KeyError("The column 'is_canceled' is not present in the test dataset.")

    model_performance = evaluate_model(best_model, X_test, y_test)

    # Step 5. Save prediction results and model
    y_pred_proba = best_model.predict_proba(X_test)[:, 1]
    save_predictions_and_model(best_model, y_pred_proba, test_df)

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
