import os
import random
import time
import json
import numpy as np
import pandas as pd
import lightgbm as lgb
from sklearn.model_selection import train_test_split, GridSearchCV, cross_val_score
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.pipeline import Pipeline

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data Retrieval
def load_data(train_path, test_path):
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    return train_df, test_df

# Data Preprocessing
def preprocess_data(train_df, test_df):
    # Handle missing values
    numerical_features = train_df.select_dtypes(include=[np.number]).columns
    categorical_features = train_df.select_dypes(include=[object]).columns

    numerical_imputer = SimpleImputer(strategy='mean')
    categorical_imputer = SimpleImputer(strategy='most_frequent')

    train_df[numerical_features] = numerical_imputer.fit_transform(train_df[numerical_features])
    train_df[categorical_features] = categorical_imputer.fit_transform(train_df[categorical_features].astype(str))

    test_df[numerical_features] = numerical_imputer.transform(test_df[numerical_features])
    test_df[categorical_features] = categorical_imputer.transform(test_df[categorical_features].astype(str))

    # Remove outliers
    def remove_outliers(df, columns):
        for col in columns:
            Q1 = df[col].quantile(0.25)
            Q3 = df[col].quantile(0.75)
            IQR = Q3 - Q1
            lower_bound = Q1 - 1.5 * IQR
            upper_bound = Q3 + 1.5 * IQR
            df = df[(df[col] >= lower_bound) & (df[col] <= upper_bound)]
        return df

    train_df = remove_outliers(train_df, ['lead_time', 'stays_in_weekend_nights', 'stays_in_week_nights'])

    # Feature Engineering
    train_df['length_of_stay'] = (pd.to_datetime(train_df['checkout_date']) - pd.to_datetime(train_df['checkin_date'])).dt.days
    train_df['booking_lead_time'] = (pd.to_datetime(train_df['checkin_date']) - pd.to_datetime(train_df['booking_date'])).dt.days
    train_df['day_of_week'] = pd.to_datetime(train_df['checkin_date']).dt.dayofweek
    train_df['month'] = pd.to_datetime(train_df['checkin_date']).dt.month

    test_df['length_of_stay'] = (pd.to_datetime(test_df['checkout_date']) - pd.to_datetime(test_df['checkin_date'])).dt.days
    test_df['booking_lead_time'] = (pd.to_datetime(test_df['checkin_date']) - pd.to_datetime(test_df['booking_date'])).dt.days
    test_df['day_of_week'] = pd.to_datetime(test_df['checkin_date']).dt.dayofweek
    test_df['month'] = pd.to_datetime(test_df['checkin_date']).dt.month

    # Convert categorical variables to numerical formats
    categorical_features = ['hotel_type', 'room_type', 'meal_plan', 'market_segment', 'distribution_channel', 'customer_type']
    preprocessor = ColumnTransformer(
        transformers=[
            ('num', StandardScaler(), numerical_features),
            ('cat', OneHotEncoder(handle_unknown='ignore'), categorical_features)
        ]
    )

    X_train = preprocessor.fit_transform(train_df.drop(columns=['is_canceled']))
    y_train = train_df['is_canceled']

    X_test = preprocessor.transform(test_df.drop(columns=['is_canceled']))

    return X_train, y_train, X_test, preprocessor

# Handle Class Imbalance
def handle_class_imbalance(X_train, y_train):
    smote = SMOTE()
    X_train_resampled, y_train_resampled = smote.fit_resample(X_train, y_train)
    return X_train_resampled, y_train_resampled

# Train the LightGBM Model
def train_model(X_train, y_train):
    params = {
        'objective': 'binary',
        'metric': 'auc',
        'boosting_type': 'gbdt',
        'learning_rate': 0.01,
        'num_leaves': 31,
        'max_depth': 10,
        'min_child_samples': 20,
        'subsample': 0.8,
        'colsample_bytree': 0.8,
        'class_weight': {0: 1, 1: 2}
    }

    lgb_train = lgb.Dataset(X_train, y_train)
    model = lgb.train(params, lgb_train, num_boost_round=1000, early_stopping_rounds=100, valid_sets=[lgb_train], valid_names=['train'])
    return model

# Evaluate the Model
def evaluate_model(model, X_test, y_test):
    y_pred = model.predict(X_test)
    y_pred_binary = (y_pred > 0.5).astype(int)
    auc_roc = roc_auc_score(y_test, y_pred)
    performance_scores = {
        'AUC-ROC': auc_roc
    }
    return performance_scores

# Analyze the Model
def analyze_model(model, preprocessor, X_train):
    feature_importance = pd.DataFrame({
        'Feature': preprocessor.get_feature_names_out(),
        'Importance': model.feature_importance()
    }).sort_values(by='Importance', ascending=False)

    print(feature_importance)

    explainer = shap.TreeExplainer(model)
    shap_values = explainer.shap_values(X_train)
    shap.summary_plot(shap_values, X_train, feature_names=preprocessor.get_feature_names_out())

# Main Function to Orchestrate the Pipeline
def main():
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/reservation_cancellation/train.csv'
    test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/reservation_cancellation/test.csv'

    # Load data
    train_df, test_df = load_data(train_path, test_path)

    # Preprocess data
    X_train, y_train, X_test, preprocessor = preprocess_data(train_df, test_df)

    # Handle class imbalance
    X_train_resampled, y_train_resampled = handle_class_imbalance(X_train, y_train)

    # Train model
    model = train_model(X_train_resampled, y_train_resampled)

    # Evaluate model
    performance_scores = evaluate_model(model, X_test, test_df['is_canceled'])

    # Save predictions
    y_pred_binary = (model.predict(X_test) > 0.5).astype(int)
    test_df['predicted_is_canceled'] = y_pred_binary
    test_df[['reservation_id', 'predicted_is_canceled']].to_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/reservation_cancellation/predictions.csv', index=False)

    # Analyze model
    analyze_model(model, preprocessor, X_train_resampled)

    # Save the trained model
    model_path = './agent_workspace/trained_models/lightgbm_model.pkl'
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    model.save_model(model_path)

    return performance_scores

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
