import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split, GridSearchCV
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.compose import ColumnTransformer
from sklearn.metrics import roc_auc_score
import lightgbm as lgb
import joblib

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data Loading and Initial Exploration
def load_datasets(train_path, test_path):
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    return train_df, test_df

def preprocess_data(train_df, test_df):
    # Initial Data Exploration
    print("Training Data Missing Values:\n", train_df.isnull().sum())
    print("Testing Data Missing Values:\n", test_df.isnull().sum())
    
    print("Training Data Summary Statistics:\n", train_df.describe())
    if 'O' in test_df.dtypes.values:
        print("Testing Data Summary Statistics:\n", test_df.describe(include=['O']))
    else:
        print("No categorical features in the testing data.")
    
    # Ensure the target variable is present in the training data
    if 'is_canceled' not in train_df.columns:
        raise KeyError("The target variable 'is_canceled' is not present in the training dataset.")
    
    print("Target Variable Distribution:\n", train_df['is_canceled'].value_counts(normalize=True))

    # Handle missing values
    train_df.fillna(train_df.mean(), inplace=True)
    train_df.fillna(train_df.mode().iloc[0], inplace=True)
    test_df.fillna(test_df.mean(), inplace=True)
    test_df.fillna(test_df.mode().iloc[0], inplace=True)

    # Feature Identification and Creation
    relevant_features = [
        'booking_lead_time', 'customer_type', 'previous_cancellations', 
        'number_of_previous_bookings', 'average_booking_duration', 'time_since_last_booking'
    ]
    
    # Create new features
    if 'customer_id' in train_df.columns:
        train_df['num_previous_bookings'] = train_df.groupby('customer_id')['customer_id'].transform('count')
        train_df['avg_booking_duration'] = train_df.groupby('customer_id')['booking_duration'].transform('mean')
        train_df['time_since_last_booking'] = train_df.groupby('customer_id')['booking_date'].transform(lambda x: (x.max() - x).dt.days)
    
    if 'customer_id' in test_df.columns:
        test_df['num_previous_bookings'] = test_df.groupby('customer_id')['customer_id'].transform('count')
        test_df['avg_booking_duration'] = test_df.groupby('customer_id')['booking_duration'].transform('mean')
        test_df['time_since_last_booking'] = test_df.groupby('customer_id')['booking_date'].transform(lambda x: (x.max() - x).dt.days)
    
    # Feature Encoding
    categorical_features = ['customer_type']
    numerical_features = [col for col in relevant_features if col not in categorical_features]
    
    preprocessor = ColumnTransformer(
        transformers=[
            ('num', StandardScaler(), numerical_features),
            ('cat', OneHotEncoder(), categorical_features)
        ]
    )
    
    # Apply preprocessing
    X_train = train_df[relevant_features]
    y_train = train_df['is_canceled']
    X_test = test_df[relevant_features]
    
    X_train_processed = preprocessor.fit_transform(X_train)
    X_test_processed = preprocessor.transform(X_test)
    
    return X_train_processed, y_train, X_test_processed, preprocessor

# Model Training and Evaluation
def train_model(X_train, y_train):
    # Initialize LightGBM Model
    model = lgb.LGBMClassifier(random_state=SEED)
    
    # Define Hyperparameter Search Space
    param_grid = {
        'learning_rate': [0.01, 0.05, 0.1],
        'num_leaves': [31, 50, 100],
        'max_depth': [5, 10, 15],
        'n_estimators': [100, 200, 300]
    }
    
    # Perform Grid Search
    grid_search = GridSearchCV(model, param_grid, cv=5, scoring='roc_auc', n_jobs=-1)
    grid_search.fit(X_train, y_train)
    
    # Train the Model with Best Hyperparameters
    best_model = grid_search.best_estimator_
    
    return best_model

def evaluate_model(model, X_val, y_val):
    y_pred_prob = model.predict_proba(X_val)[:, 1]
    auc_roc = roc_auc_score(y_val, y_pred_prob)
    return auc_roc

def generate_predictions(model, X_test, preprocessor, test_df):
    X_test_processed = preprocessor.transform(X_test)
    y_pred_prob = model.predict_proba(X_test_processed)[:, 1]
    y_pred = (y_pred_prob > 0.5).astype(int)
    
    predictions = pd.DataFrame({
        'customer_id': test_df['customer_id'],
        'is_canceled': y_pred,
        'is_canceled_prob': y_pred_prob
    })
    
    return predictions

def save_results(predictions, model, model_path, predictions_path):
    predictions.to_csv(predictions_path, index=False)
    joblib.dump(model, model_path)

# Main Function
def main():
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/reservation_cancellation/train.csv'
    test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/reservation_cancellation/test.csv'
    
    # Load datasets
    train_df, test_df = load_datasets(train_path, test_path)
    
    # Preprocess data
    X_train, y_train, X_test, preprocessor = preprocess_data(train_df, test_df)
    
    # Split training data into training and validation sets
    X_train, X_val, y_train, y_val = train_test_split(X_train, y_train, test_size=0.2, random_state=SEED)
    
    # Train the model
    model = train_model(X_train, y_train)
    
    # Evaluate the model
    auc_roc = evaluate_model(model, X_val, y_val)
    print(f"AUC-ROC on Validation Set: {auc_roc:.4f}")
    
    # Generate predictions on the test set
    predictions = generate_predictions(model, test_df[relevant_features], preprocessor, test_df)
    
    # Save results
    model_path = './agent_workspace/trained_models/lightgbm_reservation_cancellation_model.pkl'
    predictions_path = './agent_workspace/predictions/lightgbm_reservation_cancellation_predictions.csv'
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    os.makedirs(os.path.dirname(predictions_path), exist_ok=True)
    save_results(predictions, model, model_path, predictions_path)
    
    return auc_roc

if __name__ == "__main__":
    auc_roc = main()
    print(f"Model Performance on Test Set: AUC-ROC = {auc_roc:.4f}")
