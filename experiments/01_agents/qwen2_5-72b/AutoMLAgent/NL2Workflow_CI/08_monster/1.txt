######代码######

import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
from sklearn.model_selection import train_test_split, RandomizedSearchCV
from imblearn.over_sampling import SMOTE
from xgboost import XGBClassifier
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score, classification_report
import pickle

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data Retrieval and Loading
def load_data(train_path, test_path):
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    return train_df, test_df

# Data Preprocessing
def preprocess_data(train_df, test_df):
    # Ensure 'monster_type' is in the columns
    if 'monster_type' not in train_df.columns:
        raise ValueError("The 'monster_type' column is missing in the training dataset.")
    if 'monster_type' not in test_df.columns:
        raise ValueError("The 'monster_type' column is missing in the test dataset.")

    # Identify numerical and categorical columns
    numerical_features = train_df.select_dtypes(include=['int64', 'float64']).columns
    categorical_features = train_df.select_dtypes(include=['object']).columns

    # Define preprocessing for numerical data
    numerical_transformer = Pipeline(steps=[
        ('imputer', SimpleImputer(strategy='mean')),
        ('scaler', StandardScaler())
    ])

    # Define preprocessing for categorical data
    categorical_transformer = Pipeline(steps=[
        ('imputer', SimpleImputer(strategy='most_frequent')),
        ('onehot', OneHotEncoder(handle_unknown='ignore'))
    ])

    # Combine preprocessing steps
    preprocessor = ColumnTransformer(
        transformers=[
            ('num', numerical_transformer, numerical_features),
            ('cat', categorical_transformer, categorical_features)
        ])

    # Apply preprocessing to the training and test datasets
    X_train = train_df.drop(columns=['monster_type'])
    y_train = train_df['monster_type']
    X_test = test_df.drop(columns=['monster_type'])

    X_train_preprocessed = preprocessor.fit_transform(X_train)
    X_test_preprocessed = preprocessor.transform(X_test)

    return X_train_preprocessed, y_train, X_test_preprocessed, preprocessor

# Data Augmentation
def augment_data(X_train, y_train):
    smote = SMOTE()
    X_train_resampled, y_train_resampled = smote.fit_resample(X_train, y_train)
    return X_train_resampled, y_train_resampled

# Model Training and Hyperparameter Tuning
def train_model(X_train, y_train):
    xgb_model = XGBClassifier()

    # Define the hyperparameter grid
    param_grid = {
        'n_estimators': [100, 200, 300],
        'max_depth': [3, 5, 7],
        'learning_rate': [0.01, 0.1, 0.2],
        'subsample': [0.5, 0.7, 1.0]
    }

    # Perform randomized search with cross-validation
    random_search = RandomizedSearchCV(xgb_model, param_distributions=param_grid, n_iter=10, cv=5, n_jobs=-1, verbose=1)
    random_search.fit(X_train, y_train)

    # Get the best model
    best_model = random_search.best_estimator_
    return best_model

# Model Evaluation
def evaluate_model(model, X_test, y_test):
    y_pred = model.predict(X_test)

    # Calculate performance metrics
    accuracy = accuracy_score(y_test, y_pred)
    precision = precision_score(y_test, y_pred, average='weighted')
    recall = recall_score(y_test, y_pred, average='weighted')
    f1 = f1_score(y_test, y_pred, average='weighted')

    # Print the performance metrics
    print(f'Accuracy: {accuracy}')
    print(f'Precision: {precision}')
    print(f'Recall: {recall}')
    print(f'F1-Score: {f1}')

    # Print the classification report
    print(classification_report(y_test, y_pred))

    performance_scores = {
        'Accuracy': accuracy,
        'Precision': precision,
        'Recall': recall,
        'F1-Score': f1
    }
    return performance_scores

# Saving Predictions
def save_predictions(y_pred, output_path):
    predictions_df = pd.DataFrame({'monster_type': y_pred})
    predictions_df.to_csv(output_path, index=False)

# Saving the Trained Model
def save_model(model, model_path):
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    with open(model_path, 'wb') as file:
        pickle.dump(model, file)

# Main Function to Orchestrate the Pipeline
def main():
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/monster/train.csv'
    test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/monster/test.csv'

    # Load the datasets
    train_df, test_df = load_data(train_path, test_path)

    # Preprocess the datasets
    X_train, y_train, X_test, preprocessor = preprocess_data(train_df, test_df)

    # Augment the training data
    X_train_resampled, y_train_resampled = augment_data(X_train, y_train)

    # Train the model
    model = train_model(X_train_resampled, y_train_resampled)

    # Evaluate the model
    y_test = test_df['monster_type']
    performance_scores = evaluate_model(model, X_test, y_test)

    # Save the predictions
    y_pred = model.predict(X_test)
    output_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/monster/predictions.csv'
    save_predictions(y_pred, output_path)

    # Save the trained model
    model_path = '/remote-home/cs_acmis_xyt/AutomlAgent/agent_workspace/trained_models/xgboost_model.pkl'
    save_model(model, model_path)

    return performance_scores

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
