import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.metrics import accuracy_score, f1_score
from sklearn.preprocessing import StandardScaler
from sklearn.ensemble import RandomForestClassifier
from sklearn.feature_selection import SelectFromModel
from imblearn.over_sampling import SMOTE
from xgboost import XGBClassifier
import joblib

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data Retrieval and Loading
def load_datasets():
    train_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/monster/train.csv')
    test_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/monster/test.csv')
    return train_df, test_df

# Data Preprocessing
def preprocess_data(train_df, test_df):
    # Ensure 'monster_type' is in the columns
    if 'monster_type' not in train_df.columns:
        raise ValueError("The 'monster_type' column is missing in the training dataset.")
    if 'monster_type' not in test_df.columns:
        raise ValueError("The 'monster_type' column is missing in the testing dataset.")

    # Handle missing values
    for df in [train_df, test_df]:
        numerical_cols = df.select_dtypes(include=['float64', 'int64']).columns
        categorical_cols = df.select_dtypes(include=['object']).columns

        # Impute numerical columns with mean
        df[numerical_cols] = df[numerical_cols].fillna(df[numerical_cols].mean())

        # Impute categorical columns with mode
        for col in categorical_cols:
            df[col] = df[col].fillna(df[col].mode().iloc[0])

        # Drop columns with more than 30% missing values
        df.dropna(thresh=len(df) * 0.7, axis=1, inplace=True)

    # One-hot encoding
    train_df = pd.get_dummies(train_df, drop_first=True)
    test_df = pd.get_dummies(test_df, drop_first=True)

    # Ensure both datasets have the same columns
    missing_cols = set(train_df.columns) - set(test_df.columns)
    for col in missing_cols:
        test_df[col] = 0

    missing_cols = set(test_df.columns) - set(train_df.columns)
    for col in missing_cols:
        train_df[col] = 0

    # Reorder columns to match
    train_df = train_df[train_df.columns]
    test_df = test_df[train_df.columns]

    # Feature scaling
    scaler = StandardScaler()
    numerical_features = train_df.select_dtypes(include=['float64', 'int64']).columns
    train_df[numerical_features] = scaler.fit_transform(train_df[numerical_features])
    test_df[numerical_features] = scaler.transform(test_df[numerical_features])

    # Feature selection
    model = RandomForestClassifier()
    model.fit(train_df.drop('monster_type', axis=1), train_df['monster_type'])
    selector = SelectFromModel(model, prefit=True)
    train_df = train_df[train_df.columns[selector.get_support(indices=True)] + ['monster_type']]
    test_df = test_df[test_df.columns[selector.get_support(indices=True)] + ['monster_type']]

    return train_df, test_df

# Data Augmentation
def augment_data(train_df):
    smote = SMOTE()
    X_resampled, y_resampled = smote.fit_resample(train_df.drop('monster_type', axis=1), train_df['monster_type'])
    train_df = pd.concat([pd.DataFrame(X_resampled, columns=train_df.drop('monster_type', axis=1).columns), pd.Series(y_resampled, name='monster_type')], axis=1)
    return train_df

# Model Training
def train_model(train_df):
    model = XGBClassifier(n_estimators=500, max_depth=5, learning_rate=0.1, subsample=0.9)
    model.fit(train_df.drop('monster_type', axis=1), train_df['monster_type'])
    return model

# Model Evaluation
def evaluate_model(model, test_df):
    y_pred = model.predict(test_df.drop('monster_type', axis=1))
    performance_scores = {
        'ACC': accuracy_score(test_df['monster_type'], y_pred),
        'F1': f1_score(test_df['monster_type'], y_pred, average='weighted')
    }
    return performance_scores

# Main Function to Orchestrate the Pipeline
def main():
    # Step 1: Load datasets
    train_df, test_df = load_datasets()

    # Step 2: Preprocess data
    train_df, test_df = preprocess_data(train_df, test_df)

    # Step 3: Augment data
    train_df = augment_data(train_df)

    # Step 4: Train model
    model = train_model(train_df)

    # Step 5: Evaluate model
    model_performance = evaluate_model(model, test_df)

    # Step 6: Save predictions and model
    y_pred = model.predict(test_df.drop('monster_type', axis=1))
    predictions_df = pd.DataFrame({'monster_type': y_pred})
    predictions_df.to_csv('/remote-home/cs_acmis_xyt/AutomlAgent/agent_workspace/trained_models/predictions.csv', index=False)

    model_path = '/remote-home/cs_acmis_xyt/AutomlAgent/agent_workspace/trained_models/monster_type_classifier.pkl'
    joblib.dump(model, model_path)

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
