import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.model_selection import train_test_split
from imblearn.over_sampling import SMOTE
import torch
import torch.nn as nn
import torch.optim as optim
from sklearn.metrics import accuracy_score, f1_score, log_loss

SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)

# Data Retrieval and Initial Inspection
def load_data(train_path, test_path):
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    
    print("First few rows of the dataset:")
    print(train_df.head())
    print("\nData types of each column:")
    print(train_df.dtypes)
    print("\nSummary of numerical features:")
    print(train_df.describe())
    print("\nOverview of the dataset:")
    print(train_df.info())
    print("\nMissing values in the dataset:")
    print(train_df.isnull().sum())
    
    return train_df, test_df

# Feature Engineering
def feature_engineering(df):
    if 'date' in df.columns:
        df['date'] = pd.to_datetime(df['date'])
        df['day_of_week'] = df['date'].dt.dayofweek
        df['hour_of_day'] = df['date'].dt.hour
        df['month'] = df['date'].dt.month
        df['year'] = df['date'].dt.year

    if 'latitude' in df.columns and 'longitude' in df.columns:
        def haversine(lat1, lon1, lat2, lon2):
            from math import radians, sin, cos, sqrt, asin
            R = 6371  # Earth radius in kilometers
            lat1, lon1, lat2, lon2 = map(radians, [lat1, lon1, lat2, lon2])
            dlat = lat2 - lat1
            dlon = lon2 - lon1
            a = sin(dlat/2)**2 + cos(lat1) * cos(lat2) * sin(dlon/2)**2
            c = 2 * asin(sqrt(a))
            return R * c

        high_crime_area_lat, high_crime_area_lon = 37.7749, -122.4194  # Example coordinates
        df['distance_from_high_crime_area'] = df.apply(lambda row: haversine(row['latitude'], row['longitude'], high_crime_area_lat, high_crime_area_lon), axis=1)

    if 'date' in df.columns:
        df['crimes_in_past_week'] = df.groupby(['latitude', 'longitude', 'crime_category']).rolling(window='7D').count().reset_index().rename(columns={'index': 'crimes_in_past_week'})

    return df

# Data Preprocessing
def preprocess_data(train_df, test_df):
    numerical_columns = train_df.select_dtypes(include=[np.number]).columns
    categorical_columns = train_df.select_dtypes(include=[object]).columns

    # Handling Missing Values
    imputer_num = SimpleImputer(strategy='mean')
    imputer_cat = SimpleImputer(strategy='most_frequent')
    train_df[numerical_columns] = imputer_num.fit_transform(train_df[numerical_columns])
    train_df[categorical_columns] = imputer_cat.fit_transform(train_df[categorical_columns])
    test_df[numerical_columns] = imputer_num.transform(test_df[numerical_columns])
    test_df[categorical_columns] = imputer_cat.transform(test_df[categorical_columns])

    # Categorical Encoding
    encoder = OneHotEncoder(handle_unknown='ignore')
    train_encoded = encoder.fit_transform(train_df[categorical_columns])
    test_encoded = encoder.transform(test_df[categorical_columns])
    train_encoded_df = pd.DataFrame(train_encoded.toarray(), columns=encoder.get_feature_names_out(categorical_columns))
    test_encoded_df = pd.DataFrame(test_encoded.toarray(), columns=encoder.get_feature_names_out(categorical_columns))
    train_df = pd.concat([train_df.drop(columns=categorical_columns), train_encoded_df], axis=1)
    test_df = pd.concat([test_df.drop(columns=categorical_columns), test_encoded_df], axis=1)

    # Normalization/Standardization
    scaler = StandardScaler()
    train_df[numerical_columns] = scaler.fit_transform(train_df[numerical_columns])
    test_df[numerical_columns] = scaler.transform(test_df[numerical_columns])

    # Data Augmentation
    if 'crime_category' in train_df.columns:
        smote = SMOTE()
        X_resampled, y_resampled = smote.fit_resample(train_df.drop(columns=['crime_category']), train_df['crime_category'])
    else:
        X_resampled = train_df.drop(columns=['crime_category'])
        y_resampled = train_df['crime_category']

    return X_resampled, y_resampled, test_df

# Define the Model
class FeedforwardNN(nn.Module):
    def __init__(self, input_size, hidden_size, output_size):
        super(FeedforwardNN, self).__init__()
        self.layer1 = nn.Linear(input_size, hidden_size)
        self.layer2 = nn.Linear(hidden_size, hidden_size)
        self.layer3 = nn.Linear(hidden_size, output_size)
        self.relu = nn.ReLU()

    def forward(self, x):
        x = self.relu(self.layer1(x))
        x = self.relu(self.layer2(x))
        x = self.layer3(x)
        return x

# Train the Model
def train_model(model, X_train, y_train, X_val, y_val, num_epochs=100, batch_size=64, learning_rate=0.001):
    criterion = nn.CrossEntropyLoss()
    optimizer = optim.Adam(model.parameters(), lr=learning_rate)

    train_dataset = torch.utils.data.TensorDataset(X_train, y_train)
    train_loader = torch.utils.data.DataLoader(train_dataset, batch_size=batch_size, shuffle=True)

    val_dataset = torch.utils.data.TensorDataset(X_val, y_val)
    val_loader = torch.utils.data.DataLoader(val_dataset, batch_size=batch_size, shuffle=False)

    for epoch in range(num_epochs):
        model.train()
        for batch in train_loader:
            inputs, targets = batch
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, targets)
            loss.backward()
            optimizer.step()

        if (epoch + 1) % 10 == 0:
            model.eval()
            with torch.no_grad():
                val_loss = 0.0
                for batch in val_loader:
                    inputs, targets = batch
                    outputs = model(inputs)
                    loss = criterion(outputs, targets)
                    val_loss += loss.item()
                val_loss /= len(val_loader)
                print(f'Epoch [{epoch+1}/{num_epochs}], Validation Loss: {val_loss:.4f}')

    return model

# Evaluate the Model
def evaluate_model(model, X_test, y_test):
    model.eval()
    with torch.no_grad():
        test_outputs = model(X_test)
        test_loss = nn.CrossEntropyLoss()(test_outputs, y_test)
        y_pred = torch.argmax(test_outputs, dim=1).numpy()
        y_true = y_test.numpy()
        accuracy = accuracy_score(y_true, y_pred)
        f1 = f1_score(y_true, y_pred, average='weighted')
        logloss = log_loss(y_true, test_outputs.softmax(dim=1).numpy())

    performance_scores = {
        'Accuracy': accuracy,
        'F1': f1,
        'Logloss': logloss
    }

    return performance_scores

# Main function to orchestrate the pipeline
def main():
    train_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/crime_category/train.csv'
    test_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/crime_category/test.csv'

    # Step 1. Load the datasets
    train_df, test_df = load_data(train_path, test_path)

    # Step 2. Feature Engineering
    train_df = feature_engineering(train_df)
    test_df = feature_engineering(test_df)

    # Step 3. Data Preprocessing
    X_resampled, y_resampled, test_df = preprocess_data(train_df, test_df)

    # Step 4. Prepare the Data
    X_train, X_val, y_train, y_val = train_test_split(X_resampled, y_resampled, test_size=0.2, random_state=42)
    X_train = torch.tensor(X_train.values, dtype=torch.float32)
    y_train = torch.tensor(y_train.values, dtype=torch.long)
    X_val = torch.tensor(X_val.values, dtype=torch.float32)
    y_val = torch.tensor(y_val.values, dtype=torch.long)
    X_test = torch.tensor(test_df.drop(columns=['crime_category']).values, dtype=torch.float32)
    y_test = torch.tensor(test_df['crime_category'].values, dtype=torch.long)

    # Step 5. Define the Model
    input_size = X_train.shape[1]
    hidden_size = 128
    output_size = len(y_resampled.unique())
    model = FeedforwardNN(input_size, hidden_size, output_size)

    # Step 6. Train the Model
    model = train_model(model, X_train, y_train, X_val, y_val)

    # Step 7. Evaluate the Model
    model_performance = evaluate_model(model, X_test, y_test)

    # Step 8. Save the Model
    model_save_path = './agent_workspace/trained_models/fnn_model.pth'
    os.makedirs(os.path.dirname(model_save_path), exist_ok=True)
    torch.save(model.state_dict(), model_save_path)

    # Step 9. Save Predictions
    test_predictions = torch.argmax(model(X_test), dim=1).numpy()
    test_df['predicted_crime_category'] = test_predictions
    test_predictions_save_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/crime_category/test_predictions.csv'
    test_df.to_csv(test_predictions_save_path, index=False)

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
