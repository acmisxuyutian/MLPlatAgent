######代码######

import os
import random
import time
import json
import numpy as np
import pandas as pd
import torch
import torch.nn as nn
import torch.optim as optim
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import train_test_split, RandomizedSearchCV
from sklearn.metrics import accuracy_score, f1_score, log_loss
from scipy import stats
from imblearn.over_sampling import SMOTE

SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)

# Data Retrieval
def load_data(train_path, test_path):
    train_data = pd.read_csv(train_path)
    test_data = pd.read_csv(test_path)
    return train_data, test_data

# Data Preprocessing
def preprocess_data(train_data, test_data):
    # Initial Data Inspection
    print(train_data.info())
    print(train_data.describe())
    print(train_data.isnull().sum())

    # Data Cleaning
    numerical_imputer = SimpleImputer(strategy='median')
    categorical_imputer = SimpleImputer(strategy='most_frequent')

    numerical_features = train_data.select_dtypes(include=[np.number]).columns
    categorical_features = train_data.select_dtypes(include=['object']).columns

    for feature in numerical_features:
        train_data[feature] = numerical_imputer.fit_transform(train_data[[feature]])
        test_data[feature] = numerical_imputer.transform(test_data[[feature]])

    for feature in categorical_features:
        train_data[feature] = categorical_imputer.fit_transform(train_data[[feature]].values.reshape(-1, 1)).ravel()
        test_data[feature] = categorical_imputer.transform(test_data[[feature]].values.reshape(-1, 1)).ravel()

    # Remove outliers
    for feature in numerical_features:
        z_scores = stats.zscore(train_data[feature])
        train_data = train_data[(z_scores < 3) & (z_scores > -3)]

    # Feature Engineering
    train_data = pd.get_dummies(train_data, columns=categorical_features)
    test_data = pd.get_dummies(test_data, columns=categorical_features)

    if 'date' in train_data.columns:
        train_data['hour'] = pd.to_datetime(train_data['date']).dt.hour
        train_data['day_of_week'] = pd.to_datetime(train_data['date']).dt.dayofweek
        train_data['month'] = pd.to_datetime(train_data['date']).dt.month

    if 'date' in test_data.columns:
        test_data['hour'] = pd.to_datetime(test_data['date']).dt.hour
        test_data['day_of_week'] = pd.to_datetime(test_data['date']).dt.dayofweek
        test_data['month'] = pd.to_datetime(test_data['date']).dt.month

    # Feature Scaling
    scaler = StandardScaler()
    train_data[numerical_features] = scaler.fit_transform(train_data[numerical_features])
    test_data[numerical_features] = scaler.transform(test_data[numerical_features])

    # Data Splitting
    X = train_data.drop(columns=['crime_category'])
    y = train_data['crime_category']

    X_train, X_val, y_train, y_val = train_test_split(X, y, test_size=0.2, stratify=y, random_state=42)

    return X_train, X_val, y_train, y_val, test_data

# Data Augmentation
def augment_data(X_train, y_train):
    smote = SMOTE()
    X_train_resampled, y_train_resampled = smote.fit_resample(X_train, y_train)
    return X_train_resampled, y_train_resampled

# Model Definition
class MLP(nn.Module):
    def __init__(self, input_size, hidden_size, output_size):
        super(MLP, self).__init__()
        self.fc1 = nn.Linear(input_size, hidden_size)
        self.relu = nn.ReLU()
        self.fc2 = nn.Linear(hidden_size, hidden_size)
        self.fc3 = nn.Linear(hidden_size, output_size)
        self.softmax = nn.Softmax(dim=1)

    def forward(self, x):
        x = self.fc1(x)
        x = self.relu(x)
        x = self.fc2(x)
        x = self.relu(x)
        x = self.fc3(x)
        x = self.softmax(x)
        return x

# Model Training
def train_model(model, X_train, y_train, X_val, y_val, params):
    optimizer = optim.Adam(model.parameters(), lr=params['learning_rate'], weight_decay=params['l2_regularization'])
    criterion = nn.CrossEntropyLoss()

    best_val_loss = float('inf')
    best_model = None

    for epoch in range(100):
        model.train()
        for batch in range(0, len(X_train), params['batch_size']):
            X_batch = X_train[batch:batch + params['batch_size']]
            y_batch = y_train[batch:batch + params['batch_size']]
            optimizer.zero_grad()
            outputs = model(X_batch)
            loss = criterion(outputs, y_batch)
            loss.backward()
            optimizer.step()

        model.eval()
        with torch.no_grad():
            val_outputs = model(X_val)
            val_loss = log_loss(y_val, val_outputs)
            if val_loss < best_val_loss:
                best_val_loss = val_loss
                best_model = model

    return best_model

# Model Evaluation
def evaluate_model(model, X_test, y_test):
    model.eval()
    with torch.no_grad():
        test_outputs = model(X_test)
        test_loss = log_loss(y_test, test_outputs)
        y_pred = test_outputs.argmax(dim=1)
        performance_scores = {
            'ACC': accuracy_score(y_test, y_pred),
            'F1': f1_score(y_test, y_pred, average='weighted')
        }
        print(f'Test Logloss: {test_loss}')
        print(f'Test Accuracy: {performance_scores["ACC"]}')
        print(f'Test F1 Score: {performance_scores["F1"]}')
    return performance_scores

# Main Function
def main():
    train_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/crime_category/train.csv'
    test_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/crime_category/test.csv'

    # Step 1. Load the dataset
    train_data, test_data = load_data(train_path, test_path)

    # Step 2. Preprocess the data
    X_train, X_val, y_train, y_val, test_data = preprocess_data(train_data, test_data)

    # Step 3. Augment the data
    X_train_resampled, y_train_resampled = augment_data(X_train, y_train)

    # Step 4. Define the model
    input_size = X_train.shape[1]
    hidden_size = 128
    output_size = len(y_train.unique())

    model = MLP(input_size, hidden_size, output_size)

    # Step 5. Hyperparameter Tuning
    param_distributions = {
        'learning_rate': [0.0001, 0.001, 0.01],
        'batch_size': [32, 64, 128],
        'l2_regularization': [0.0001, 0.001, 0.01]
    }

    def train_and_evaluate(params):
        model = MLP(input_size, hidden_size, output_size)
        optimizer = optim.Adam(model.parameters(), lr=params['learning_rate'], weight_decay=params['l2_regularization'])
        criterion = nn.CrossEntropyLoss()

        best_val_loss = float('inf')
        best_model = None

        for epoch in range(100):
            model.train()
            for batch in range(0, len(X_train_resampled), params['batch_size']):
                X_batch = X_train_resampled[batch:batch + params['batch_size']]
                y_batch = y_train_resampled[batch:batch + params['batch_size']]
                optimizer.zero_grad()
                outputs = model(X_batch)
                loss = criterion(outputs, y_batch)
                loss.backward()
                optimizer.step()

            model.eval()
            with torch.no_grad():
                val_outputs = model(X_val)
                val_loss = log_loss(y_val, val_outputs)
                if val_loss < best_val_loss:
                    best_val_loss = val_loss
                    best_model = model

        return best_val_loss

    random_search = RandomizedSearchCV(train_and_evaluate, param_distributions, n_iter=10, cv=5)
    random_search.fit(X_train_resampled, y_train_resampled)
    best_params = random_search.best_params_

    # Step 6. Train the final model with the best hyperparameters
    best_model = MLP(input_size, hidden_size, output_size)
    best_model = train_model(best_model, X_train_resampled, y_train_resampled, X_val, y_val, best_params)

    # Step 7. Evaluate the model
    X_test = test_data.drop(columns=['crime_category'])
    y_test = test_data['crime_category']
    model_performance = evaluate_model(best_model, X_test, y_test)

    # Step 8. Save the model and predictions
    model_save_path = "./agent_workspace/trained_models/best_model.pth"
    os.makedirs(os.path.dirname(model_save_path), exist_ok=True)
    torch.save(best_model.state_dict(), model_save_path)

    test_predictions = best_model(X_test).argmax(dim=1)
    test_predictions_df = pd.DataFrame({'id': test_data['id'], 'predicted_category': test_predictions})
    test_predictions_df.to_csv('./agent_workspace/trained_models/test_predictions.csv', index=False)

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
