######代码######

import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import log_loss
from imblearn.over_sampling import SMOTE
from xgboost import XGBClassifier
import joblib

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data Retrieval
def load_datasets(train_path, test_path):
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    return train_df, test_df

# Data Preprocessing
def preprocess_data(train_df, test_df):
    # Handle Missing Values
    for df in [train_df, test_df]:
        for column in df.columns:
            if df[column].dtype == 'object':
                df[column].fillna(df[column].mode().iloc[0], inplace=True)
            else:
                df[column].fillna(df[column].median(), inplace=True)
    
    # Remove Duplicates
    train_df.drop_duplicates(inplace=True)
    test_df.drop_duplicates(inplace=True)
    
    # Categorical Encoding
    train_df = pd.get_dummies(train_df)
    test_df = pd.get_dummies(test_df)
    
    # Ensure the same columns are present in both datasets
    train_columns = set(train_df.columns)
    test_columns = set(test_df.columns)
    common_columns = list(train_columns.intersection(test_columns))
    
    # Add missing columns to test_df with value 0
    missing_columns = train_columns - test_columns
    for col in missing_columns:
        test_df[col] = 0
    
    # Add missing columns to train_df with value 0
    missing_columns = test_columns - train_columns
    for col in missing_columns:
        train_df[col] = 0
    
    # Reorder columns to match
    train_df = train_df[common_columns]
    test_df = test_df[common_columns]
    
    # Numerical Normalization
    scaler = StandardScaler()
    numerical_features = train_df.select_dtypes(include=['float64', 'int64']).columns
    train_df[numerical_features] = scaler.fit_transform(train_df[numerical_features])
    test_df[numerical_features] = scaler.transform(test_df[numerical_features])
    
    return train_df, test_df

# Data Splitting
def split_data(train_df):
    if 'crime_category' not in train_df.columns:
        raise ValueError("Target variable 'crime_category' not found in the training dataset.")
    
    X_train = train_df.drop(columns=['crime_category'], errors='ignore')
    y_train = train_df['crime_category']
    
    X_train, X_val, y_train, y_val = train_test_split(X_train, y_train, test_size=0.2, stratify=y_train, random_state=42)
    return X_train, X_val, y_train, y_val

# Data Augmentation
def augment_data(X_train, y_train):
    smote = SMOTE(random_state=42)
    X_train, y_train = smote.fit_resample(X_train, y_train)
    return X_train, y_train

# Model Training
def train_model(X_train, y_train):
    xgb_model = XGBClassifier(n_estimators=200, max_depth=9, learning_rate=0.1, random_state=42)
    xgb_model.fit(X_train, y_train)
    return xgb_model

# Model Evaluation
def evaluate_model(model, X_val, y_val):
    y_val_pred = model.predict_proba(X_val)
    val_logloss = log_loss(y_val, y_val_pred)
    print(f'Validation Logloss: {val_logloss}')
    return val_logloss

# Final Evaluation on Test Set
def final_evaluation(model, test_df):
    if 'crime_category' not in test_df.columns:
        raise ValueError("Target variable 'crime_category' not found in the test dataset.")
    
    y_test_pred = model.predict_proba(test_df.drop(columns=['crime_category'], errors='ignore'))
    test_logloss = log_loss(test_df['crime_category'], y_test_pred)
    print(f'Test Logloss: {test_logloss}')
    
    # Save predictions to a CSV file
    predictions_df = pd.DataFrame(y_test_pred, columns=model.classes_)
    predictions_df.to_csv('predictions.csv', index=False)
    
    return test_logloss

# Model Saving
def save_model(model, model_path):
    joblib.dump(model, model_path)

# Main function to orchestrate the pipeline
def main():
    train_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/crime_category/train.csv'
    test_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/crime_category/test.csv'
    
    # Step 1: Load datasets
    train_df, test_df = load_datasets(train_path, test_path)
    
    # Step 2: Preprocess data
    train_df, test_df = preprocess_data(train_df, test_df)
    
    # Step 3: Split data
    X_train, X_val, y_train, y_val = split_data(train_df)
    
    # Step 4: Augment data
    X_train, y_train = augment_data(X_train, y_train)
    
    # Step 5: Train model
    model = train_model(X_train, y_train)
    
    # Step 6: Evaluate model on validation set
    val_logloss = evaluate_model(model, X_val, y_val)
    
    # Step 7: Final evaluation on test set
    test_logloss = final_evaluation(model, test_df)
    
    # Step 8: Save model
    model_path = r'./agent_workspace/trained_models/xgboost_crime_category_model.pkl'
    save_model(model, model_path)
    
    return test_logloss

if __name__ == "__main__":
    test_logloss = main()
    print(f"Model Performance on Test Set: Test Logloss: {test_logloss}")
