import os
import pandas as pd
import numpy as np
from sklearn.impute import SimpleImputer
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import mean_absolute_error
import joblib

# Set random seed for reproducibility
SEED = 42
np.random.seed(SEED)

# Paths to the datasets
train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/crab_age/train.csv'
test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/crab_age/test.csv'
predictions_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/crab_age/predictions.csv'
model_path = './agent_workspace/trained_models/crab_age_random_forest.pkl'

# Ensure the directory for saving the model exists
os.makedirs(os.path.dirname(model_path), exist_ok=True)

# Load the datasets
def load_data(train_path, test_path):
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    
    assert not train_df.empty, "Training dataset is empty"
    assert not test_df.empty, "Test dataset is empty"
    
    return train_df, test_df

# Preprocess the datasets
def preprocess_data(train_df, test_df):
    # Identify categorical variables
    categorical_features = train_df.select_dtypes(include=['object']).columns
    
    # Apply one-hot encoding
    train_df = pd.get_dummies(train_df, columns=categorical_features)
    test_df = pd.get_dummies(test_df, columns=categorical_features)
    
    # Ensure the same columns are present in both datasets
    train_df, test_df = train_df.align(test_df, join='inner', axis=1)
    
    # Handle missing values
    imputer = SimpleImputer(strategy='median')
    train_df = pd.DataFrame(imputer.fit_transform(train_df), columns=train_df.columns)
    test_df = pd.DataFrame(imputer.transform(test_df), columns=test_df.columns)
    
    return train_df, test_df

# Train the model
def train_model(X_train, y_train):
    model = RandomForestRegressor(
        n_estimators=200,
        max_depth=20,
        min_samples_split=5,
        min_samples_leaf=2,
        random_state=42
    )
    model.fit(X_train, y_train)
    return model

# Evaluate the model
def evaluate_model(model, X_test, y_test):
    y_pred = model.predict(X_test)
    mae = mean_absolute_error(y_test, y_pred)
    return mae

# Save the predictions
def save_predictions(y_pred, predictions_path):
    predictions_df = pd.DataFrame({'age': y_pred})
    predictions_df.to_csv(predictions_path, index=False)

# Save the model
def save_model(model, model_path):
    joblib.dump(model, model_path)

# Main function to orchestrate the entire pipeline
def main():
    # Load the datasets
    train_df, test_df = load_data(train_path, test_path)
    
    # Preprocess the datasets
    train_df, test_df = preprocess_data(train_df, test_df)
    
    # Split the datasets into features and target variable
    if 'age' in train_df.columns:
        X_train = train_df.drop(columns=['age'])
        y_train = train_df['age']
    else:
        raise KeyError("Column 'age' not found in the training dataset")
    
    if 'age' in test_df.columns:
        X_test = test_df.drop(columns=['age'])
        y_test = test_df['age']
    else:
        raise KeyError("Column 'age' not found in the test dataset")
    
    # Train the model
    model = train_model(X_train, y_train)
    
    # Evaluate the model
    mae = evaluate_model(model, X_test, y_test)
    
    # Save the model
    save_model(model, model_path)
    
    # Save the predictions
    y_pred = model.predict(X_test)
    save_predictions(y_pred, predictions_path)
    
    return mae

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
