import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import train_test_split, GridSearchCV
from sklearn.metrics import mean_squared_log_error
import torch
import torch.nn as nn
import torch.optim as optim

# Set the random seed for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)

# Data Retrieval and Collection
train_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/abalone/train.csv'
test_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/abalone/test.csv'

train_df = pd.read_csv(train_path)
test_df = pd.read_csv(test_path)

# Data Preprocessing
def preprocess_data(train_df, test_df):
    # Data Inspection
    print(train_df.head())
    print(train_df.info())
    print(train_df.describe())
    print(train_df.isnull().sum())

    # One-Hot Encoding
    categorical_cols = ['Sex']
    train_df = pd.get_dummies(train_df, columns=categorical_cols)
    test_df = pd.get_dummies(test_df, columns=categorical_cols)

    # Feature Scaling
    numerical_cols = ['Length', 'Diameter', 'Height', 'Whole weight', 'Shucked weight', 'Viscera weight', 'Shell weight']
    scaler = StandardScaler()

    # Check if all numerical columns exist in the DataFrame
    for col in numerical_cols:
        if col not in train_df.columns:
            raise KeyError(f"Column {col} not found in the training dataset.")
        if col not in test_df.columns:
            raise KeyError(f"Column {col} not found in the test dataset.")

    train_df[numerical_cols] = scaler.fit_transform(train_df[numerical_cols])
    test_df[numerical_cols] = scaler.transform(test_df[numerical_cols])

    # Data Splitting
    X_train = train_df.drop(columns=['Age'])
    y_train = train_df['Age']

    X_test = test_df.drop(columns=['Age'])
    y_test = test_df['Age']

    X_train, X_val, y_train, y_val = train_test_split(X_train, y_train, test_size=0.2, random_state=42)

    return X_train, y_train, X_val, y_val, X_test, y_test

# Model Building and Training
class FeedforwardNN(nn.Module):
    def __init__(self, input_size, hidden_size, output_size):
        super(FeedforwardNN, self).__init__()
        self.fc1 = nn.Linear(input_size, hidden_size)
        self.relu = nn.ReLU()
        self.fc2 = nn.Linear(hidden_size, hidden_size)
        self.fc3 = nn.Linear(hidden_size, output_size)

    def forward(self, x):
        out = self.fc1(x)
        out = self.relu(out)
        out = self.fc2(out)
        out = self.relu(out)
        out = self.fc3(out)
        return out

def train_and_evaluate(params, X_train_tensor, y_train_tensor, X_val_tensor, y_val_tensor):
    model = FeedforwardNN(input_size, params['hidden_size'], output_size)
    criterion = nn.MSELoss()
    optimizer = optim.Adam(model.parameters(), lr=params['learning_rate'])

    for epoch in range(params['epochs']):
        model.train()
        optimizer.zero_grad()
        outputs = model(X_train_tensor)
        loss = criterion(outputs, y_train_tensor)
        loss.backward()
        optimizer.step()

    model.eval()
    with torch.no_grad():
        val_outputs = model(X_val_tensor)
        val_loss = criterion(val_outputs, y_val_tensor)
        val_rmlse = mean_squared_log_error(y_val_tensor.numpy(), val_outputs.numpy(), squared=False)

    return val_rmlse

def train_model(X_train_tensor, y_train_tensor, best_params):
    model = FeedforwardNN(input_size, best_params['hidden_size'], output_size)
    criterion = nn.MSELoss()
    optimizer = optim.Adam(model.parameters(), lr=best_params['learning_rate'])

    for epoch in range(best_params['epochs']):
        model.train()
        optimizer.zero_grad()
        outputs = model(X_train_tensor)
        loss = criterion(outputs, y_train_tensor)
        loss.backward()
        optimizer.step()

        if (epoch + 1) % 10 == 0:
            print(f'Epoch [{epoch+1}/{best_params["epochs"]}], Loss: {loss.item():.4f}')

    return model

# Model Evaluation
def evaluate_model(model, X_test_tensor, y_test_tensor):
    model.eval()
    with torch.no_grad():
        test_outputs = model(X_test_tensor)
        test_rmlse = mean_squared_log_error(y_test_tensor.numpy(), test_outputs.numpy(), squared=False)

    return test_rmlse

# Main Function to Orchestrate the Pipeline
def main():
    """
    Main function to execute the tabular regression pipeline.
    """
    # Data Retrieval and Collection
    train_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/abalone/train.csv'
    test_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/abalone/test.csv'

    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)

    # Data Preprocessing
    X_train, y_train, X_val, y_val, X_test, y_test = preprocess_data(train_df, test_df)

    # Convert DataFrames to Tensors
    X_train_tensor = torch.tensor(X_train.values, dtype=torch.float32)
    y_train_tensor = torch.tensor(y_train.values, dtype=torch.float32).view(-1, 1)
    X_val_tensor = torch.tensor(X_val.values, dtype=torch.float32)
    y_val_tensor = torch.tensor(y_val.values, dtype=torch.float32).view(-1, 1)
    X_test_tensor = torch.tensor(X_test.values, dtype=torch.float32)
    y_test_tensor = torch.tensor(y_test.values, dtype=torch.float32).view(-1, 1)

    # Model Building and Training
    input_size = X_train.shape[1]
    hidden_size = 128
    output_size = 1

    param_grid = {
        'hidden_size': [64, 128, 256],
        'learning_rate': [0.001, 0.01, 0.1],
        'epochs': [50, 100, 200]
    }

    # Define a function to wrap the training and evaluation for GridSearchCV
    def train_and_evaluate_wrapper(params):
        return train_and_evaluate(params, X_train_tensor, y_train_tensor, X_val_tensor, y_val_tensor)

    grid_search = GridSearchCV(train_and_evaluate_wrapper, param_grid, scoring='neg_mean_squared_log_error', cv=5)
    grid_search.fit(X_train_tensor, y_train_tensor)

    best_params = grid_search.best_params_
    print("Best Parameters:", best_params)

    best_model = train_model(X_train_tensor, y_train_tensor, best_params)

    # Model Evaluation
    test_rmlse = evaluate_model(best_model, X_test_tensor, y_test_tensor)
    print(f'Test RMLSE: {test_rmlse:.4f}')

    # Save the Model
    model_save_path = os.path.join("./agent_workspace/trained_models", 'abalone_age_prediction_model.pth')
    os.makedirs(os.path.dirname(model_save_path), exist_ok=True)
    torch.save(best_model.state_dict(), model_save_path)

    return test_rmlse

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
