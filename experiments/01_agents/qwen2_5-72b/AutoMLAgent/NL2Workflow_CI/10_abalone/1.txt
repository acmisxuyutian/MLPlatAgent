import os
import random
import numpy as np
import pandas as pd
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import train_test_split
import xgboost as xgb
from sklearn.metrics import mean_squared_log_error
import joblib

# Set the random seed for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data Retrieval and Collection
def load_data():
    train_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/abalone/train.csv'
    test_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/abalone/test.csv'

    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)

    return train_df, test_df

# Data Preprocessing
def preprocess_data(train_df, test_df):
    print("Inspecting the training data:")
    print(train_df.head())
    print(train_df.info())
    print(train_df.describe())

    if 'Sex' in train_df.columns:
        categorical_columns = ['Sex']
        train_df = pd.get_dummies(train_df, columns=categorical_columns)
        test_df = pd.get_dummies(test_df, columns=categorical_columns)
    else:
        print("Warning: 'Sex' column not found in the dataset.")

    numerical_columns = train_df.select_dtypes(include=['float64', 'int64']).columns.tolist()
    scaler = StandardScaler()
    train_df[numerical_columns] = scaler.fit_transform(train_df[numerical_columns])
    test_df[numerical_columns] = scaler.transform(test_df[numerical_columns])

    if 'Rings' in train_df.columns and 'Rings' in test_df.columns:
        X_train = train_df.drop(columns=['Rings'])
        y_train = train_df['Rings']

        X_test = test_df.drop(columns=['Rings'])
        y_test = test_df['Rings']

        X_train, X_val, y_train, y_val = train_test_split(X_train, y_train, test_size=0.2, random_state=42)
    else:
        raise KeyError("The 'Rings' column is missing in the dataset.")

    return X_train, X_val, y_train, y_val, X_test, y_test

# Model Training
def train_model(X_train, y_train):
    model = xgb.XGBRegressor(
        n_estimators=300,
        max_depth=7,
        learning_rate=0.1,
        subsample=0.8,
        colsample_bytree=0.8,
        random_state=42
    )

    model.fit(X_train, y_train)
    return model

# Model Evaluation
def evaluate_model(model, X_val, y_val, X_test, y_test):
    y_val_pred = model.predict(X_val)
    y_test_pred = model.predict(X_test)

    # Ensure predictions and actual values are non-negative for MSLE
    y_val_pred = np.maximum(y_val_pred, 0)
    y_test_pred = np.maximum(y_test_pred, 0)
    y_val = np.maximum(y_val, 0)
    y_test = np.maximum(y_test, 0)

    rmlse_val = np.sqrt(mean_squared_log_error(y_val, y_val_pred))
    print(f'Validation RMLSE: {rmlse_val}')

    rmlse_test = np.sqrt(mean_squared_log_error(y_test, y_test_pred))
    print(f'Test RMLSE: {rmlse_test}')

    performance_scores = {
        'Validation RMLSE': rmlse_val,
        'Test RMLSE': rmlse_test
    }

    return performance_scores

# Model Saving
def save_model(model, model_path):
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    joblib.dump(model, model_path)

# Main Function
def main():
    # Step 1. Retrieve or load a dataset from the specified source
    train_df, test_df = load_data()

    # Step 2. Preprocess the data
    X_train, X_val, y_train, y_val, X_test, y_test = preprocess_data(train_df, test_df)

    # Step 3. Train the model
    model = train_model(X_train, y_train)

    # Step 4. Evaluate the model
    performance_scores = evaluate_model(model, X_val, y_val, X_test, y_test)

    # Step 5. Save the trained model
    model_path = r'./agent_workspace/trained_models/abalone/xgboost_model.pkl'
    save_model(model, model_path)

    return performance_scores

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
