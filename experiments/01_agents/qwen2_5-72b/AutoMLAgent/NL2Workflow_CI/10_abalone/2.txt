import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_log_error
from pytorch_tabnet.tab_model import TabNetRegressor
import torch

# Set the random seed for reproducibility
SEED = 42
random.seed(SEED)
np.random.seed(SEED)
torch.manual_seed(SEED)

# Data Retrieval and Collection
train_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/abalone/train.csv'
test_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/abalone/test.csv'

train_df = pd.read_csv(train_path)
test_df = pd.read_csv(test_path)

# Data Preprocessing
def preprocess_data(train_df, test_df):
    # Data Inspection
    print("Training Data Info:")
    print(train_df.info())
    print(train_df.describe())
    print(train_df.isnull().sum())

    # One-Hot Encoding
    train_df = pd.get_dummies(train_df, columns=['Sex'], drop_first=True)
    test_df = pd.get_dummies(test_df, columns=['Sex'], drop_first=True)

    # Feature Scaling
    numerical_features = ['Length', 'Diameter', 'Height', 'Whole weight', 'Shell weight']
    scaler = StandardScaler()
    train_df[numerical_features] = scaler.fit_transform(train_df[numerical_features])
    test_df[numerical_features] = scaler.transform(test_df[numerical_features])

    # Data Splitting
    X_train = train_df.drop(columns=['Rings'])
    y_train = train_df['Rings']
    X_val, X_test, y_val, y_test = train_test_split(X_train, y_train, test_size=0.2, random_state=42)

    return X_train, y_train, X_val, y_val, X_test, y_test, test_df, scaler

X_train, y_train, X_val, y_val, X_test, y_test, test_df, scaler = preprocess_data(train_df, test_df)

# Model Training and Optimization
def train_model(X_train, y_train, X_val, y_val):
    model = TabNetRegressor(
        n_d=64, n_a=64, n_steps=5, gamma=1.5, n_independent=2, n_shared=2,
        mask_type='sparsemax', optimizer_fn=torch.optim.Adam,
        optimizer_params=dict(lr=2e-2, weight_decay=1e-5),
        scheduler_params=dict(mode="min", patience=5, min_lr=1e-5, factor=0.5),
        scheduler_fn=torch.optim.lr_scheduler.ReduceLROnPlateau,
        verbose=1
    )

    # Reshape the target variables to 2D
    y_train = y_train.values.reshape(-1, 1)
    y_val = y_val.values.reshape(-1, 1)

    # Set the device to CPU to avoid CUDA out of memory error
    device = torch.device('cpu')
    model.network.to(device)

    model.fit(
        X_train=X_train.values, y_train=y_train,
        eval_set=[(X_train.values, y_train), (X_val.values, y_val)],
        eval_name=['train', 'valid'],
        eval_metric=['rmse'],
        max_epochs=1000, patience=50,
        batch_size=1024, virtual_batch_size=128,
        num_workers=0,
        drop_last=False
    )

    return model

model = train_model(X_train, y_train, X_val, y_val)

# Model Evaluation
def evaluate_model(model, X_test, y_test):
    y_pred = model.predict(X_test.values)
    rmlse = np.sqrt(mean_squared_log_error(y_test, y_pred))
    print(f'RMLSE on test set: {rmlse}')
    return rmlse

rmlse = evaluate_model(model, X_test, y_test)

# Model Saving and Deployment
model_save_path = './agent_workspace/trained_models/abalone_tabnet_model'
os.makedirs(os.path.dirname(model_save_path), exist_ok=True)
model.save_model(model_save_path)

# Model Loading
loaded_model = TabNetRegressor()
loaded_model.load_model(model_save_path)

# Inference on new data
def predict_new_data(loaded_model, new_data_path, scaler):
    new_data = pd.read_csv(new_data_path)
    new_data = pd.get_dummies(new_data, columns=['Sex'], drop_first=True)
    new_data[numerical_features] = scaler.transform(new_data[numerical_features])
    predictions = loaded_model.predict(new_data.values)
    print("Predictions on new data:", predictions)

# Example usage of predict_new_data function
# predict_new_data(loaded_model, 'path_to_new_data.csv', scaler)

print("Model Performance on Test Set:", rmlse)
