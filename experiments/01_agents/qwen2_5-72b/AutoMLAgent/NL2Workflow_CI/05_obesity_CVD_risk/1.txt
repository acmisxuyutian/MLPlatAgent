import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.impute import KNNImputer
from sklearn.preprocessing import StandardScaler, PolynomialFeatures
from sklearn.feature_selection import mutual_info_classif
from sklearn.ensemble import RandomForestClassifier, GradientBoostingClassifier
from sklearn.model_selection import train_test_split, cross_val_score
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score
from imblearn.over_sampling import SMOTE
import joblib
import shap

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
def load_datasets(train_path, test_path):
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    return train_df, test_df

# Step 2. Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
def split_data(train_df, test_size=0.2, random_state=SEED, stratify=None):
    train_df, valid_df = train_test_split(train_df, test_size=test_size, random_state=random_state, stratify=stratify)
    return train_df, valid_df

# Step 3. With the split dataset, run data preprocessing and feature engineering (if applicable) using the "preprocess_data" function you defined
def preprocess_data(train_df, test_df):
    # Handle Missing Values
    imputer = KNNImputer(n_neighbors=5)
    train_df = pd.DataFrame(imputer.fit_transform(train_df), columns=train_df.columns)
    test_df = pd.DataFrame(imputer.transform(test_df), columns=test_df.columns)

    # Encode Categorical Variables
    train_df = pd.get_dummies(train_df)
    test_df = pd.get_dummies(test_df)

    # Scale Numerical Features
    scaler = StandardScaler()
    numerical_columns = train_df.select_dtypes(include=['int64', 'float64']).columns
    train_df[numerical_columns] = scaler.fit_transform(train_df[numerical_columns])
    test_df[numerical_columns] = scaler.transform(test_df[numerical_columns])

    # Create Interaction Features
    interaction = PolynomialFeatures(degree=2, interaction_only=True, include_bias=False)
    interaction_features = interaction.fit_transform(train_df[numerical_columns])
    interaction_df = pd.DataFrame(interaction_features, columns=interaction.get_feature_names_out(numerical_columns))
    train_df = pd.concat([train_df, interaction_df], axis=1)

    interaction_features_test = interaction.transform(test_df[numerical_columns])
    interaction_df_test = pd.DataFrame(interaction_features_test, columns=interaction.get_feature_names_out(numerical_columns))
    test_df = pd.concat([test_df, interaction_df_test], axis=1)

    return train_df, test_df

# Step 4. Augment the training data
def augment_data(train_df):
    smote = SMOTE()
    X_train, y_train = smote.fit_resample(train_df.drop(columns=['obesity_risk']), train_df['obesity_risk'])
    train_df = pd.concat([pd.DataFrame(X_train, columns=train_df.drop(columns=['obesity_risk']).columns), pd.Series(y_train, name='obesity_risk')], axis=1)
    return train_df

# Step 5. Analyze the data
def analyze_data(train_df):
    summary_stats = train_df.describe()
    correlation_matrix = train_df.corr()
    class_distribution = train_df['obesity_risk'].value_counts()

    # Feature Importance Analysis
    importance_scores = mutual_info_classif(train_df.drop(columns=['obesity_risk']), train_df['obesity_risk'])
    feature_importance = pd.Series(importance_scores, index=train_df.drop(columns=['obesity_risk']).columns)

    model = RandomForestClassifier()
    model.fit(train_df.drop(columns=['obesity_risk']), train_df['obesity_risk'])
    feature_importance_rf = pd.Series(model.feature_importances_, index=train_df.drop(columns=['obesity_risk']).columns)

    return summary_stats, correlation_matrix, class_distribution, feature_importance, feature_importance_rf

# Step 6. Define required model
def prepare_data(train_df, valid_df, test_df):
    X_train = train_df.drop(columns=['obesity_risk'])
    y_train = train_df['obesity_risk']
    X_valid = valid_df.drop(columns=['obesity_risk'])
    y_valid = valid_df['obesity_risk']
    X_test = test_df.drop(columns=['obesity_risk'])
    y_test = test_df['obesity_risk']
    return X_train, y_train, X_valid, y_valid, X_test, y_test

# Step 7. Train the model
def train_model(X_train, y_train):
    gb_model = GradientBoostingClassifier(n_estimators=400, learning_rate=0.05, max_depth=7, subsample=0.8)
    gb_model.fit(X_train, y_train)
    return gb_model

# Step 8. Evaluate the model
def evaluate_model(model, X_test, y_test):
    y_pred = model.predict(X_test)
    accuracy = accuracy_score(y_test, y_pred)
    precision = precision_score(y_test, y_pred, average='weighted')
    recall = recall_score(y_test, y_pred, average='weighted')
    f1 = f1_score(y_test, y_pred, average='weighted')
    performance_scores = {
        'Accuracy': accuracy,
        'Precision': precision,
        'Recall': recall,
        'F1-score': f1
    }
    return performance_scores

# Step 9. Save the model
def save_model(model, model_path):
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    joblib.dump(model, model_path)

# Step 10. Load the model for inference
def load_model(model_path):
    return joblib.load(model_path)

# Step 11. Model Interpretability
def model_interpretability(model, X_test):
    explainer = shap.TreeExplainer(model)
    shap_values = explainer.shap_values(X_test)
    shap.summary_plot(shap_values, X_test, plot_type="bar")

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/obesity_CVD_risk/train.csv'
    test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/obesity_CVD_risk/test.csv'
    train_df, test_df = load_datasets(train_path, test_path)

    # Check if 'obesity_risk' is in the dataset
    if 'obesity_risk' not in train_df.columns:
        raise KeyError("The target variable 'obesity_risk' is not present in the training dataset.")
    if 'obesity_risk' not in test_df.columns:
        raise KeyError("The target variable 'obesity_risk' is not present in the test dataset.")

    # Step 2. Create a train-valid-test split of the data by splitting the `dataset` into train_loader, valid_loader, and test_loader.
    train_df, valid_df = split_data(train_df, test_size=0.2, random_state=SEED, stratify=train_df['obesity_risk'])

    # Step 3. With the split dataset, run data preprocessing and feature engineering (if applicable) using the "preprocess_data" function you defined
    train_df, test_df = preprocess_data(train_df, test_df)
    valid_df, _ = preprocess_data(valid_df, valid_df)  # Preprocess validation data

    # Step 4. Augment the training data
    train_df = augment_data(train_df)

    # Step 5. Analyze the data
    summary_stats, correlation_matrix, class_distribution, feature_importance, feature_importance_rf = analyze_data(train_df)

    # Step 6. Define required model
    X_train, y_train, X_valid, y_valid, X_test, y_test = prepare_data(train_df, valid_df, test_df)

    # Step 7. Train the model
    gb_model = train_model(X_train, y_train)

    # Step 8. Evaluate the model
    model_performance = evaluate_model(gb_model, X_test, y_test)

    # Step 9. Save the model
    model_path = './agent_workspace/trained_models/obesity_risk_gradient_boosting.pkl'
    save_model(gb_model, model_path)

    # Step 10. Load the model for inference
    gb_model = load_model(model_path)

    # Step 11. Model Interpretability
    model_interpretability(gb_model, X_test)

    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
