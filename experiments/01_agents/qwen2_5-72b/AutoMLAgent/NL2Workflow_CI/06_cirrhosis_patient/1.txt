import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.model_selection import train_test_split
from sklearn.ensemble import GradientBoostingClassifier
from sklearn.metrics import log_loss
import pickle
import shap

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
def load_datasets():
    train_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/cirrhosis_patient/train.csv')
    test_df = pd.read_csv('/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/cirrhosis_patient/test.csv')
    return train_df, test_df

# Step 2. Preprocess the data
def preprocess_data(train_df, test_df):
    # Identify numerical and categorical features
    numerical_features = [col for col in train_df.columns if train_df[col].dtype != 'object' and col != 'survival_status']
    categorical_features = [col for col in train_df.columns if train_df[col].dtype == 'object']

    # Impute missing values in numerical features
    numerical_imputer = SimpleImputer(strategy='mean')
    train_df[numerical_features] = numerical_imputer.fit_transform(train_df[numerical_features])
    test_df[numerical_features] = numerical_imputer.transform(test_df[numerical_features])

    # Impute missing values in categorical features
    categorical_imputer = SimpleImputer(strategy='most_frequent')
    train_df[categorical_features] = categorical_imputer.fit_transform(train_df[categorical_features])
    test_df[categorical_features] = categorical_imputer.transform(test_df[categorical_features])

    # Standardize numerical features
    scaler = StandardScaler()
    train_df[numerical_features] = scaler.fit_transform(train_df[numerical_features])
    test_df[numerical_features] = scaler.transform(test_df[numerical_features])

    # One-hot encode categorical features
    encoder = OneHotEncoder(drop='first', sparse_output=False)
    encoded_train = encoder.fit_transform(train_df[categorical_features])
    encoded_test = encoder.transform(test_df[categorical_features])

    # Create new DataFrame with encoded features
    encoded_train_df = pd.DataFrame(encoded_train, columns=encoder.get_feature_names_out(categorical_features))
    encoded_test_df = pd.DataFrame(encoded_test, columns=encoder.get_feature_names_out(categorical_features))

    # Concatenate encoded features with numerical features
    train_df = pd.concat([train_df[numerical_features], encoded_train_df], axis=1)
    test_df = pd.concat([test_df[numerical_features], encoded_test_df], axis=1)

    return train_df, test_df

# Step 3. Split the data into training and validation sets
def split_data(train_df):
    if 'survival_status' not in train_df.columns:
        raise KeyError("The column 'survival_status' is not present in the training dataset.")
    
    X_train, X_val, y_train, y_val = train_test_split(
        train_df.drop(columns=['survival_status'], errors='ignore'), 
        train_df['survival_status'], 
        test_size=0.2, 
        random_state=42
    )
    return X_train, X_val, y_train, y_val

# Step 4. Train the model
def train_model(X_train, y_train):
    gbm_model = GradientBoostingClassifier(n_estimators=100, learning_rate=0.1, max_depth=5, subsample=0.8, random_state=42)
    gbm_model.fit(X_train, y_train)
    return gbm_model

# Step 5. Evaluate the model on the validation set
def evaluate_model(model, X_val, y_val):
    y_val_pred_proba = model.predict_proba(X_val)
    val_logloss = log_loss(y_val, y_val_pred_proba)
    return val_logloss

# Step 6. Evaluate the model on the test set
def evaluate_test_model(model, test_df):
    if 'survival_status' not in test_df.columns:
        raise KeyError("The column 'survival_status' is not present in the test dataset.")
    
    y_test_pred_proba = model.predict_proba(test_df.drop(columns=['survival_status'], errors='ignore'))
    test_logloss = log_loss(test_df['survival_status'], y_test_pred_proba)
    return test_logloss

# Step 7. Feature importance
def feature_importance(model, X_train):
    feature_importances = model.feature_importances_
    important_features = pd.DataFrame({'Feature': X_train.columns, 'Importance': feature_importances})
    important_features = important_features.sort_values(by='Importance', ascending=False)
    return important_features

# Step 8. Model explainability
def model_explainability(model, X_train):
    explainer = shap.TreeExplainer(model)
    shap_values = explainer.shap_values(X_train)
    shap.summary_plot(shap_values, X_train, plot_type="bar")

# Step 9. Save the trained model
def save_model(model):
    model_path = "./agent_workspace/trained_models/gbm_model.pkl"
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    with open(model_path, 'wb') as f:
        pickle.dump(model, f)

# The main function to orchestrate the data loading, data preprocessing, feature engineering, model training, model preparation, and model evaluation
def main():
    # Step 1. Retrieve or load a dataset from hub (if available) or user's local storage (if given)
    train_df, test_df = load_datasets()

    # Step 2. Preprocess the data
    train_df, test_df = preprocess_data(train_df, test_df)

    # Step 3. Split the data into training and validation sets
    X_train, X_val, y_train, y_val = split_data(train_df)

    # Step 4. Train the model
    model = train_model(X_train, y_train)

    # Step 5. Evaluate the model on the validation set
    val_logloss = evaluate_model(model, X_val, y_val)

    # Step 6. Evaluate the model on the test set
    test_logloss = evaluate_test_model(model, test_df)

    # Step 7. Feature importance
    important_features = feature_importance(model, X_train)

    # Step 8. Model explainability
    model_explainability(model, X_train)

    # Step 9. Save the trained model
    save_model(model)

    # Return the performance scores
    return {
        'Validation Logloss': val_logloss,
        'Test Logloss': test_logloss,
        'Feature Importance': important_features
    }

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
