import os
import random
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler, OneHotEncoder, PolynomialFeatures
from sklearn.model_selection import train_test_split, GridSearchCV
from sklearn.ensemble import GradientBoostingClassifier
from sklearn.metrics import log_loss
import joblib

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data Loading and Preprocessing
def load_data():
    train_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/cirrhosis_patient/train.csv'
    test_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/cirrhosis_patient/test.csv'

    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)

    print("Training Data:")
    print(train_df.head())
    print(train_df.columns)
    print("Test Data:")
    print(test_df.head())
    print(test_df.columns)

    return train_df, test_df

def preprocess_data(train_df, test_df):
    # Handling Missing Values
    numerical_columns = train_df.select_dtypes(include=[np.number]).columns
    categorical_columns = train_df.select_dtypes(include=[object]).columns

    numerical_imputer = SimpleImputer(strategy='mean')
    train_df[numerical_columns] = numerical_imputer.fit_transform(train_df[numerical_columns])
    test_df[numerical_columns] = numerical_imputer.transform(test_df[numerical_columns])

    categorical_imputer = SimpleImputer(strategy='most_frequent')
    train_df[categorical_columns] = categorical_imputer.fit_transform(train_df[categorical_columns])
    test_df[categorical_columns] = categorical_imputer.transform(test_df[categorical_columns])

    # Feature Scaling
    scaler = StandardScaler()
    train_df[numerical_columns] = scaler.fit_transform(train_df[numerical_columns])
    test_df[numerical_columns] = scaler.transform(test_df[numerical_columns])

    # Feature Engineering
    interaction_features = PolynomialFeatures(interaction_only=True, include_bias=False)
    train_interactions = interaction_features.fit_transform(train_df[numerical_columns])
    test_interactions = interaction_features.transform(test_df[numerical_columns])

    polynomial_features = PolynomialFeatures(degree=2, include_bias=False)
    train_polynomials = polynomial_features.fit_transform(train_df[numerical_columns])
    test_polynomials = polynomial_features.transform(test_df[numerical_columns])

    train_df = pd.concat([train_df, pd.DataFrame(train_interactions), pd.DataFrame(train_polynomials)], axis=1)
    test_df = pd.concat([test_df, pd.DataFrame(test_interactions), pd.DataFrame(test_polynomials)], axis=1)

    # Categorical Encoding
    encoder = OneHotEncoder(handle_unknown='ignore', sparse_output=False)
    train_encoded = encoder.fit_transform(train_df[categorical_columns])
    test_encoded = encoder.transform(test_df[categorical_columns])

    train_df = pd.concat([train_df, pd.DataFrame(train_encoded)], axis=1)
    test_df = pd.concat([test_df, pd.DataFrame(test_encoded)], axis=1)

    # Data Splitting
    if 'survival_status' not in train_df.columns:
        raise KeyError("Column 'survival_status' not found in the training dataset.")

    X_train, X_val, y_train, y_val = train_test_split(
        train_df.drop(columns=['survival_status']),
        train_df['survival_status'],
        test_size=0.2,
        stratify=train_df['survival_status'],
        random_state=42
    )

    return X_train, X_val, y_train, y_val, test_df

# Model Training and Hyperparameter Tuning
def train_model(X_train, y_train):
    model = GradientBoostingClassifier()
    model.fit(X_train, y_train)

    param_grid = {
        'n_estimators': [100, 200, 300],
        'learning_rate': [0.01, 0.1, 0.2],
        'max_depth': [3, 5, 7]
    }

    grid_search = GridSearchCV(model, param_grid, cv=5, scoring='neg_log_loss', n_jobs=-1)
    grid_search.fit(X_train, y_train)

    best_model = grid_search.best_estimator_
    return best_model

# Model Evaluation
def evaluate_model(model, X_val, y_val, test_df):
    y_val_pred = model.predict_proba(X_val)
    val_logloss = log_loss(y_val, y_val_pred)
    print(f'Validation Logloss: {val_logloss}')

    y_test_pred = model.predict_proba(test_df.drop(columns=['survival_status']))
    test_logloss = log_loss(test_df['survival_status'], y_test_pred)
    print(f'Test Logloss: {test_logloss}')

    performance_scores = {
        'Validation Logloss': val_logloss,
        'Test Logloss': test_logloss
    }
    return performance_scores

# Model Saving and Deployment
def save_model(model, model_path):
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    joblib.dump(model, model_path)

def load_model(model_path):
    return joblib.load(model_path)

# The main function to orchestrate the entire pipeline
def main():
    train_df, test_df = load_data()
    X_train, X_val, y_train, y_val, test_df = preprocess_data(train_df, test_df)
    model = train_model(X_train, y_train)
    performance_scores = evaluate_model(model, X_val, y_val, test_df)
    model_path = './agent_workspace/trained_models/cirrhosis_patient_model.pkl'
    save_model(model, model_path)
    return performance_scores

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
