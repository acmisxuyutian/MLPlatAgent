######代码######

import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split, cross_val_score
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import OneHotEncoder, StandardScaler
from sklearn.metrics import log_loss, confusion_matrix, accuracy_score, precision_score, recall_score, f1_score
import xgboost as xgb
import joblib
import shap

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data Retrieval
def load_datasets(train_path, test_path):
    # Load the training dataset
    train_df = pd.read_csv(train_path)
    
    # Load the test dataset
    test_df = pd.read_csv(test_path)
    
    return train_df, test_df

# Data Preprocessing
def preprocess_data(train_df, test_df):
    # Ensure the target column exists
    if 'survival_status' not in train_df.columns:
        raise ValueError("The 'survival_status' column is missing in the training dataset.")
    
    # Split the training dataset into training and validation sets
    X_train, X_val, y_train, y_val = train_test_split(
        train_df.drop(columns=['survival_status']),
        train_df['survival_status'],
        test_size=0.2,
        stratify=train_df['survival_status'],
        random_state=42
    )
    
    # Impute missing values in numerical features
    num_imputer = SimpleImputer(strategy='median')
    X_train_num = num_imputer.fit_transform(X_train.select_dtypes(include=['number']))
    X_val_num = num_imputer.transform(X_val.select_dtypes(include=['number']))
    X_test_num = num_imputer.transform(test_df.select_dtypes(include=['number']))
    
    # Impute missing values in categorical features
    cat_imputer = SimpleImputer(strategy='most_frequent')
    X_train_cat = cat_imputer.fit_transform(X_train.select_dtypes(include=['object']))
    X_val_cat = cat_imputer.transform(X_val.select_dtypes(include=['object']))
    X_test_cat = cat_imputer.transform(test_df.select_dtypes(include=['object']))
    
    # Encode categorical features
    cat_encoder = OneHotEncoder(handle_unknown='ignore')
    X_train_cat_encoded = cat_encoder.fit_transform(X_train_cat)
    X_val_cat_encoded = cat_encoder.transform(X_val_cat)
    X_test_cat_encoded = cat_encoder.transform(X_test_cat)
    
    # Scale numerical features
    scaler = StandardScaler()
    X_train_num_scaled = scaler.fit_transform(X_train_num)
    X_val_num_scaled = scaler.transform(X_val_num)
    X_test_num_scaled = scaler.transform(X_test_num)
    
    # Combine numerical and categorical features
    X_train_final = np.hstack((X_train_num_scaled, X_train_cat_encoded.toarray()))
    X_val_final = np.hstack((X_val_num_scaled, X_val_cat_encoded.toarray()))
    X_test_final = np.hstack((X_test_num_scaled, X_test_cat_encoded.toarray()))
    
    return X_train_final, y_train, X_val_final, y_val, X_test_final, test_df['survival_status'], num_imputer, cat_imputer, cat_encoder, scaler

# Model Training
def train_model(X_train, y_train):
    # Initialize the XGBoost classifier
    xgb_model = xgb.XGBClassifier(
        n_estimators=100,
        learning_rate=0.1,
        max_depth=3,
        subsample=0.8,
        random_state=42
    )
    
    # Train the model using 5-fold cross-validation
    cv_scores = cross_val_score(xgb_model, X_train, y_train, cv=5, scoring='neg_log_loss')
    print(f'Cross-validation logloss scores: {cv_scores}')
    print(f'Mean cross-validation logloss: {-cv_scores.mean()}')
    
    # Train the final model on the entire training set
    xgb_model.fit(X_train, y_train)
    
    return xgb_model

# Model Evaluation
def evaluate_model(model, X_val, y_val):
    # Predict on the validation set
    y_val_pred = model.predict(X_val)
    y_val_pred_proba = model.predict_proba(X_val)
    
    # Evaluate the model
    val_logloss = log_loss(y_val, y_val_pred_proba)
    val_conf_matrix = confusion_matrix(y_val, y_val_pred)
    val_accuracy = accuracy_score(y_val, y_val_pred)
    val_precision = precision_score(y_val, y_val_pred)
    val_recall = recall_score(y_val, y_val_pred)
    val_f1 = f1_score(y_val, y_val_pred)
    
    print(f'Validation logloss: {val_logloss}')
    print(f'Validation confusion matrix:\n{val_conf_matrix}')
    print(f'Validation accuracy: {val_accuracy}')
    print(f'Validation precision: {val_precision}')
    print(f'Validation recall: {val_recall}')
    print(f'Validation F1-score: {val_f1}')
    
    performance_scores = {
        'logloss': val_logloss,
        'accuracy': val_accuracy,
        'precision': val_precision,
        'recall': val_recall,
        'f1_score': val_f1
    }
    
    return performance_scores

# Model Testing
def test_model(model, X_test, y_test):
    # Predict on the test set
    y_test_pred = model.predict(X_test)
    y_test_pred_proba = model.predict_proba(X_test)
    
    # Evaluate the model on the test set
    test_logloss = log_loss(y_test, y_test_pred_proba)
    test_conf_matrix = confusion_matrix(y_test, y_test_pred)
    test_accuracy = accuracy_score(y_test, y_test_pred)
    test_precision = precision_score(y_test, y_test_pred)
    test_recall = recall_score(y_test, y_test_pred)
    test_f1 = f1_score(y_test, y_test_pred)
    
    print(f'Test logloss: {test_logloss}')
    print(f'Test confusion matrix:\n{test_conf_matrix}')
    print(f'Test accuracy: {test_accuracy}')
    print(f'Test precision: {test_precision}')
    print(f'Test recall: {test_recall}')
    print(f'Test F1-score: {test_f1}')
    
    performance_scores = {
        'logloss': test_logloss,
        'accuracy': test_accuracy,
        'precision': test_precision,
        'recall': test_recall,
        'f1_score': test_f1
    }
    
    return performance_scores

# Model Saving
def save_model(model, num_imputer, cat_imputer, cat_encoder, scaler, model_dir):
    os.makedirs(model_dir, exist_ok=True)
    joblib.dump(model, os.path.join(model_dir, 'cirrhosis_patient_model.pkl'))
    joblib.dump(num_imputer, os.path.join(model_dir, 'num_imputer.pkl'))
    joblib.dump(cat_imputer, os.path.join(model_dir, 'cat_imputer.pkl'))
    joblib.dump(cat_encoder, os.path.join(model_dir, 'cat_encoder.pkl'))
    joblib.dump(scaler, os.path.join(model_dir, 'scaler.pkl'))

# Model Interpretability
def interpret_model(model, X_train, feature_names):
    # Create a SHAP explainer
    explainer = shap.Explainer(model)
    shap_values = explainer(X_train)
    
    # Plot SHAP summary
    shap.summary_plot(shap_values, X_train, feature_names=feature_names)

# Main function to orchestrate the pipeline
def main():
    # Load datasets
    train_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/cirrhosis_patient/train.csv'
    test_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/cirrhosis_patient/test.csv'
    train_df, test_df = load_datasets(train_path, test_path)
    
    # Preprocess data
    X_train, y_train, X_val, y_val, X_test, y_test, num_imputer, cat_imputer, cat_encoder, scaler = preprocess_data(train_df, test_df)
    
    # Train model
    model = train_model(X_train, y_train)
    
    # Evaluate model on validation set
    val_performance = evaluate_model(model, X_val, y_val)
    
    # Test model on test set
    test_performance = test_model(model, X_test, y_test)
    
    # Save model and preprocessing steps
    model_dir = r'./agent_workspace/trained_models'
    save_model(model, num_imputer, cat_imputer, cat_encoder, scaler, model_dir)
    
    # Interpret model
    feature_names = train_df.drop(columns=['survival_status']).columns.tolist()
    interpret_model(model, X_train, feature_names)
    
    return test_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
