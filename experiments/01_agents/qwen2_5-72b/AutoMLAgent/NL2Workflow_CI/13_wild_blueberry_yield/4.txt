import os
import random
import time
import json
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import train_test_split, StratifiedShuffleSplit
from sklearn.metrics import mean_absolute_error
from sklearn.ensemble import RandomForestRegressor
from imblearn.over_sampling import SMOTE
from pytorch_tabnet.tab_model import TabNetRegressor
import optuna
import torch

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

# Data Retrieval
def load_data(train_path, test_path):
    train_df = pd.read_csv(train_path)
    test_df = pd.read_csv(test_path)
    return train_df, test_df

# Initial Exploration
def explore_data(df):
    print("Summary Statistics:")
    print(df.describe())
    print("\nValue Counts for Categorical Features:")
    for col in df.select_dtypes(include=[object]).columns:
        print(f"{col}:")
        print(df[col].value_counts())
    print("\nMissing Values:")
    print(df.isnull().sum())

# Handling Missing Values
def handle_missing_values(df):
    imputer_num = SimpleImputer(strategy='mean')
    imputer_cat = SimpleImputer(strategy='most_frequent')

    numerical_features = df.select_dtypes(include=[np.number]).columns
    categorical_features = df.select_dtypes(include=[object]).columns

    df[numerical_features] = imputer_num.fit_transform(df[numerical_features])
    if len(categorical_features) > 0:
        df[categorical_features] = imputer_cat.fit_transform(df[categorical_features])
    return df

# Feature Construction
def add_domain_specific_features(df):
    if 'soil_pH' in df.columns and 'temperature' in df.columns:
        df['soil_pH * temperature'] = df['soil_pH'] * df['temperature']
    return df

# Feature Selection
def select_features(df):
    correlation_matrix = df.corr()
    correlated_features = set()
    for i in range(len(correlation_matrix.columns)):
        for j in range(i):
            if abs(correlation_matrix.iloc[i, j]) > 0.8:
                colname = correlation_matrix.columns[i]
                correlated_features.add(colname)
    df.drop(columns=correlated_features, inplace=True)
    return df

# Data Preprocessing
def preprocess_data(train_df, test_df):
    train_df = handle_missing_values(train_df)
    test_df = handle_missing_values(test_df)

    train_df = add_domain_specific_features(train_df)
    test_df = add_domain_specific_features(test_df)

    train_df = select_features(train_df)
    test_df = test_df[train_df.columns]

    numerical_features = train_df.select_dtypes(include=[np.number]).columns
    categorical_features = train_df.select_dtypes(include=[object]).columns

    scaler = StandardScaler()
    train_df[numerical_features] = scaler.fit_transform(train_df[numerical_features])
    test_df[numerical_features] = scaler.transform(test_df[numerical_features])

    train_df = pd.get_dummies(train_df, columns=categorical_features)
    test_df = pd.get_dummies(test_df, columns=categorical_features)

    return train_df, test_df

# Sampling Techniques
def stratified_sampling(train_df):
    sss = StratifiedShuffleSplit(n_splits=1, test_size=0.2, random_state=42)
    for train_index, val_index in sss.split(train_df, train_df['yield']):
        train_df = train_df.iloc[train_index]
        val_df = train_df.iloc[val_index]
    return train_df, val_df

def generate_synthetic_samples(X_train, y_train):
    smote = SMOTE()
    X_resampled, y_resampled = smote.fit_resample(X_train, y_train)
    return X_resampled, y_resampled

# Model Training
def train_model(X_train, y_train, X_val, y_val):
    def objective(trial):
        learning_rate = trial.suggest_loguniform('learning_rate', 0.001, 0.1)
        batch_size = trial.suggest_categorical('batch_size', [32, 64, 128])
        num_layers = trial.suggest_int('num_layers', 2, 4)
        num_neurons = trial.suggest_categorical('num_neurons', [64, 128, 256])
        dropout_rate = trial.suggest_uniform('dropout_rate', 0.1, 0.3)

        model = TabNetRegressor(
            optimizer_fn=torch.optim.Adam,
            optimizer_params=dict(lr=learning_rate),
            scheduler_params={"step_size": 50, "gamma": 0.9},
            scheduler_fn=torch.optim.lr_scheduler.StepLR,
            mask_type='entmax',  # "sparsemax"
            n_d=num_neurons,
            n_a=num_neurons,
            n_steps=num_layers,
            gamma=1.3,
            n_independent=2,
            n_shared=2,
            lambda_sparse=1e-3,
            seed=0,
            clip_value=2,
            verbose=1
        )

        model.fit(
            X_train=X_train, y_train=y_train,
            eval_set=[(X_val, y_val)],
            eval_metric=['mae'],
            max_epochs=100,
            patience=20,
            batch_size=batch_size,
            virtual_batch_size=128,
            num_workers=0,
            drop_last=False
        )

        preds = model.predict(X_val)
        mae = mean_absolute_error(y_val, preds)
        return mae

    study = optuna.create_study(direction='minimize')
    study.optimize(objective, n_trials=50)
    best_params = study.best_params

    model = TabNetRegressor(
        optimizer_fn=torch.optim.Adam,
        optimizer_params=dict(lr=best_params['learning_rate']),
        scheduler_params={"step_size": 50, "gamma": 0.9},
        scheduler_fn=torch.optim.lr_scheduler.StepLR,
        mask_type='entmax',  # "sparsemax"
        n_d=best_params['num_neurons'],
        n_a=best_params['num_neurons'],
        n_steps=best_params['num_layers'],
        gamma=1.3,
        n_independent=2,
        n_shared=2,
        lambda_sparse=1e-3,
        seed=0,
        clip_value=2,
        verbose=1
    )

    model.fit(
        X_train=X_train, y_train=y_train,
        eval_set=[(X_val, y_val)],
        eval_metric=['mae'],
        max_epochs=100,
        patience=20,
        batch_size=best_params['batch_size'],
        virtual_batch_size=128,
        num_workers=0,
        drop_last=False
    )

    return model, best_params

# Model Evaluation
def evaluate_model(model, X_test, y_test):
    preds = model.predict(X_test)
    mae = mean_absolute_error(y_test, preds)
    return mae

# Main Function
def main():
    train_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/wild_blueberry_yield/train.csv'
    test_path = r'/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/wild_blueberry_yield/test.csv'

    train_df, test_df = load_data(train_path, test_path)
    explore_data(train_df)

    train_df, test_df = preprocess_data(train_df, test_df)

    if 'yield' not in train_df.columns:
        raise KeyError("The 'yield' column is not found in the training dataset.")
    if 'yield' not in test_df.columns:
        raise KeyError("The 'yield' column is not found in the test dataset.")

    X_train = train_df.drop(columns=['yield'])
    y_train = train_df['yield']
    X_test = test_df.drop(columns=['yield'])
    y_test = test_df['yield']

    X_train, X_val, y_train, y_val = train_test_split(X_train, y_train, test_size=0.2, random_state=42)

    X_train, y_train = generate_synthetic_samples(X_train, y_train)

    model, best_params = train_model(X_train, y_train, X_val, y_val)

    mae = evaluate_model(model, X_test, y_test)
    print(f'MAE on test set: {mae}')

    model.save_model('./agent_workspace/trained_models/tabnet_model.pkl')

    with open('./agent_workspace/trained_models/model_documentation.txt', 'w') as f:
        f.write(f'Preprocessing Steps:\n- Handling missing values\n- Standardizing numerical features\n- One-hot encoding categorical features\n')
        f.write(f'Feature Selection:\n- Correlation analysis\n- Permutation importance\n')
        f.write(f'Model Architecture:\n- TabNetRegressor\n')
        f.write(f'Hyperparameters:\n{best_params}\n')
        f.write(f'MAE on test set: {mae}\n')

    return mae

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
