import os
import random
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler
from sklearn.feature_selection import RFE
from sklearn.ensemble import RandomForestClassifier
from imblearn.over_sampling import SMOTE
from xgboost import XGBClassifier
from sklearn.model_selection import cross_val_score
from sklearn.metrics import roc_auc_score
import joblib

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

def load_data(file_path):
    """Load data from a CSV file."""
    return pd.read_csv(file_path)

def preprocess_data(train_data, test_data):
    """Preprocess the data including handling missing values, feature scaling, feature engineering, and feature selection."""
    
    # Check for missing values
    missing_values_train = train_data.isnull().sum()
    print("Missing Values in Train Data:\n", missing_values_train)
    
    missing_values_test = test_data.isnull().sum()
    print("Missing Values in Test Data:\n", missing_values_test)
    
    # Check data types
    data_types_train = train_data.dtypes
    print("Data Types in Train Data:\n", data_types_train)
    
    data_types_test = test_data.dtypes
    print("Data Types in Test Data:\n", data_types_test)
    
    # Summarize the dataset
    data_summary_train = train_data.describe()
    print("Data Summary in Train Data:\n", data_summary_train)
    
    data_summary_test = test_data.describe()
    print("Data Summary in Test Data:\n", data_summary_test)
    
    # Summarize the target variable
    if 'has_kidney_stone' in train_data.columns:
        target_summary_train = train_data['has_kidney_stone'].value_counts()
        print("Target Variable Summary in Train Data:\n", target_summary_train)
    else:
        raise KeyError("The target variable 'has_kidney_stone' is not found in the train dataset.")
    
    if 'has_kidney_stone' in test_data.columns:
        target_summary_test = test_data['has_kidney_stone'].value_counts()
        print("Target Variable Summary in Test Data:\n", target_summary_test)
    else:
        raise KeyError("The target variable 'has_kidney_stone' is not found in the test dataset.")
    
    # Impute missing values for numerical features using the mean
    numerical_features = train_data.select_dtypes(include=['float64', 'int64']).columns
    numerical_imputer = SimpleImputer(strategy='mean')
    train_data[numerical_features] = numerical_imputer.fit_transform(train_data[numerical_features])
    test_data[numerical_features] = numerical_imputer.transform(test_data[numerical_features])
    
    # Impute missing values for categorical features using the mode
    categorical_features = train_data.select_dtypes(include=['object']).columns
    categorical_imputer = SimpleImputer(strategy='most_frequent')
    train_data[categorical_features] = categorical_imputer.fit_transform(train_data[categorical_features])
    test_data[categorical_features] = categorical_imputer.transform(test_data[categorical_features])
    
    # Standardize numerical features
    scaler = StandardScaler()
    train_data[numerical_features] = scaler.fit_transform(train_data[numerical_features])
    test_data[numerical_features] = scaler.transform(test_data[numerical_features])
    
    # Create interaction terms
    train_data['age_blood_pressure'] = train_data['age'] * train_data['blood_pressure']
    test_data['age_blood_pressure'] = test_data['age'] * test_data['blood_pressure']
    
    # Feature selection
    model = RandomForestClassifier()
    rfe = RFE(estimator=model, n_features_to_select=10)
    data_selected = rfe.fit_transform(train_data[numerical_features], train_data['has_kidney_stone'])
    selected_features = train_data.columns[rfe.support_]
    print("Selected Features:\n", selected_features)
    
    # Apply SMOTE to balance the class distribution
    smote = SMOTE()
    X_resampled, y_resampled = smote.fit_resample(train_data[selected_features], train_data['has_kidney_stone'])
    data_resampled = pd.DataFrame(X_resampled, columns=selected_features)
    data_resampled['has_kidney_stone'] = y_resampled
    
    return data_resampled, numerical_imputer, categorical_imputer, scaler, selected_features

def train_model(data_resampled, selected_features):
    """Train the XGBoost model."""
    model = XGBClassifier(n_estimators=200, max_depth=5, learning_rate=0.1)
    model.fit(data_resampled[selected_features], data_resampled['has_kidney_stone'])
    return model

def evaluate_model(model, test_data, numerical_imputer, categorical_imputer, scaler, selected_features):
    """Evaluate the model on the test set."""
    # Preprocess the test dataset
    test_data[selected_features] = numerical_imputer.transform(test_data[selected_features])
    test_data[selected_features] = scaler.transform(test_data[selected_features])
    
    # Predict on the test set
    y_pred = model.predict_proba(test_data[selected_features])[:, 1]
    
    # Calculate AUC-ROC
    auc_roc = roc_auc_score(test_data['has_kidney_stone'], y_pred)
    print(f'Test AUC-ROC: {auc_roc}')
    
    performance_scores = {
        'AUC-ROC': auc_roc
    }
    return performance_scores

def main():
    """Main function to orchestrate the data loading, preprocessing, model training, and evaluation."""
    # Step 1. Retrieve or load datasets from user's local storage
    train_file_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/kidney_stone/train.csv'
    test_file_path = '/remote-home/cs_acmis_xyt/AutomlAgent/benchmark/NL2Workflow/datasets/tabular/kidney_stone/test.csv'
    train_data = load_data(train_file_path)
    test_data = load_data(test_file_path)
    
    # Step 2. Preprocess the data
    data_resampled, numerical_imputer, categorical_imputer, scaler, selected_features = preprocess_data(train_data, test_data)
    
    # Step 3. Train the model
    model = train_model(data_resampled, selected_features)
    
    # Step 4. Evaluate the model
    model_performance = evaluate_model(model, test_data, numerical_imputer, categorical_imputer, scaler, selected_features)
    
    # Step 5. Save the model
    model_path = './agent_workspace/trained_models/kidney_stone_model.pkl'
    os.makedirs(os.path.dirname(model_path), exist_ok=True)
    joblib.dump(model, model_path)
    
    return model_performance

if __name__ == "__main__":
    model_performance = main()
    print("Model Performance on Test Set:", model_performance)
