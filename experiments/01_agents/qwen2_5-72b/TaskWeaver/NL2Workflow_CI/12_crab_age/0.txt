import pandas as pd
from sklearn.preprocessing import OneHotEncoder

# Load the datasets
train_data = pd.read_csv(r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/crab_age/train.csv')
test_data = pd.read_csv(r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/crab_age/test.csv')

# Perform one-hot encoding
encoder = OneHotEncoder()
encoded_train = encoder.fit_transform(train_data.select_dtypes(include=['object']))
encoded_test = encoder.transform(test_data.select_dtypes(include=['object']))

# Convert encoded data back to DataFrame
encoded_train_df = pd.DataFrame(encoded_train.toarray(), columns=encoder.get_feature_names_out())
encoded_test_df = pd.DataFrame(encoded_test.toarray(), columns=encoder.get_feature_names_out())

# Concatenate the original numerical features with the encoded categorical features
train_data_encoded = pd.concat([train_data.select_dtypes(exclude=['object']), encoded_train_df], axis=1)
test_data_encoded = pd.concat([test_data.select_dtypes(exclude=['object']), encoded_test_df], axis=1)
