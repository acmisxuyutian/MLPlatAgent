import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import train_test_split
from sklearn.metrics import roc_auc_score
from sklearn.ensemble import RandomForestClassifier

# Load the train and eval data
train_data_path = 'D:\\LLM-Driven_AI-Studio\\MLAgent\\data\\benchmark\\ml_benchmark\\06_santander_customer\\split_train.csv'
eval_data_path = 'D:\\LLM-Driven_AI-Studio\\MLAgent\\data\\benchmark\\ml_benchmark\\06_santander_customer\\split_eval.csv'

train_data = pd.read_csv(train_data_path)
eval_data = pd.read_csv(eval_data_path)

# Exploratory Data Analysis (EDA)
# Check the distribution of the target variable
sns.countplot(x='target', data=train_data)
plt.title('Target Variable Distribution')
plt.show()

# Check the distribution of the features
train_data.hist(bins=50, figsize=(20, 15))
plt.suptitle('Feature Distributions')
plt.show()

# Check for missing values
missing_values = train_data.isnull().sum()
print('Missing Values:\n', missing_values)

# Data Preprocessing
# Handle missing values (if any)
train_data.fillna(train_data.mean(), inplace=True)
eval_data.fillna(eval_data.mean(), inplace=True)

# Scale numerical features
scaler = StandardScaler()
train_data_scaled = scaler.fit_transform(train_data.drop(columns=['target']))
eval_data_scaled = scaler.transform(eval_data.drop(columns=['target']))

# Feature Engineering (if needed)
# For now, we will use the scaled features as they are

# Modeling
# Split the training data into training and validation sets
X_train, X_val, y_train, y_val = train_test_split(
    train_data_scaled, train_data['target'], test_size=0.2, random_state=42)

# Train a Random Forest Classifier
model = RandomForestClassifier(n_estimators=100, random_state=42)
model.fit(X_train, y_train)

# Predict probabilities on the validation set
y_val_pred_proba = model.predict_proba(X_val)[:, 1]

# Evaluate the model using AUC on the validation set
auc_val = roc_auc_score(y_val, y_val_pred_proba)
print(f'AUC on Validation Set: {auc_val:.4f}')

# Predict probabilities on the eval set
y_eval_pred_proba = model.predict_proba(eval_data_scaled)[:, 1]

# Evaluate the model using AUC on the eval set
auc_eval = roc_auc_score(eval_data['target'], y_eval_pred_proba)
print(f'AUC on Eval Set: {auc_eval:.4f}')

# Result variables
train_data, eval_data, auc_val, auc_eval
