encoded_test_df = pd.DataFrame(encoded_test, columns=encoder.get_feature_names_out(['Geography', 'Card Type']))

train_data = pd.concat([train_data.drop(columns=['Geography', 'Card Type']), encoded_train_df], axis=1)
