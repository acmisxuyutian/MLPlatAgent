from sklearn.linear_model import LogisticRegression
from sklearn.metrics import accuracy_score, classification_report

# Load the training and test datasets
train_data_path = 'D:\\LLM-Driven_AI-Studio\\MLAgent\\data\\benchmark/DSEval/datasets/05_patient_profile/train.csv'
test_data_path = 'D:\\LLM-Driven_AI-Studio\\MLAgent\\data\\benchmark/DSEval/datasets/05_patient_profile/test.csv'

# Load the datasets
train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# One-hot encode categorical features (except 'Disease')
categorical_features = ['Fever', 'Cough', 'Fatigue', 'Difficulty Breathing', 'Gender', 'Blood Pressure', 'Cholesterol Level']

# One-hot encode the training data
encoder = OneHotEncoder()
encoded_train = encoder.fit_transform(train_df[categorical_features])
encoded_train_df = pd.DataFrame(encoded_train.toarray(), columns=encoder.get_feature_names_out(categorical_features))

# Combine the encoded features with the remaining numerical features
train_df_encoded = pd.concat([train_df.drop(columns=categorical_features), encoded_train_df], axis=1)

# One-hot encode the test data
encoded_test = encoder.transform(test_df[categorical_features])
encoded_test_df = pd.DataFrame(encoded_test.toarray(), columns=encoder.get_feature_names_out(categorical_features))

# Combine the encoded features with the remaining numerical features
test_df_encoded = pd.concat([test_df.drop(columns=categorical_features), encoded_test_df], axis=1)

# Feature selection: Select the most important 5 features
X_train_all = train_df_encoded.drop(columns=['Disease', 'Outcome Variable'])
y_train = train_df_encoded['Outcome Variable']

selector = SelectKBest(score_func=chi2, k=5)
selector.fit(X_train_all, y_train)
selected_features = X_train_all.columns[selector.get_support()]

# Build logistic regression model1 using all one-hot encoded features
model1 = LogisticRegression()
model1.fit(X_train_all, y_train)

# Build logistic regression model2 using only the most important 5 features
X_train_selected = train_df_encoded[selected_features]
model2 = LogisticRegression()
model2.fit(X_train_selected, y_train)

# Prepare the test data
X_test_all = test_df_encoded.drop(columns=['Disease', 'Outcome Variable'])
X_test_selected = test_df_encoded[selected_features]
y_test = test_df_encoded['Outcome Variable']

# Evaluate the performance of model1 on the test set
y_pred1 = model1.predict(X_test_all)
