import pandas as pd
from sklearn.ensemble import RandomForestClassifier

# Load the training dataset

train_df = pd.read_csv(train_data_path)

# One-hot encode categorical columns
train_df_encoded = pd.get_dummies(train_df, columns=['Drug', 'Sex', 'Ascites', 'Hepatomegaly', 'Spiders', 'Edema', 'Status'])

# Prepare the features and target
X = train_df_encoded.drop(['id', 'Status_CL', 'Status_C', 'Status_D'], axis=1)
y = train_df_encoded[['Status_CL', 'Status_C', 'Status_D']]

# Convert multi-label to single label
y_single_label = y.idxmax(axis=1).map({'Status_CL': 0, 'Status_C': 1, 'Status_D': 2})

# Split the data into training and validation sets

# Initialize the model

# Train the model
model.fit(X_train, y_train)

# Predict probabilities on the validation set
y_pred_proba = model.predict_proba(X_val)

# Calculate log loss
