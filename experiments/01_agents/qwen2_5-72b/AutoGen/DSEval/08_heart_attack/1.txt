import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import roc_auc_score
from sklearn.preprocessing import LabelEncoder

# Step 1: Load the data
train_data_path = 'D:\\LLM-Driven_AI-Studio\\MLAgent\\data\\benchmark\\DSEval\\datasets\\08_heart_attack\\train.csv'
test_data_path = 'D:\\LLM-Driven_AI-Studio\\MLAgent\\data\\benchmark\\DSEval\\datasets\\08_heart_attack\\test.csv'

train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# Step 2: Convert the values in the 'Diet' column
diet_mapping = {'Healthy': 1, 'Average': 2, 'Unhealthy': 3}
train_df['Diet'] = train_df['Diet'].map(diet_mapping)
test_df['Diet'] = test_df['Diet'].map(diet_mapping)

# Step 3: Identify and handle non-numeric columns
def handle_non_numeric_columns(df):
    for column in df.select_dtypes(include=['object']).columns:
        if df[column].nunique() > 2:
            le = LabelEncoder()
            df[column] = le.fit_transform(df[column])
        else:
            df[column] = df[column].astype('category').cat.codes
    return df

train_df = handle_non_numeric_columns(train_df)
test_df = handle_non_numeric_columns(test_df)

# Step 4: Prepare the data for the RandomForest model
X_train = train_df.drop(columns=['Heart Attack Risk'])
y_train = train_df['Heart Attack Risk']

X_test = test_df.drop(columns=['Heart Attack Risk'])
y_test = test_df['Heart Attack Risk']

# Step 5: Build and fit the RandomForest model
model = RandomForestClassifier(random_state=42)
model.fit(X_train, y_train)

# Step 6: Make predictions on the test set
y_pred_proba = model.predict_proba(X_test)[:, 1]

# Step 7: Evaluate the model using the ROC AUC score
roc_auc = roc_auc_score(y_test, y_pred_proba)

# Return the ROC AUC score in a tuple
