import pandas as pd

# Load the training and test datasets
train_path = 'D:\\LLM-Driven_AI-Studio\\MLAgent\\data\\benchmark\\DSEval\\datasets\\10_life_expectancy\\train.csv'
test_path = 'D:\\LLM-Driven_AI-Studio\\MLAgent\\data\\benchmark\\DSEval\\datasets\\10_life_expectancy\\test.csv'

train_df = pd.read_csv(train_path)
test_df = pd.read_csv(test_path)

# Define the GDP bins
def categorize_gdp(gdp):
    if gdp <= 1000:
        return 'Low GDP'
    elif 1000 < gdp <= 10000:
        return 'Medium GDP'
    else:
        return 'High GDP'

# Apply the function to create the new feature "GDP_bins"
train_df['GDP_bins'] = train_df['GDP'].apply(categorize_gdp)
test_df['GDP_bins'] = test_df['GDP'].apply(categorize_gdp)

# Convert the categorical feature "GDP_bins" to dummy variables
train_df = pd.get_dummies(train_df, columns=['GDP_bins'], drop_first=True)
test_df = pd.get_dummies(test_df, columns=['GDP_bins'], drop_first=True)

# Display the first few rows of the training and test datasets
print(train_df.head())
print(test_df.head())
