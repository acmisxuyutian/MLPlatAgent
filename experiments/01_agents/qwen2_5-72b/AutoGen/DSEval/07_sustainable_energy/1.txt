import pandas as pd

# Load the training and test datasets
train_data_path = 'D:\\LLM-Driven_AI-Studio\\MLAgent\\data\\benchmark\\DSEval\\datasets\\07_sustainable_energy\\train.csv'

train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# Display the first few rows of the training and test datasets
print("Training Data:")
print(train_df.head())
print("\nTest Data:")
print(test_df.head())

# Define the target variable and features
target = 'Renewable energy share in the total final energy consumption (%)'
features = [col for col in train_df.columns if col not in ['Entity', 'Year', target]]

# Separate the features and target variable for the training set
X_train = train_df[features]
y_train = train_df[target]

# Separate the features for the test set
X_test = test_df[features]

# Display the first few rows of the features and target variable
print("Training Features:")
print(X_train.head())
print("\nTraining Target:")
print(y_train.head())
print("\nTest Features:")
print(X_test.head())

from sklearn.linear_model import LinearRegression

# Initialize the linear regression model
model = LinearRegression()

# Fit the model on the training data
model.fit(X_train, y_train)

# Make predictions on the test set
y_pred = model.predict(X_test)

# Display the first few predictions
print("Predictions on the Test Set:")
print(y_pred[:5])

from sklearn.metrics import mean_squared_error
import numpy as np

# Load the actual target values from the test set
y_test = test_df[target]

# Calculate the RMSE
rmse = np.sqrt(mean_squared_error(y_test, y_pred))

# Display the RMSE
print(f"Root Mean Squared Error (RMSE): {rmse}")
