cat_features = train_df.select_dtypes(include=['object']).columns.tolist()

num_features.remove('SalePrice')

# Create a column transformer
preprocessor = ColumnTransformer(
    transformers=[
        ('num', Pipeline(steps=[('imputer', imputer_num), ('scaler', StandardScaler())]), num_features),
        ('cat', Pipeline(steps=[('imputer', imputer_cat), ('onehot', OneHotEncoder(handle_unknown='ignore'))]), cat_features)
    ])

# Apply the preprocessor to the training and evaluation datasets
X_train = train_df.drop(columns=['SalePrice'])
y_train = train_df['SalePrice']
X_eval = eval_df.drop(columns=['SalePrice'])
y_eval = eval_df['SalePrice']

X_eval_preprocessed = preprocessor.transform(X_eval)

# Convert the preprocessed data back to DataFrames for easier handling

print("Preprocessed Training Data:")
print(X_train_preprocessed_df.head())
