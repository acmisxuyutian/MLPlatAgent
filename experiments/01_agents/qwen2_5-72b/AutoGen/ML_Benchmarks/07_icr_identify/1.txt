from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline

# Define the columns for transformation
numerical_columns = train_df.select_dtypes(include=['float64']).columns
categorical_columns = train_df.select_dtypes(include=['object']).columns
categorical_columns = [col for col in categorical_columns if col != 'Id']

# Create the column transformer
preprocessor = ColumnTransformer(
    transformers=[
        ('num', StandardScaler(), numerical_columns),
        ('cat', OneHotEncoder(), categorical_columns)
    ])

# Create the preprocessing pipeline
preprocessing_pipeline = Pipeline(steps=[
    ('preprocessor', preprocessor)
])

# Fit and transform the train dataset
X_train = train_df.drop(columns=['Id', 'Class'])
y_train = train_df['Class']
X_train_preprocessed = preprocessing_pipeline.fit_transform(X_train)

# Transform the eval dataset
X_eval = eval_df.drop(columns=['Id', 'Class'])
y_eval = eval_df['Class']
X_eval_preprocessed = preprocessing_pipeline.transform(X_eval)

# Verify the shape of the preprocessed data
print("Shape of preprocessed train data:", X_train_preprocessed.shape)
print("Shape of preprocessed eval data:", X_eval_preprocessed.shape)
