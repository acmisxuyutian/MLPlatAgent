import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import accuracy_score

# Load the datasets
train_path = 'D:\\LLM-Driven_AI-Studio\\MLAgent\\data\\benchmark\\ml_benchmark\\04_titanic\\split_train.csv'
eval_path = 'D:\\LLM-Driven_AI-Studio\\MLAgent\\data\\benchmark\\ml_benchmark\\04_titanic\\split_eval.csv'

train_df = pd.read_csv(train_path)
eval_df = pd.read_csv(eval_path)

# Handling missing values
train_df['Age'].fillna(train_df['Age'].median(), inplace=True)
train_df['Embarked'].fillna(train_df['Embarked'].mode()[0], inplace=True)
train_df.drop(columns=['Cabin'], inplace=True)

eval_df['Age'].fillna(eval_df['Age'].median(), inplace=True)
eval_df.drop(columns=['Cabin'], inplace=True)

# Encoding categorical variables
train_df['Sex'] = train_df['Sex'].map({'male': 0, 'female': 1})
train_df = pd.get_dummies(train_df, columns=['Embarked'], drop_first=True)

eval_df['Sex'] = eval_df['Sex'].map({'male': 0, 'female': 1})
eval_df = pd.get_dummies(eval_df, columns=['Embarked'], drop_first=True)

# Dropping unnecessary columns
train_df.drop(columns=['PassengerId', 'Name', 'Ticket'], inplace=True)
eval_df.drop(columns=['PassengerId', 'Name', 'Ticket'], inplace=True)

# Feature engineering
train_df['FamilySize'] = train_df['SibSp'] + train_df['Parch'] + 1
eval_df['FamilySize'] = eval_df['SibSp'] + eval_df['Parch'] + 1

# Split the training data into training and validation sets
X_train = train_df.drop(columns=['Survived'])
y_train = train_df['Survived']
X_eval = eval_df.drop(columns=['Survived'])
y_eval = eval_df['Survived']

# Train a logistic regression model
model = LogisticRegression(max_iter=1000)
model.fit(X_train, y_train)

# Predict on the evaluation set
y_pred = model.predict(X_eval)

# Calculate accuracy
accuracy = accuracy_score(y_eval, y_pred)
print(f"Accuracy on the evaluation set: {accuracy:.4f}")
