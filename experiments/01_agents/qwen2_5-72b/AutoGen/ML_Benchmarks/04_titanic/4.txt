import pandas as pd
from sklearn.preprocessing import LabelEncoder, StandardScaler

# Load the datasets
train_path = 'D:\\LLM-Driven_AI-Studio\\MLAgent\\data\\benchmark\\ml_benchmark\\04_titanic\\split_train.csv'
eval_path = 'D:\\LLM-Driven_AI-Studio\\MLAgent\\data\\benchmark\\ml_benchmark\\04_titanic\\split_eval.csv'

train_df = pd.read_csv(train_path)
eval_df = pd.read_csv(eval_path)

# Handle missing values
train_df['Age'].fillna(train_df['Age'].median(), inplace=True)
train_df['Cabin'].fillna('Unknown', inplace=True)
train_df['Embarked'].fillna(train_df['Embarked'].mode()[0], inplace=True)

eval_df['Age'].fillna(eval_df['Age'].median(), inplace=True)
eval_df['Cabin'].fillna('Unknown', inplace=True)
eval_df['Embarked'].fillna(eval_df['Embarked'].mode()[0], inplace=True)

# Encode categorical variables
le = LabelEncoder()
train_df['Sex'] = le.fit_transform(train_df['Sex'])
train_df['Embarked'] = le.fit_transform(train_df['Embarked'])

eval_df['Sex'] = le.fit_transform(eval_df['Sex'])
eval_df['Embarked'] = le.fit_transform(eval_df['Embarked'])

# Feature engineering
train_df['Title'] = train_df['Name'].apply(lambda x: x.split(',')[1].split('.')[0].strip())
train_df['FamilySize'] = train_df['SibSp'] + train_df['Parch'] + 1

eval_df['Title'] = eval_df['Name'].apply(lambda x: x.split(',')[1].split('.')[0].strip())
eval_df['FamilySize'] = eval_df['SibSp'] + eval_df['Parch'] + 1

# Normalize numerical features
scaler = StandardScaler()
train_df[['Age', 'Fare']] = scaler.fit_transform(train_df[['Age', 'Fare']])
eval_df[['Age', 'Fare']] = scaler.transform(eval_df[['Age', 'Fare']])

# Drop unnecessary columns
train_df.drop(['Name', 'Ticket', 'Cabin'], axis=1, inplace=True)
eval_df.drop(['Name', 'Ticket', 'Cabin'], axis=1, inplace=True)

# Display the first few rows of the preprocessed training dataset
print("Preprocessed Training Data:")
print(train_df.head())

# Display the first few rows of the preprocessed evaluation dataset
print("\nPreprocessed Evaluation Data:")
print(eval_df.head())
