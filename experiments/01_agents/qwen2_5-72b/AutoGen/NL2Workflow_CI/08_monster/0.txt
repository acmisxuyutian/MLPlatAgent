import pandas as pd
from sklearn.preprocessing import LabelEncoder

# 重新加载训练集和测试集
train_path = r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/monster/train.csv'
test_path = r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/monster/test.csv'

train_df = pd.read_csv(train_path)
test_df = pd.read_csv(test_path)

# 检查缺失值
print("训练集缺失值情况:")
print(train_df.isnull().sum())
print("\n测试集缺失值情况:")
print(test_df.isnull().sum())

# 合并训练集和测试集的颜色
all_colors = pd.concat([train_df['color'], test_df['color']]).unique()

# 编码分类变量
label_encoder = LabelEncoder()
label_encoder.fit(all_colors)

# 编码训练集的 color 和 type
train_df['color'] = label_encoder.transform(train_df['color'])
train_df['type'] = label_encoder.fit_transform(train_df['type'])

# 编码测试集的 color
test_df['color'] = label_encoder.transform(test_df['color'])

# 查看编码后的数据
print("\n编码后的训练集前几行:")
print(train_df.head())
print("\n编码后的测试集前几行:")
