from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score

# 分离特征和目标变量
X_train = train_df.drop(columns=['id', 'type'])
y_train = train_df['type']

X_test = test_df.drop(columns=['id', 'type'])
y_test = test_df['type']

# 初始化随机森林分类器
rf_classifier = RandomForestClassifier(n_estimators=100, random_state=42)

# 训练模型
rf_classifier.fit(X_train, y_train)

# 在测试集上进行预测
y_pred = rf_classifier.predict(X_test)

# 计算准确率
accuracy = accuracy_score(y_test, y_pred)
print(f"模型在测试集上的准确率: {accuracy:.4f}")

# 保存预测结果
test_df['predicted_type'] = y_pred
test_df['predicted_type'] = le_type.inverse_transform(test_df['predicted_type'])

# 保存到文件
output_path = r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/monster/predictions.csv'
test_df.to_csv(output_path, index=False)

print(f"预测结果已保存到: {output_path}")
