import pandas as pd
import numpy as np
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import mean_absolute_error

# 加载训练集和测试集
train_path = r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/wild_blueberry_yield/train.csv'
test_path = r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/wild_blueberry_yield/test.csv'

train_df = pd.read_csv(train_path)
test_df = pd.read_csv(test_path)

# 特征构建
train_df['TempRangeDiff'] = train_df['MaxOfUpperTRange'] - train_df['MinOfLowerTRange']
train_df['TempRangeAvg'] = (train_df['MaxOfUpperTRange'] + train_df['MinOfLowerTRange']) / 2
test_df['TempRangeDiff'] = test_df['MaxOfUpperTRange'] - test_df['MinOfLowerTRange']
test_df['TempRangeAvg'] = (test_df['MaxOfUpperTRange'] + test_df['MinOfLowerTRange']) / 2

# 特征选择
features = ['clonesize', 'honeybee', 'bumbles', 'andrena', 'osmia', 'RainingDays', 'AverageRainingDays', 'fruitset', 'fruitmass', 'seeds', 'TempRangeDiff', 'TempRangeAvg']
target = 'yield'

X_train = train_df[features]
y_train = train_df[target]
X_test = test_df[features]
y_test = test_df[target]

# 数据标准化
scaler = StandardScaler()
X_train_scaled = scaler.fit_transform(X_train)
X_test_scaled = scaler.transform(X_test)

# 数据抽样
X_train_sample, X_val, y_train_sample, y_val = train_test_split(X_train_scaled, y_train, test_size=0.2, random_state=42)

# 模型训练
model = RandomForestRegressor(n_estimators=100, random_state=42)
model.fit(X_train_sample, y_train_sample)

# 模型评估
y_pred = model.predict(X_test_scaled)
mae = mean_absolute_error(y_test, y_pred)
print(f"测试集上的MAE: {mae}")
