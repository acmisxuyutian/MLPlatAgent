import pandas as pd
import numpy as np
from sklearn.preprocessing import StandardScaler, LabelEncoder
from sklearn.metrics import median_absolute_error
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, TensorDataset

train_path = r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/mohs_hardness/train.csv'
test_path = r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/mohs_hardness/test.csv'

train_df = pd.read_csv(train_path)
test_df = pd.read_csv(test_path)

# 假设特征列名为 'feature1', 'feature2', ..., 'featureN'，标签列为 'hardness'
feature_columns = train_df.columns[:-1]
label_column = 'hardness'

# 特征缩放
scaler = StandardScaler()
train_features = scaler.fit_transform(train_df[feature_columns])
test_features = scaler.transform(test_df[feature_columns])

# 标签编码
label_encoder = LabelEncoder()
train_labels = label_encoder.fit_transform(train_df[label_column])
test_labels = label_encoder.transform(test_df[label_column])

# 转换为 PyTorch 张量
train_features = torch.tensor(train_features, dtype=torch.float32)
train_labels = torch.tensor(train_labels, dtype=torch.float32).view(-1, 1)
test_features = torch.tensor(test_features, dtype=torch.float32)
test_labels = torch.tensor(test_labels, dtype=torch.float32).view(-1, 1)

# 创建数据加载器
train_dataset = TensorDataset(train_features, train_labels)
train_loader = DataLoader(train_dataset, batch_size=32, shuffle=True)

class MohsHardnessModel(nn.Module):
    def __init__(self, input_size):
        super(MohsHardnessModel, self).__init__()
        self.fc1 = nn.Linear(input_size, 64)
        self.fc2 = nn.Linear(64, 32)
        self.fc3 = nn.Linear(32, 1)
        self.tanh = nn.Tanh()

    def forward(self, x):
        x = self.tanh(self.fc1(x))
        x = self.tanh(self.fc2(x))
        x = self.fc3(x)
        return x

input_size = train_features.shape[1]
model = MohsHardnessModel(input_size)

criterion = nn.MSELoss()
optimizer = optim.SGD(model.parameters(), lr=0.01)

num_epochs = 100
for epoch in range(num_epochs):
    for inputs, labels in train_loader:
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
    
    if (epoch + 1) % 10 == 0:
        print(f'Epoch [{epoch+1}/{num_epochs}], Loss: {loss.item():.4f}')

model.eval()
with torch.no_grad():
    test_outputs = model(test_features)
    test_predictions = test_outputs.numpy().flatten()
    test_labels_np = test_labels.numpy().flatten()
    mdae = median_absolute_error(test_labels_np, test_predictions)
    print(f'Median Absolute Error (MAE) on test set: {mdae:.4f}')
