import pandas as pd
import lightgbm as lgb
from sklearn.model_selection import train_test_split
from sklearn.metrics import roc_auc_score
from sklearn.preprocessing import LabelEncoder

# 加载数据
train_path = r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/reservation_cancellation/train.csv'
test_path = r'D:\LLM-Driven_AI-Studio\MLAgent\data\benchmark/NL2Workflow/datasets/tabular/reservation_cancellation/test.csv'

train_df = pd.read_csv(train_path)
test_df = pd.read_csv(test_path)

# 查看数据的基本信息
print("Training data shape:", train_df.shape)
print("Testing data shape:", test_df.shape)
print(train_df.head())
print(test_df.head())

# 检查缺失值
print("Missing values in training data:\n", train_df.isnull().sum())
print("Missing values in testing data:\n", test_df.isnull().sum())

# 处理缺失值
# 假设我们用中位数填充数值型特征，用众数填充分类特征
for column in train_df.columns:
    if train_df[column].dtype == 'object':
        mode = train_df[column].mode()[0]
        train_df[column].fillna(mode, inplace=True)
        test_df[column].fillna(mode, inplace=True)
    else:
        median = train_df[column].median()
        train_df[column].fillna(median, inplace=True)
        test_df[column].fillna(median, inplace=True)

# 编码分类变量
label_encoders = {}
for column in train_df.select_dtypes(include=['object']).columns:
    le = LabelEncoder()
    train_df[column] = le.fit_transform(train_df[column])
    test_df[column] = le.transform(test_df[column])
    label_encoders[column] = le

# 分离特征和目标变量
X_train = train_df.drop(columns=['is_canceled'])
y_train = train_df['is_canceled']
X_test = test_df.drop(columns=['is_canceled'])
y_test = test_df['is_canceled']

# 训练LightGBM模型
train_data = lgb.Dataset(X_train, label=y_train)
params = {
    'objective': 'binary',
    'metric': 'auc',
    'boosting_type': 'gbdt',
    'num_leaves': 31,
    'learning_rate': 0.05,
    'feature_fraction': 0.9,
    'bagging_fraction': 0.8,
    'bagging_freq': 5,
    'verbose': 0
}

model = lgb.train(params, train_data, num_boost_round=100)

# 预测
y_pred = model.predict(X_test)

# 计算AUC-ROC
auc_roc = roc_auc_score(y_test, y_pred)
print("AUC-ROC on test set:", auc_roc)

# 保存预测结果
test_df['predicted_is_canceled'] = y_pred
test_df.to_csv('predictions.csv', index=False)
