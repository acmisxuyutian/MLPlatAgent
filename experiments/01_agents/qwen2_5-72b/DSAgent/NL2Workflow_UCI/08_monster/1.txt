import pandas as pd
from sklearn.metrics import accuracy_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import GridSearchCV
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = accuracy_score(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    # Define the scaler
    scaler = StandardScaler()

    # Scale the features
    X_train_scaled = scaler.fit_transform(X_train)
    X_valid_scaled = scaler.transform(X_valid)

    # Define the logistic regression model
    model = LogisticRegression(random_state=SEED)

    # Define the parameter grid
    param_grid = {
        'C': [0.01, 0.1, 1, 10, 100],
        'solver': ['liblinear', 'saga']
    }

    # Define the grid search
    grid_search = GridSearchCV(model, param_grid, cv=5, scoring='accuracy')

    # Perform the grid search
    grid_search.fit(X_train_scaled, y_train)

    # Get the best model
    best_model = grid_search.best_estimator_

    return best_model, scaler


def predict(model, X, scaler):
    # Scale the features
    X_scaled = scaler.transform(X)

    # Predict the model
    y_pred = model.predict(X_scaled)
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['type'], axis=1)
    y = data_df['type'].to_numpy()

    # Apply one-hot encoding to convert categorical columns to numeric
    X = pd.get_dummies(X)

    # Apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    best_model, scaler = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the model on the validation set using compute_metrics and print the results
    y_valid_pred = predict(best_model, X_valid, scaler)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final accuracy on validation set after scaling and hyperparameter tuning: ", score)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['type'], axis=1)
    X_test = pd.get_dummies(X_test)
    y_test_pred = predict(best_model, X_test, scaler)
    submit_predictions_for_test_set(y_test_pred)
