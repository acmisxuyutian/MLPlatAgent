import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import Pipeline
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    # Define the pipeline with StandardScaler and LogisticRegression
    model = Pipeline([
        ('scaler', StandardScaler()),
        ('classifier', LogisticRegression(random_state=SEED, max_iter=1000))
    ])
    
    # Train the model
    model.fit(X_train, y_train)
    
    return model


def predict(model, X):
    # Generate predictions
    y_pred = model.predict_proba(X)[:, 1]
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['Exited'], axis=1)
    y = data_df['Exited'].to_numpy()

    # Apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the model on the validation set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final auc_roc on validation set: ", score)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['Exited'], axis=1)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)

import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import Pipeline
from submission import submit_predictions_for_test_set

    score = roc_auc_score(y_test, y_test_pred)
    return score


def preprocess_data(df, train_columns=None):
    # Convert categorical columns to one-hot encoded columns
    df = pd.get_dummies(df, drop_first=True)
    
    # Align columns with the training set if train_columns is provided
    if train_columns is not None:
        df = df.reindex(columns=train_columns, fill_value=0)
    
    return df


def train_model(X_train, y_train, X_valid, y_valid):
    # Preprocess the data
    X_train = preprocess_data(X_train)
    X_valid = preprocess_data(X_valid, X_train.columns)
    
    # Define the pipeline with StandardScaler and LogisticRegression
    model = Pipeline([
        ('scaler', StandardScaler()),
        ('classifier', LogisticRegression(random_state=SEED, max_iter=1000))
    ])
    
    # Train the model
    model.fit(X_train, y_train)
    
    return model


def predict(model, X, train_columns):
    # Preprocess the data
    X = preprocess_data(X, train_columns)
    
    # Generate predictions
    y_pred = model.predict_proba(X)[:, 1]
    return y_pred
