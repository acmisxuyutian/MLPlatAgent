import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LogisticRegression

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score

    model = Pipeline([

    return model

import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import StandardScaler
from xgboost import XGBClassifier

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score

def train_model(X_train, y_train, X_valid, y_valid):
    # Select only numeric columns
    numeric_features = X_train.select_dtypes(include=[np.number]).columns
    X_train_numeric = X_train[numeric_features]
    X_valid_numeric = X_valid[numeric_features]

    # Define the pipeline with StandardScaler and XGBClassifier
    best_score = 0
    best_params = {}
    best_model = None

    # Hyperparameters to tune
    n_estimators = [50, 100, 200]
    max_depth = [3, 5, 7]

    for n in n_estimators:
        for d in max_depth:
            model = Pipeline([
                ('scaler', StandardScaler()),
                ('classifier', XGBClassifier(n_estimators=n, max_depth=d, random_state=SEED, use_label_encoder=False, eval_metric='logloss'))
            ])
            # Train the model
            model.fit(X_train_numeric, y_train)
            # Predict on the validation set
            y_valid_pred = model.predict_proba(X_valid_numeric)[:, 1]
            # Compute the AUC-ROC score
            score = compute_metrics(y_valid, y_valid_pred)
            # Update the best score and parameters
            if score > best_score:
                best_score = score
                best_params = {'n_estimators': n, 'max_depth': d}
                best_model = model

    print(f"Best AUC-ROC score: {best_score}")
    print(f"Best hyperparameters: {best_params}")
    return best_model

def predict(model, X):
    # Select only numeric columns
    numeric_features = X.select_dtypes(include=[np.number]).columns
    X_numeric = X[numeric_features]
    # Predict the probabilities for the positive class
    y_pred = model.predict_proba(X_numeric)[:, 1]
    return y_pred

if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['Exited'], axis=1)
    y = data_df['Exited'].to_numpy()

    # apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final auc_roc on validation set: ", score)

    # submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['Exited'], axis=1)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)
