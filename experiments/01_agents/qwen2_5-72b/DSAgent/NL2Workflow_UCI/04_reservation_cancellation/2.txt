import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import Pipeline
from sklearn.model_selection import GridSearchCV

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    # Define the feature scaling and model pipeline
    scaler = StandardScaler()
    model = LogisticRegression(random_state=SEED)

    # Define the hyperparameter grid
    param_grid = {
        'C': [0.01, 0.1, 1, 10, 100],
        'tol': [1e-4, 1e-3, 1e-2]
    }

    # Create the pipeline
    pipeline = Pipeline([
        ('scaler', scaler),
        ('model', model)
    ])

    # Perform grid search to find the best hyperparameters
    grid_search = GridSearchCV(pipeline, param_grid, cv=5, scoring='roc_auc', n_jobs=-1)
    grid_search.fit(X_train, y_train)

    # Get the best model
    best_model = grid_search.best_estimator_

    return best_model


def predict(model, X):
    y_pred = model.predict_proba(X)[:, 1]
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['booking_status'], axis=1)
    y = data_df['booking_status'].to_numpy()

    # apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final auc_roc on validation set with feature scaling and hyperparameter tuning: ", score)

    # submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['booking_status'], axis=1)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)
