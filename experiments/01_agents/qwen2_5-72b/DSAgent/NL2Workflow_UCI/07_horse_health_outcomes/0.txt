import pandas as pd
from sklearn.metrics import f1_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = f1_score(y_test, y_test_pred, average='micro')
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    # Define and train the logistic regression model
    model = LogisticRegression(random_state=SEED, max_iter=1000)
    model.fit(X_train, y_train)
    return model


def predict(model, X):
    # Generate predictions using the trained model
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':

import pandas as pd
from sklearn.metrics import f1_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
import lightgbm as lgb
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = f1_score(y_test, y_test_pred, average='micro')
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    # Convert categorical variables to numeric types
    X_train = pd.get_dummies(X_train)
    X_valid = pd.get_dummies(X_valid)
    
    # Ensure the same columns in both training and validation sets
    X_train, X_valid = X_train.align(X_valid, join='outer', axis=1, fill_value=0)
    
    # Define the LightGBM model with the specified hyperparameters
    model = lgb.LGBMClassifier(
        boosting_type='gbdt',
        objective='multiclass',
        num_class=3,
        n_estimators=100,
        learning_rate=0.1,
        num_leaves=31,
        max_depth=5,
        min_data_in_leaf=20,
        feature_fraction=0.9,
        lambda_l1=0.1,
        lambda_l2=0.1,
        max_bin=255,
        verbose=-1,
        force_col_wise=True,
        n_jobs=-1
    )
    
    # Train the model with callbacks for early stopping and logging
    model.fit(
        X_train, y_train,
        eval_set=[(X_valid, y_valid)],
        callbacks=[lgb.early_stopping(stopping_rounds=10), lgb.log_evaluation(period=0)]
    )
    
    return model


def predict(model, X):
    # Convert categorical variables to numeric types
    X = pd.get_dummies(X)
    
    # Ensure the same columns as in the training set
    X = X.reindex(columns=model.booster_.feature_name(), fill_value=0)
    
    # Predict the model
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['outcome'], axis=1)
    y = data_df['outcome'].to_numpy()

    # Apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the model on the validation set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final f1 on validation set: ", score)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['outcome'], axis=1)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)
