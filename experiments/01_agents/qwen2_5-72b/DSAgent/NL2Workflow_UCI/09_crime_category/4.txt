import pandas as pd
from sklearn.metrics import log_loss
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = log_loss(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    model = LogisticRegression(max_iter=1000, random_state=SEED)
    model.fit(X_train, y_train)
    return model


def predict(model, X):
    y_pred = model.predict_proba(X)
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    # evaluate the model on the valid set using compute_metrics and print the results

import pandas as pd
from sklearn.metrics import log_loss
import numpy as np
import random

    categorical_cols = X.select_dtypes(include=['object']).columns

    X_numeric = X.drop(categorical_cols, axis=1)
    # Concatenate the numeric and encoded data
    X_processed = pd.concat([X_numeric, X_encoded_df], axis=1)
    return X_processed
