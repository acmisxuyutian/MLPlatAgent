===================生成的代码========================:
import pandas as pd
from sklearn.metrics import median_absolute_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from lightgbm import LGBMRegressor
from hyperopt import fmin, tpe, hp, Trials

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

def compute_metrics(y_test, y_test_pred):
    score = median_absolute_error(y_test, y_test_pred)
    return score

def train_model(X_train, y_train, X_valid, y_valid, params):
    model = LGBMRegressor(**params)
    model.fit(X_train, y_train, eval_set=[(X_valid, y_valid)], early_stopping_rounds=10, verbose=False)
    return model

def predict(model, X):
    y_pred = model.predict(X)
    return y_pred

def objective(params):
    model = train_model(X_train, y_train, X_valid, y_valid, params)
    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    return score

if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['Hardness'], axis=1)
    y = data_df['Hardness'].to_numpy()

    # apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # define the hyperparameter space
    space = {
        'n_estimators': hp.choice('n_estimators', [100, 200, 300, 400, 500]),
        'learning_rate': hp.loguniform('learning_rate', np.log(0.01), np.log(0.5))
    }

    # perform hyperparameter tuning
    trials = Trials()
    best = fmin(fn=objective, space=space, algo=tpe.suggest, max_evals=50, trials=trials)

    # train the model with the best hyperparameters
    best_params = {
        'n_estimators': best['n_estimators'],
        'learning_rate': best['learning_rate']
    }
    model = train_model(X_train, y_train, X_valid, y_valid, best_params)

    # evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final mae on validation set: ", score)

    # submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['Hardness'], axis=1)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)
