import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import StandardScaler
import lightgbm as lgb

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    # Scale the features
    scaler = StandardScaler()
    X_train_scaled = scaler.fit_transform(X_train)
    X_valid_scaled = scaler.transform(X_valid)
    
    # Define the LightGBM dataset
    train_data = lgb.Dataset(X_train_scaled, label=y_train)
    valid_data = lgb.Dataset(X_valid_scaled, label=y_valid, reference=train_data)
    
    # Define the parameters for the LightGBM model
    params = {
        'objective': 'binary',
        'metric': 'auc',
        'boosting_type': 'gbdt',
        'learning_rate': 0.1,
        'num_leaves': 31,
        'random_state': SEED
    }
    
    # Perform hyperparameter tuning
    best_params = {}
    for learning_rate in [0.01, 0.1, 0.2]:
        for num_leaves in [15, 31, 63]:
            params['learning_rate'] = learning_rate
            params['num_leaves'] = num_leaves
            model = lgb.train(params, train_data, valid_sets=[valid_data], early_stopping_rounds=10, verbose_eval=False)
            y_valid_pred = model.predict(X_valid_scaled, num_iteration=model.best_iteration)
            score = compute_metrics(y_valid, y_valid_pred)
            if 'best_score' not in best_params or score > best_params['best_score']:
                best_params['best_score'] = score
                best_params['learning_rate'] = learning_rate
                best_params['num_leaves'] = num_leaves
    
    # Train the final model with the best hyperparameters
    params['learning_rate'] = best_params['learning_rate']
    params['num_leaves'] = best_params['num_leaves']
    model = lgb.train(params, train_data, valid_sets=[valid_data], early_stopping_rounds=10, verbose_eval=False)
    
    return model, scaler


def predict(model, X, scaler):
    # Scale the features
    X_scaled = scaler.transform(X)
    
    # Predict the probabilities
    y_pred = model.predict(X_scaled, num_iteration=model.best_iteration)
    
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['smoking'], axis=1)
    y = data_df['smoking'].to_numpy()

    # apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # define and train the model
    model, scaler = train_model(X_train, y_train, X_valid, y_valid)

    # evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid, scaler)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final auc_roc on validation set: ", score)

    # submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['smoking'], axis=1)
    y_test_pred = predict(model, X_test, scaler)
    submit_predictions_for_test_set(y_test_pred)
