import pandas as pd
from sklearn.metrics import root_mean_squared_log_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split, GridSearchCV
from lightgbm import LGBMRegressor
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = root_mean_squared_log_error(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    # Define the parameter grid for hyperparameter tuning
    param_grid = {
        'learning_rate': [0.01, 0.05, 0.1],
        'n_estimators': [100, 200, 300]
    }

    # Initialize the LightGBM model with the Tweedie objective function
    model = LGBMRegressor(objective='tweedie', random_state=SEED)

    # Perform grid search with cross-validation
    grid_search = GridSearchCV(estimator=model, param_grid=param_grid, scoring='neg_root_mean_squared_log_error', cv=3, verbose=1, n_jobs=-1)
    grid_search.fit(X_train, y_train)

    # Get the best model and its parameters
    best_model = grid_search.best_estimator_
    best_params = grid_search.best_params_

    # Evaluate the best model on the validation set
    y_valid_pred = best_model.predict(X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Best RMSLE on validation set: ", score)
    print("Best hyperparameters: ", best_params)

    return best_model


def predict(model, X):
    # Use the trained model to make predictions
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['cost'], axis=1)
    y = data_df['cost'].to_numpy()

    # Apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['cost'], axis=1)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)
