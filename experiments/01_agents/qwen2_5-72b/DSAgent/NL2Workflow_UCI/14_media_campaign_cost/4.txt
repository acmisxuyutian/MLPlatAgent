import pandas as pd
from sklearn.metrics import root_mean_squared_log_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from lightgbm import LGBMRegressor
from hyperopt import fmin, tpe, hp, Trials
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = root_mean_squared_log_error(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    # Define the hyperparameter space
    space = {
        'n_estimators': hp.choice('n_estimators', [100, 200, 300, 400, 500]),
        'learning_rate': hp.loguniform('learning_rate', -5, 0)
    }

    # Define the objective function for hyperopt
    def objective(params):
        model = LGBMRegressor(objective='tweedie', random_state=SEED, **params)
        model.fit(X_train, y_train, eval_set=[(X_valid, y_valid)], early_stopping_rounds=10, verbose=10)
        y_valid_pred = model.predict(X_valid)
        score = compute_metrics(y_valid, y_valid_pred)
        return score

    # Perform hyperparameter tuning
    trials = Trials()
    best = fmin(fn=objective, space=space, algo=tpe.suggest, max_evals=50, trials=trials)

    # Train the model with the best hyperparameters
    best_params = {
        'n_estimators': best['n_estimators'],
        'learning_rate': best['learning_rate']
    }
    model = LGBMRegressor(objective='tweedie', random_state=SEED, **best_params)
    model.fit(X_train, y_train, eval_set=[(X_valid, y_valid)], early_stopping_rounds=10, verbose=10)
    return model


def predict(model, X):
    # Use the trained model to make predictions
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['cost'], axis=1)
    y = data_df['cost'].to_numpy()

    # Apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the model on the validation set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final RMSLE on validation set: ", score)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['cost'], axis=1)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)
