import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LogisticRegression
from submission import submit_predictions_for_test_set
import optuna

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score

def train_model(X_train, y_train, X_valid, y_valid, **params):
    # Scale the features
    scaler = StandardScaler()
    X_train_scaled = scaler.fit_transform(X_train)
    X_valid_scaled = scaler.transform(X_valid)
    
    # Train the Logistic Regression model with the best hyperparameters
    model = LogisticRegression(**params, random_state=SEED)
    model.fit(X_train_scaled, y_train)
    
    return model, scaler

def predict(model, X, scaler):
    # Scale the input features
    X_scaled = scaler.transform(X)
    # Make predictions
    y_pred = model.predict_proba(X_scaled)[:, 1]
    return y_pred

def objective(trial):
    # Define the hyperparameters to tune
    C = trial.suggest_loguniform('C', 1e-4, 1e4)
    solver = trial.suggest_categorical('solver', ['newton-cg', 'lbfgs', 'liblinear', 'sag', 'saga'])
    
    # Scale the features
    scaler = StandardScaler()
    X_train_scaled = scaler.fit_transform(X_train)
    X_valid_scaled = scaler.transform(X_valid)
    
    # Train the Logistic Regression model with the suggested hyperparameters
    model = LogisticRegression(C=C, solver=solver, random_state=SEED)
    model.fit(X_train_scaled, y_train)
    
    # Make predictions on the validation set
    y_valid_pred = model.predict_proba(X_valid_scaled)[:, 1]
    
    # Compute the AUC-ROC score
    score = roc_auc_score(y_valid, y_valid_pred)
    
    return score

if __name__ == '__main__':
    # Load the data
    data_df = pd.read_csv('train.csv')
    X = data_df.drop(['target'], axis=1)
    y = data_df['target'].to_numpy()
    
    # Split the data
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)
    
    # Perform hyperparameter tuning
    study = optuna.create_study(direction='maximize')
    study.optimize(objective, n_trials=50)
    
    # Get the best hyperparameters
    best_params = study.best_params
    print("Best hyperparameters: ", best_params)
    
    # Retrain the model with the best hyperparameters
    best_model, scaler = train_model(X_train, y_train, X_valid, y_valid, **best_params)
    
    # Evaluate the model on the validation set
    y_valid_pred = predict(best_model, X_valid, scaler)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final auc_roc on validation set with tuned hyperparameters: ", score)
    
    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['target'], axis=1)
    y_test_pred = predict(best_model, X_test, scaler)
    submit_predictions_for_test_set(y_test_pred)
