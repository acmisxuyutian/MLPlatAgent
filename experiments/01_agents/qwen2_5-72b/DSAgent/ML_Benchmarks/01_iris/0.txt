import random

    model.fit(X_train, y_train)
    return model


def predict(model, X):
    y_pred = model.predict(X)
    return y_pred

    X_test = submission_df.drop(['accuracy'], axis=1)

import pandas as pd
from sklearn.metrics import accuracy_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.tree import DecisionTreeClassifier
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = accuracy_score(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid, model_type='logistic'):
    if model_type == 'logistic':
        model = LogisticRegression(random_state=SEED)
    elif model_type == 'decision_tree':
        model = DecisionTreeClassifier(random_state=SEED)
    model.fit(X_train, y_train)
    return model


def predict(model, X):
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':
    try:
        data_df = pd.read_csv('train.csv')
    except FileNotFoundError:
        print("Error: 'train.csv' not found. Please ensure the file is in the correct directory.")
        exit(1)

    X = data_df.drop(['accuracy'], axis=1)
    y = data_df['accuracy'].to_numpy()

    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Train and evaluate logistic regression model
    logistic_model = train_model(X_train, y_train, X_valid, y_valid, model_type='logistic')
    y_valid_pred_logistic = predict(logistic_model, X_valid)
    score_logistic = compute_metrics(y_valid, y_valid_pred_logistic)
    print("Logistic Regression accuracy on validation set: ", score_logistic)

    # Train and evaluate decision tree model
    decision_tree_model = train_model(X_train, y_train, X_valid, y_valid, model_type='decision_tree')
    y_valid_pred_decision_tree = predict(decision_tree_model, X_valid)
    score_decision_tree = compute_metrics(y_valid, y_valid_pred_decision_tree)
    print("Decision Tree accuracy on validation set: ", score_decision_tree)

    try:
        submission_df = pd.read_csv('test.csv')
    except FileNotFoundError:
        print("Error: 'test.csv' not found. Please ensure the file is in the correct directory.")
        exit(1)

    X_test = submission_df.drop(['accuracy'], axis=1)
    y_test_pred = predict(decision_tree_model, X_test)
    submit_predictions_for_test_set(y_test_pred)
