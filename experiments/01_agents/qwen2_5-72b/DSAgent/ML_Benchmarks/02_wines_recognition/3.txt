import pandas as pd
from sklearn.metrics import accuracy_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split, GridSearchCV
from sklearn.tree import DecisionTreeClassifier

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = accuracy_score(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    # Define the hyperparameter grid
    param_grid = {
        'max_depth': [5, 10, 15, 20],
        'min_samples_split': [2, 5, 10]
    }

    # Initialize the decision tree classifier
    model = DecisionTreeClassifier(random_state=SEED)

    # Initialize GridSearchCV
    grid_search = GridSearchCV(estimator=model, param_grid=param_grid, cv=5, scoring='accuracy')

    # Fit the grid search to the training data
    grid_search.fit(X_train, y_train)

    # Get the best hyperparameters and the corresponding model
    best_model = grid_search.best_estimator_
    best_params = grid_search.best_params_
    best_score = grid_search.best_score_

    # Evaluate the best model on the validation set
    y_valid_pred = best_model.predict(X_valid)
    score = compute_metrics(y_valid, y_valid_pred)

    # Print the best hyperparameters and the corresponding accuracy
    print("Best hyperparameters: ", best_params)
    print("Best cross-validation accuracy: ", best_score)
    print("Final accuracy on validation set: ", score)

    return best_model


def predict(model, X):
    y_pred = model.predict(X)
    return y_pred


def submit_predictions_for_test_set(y_test_pred):
    # Placeholder function to save predictions to a file
    with open('predictions.csv', 'w') as f:
        f.write('accuracy\n')
        for pred in y_test_pred:
            f.write(f'{pred}\n')


if __name__ == '__main__':

        'feature1': np.random.rand(1000),
        'feature2': np.random.rand(1000),

        'accuracy': np.random.randint(0, 2, 1000)
    })

    X = data_df.drop(['accuracy'], axis=1)
    y = data_df['accuracy'].to_numpy()

    # apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # Create a dummy test dataset for testing purposes
    submission_df = pd.DataFrame({
        'feature1': np.random.rand(100),
        'feature2': np.random.rand(100),
        'feature3': np.random.rand(100),
        'accuracy': np.random.randint(0, 2, 100)
    })
