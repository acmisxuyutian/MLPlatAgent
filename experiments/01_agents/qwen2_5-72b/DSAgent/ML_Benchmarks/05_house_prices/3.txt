import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from lightgbm import LGBMRegressor
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

def compute_metrics(y_test, y_test_pred):
    score = mean_squared_error(y_test, y_test_pred, squared=False)
    return score

def train_model(X_train, y_train, X_valid, y_valid):
    # Define and train the LightGBM model with default hyperparameters
    model = LGBMRegressor()
    model.fit(X_train, np.log1p(y_train))
    return model

def predict(model, X):
    # Make predictions using the trained model
    y_pred = model.predict(X)
    return np.expm1(y_pred)

if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['SalePrice'], axis=1)
    y = data_df['SalePrice'].to_numpy()

    # apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(np.log1p(y_valid), np.log1p(y_valid_pred))
    print("Final rmse on validation set: ", score)

    # submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['SalePrice'], axis=1)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)

import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from lightgbm import LGBMRegressor
from submission import submit_predictions_for_test_set

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

def compute_metrics(y_test, y_test_pred):
    score = np.sqrt(mean_squared_error(y_test, y_test_pred))
    return score

def train_model(X_train, y_train, X_valid, y_valid):
    # Define and train the LightGBM model with tuned hyperparameters
    model = LGBMRegressor(learning_rate=0.05, num_leaves=31)
    model.fit(X_train, np.log1p(y_train))
    return model

def predict(model, X):
    # Make predictions using the trained model
    y_pred = model.predict(X)
    return np.expm1(y_pred)

    # Apply one-hot encoding to categorical features
    X = pd.get_dummies(X)

    # apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # evaluate the model on the valid set using compute_metrics and print the results
