import pandas as pd
from sklearn.metrics import log_loss
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import OneHotEncoder
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = log_loss(y_test, y_test_pred)
    return score


def preprocess_data(X_train, X_valid, X_test):
    # Identify categorical columns
    categorical_cols = X_train.select_dtypes(include=['object']).columns

    # Initialize the OneHotEncoder
    encoder = OneHotEncoder(handle_unknown='ignore', sparse=False)

    # Fit the encoder on the training data
    encoder.fit(X_train[categorical_cols])

    # Transform the training, validation, and test data
    X_train_encoded = encoder.transform(X_train[categorical_cols])
    X_valid_encoded = encoder.transform(X_valid[categorical_cols])
    X_test_encoded = encoder.transform(X_test[categorical_cols])

    # Get the column names for the encoded features
    encoded_feature_names = encoder.get_feature_names_out(categorical_cols)

    # Combine the encoded features with the numerical features
    X_train_final = pd.concat([X_train.drop(categorical_cols, axis=1), pd.DataFrame(X_train_encoded, columns=encoded_feature_names)], axis=1)
    X_valid_final = pd.concat([X_valid.drop(categorical_cols, axis=1), pd.DataFrame(X_valid_encoded, columns=encoded_feature_names)], axis=1)
    X_test_final = pd.concat([X_test.drop(categorical_cols, axis=1), pd.DataFrame(X_test_encoded, columns=encoded_feature_names)], axis=1)

    return X_train_final, X_valid_final, X_test_final


def train_model(X_train, y_train, X_valid, y_valid):
    # Define and train the logistic regression model
    model = LogisticRegression(random_state=SEED)
    model.fit(X_train, y_train)
    return model


def predict(model, X):
    # Generate predictions using the trained model
    y_pred = model.predict_proba(X)[:, 1]
    return y_pred

    # Preprocess the data
