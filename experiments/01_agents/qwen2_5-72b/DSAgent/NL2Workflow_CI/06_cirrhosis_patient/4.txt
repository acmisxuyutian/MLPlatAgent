import pandas as pd
from sklearn.metrics import log_loss
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

def compute_metrics(y_test, y_test_pred):
    score = log_loss(y_test, y_test_pred)
    return score

from sklearn.linear_model import LogisticRegression

def train_model(X_train, y_train, X_valid, y_valid):
    model = LogisticRegression(random_state=SEED)
    model.fit(X_train, y_train)
    return model

def predict(model, X):
    y_pred = model.predict_proba(X)[:, 1]
    return y_pred

if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['Status'], axis=1)
    y = data_df['Status'].to_numpy()

    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    model = train_model(X_train, y_train, X_valid, y_valid)

    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final logloss on validation set: ", score)

    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['Status'], axis=1)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)

import pandas as pd
from sklearn.metrics import log_loss
import numpy as np
import random
import torch
from sklearn.linear_model import LogisticRegression
from sklearn.model_selection import train_test_split

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = log_loss(y_test, y_test_pred)
    return score


def submit_predictions_for_test_set(y_test_pred, output_file='submission.csv'):
    submission_df = pd.DataFrame(y_test_pred, columns=['Status'])
    submission_df.index.name = 'ID'
    submission_df.to_csv(output_file)


def train_model(X_train, y_train, X_valid, y_valid):
    model = LogisticRegression(random_state=SEED, max_iter=1000)
    model.fit(X_train, y_train)
    return model


def predict(model, X):
    y_pred = model.predict_proba(X)[:, 1]
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['Status'], axis=1)
    y = data_df['Status'].to_numpy()

    # apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final logloss on validation set: ", score)

    # submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['ID'], axis=1)  # Assuming 'ID' is the column name for the test set
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)
