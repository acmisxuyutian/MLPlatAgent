import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split, GridSearchCV
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    # Scale the features using StandardScaler
    scaler = StandardScaler()
    X_train_scaled = scaler.fit_transform(X_train)
    X_valid_scaled = scaler.transform(X_valid)

    # Define the hyperparameters to tune
    param_grid = {
        'C': [0.01, 0.1, 1, 10, 100],
        'solver': ['liblinear', 'saga']
    }

    # Perform grid search with cross-validation
    grid_search = GridSearchCV(LogisticRegression(random_state=SEED), param_grid, cv=5, scoring='roc_auc')
    grid_search.fit(X_train_scaled, y_train)

    # Get the best model and its hyperparameters
    best_model = grid_search.best_estimator_
    best_params = grid_search.best_params_

    return best_model, scaler, best_params


def predict(model, scaler, X):
    # Scale the features using the same scaler
    X_scaled = scaler.transform(X)
    
    # Predict the model
    y_pred = model.predict_proba(X_scaled)[:, 1]
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['target'], axis=1)
    y = data_df['target'].to_numpy()

    # apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # define and train the model
    best_model, scaler, best_params = train_model(X_train, y_train, X_valid, y_valid)

    # evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(best_model, scaler, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Best hyperparameters:", best_params)
    print("AUC-ROC on validation set with best hyperparameters:", score)

    # submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['target'], axis=1)
    y_test_pred = predict(best_model, scaler, X_test)
    submit_predictions_for_test_set(y_test_pred)
