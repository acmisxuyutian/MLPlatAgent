import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set

# Install LightGBM if not already installed
!pip install lightgbm

import lightgbm as lgb

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score

def train_model(X_train, y_train, X_valid, y_valid):
    # Define the LightGBM dataset
    train_data = lgb.Dataset(X_train, label=y_train)
    valid_data = lgb.Dataset(X_valid, label=y_valid, reference=train_data)
    
    # Define the parameters for the LightGBM model
    params = {
        'objective': 'binary',
        'metric': 'auc',
        'boosting_type': 'gbdt',
        'num_leaves': 31,
        'learning_rate': 0.05,
        'feature_fraction': 0.9,
        'bagging_fraction': 0.8,
        'bagging_freq': 5,
        'verbose': 0
    }
    
    # Train the model
    model = lgb.train(params, train_data, valid_sets=[train_data, valid_data], num_boost_round=100, early_stopping_rounds=10)
    return model

def predict(model, X):
    # Generate predictions
    y_pred = model.predict(X, num_iteration=model.best_iteration)
    return y_pred

if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['booking_status'], axis=1)
    y = data_df['booking_status'].to_numpy()

    # Apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final auc_roc on validation set: ", score)

    # Save the trained model and the validation set predictions
    model.save_model('lightgbm_model.txt')
    np.save('y_valid_pred.npy', y_valid_pred)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['booking_status'], axis=1)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)

import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set

# Check if LightGBM is installed and install it if not
try:
    import lightgbm as lgb
except ImportError:
    import subprocess
    import sys
    subprocess.check_call([sys.executable, "-m", "pip", "install", "lightgbm"])
    import lightgbm as lgb

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score

def train_model(X_train, y_train, X_valid, y_valid, params):
    # Define the LightGBM dataset
    train_data = lgb.Dataset(X_train, label=y_train)
    valid_data = lgb.Dataset(X_valid, label=y_valid, reference=train_data)
    
    # Train the model
    model = lgb.train(params, train_data, valid_sets=[train_data, valid_data], num_boost_round=100, early_stopping_rounds=10)
    return model

def predict(model, X):
    # Generate predictions
    y_pred = model.predict(X, num_iteration=model.best_iteration)
    return y_pred

if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['booking_status'], axis=1)
    y = data_df['booking_status'].to_numpy()

    # Apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define the grid of hyperparameters to search
    learning_rates = [0.01, 0.05, 0.1]
    num_leaves = [20, 31, 40]

    best_score = 0
    best_params = None
    best_model = None

    for lr in learning_rates:
        for nl in num_leaves:
            # Define the parameters for the LightGBM model
            params = {
                'objective': 'binary',
                'metric': 'auc',
                'boosting_type': 'gbdt',
                'num_leaves': nl,

                'feature_fraction': 0.9,

            # Define and train the model

            # Evaluate the model on the valid set using compute_metrics

            print(f"Validation AUC-ROC for learning_rate={lr}, num_leaves={nl}: {score}")

            # Update the best model and parameters if this score is better
            if score > best_score:
                best_score = score
                best_params = params
                best_model = model

    print(f"Best AUC-ROC: {best_score}")
    print(f"Best parameters: {best_params}")

    # Save the best trained model and the validation set predictions
    best_model.save_model('lightgbm_model.txt')
    y_valid_pred = predict(best_model, X_valid)
    np.save('y_valid_pred.npy', y_valid_pred)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['booking_status'], axis=1)
    y_test_pred = predict(best_model, X_test)
    submit_predictions_for_test_set(y_test_pred)
