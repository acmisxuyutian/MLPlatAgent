===================生成的代码========================:
import pandas as pd
from sklearn.metrics import root_mean_squared_log_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from lightgbm import LGBMRegressor, early_stopping, log_evaluation
from submission import submit_predictions_for_test_set
import hyperopt
from hyperopt import hp, fmin, tpe, Trials

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = root_mean_squared_log_error(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid, params=None):
    if params is None:
        params = {}
    model = LGBMRegressor(
        n_estimators=params.get('n_estimators', 100),
        learning_rate=params.get('learning_rate', 0.1),
        random_state=SEED
    )
    model.fit(
        X_train, y_train,
        eval_set=[(X_valid, y_valid)],
        callbacks=[early_stopping(10), log_evaluation(period=0)]
    )
    return model


def predict(model, X):
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    # Convert the 'Sex' column to numerical values using label encoding
    data_df['Sex'] = data_df['Sex'].map({'M': 0, 'F': 1, 'I': 2})

    X = data_df.drop(['Rings'], axis=1)
    y = data_df['Rings'].to_numpy()

    # Apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define the search space for hyperparameters
    space = {
        'n_estimators': hp.choice('n_estimators', range(50, 500, 50)),
        'learning_rate': hp.loguniform('learning_rate', -5, 0)
    }

    # Define the objective function
    def objective(params):
        model = train_model(X_train, y_train, X_valid, y_valid, params)
        y_valid_pred = predict(model, X_valid)
        score = compute_metrics(y_valid, y_valid_pred)
        return score

    # Perform hyperparameter tuning
    trials = Trials()
    best = fmin(
        fn=objective,
        space=space,
        algo=tpe.suggest,
        max_evals=50,
        trials=trials
    )

    # Train the final model with the best hyperparameters
    best_model = train_model(X_train, y_train, X_valid, y_valid, best)

    # Evaluate the final model on the validation set
    y_valid_pred = predict(best_model, X_valid)
    final_score = compute_metrics(y_valid, y_valid_pred)
    print("Final rmsle on validation set with tuned hyperparameters: ", final_score)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    # Convert the 'Sex' column to numerical values using label encoding
    submission_df['Sex'] = submission_df['Sex'].map({'M': 0, 'F': 1, 'I': 2})
    X_test = submission_df.drop(['Rings'], axis=1)
    y_test_pred = predict(best_model, X_test)
    submit_predictions_for_test_set(y_test_pred)
