import pandas as pd
from sklearn.metrics import median_absolute_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split, GridSearchCV
from lightgbm import LGBMRegressor
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

def compute_metrics(y_test, y_test_pred):
    score = median_absolute_error(y_test, y_test_pred)
    return score

def train_model(X_train, y_train, X_valid, y_valid):
    # Define the hyperparameter grid
    param_grid = {
        'n_estimators': [50, 100, 200],
        'learning_rate': [0.01, 0.1, 0.2]
    }

    # Define the LightGBM model
    model = LGBMRegressor(random_state=SEED)

    # Perform grid search for hyperparameter tuning
    grid_search = GridSearchCV(estimator=model, param_grid=param_grid, scoring='neg_median_absolute_error', cv=3, n_jobs=-1)
    grid_search.fit(X_train, y_train)

    # Get the best model
    best_model = grid_search.best_estimator_
    return best_model

def predict(model, X):
    y_pred = model.predict(X)
    return y_pred

if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['yield'], axis=1)
    y = data_df['yield'].to_numpy()

    # apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final mae on validation set: ", score)

    # submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['yield'], axis=1)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)
