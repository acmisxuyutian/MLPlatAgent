import pandas as pd
from sklearn.metrics import f1_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = f1_score(y_test, y_test_pred, average='micro')
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    model = RandomForestClassifier(random_state=SEED)
    model.fit(X_train, y_train)
    return model


def predict(model, X):
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['outcome'], axis=1)
    y = data_df['outcome'].to_numpy()

    # apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final f1 on validation set: ", score)

    # submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['outcome'], axis=1)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)

import pandas as pd
from sklearn.metrics import f1_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from submission import submit_predictions_for_test_set

torch.manual_seed(SEED)

    score = f1_score(y_test, y_test_pred, average='micro')
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    model = RandomForestClassifier(random_state=SEED)
    model.fit(X_train, y_train)
    return model


def predict(model, X):
    y_pred = model.predict(X)
    return y_pred


def preprocess_data(df, training=True):
    # Convert categorical columns to numeric using one-hot encoding
    if training:
        df = pd.get_dummies(df)
    else:
        # Ensure the test data has the same columns as the training data
        train_df = pd.read_csv('train.csv')
        train_df = pd.get_dummies(train_df)
        test_df = pd.get_dummies(df)
        # Align columns
        train_columns = set(train_df.columns)
        test_columns = set(test_df.columns)
        missing_columns = train_columns - test_columns
        for col in missing_columns:
            test_df[col] = 0
        test_df = test_df[train_df.columns]
        df = test_df
    return df
