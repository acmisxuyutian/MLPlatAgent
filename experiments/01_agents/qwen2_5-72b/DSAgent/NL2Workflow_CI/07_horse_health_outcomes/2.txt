import random

np.random.seed(SEED)

def preprocess_data(df):
    # Convert categorical columns to numeric using LabelEncoder
    le = LabelEncoder()
    for column in df.select_dtypes(include=['object']).columns:
        df[column] = le.fit_transform(df[column])
    return df

def compute_metrics(y_test, y_test_pred):
    score = f1_score(y_test, y_test_pred, average='micro')
    return score

def train_model(X_train, y_train, X_valid, y_valid):
    best_score = 0
    best_model = None
    best_params = {}

    # Define the hyperparameters to tune
    Cs = [0.001, 0.01, 0.1, 1, 10, 100]
    solvers = ['liblinear', 'saga']

    for C in Cs:
        for solver in solvers:
            model = LogisticRegression(C=C, solver=solver, random_state=SEED, max_iter=2000)
            model.fit(X_train, y_train)
            y_valid_pred = model.predict(X_valid)
            score = compute_metrics(y_valid, y_valid_pred)
            if score > best_score:
                best_score = score
                best_model = model
                best_params = {'C': C, 'solver': solver}

    print(f"Best F1 score: {best_score}")
    print(f"Best parameters: {best_params}")
    return best_model

def predict(model, X):
    y_pred = model.predict(X)
    return y_pred
