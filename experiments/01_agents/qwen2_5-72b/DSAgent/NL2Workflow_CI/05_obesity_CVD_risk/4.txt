import pandas as pd
from sklearn.metrics import accuracy_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LogisticRegression
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

def compute_metrics(y_test, y_test_pred):
    score = accuracy_score(y_test, y_test_pred)
    return score

def train_model(X_train, y_train, X_valid, y_valid):
    # Standardize the features
    scaler = StandardScaler()
    X_train_scaled = scaler.fit_transform(X_train)
    X_valid_scaled = scaler.transform(X_valid)
    
    # Define and train the logistic regression model
    model = LogisticRegression(random_state=SEED)
    model.fit(X_train_scaled, y_train)
    
    return model, scaler

def predict(model, X, scaler):
    # Standardize the features
    X_scaled = scaler.transform(X)
    
    # Predict the model
    y_pred = model.predict(X_scaled)
    return y_pred

if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['NObeyesdad'], axis=1)
    y = data_df['NObeyesdad'].to_numpy()

    # Apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model, scaler = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid, scaler)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final accuracy on validation set: ", score)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['NObeyesdad'], axis=1)
    y_test_pred = predict(model, X_test, scaler)
    submit_predictions_for_test_set(y_test_pred)

import pandas as pd
from sklearn.metrics import accuracy_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler, LabelEncoder
from sklearn.linear_model import LogisticRegression
from submission import submit_predictions_for_test_set
import optuna

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

def compute_metrics(y_test, y_test_pred):
    score = accuracy_score(y_test, y_test_pred)
    return score

def preprocess_data(X):
    # Convert categorical columns to numeric
    le = LabelEncoder()
    for col in X.select_dtypes(include=['object']).columns:
        X[col] = le.fit_transform(X[col])
    return X

def objective(trial):
    # Define the hyperparameters to tune
    C = trial.suggest_loguniform('C', 1e-4, 1e4)
    solver = trial.suggest_categorical('solver', ['newton-cg', 'lbfgs', 'liblinear', 'sag', 'saga'])
    
    # Define and train the logistic regression model with the suggested hyperparameters
    model = LogisticRegression(C=C, solver=solver, random_state=SEED)
    model.fit(X_train_scaled, y_train)
    
    # Evaluate the model on the validation set
    y_valid_pred = model.predict(X_valid_scaled)
    score = compute_metrics(y_valid, y_valid_pred)
    
    return score

if __name__ == '__main__':
    # Load the data
    data_df = pd.read_csv('train.csv')
    X = data_df.drop(['NObeyesdad'], axis=1)
    y = data_df['NObeyesdad'].to_numpy()
    
    # Split the data
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)
    
    # Preprocess the data
    X_train = preprocess_data(X_train)
    X_valid = preprocess_data(X_valid)
    
    # Standardize the features
    scaler = StandardScaler()
    X_train_scaled = scaler.fit_transform(X_train)
    X_valid_scaled = scaler.transform(X_valid)
    
    # Perform hyperparameter tuning using Optuna
    study = optuna.create_study(direction='maximize')
    study.optimize(objective, n_trials=50)
    
    # Get the best hyperparameters
    best_params = study.best_params
    best_model = LogisticRegression(C=best_params['C'], solver=best_params['solver'], random_state=SEED)
    best_model.fit(X_train_scaled, y_train)
    
    # Evaluate the best model on the validation set
    y_valid_pred = best_model.predict(X_valid_scaled)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Best model accuracy on validation set: ", score)
    
    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['NObeyesdad'], axis=1)
    y_test_pred = predict(best_model, X_test, scaler)
    submit_predictions_for_test_set(y_test_pred)
