import pandas as pd
from sklearn.metrics import accuracy_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split

from sklearn.linear_model import LogisticRegression
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

def compute_metrics(y_test, y_test_pred):
    score = accuracy_score(y_test, y_test_pred)
    return score

    # Standardize the features
    scaler = StandardScaler()
    X_train_scaled = scaler.fit_transform(X_train)
    X_valid_scaled = scaler.transform(X_valid)
    
    # Define and train the logistic regression model

    return X, label_encoders


def train_model(X_train, y_train, X_valid, y_valid):

    X_train, label_encoders = preprocess_data(X_train)
    X_valid, _ = preprocess_data(X_valid, label_encoders)
    
    # Define the scaler and model
    scaler = StandardScaler()
    model = LogisticRegression(random_state=SEED, max_iter=1000)
    
    # Scale the features
    X_train_scaled = scaler.fit_transform(X_train)
    X_valid_scaled = scaler.transform(X_valid)
    
    # Train the model
    model.fit(X_train_scaled, y_train)
    
    # Evaluate the model on the validation set
    y_valid_pred = model.predict(X_valid_scaled)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Validation accuracy: ", score)
    
    return model, scaler, label_encoders

    # Scale the features
    X_scaled = scaler.transform(X)
    
    # Predict the model
