import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
import xgboost as xgb  # Import the xgboost module

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score

def train_model(X_train, y_train, X_valid, y_valid, max_depth, n_estimators):
    # Define the XGBoost model with DART booster and specified hyperparameters
    model = xgb.XGBClassifier(
        booster='dart',
        learning_rate=0.1,
        n_estimators=n_estimators,
        max_depth=max_depth,
        subsample=0.8,
        random_state=SEED
    )
    # Train the model
    model.fit(X_train, y_train, eval_set=[(X_valid, y_valid)], verbose=False)
    return model

def predict(model, X):
    # Predict probabilities for the positive class
    y_pred = model.predict_proba(X)[:, 1]
    return y_pred

if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['smoking'], axis=1)
    y = data_df['smoking'].to_numpy()

    # apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define the hyperparameter combinations to try
    max_depth_values = [3, 5, 7]
    n_estimators_values = [100, 200, 300]

    best_score = 0
    best_params = {}

    for max_depth in max_depth_values:
        for n_estimators in n_estimators_values:
            # Train the model with the current hyperparameters
            model = train_model(X_train, y_train, X_valid, y_valid, max_depth, n_estimators)

            # Evaluate the model on the validation set
            y_valid_pred = predict(model, X_valid)
            score = compute_metrics(y_valid, y_valid_pred)

            print(f"max_depth: {max_depth}, n_estimators: {n_estimators}, AUC-ROC: {score}")

            # Update the best score and parameters if the current score is better
            if score > best_score:
                best_score = score
                best_params = {'max_depth': max_depth, 'n_estimators': n_estimators}

    print(f"Best AUC-ROC: {best_score} with parameters: {best_params}")

    # Train the final model with the best hyperparameters
    final_model = train_model(X_train, y_train, X_valid, y_valid, best_params['max_depth'], best_params['n_estimators'])

    # submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['smoking'], axis=1)
    y_test_pred = predict(final_model, X_test)
    submit_predictions_for_test_set(y_test_pred)
