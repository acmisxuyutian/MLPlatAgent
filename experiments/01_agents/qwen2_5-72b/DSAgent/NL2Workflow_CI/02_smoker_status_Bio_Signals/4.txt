import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from xgboost import XGBClassifier
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    model = XGBClassifier(
        booster='dart',
        learning_rate=0.1,
        max_depth=3,
        n_estimators=100,
        random_state=SEED
    )
    model.fit(X_train, y_train, eval_set=[(X_valid, y_valid)], early_stopping_rounds=10, verbose=False)
    return model


def predict(model, X):
    y_pred = model.predict_proba(X)[:, 1]
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['smoking'], axis=1)
    y = data_df['smoking'].to_numpy()

    # apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final auc_roc on validation set: ", score)

    # submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['smoking'], axis=1)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)

import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split, GridSearchCV
from xgboost import XGBClassifier
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    # Define the parameter grid
    param_grid = {
        'max_depth': [3, 5, 7],
        'n_estimators': [100, 200, 300]
    }

    # Initialize the XGBoost model
    model = XGBClassifier(
        booster='dart',
        learning_rate=0.1,
        random_state=SEED
    )

    # Perform grid search
    grid_search = GridSearchCV(estimator=model, param_grid=param_grid, scoring='roc_auc', cv=3, verbose=1, n_jobs=-1)
    grid_search.fit(X_train, y_train, eval_set=[(X_valid, y_valid)], verbose=False)

    # Get the best model
    best_model = grid_search.best_estimator_

    return best_model


def predict(model, X):
    y_pred = model.predict_proba(X)[:, 1]
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['smoking'], axis=1)
    y = data_df['smoking'].to_numpy()

    # apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Best auc_roc on validation set: ", score)
    print("Best parameters: ", grid_search.best_params_)

    # submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['smoking'], axis=1)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)

def compute_metrics(y_test, y_test_pred):

def train_model(X_train, y_train, X_valid, y_valid):
    # Define the parameter grid
    param_grid = {
        'max_depth': [3, 5, 7],
        'n_estimators': [100, 200, 300]
    }

    # Initialize the XGBoost model
    model = XGBClassifier(
        booster='dart',
        learning_rate=0.1,
        random_state=SEED
    )

    # Perform grid search
    grid_search = GridSearchCV(estimator=model, param_grid=param_grid, scoring='roc_auc', cv=3, verbose=1, n_jobs=-1)
    grid_search.fit(X_train, y_train, eval_set=[(X_valid, y_valid)], verbose=False)

    # Get the best model
    best_model = grid_search.best_estimator_

    return best_model, grid_search
