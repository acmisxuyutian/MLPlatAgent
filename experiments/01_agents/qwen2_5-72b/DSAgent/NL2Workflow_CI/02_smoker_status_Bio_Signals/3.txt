import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split, GridSearchCV
from xgboost import XGBClassifier
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    # Define the parameter grid for hyperparameter tuning
    param_grid = {
        'max_depth': [3, 5, 7],
        'subsample': [0.8, 0.9, 1.0]
    }

    # Define the XGBoost model with the specified parameters
    model = XGBClassifier(booster='dart', learning_rate=0.1, random_state=SEED)

    # Perform grid search with cross-validation
    grid_search = GridSearchCV(estimator=model, param_grid=param_grid, scoring='roc_auc', cv=3, n_jobs=-1)
    grid_search.fit(X_train, y_train)

    # Get the best model and its parameters
    best_model = grid_search.best_estimator_
    best_params = grid_search.best_params_

    # Evaluate the best model on the validation set
    y_valid_pred = predict(best_model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Best parameters found: ", best_params)
    print("Final auc_roc on validation set with best parameters: ", score)

    return best_model


def predict(model, X):
    # Predict probabilities for the positive class
    y_pred = model.predict_proba(X)[:, 1]
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['smoking'], axis=1)
    y = data_df['smoking'].to_numpy()

    # Apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    best_model = train_model(X_train, y_train, X_valid, y_valid)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['smoking'], axis=1)
    y_test_pred = predict(best_model, X_test)
    submit_predictions_for_test_set(y_test_pred)
