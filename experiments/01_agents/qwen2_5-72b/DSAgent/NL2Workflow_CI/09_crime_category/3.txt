import pandas as pd
from sklearn.metrics import log_loss
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
import lightgbm as lgb
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = log_loss(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    # Define the LightGBM model with the specified hyperparameters
    model = lgb.LGBMClassifier(
        boosting_type='gbdt',
        objective='multiclass',
        num_class=39,
        n_estimators=100,
        learning_rate=0.1,
        num_leaves=31,
        max_depth=5,
        min_data_in_leaf=20,
        feature_fraction=0.9,
        lambda_l1=0.1,
        lambda_l2=0.1,
        max_bin=255,
        verbose=-1,
        force_col_wise=True,
        n_jobs=-1
    )
    
    # Train the model
    model.fit(X_train, y_train, eval_set=[(X_valid, y_valid)], early_stopping_rounds=10, verbose=False)
    
    return model


def predict(model, X):
    # Predict the probabilities for the classes
    y_pred = model.predict_proba(X)
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['Category'], axis=1)
    y = data_df['Category'].to_numpy()

    # Apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the model on the validation set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final logloss on validation set: ", score)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['Category'], axis=1)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)

import pandas as pd
from sklearn.metrics import log_loss
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from lightgbm import LGBMClassifier, early_stopping, log_evaluation
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred, labels):
    score = log_loss(y_test, y_test_pred, labels=labels)
    return score


def preprocess_data(X):
    # Convert categorical columns to numerical using one-hot encoding
    X = pd.get_dummies(X, columns=['DayOfWeek', 'PdDistrict'])
    return X


def train_model(X_train, y_train, X_valid, y_valid):
    model = LGBMClassifier(random_state=SEED)
    model.fit(
        X_train, y_train,
        eval_set=[(X_valid, y_valid)],
        eval_metric='multi_logloss',
        callbacks=[early_stopping(50), log_evaluation(10)]
    )
    return model


def predict(model, X):
    y_pred = model.predict_proba(X)
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['Category'], axis=1)
    y = data_df['Category'].to_numpy()

    # apply preprocessing
    X = preprocess_data(X)
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid)
    labels = np.unique(y)
    score = compute_metrics(y_valid, y_valid_pred, labels)
    print("Final logloss on validation set: ", score)

    # submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['Category'], axis=1)
    X_test = preprocess_data(X_test)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)
