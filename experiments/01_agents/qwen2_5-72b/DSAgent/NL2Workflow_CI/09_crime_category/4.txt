import pandas as pd
from sklearn.metrics import log_loss
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
import lightgbm as lgb

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = log_loss(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    # Define the LightGBM dataset
    train_data = lgb.Dataset(X_train, label=y_train)
    valid_data = lgb.Dataset(X_valid, label=y_valid, reference=train_data)
    
    # Define the parameters for the LightGBM model
    params = {
        'objective': 'multiclass',
        'num_class': len(np.unique(y_train)),
        'metric': 'multi_logloss',
        'boosting_type': 'gbdt',
        'num_leaves': 31,
        'learning_rate': 0.05,
        'feature_fraction': 0.9,
        'bagging_fraction': 0.8,
        'bagging_freq': 5,
        'verbose': 0
    }
    
    # Train the model
    model = lgb.train(params, train_data, valid_sets=[train_data, valid_data], num_boost_round=100, early_stopping_rounds=10)
    return model


def predict(model, X):
    # Make predictions
    y_pred = model.predict(X, num_iteration=model.best_iteration)
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    model = train_model(X_train, y_train, X_valid, y_valid)

    # evaluate the model on the valid set using compute_metrics and print the results
    # should fill out the predict function
    y_valid_pred = predict(model, X_valid)

    # submit predictions for the test set

import pandas as pd
from sklearn.metrics import log_loss
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import LabelEncoder
from submission import submit_predictions_for_test_set
import lightgbm as lgb

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = log_loss(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    # Convert labels to numerical values
    label_encoder = LabelEncoder()
    y_train = label_encoder.fit_transform(y_train)
    y_valid = label_encoder.transform(y_valid)

    # Define the LightGBM dataset
    train_data = lgb.Dataset(X_train, label=y_train)
    valid_data = lgb.Dataset(X_valid, label=y_valid, reference=train_data)
    
    # Define the parameters for the LightGBM model
    params = {
        'objective': 'multiclass',
        'num_class': len(np.unique(y_train)),
        'metric': 'multi_logloss',
        'boosting_type': 'gbdt',
        'num_leaves': 31,
        'learning_rate': 0.05,
        'feature_fraction': 0.9,
        'bagging_fraction': 0.8,
        'bagging_freq': 5,
        'verbose': 0
    }
    
    # Train the model
    model = lgb.train(params, train_data, valid_sets=[train_data, valid_data], num_boost_round=100, callbacks=[lgb.early_stopping(10)])
    return model, label_encoder


def predict(model, X, label_encoder):
    # Make predictions
    y_pred = model.predict(X, num_iteration=model.best_iteration)
    # Convert predictions back to original labels
    y_pred_labels = label_encoder.inverse_transform(np.argmax(y_pred, axis=1))
    return y_pred_labels


if __name__ == '__main__':

    # define and train the model
    model, label_encoder = train_model(X_train, y_train, X_valid, y_valid)

    # evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid, label_encoder)
    score = compute_metrics(y_valid, y_valid_pred)

def preprocess_data(X):
    # Convert categorical columns to numerical using one-hot encoding
    categorical_cols = ['DayOfWeek', 'PdDistrict']
    X = pd.get_dummies(X, columns=categorical_cols)
    return X


def train_model(X_train, y_train, X_valid, y_valid):
    # Define the LightGBM dataset
    lgb_train = lgb.Dataset(X_train, y_train)
    lgb_valid = lgb.Dataset(X_valid, y_valid, reference=lgb_train)
    
    # Define the parameters for the LightGBM model
    params = {
        'objective': 'multiclass',
        'num_class': len(np.unique(y_train)),
        'metric': 'multi_logloss',
        'boosting_type': 'gbdt',
        'verbose': -1
    }
    
    # Train the model
    model = lgb.train(params, lgb_train, valid_sets=[lgb_train, lgb_valid], num_boost_round=100, callbacks=[lgb.early_stopping(10)])
    
    return model


def predict(model, X):
    # Make predictions
    y_pred = model.predict(X, num_iteration=model.best_iteration)
