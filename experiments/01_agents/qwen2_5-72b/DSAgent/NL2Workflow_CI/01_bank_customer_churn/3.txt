import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.preprocessing import StandardScaler
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    # Apply feature scaling
    scaler = StandardScaler()
    X_train_scaled = scaler.fit_transform(X_train)
    X_valid_scaled = scaler.transform(X_valid)

    # Define and train the model
    model = RandomForestClassifier(n_estimators=100, random_state=SEED)
    model.fit(X_train_scaled, y_train)

    return model, scaler


def predict(model, X, scaler):
    # Scale the features
    X_scaled = scaler.transform(X)
    # Predict the model
    y_pred = model.predict_proba(X_scaled)[:, 1]
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['Exited'], axis=1)
    y = data_df['Exited'].to_numpy()

    # Apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model, scaler = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid, scaler)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final auc_roc on validation set: ", score)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['Exited'], axis=1)
    y_test_pred = predict(model, X_test, scaler)
    submit_predictions_for_test_set(y_test_pred)

import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.compose import ColumnTransformer
from lightgbm import LGBMClassifier
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    # Identify categorical and numerical columns
    categorical_features = X_train.select_dtypes(include=['object', 'category']).columns
    numerical_features = X_train.select_dtypes(include=['int64', 'float64']).columns

    # Define the preprocessing for numerical and categorical data
    preprocessor = ColumnTransformer(
        transformers=[
            ('num', StandardScaler(), numerical_features),
            ('cat', OneHotEncoder(handle_unknown='ignore'), categorical_features)
        ]
    )

    # Apply feature scaling and encoding
    X_train_processed = preprocessor.fit_transform(X_train)
    X_valid_processed = preprocessor.transform(X_valid)

    # Define and train the Random Forest model
    rf_model = RandomForestClassifier(n_estimators=100, random_state=SEED)
    rf_model.fit(X_train_processed, y_train)

    # Define and train the LGBM model
    lgbm_model = LGBMClassifier(n_estimators=100, random_state=SEED)
    lgbm_model.fit(X_train_processed, y_train)

    return rf_model, lgbm_model, preprocessor


def predict(models, X, preprocessor):
    # Apply the same preprocessing to the test data
    X_processed = preprocessor.transform(X)
    
    # Predict the model
    y_pred_rf = models[0].predict_proba(X_processed)[:, 1]
    y_pred_lgbm = models[1].predict_proba(X_processed)[:, 1]
    
    # Combine predictions using simple averaging
    y_pred_combined = (y_pred_rf + y_pred_lgbm) / 2.0
    return y_pred_combined


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['Exited'], axis=1)
    y = data_df['Exited'].to_numpy()

    # Apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define and train the models
    rf_model, lgbm_model, preprocessor = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the combined model on the valid set using compute_metrics and print the results
    y_valid_pred = predict((rf_model, lgbm_model), X_valid, preprocessor)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final auc_roc on validation set: ", score)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['Exited'], axis=1)
    y_test_pred = predict((rf_model, lgbm_model), X_test, preprocessor)
    submit_predictions_for_test_set(y_test_pred)
