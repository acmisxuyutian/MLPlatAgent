import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.preprocessing import StandardScaler
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score

def train_model(X_train, y_train, X_valid, y_valid):
    # Feature scaling
    scaler = StandardScaler()
    X_train_scaled = scaler.fit_transform(X_train)
    X_valid_scaled = scaler.transform(X_valid)

    # Define and train the model
    model = RandomForestClassifier(random_state=SEED)
    model.fit(X_train_scaled, y_train)
    return model, scaler

def predict(model, X, scaler):
    # Scale the features
    X_scaled = scaler.transform(X)
    # Predict the probabilities for the positive class
    y_pred = model.predict_proba(X_scaled)[:, 1]
    return y_pred

if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['Exited'], axis=1)
    y = data_df['Exited'].to_numpy()

    # Apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model, scaler = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid, scaler)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final auc_roc on validation set: ", score)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['Exited'], axis=1)
    y_test_pred = predict(model, X_test, scaler)
    submit_predictions_for_test_set(y_test_pred)

import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from lightgbm import LGBMClassifier
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score

def train_model(X_train, y_train, X_valid, y_valid, model_type='random_forest'):
    # Identify categorical and numerical columns
    categorical_features = X_train.select_dtypes(include=['object', 'category']).columns
    numerical_features = X_train.select_dtypes(include=['int64', 'float64']).columns

    # Define the preprocessing for numerical and categorical data
    preprocessor = ColumnTransformer(
        transformers=[
            ('num', StandardScaler(), numerical_features),
            ('cat', OneHotEncoder(handle_unknown='ignore'), categorical_features)
        ])

    # Define the model
    if model_type == 'random_forest':
        model = RandomForestClassifier(random_state=SEED)
    elif model_type == 'lgbm':
        model = LGBMClassifier(random_state=SEED)

    # Create a pipeline that first preprocesses the data and then trains the model
    pipeline = Pipeline(steps=[('preprocessor', preprocessor),
                               ('model', model)])

    # Train the model
    pipeline.fit(X_train, y_train)

    return pipeline

def predict(model, X):
    # Predict the probabilities for the positive class
    y_pred = model.predict_proba(X)[:, 1]
    return y_pred

if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['Exited'], axis=1)
    y = data_df['Exited'].to_numpy()

    # Apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define and train the RandomForest model
    rf_model = train_model(X_train, y_train, X_valid, y_valid, model_type='random_forest')

    # Define and train the LGBM model
    lgbm_model = train_model(X_train, y_train, X_valid, y_valid, model_type='lgbm')

    # Predict the probabilities for the positive class on the validation set using both models
    y_valid_pred_rf = predict(rf_model, X_valid)
    y_valid_pred_lgbm = predict(lgbm_model, X_valid)

    # Combine the predictions by averaging the probabilities
    y_valid_pred_combined = (y_valid_pred_rf + y_valid_pred_lgbm) / 2

    # Compute the AUC-ROC score on the validation set
    score = compute_metrics(y_valid, y_valid_pred_combined)
    print("Final auc_roc on validation set (combined): ", score)

    # If the performance improves, submit the combined predictions for the test set
    if score > 0.5:  # Assuming 0.5 as a baseline, you can adjust this threshold
        submission_df = pd.read_csv('test.csv')
        X_test = submission_df.drop(['Exited'], axis=1)
        y_test_pred_rf = predict(rf_model, X_test)
        y_test_pred_lgbm = predict(lgbm_model, X_test)
        y_test_pred_combined = (y_test_pred_rf + y_test_pred_lgbm) / 2
        submit_predictions_for_test_set(y_test_pred_combined)
        print("Combined predictions submitted for the test set with AUC-ROC score: ", score)
