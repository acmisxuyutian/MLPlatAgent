import pandas as pd
from sklearn.metrics import median_absolute_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.ensemble import RandomForestRegressor

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = median_absolute_error(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid, n_estimators, max_depth):
    model = RandomForestRegressor(n_estimators=n_estimators, max_depth=max_depth, random_state=SEED)
    model.fit(X_train, y_train)
    return model


def predict(model, X):
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    # Convert categorical data to numerical data
    data_df = pd.get_dummies(data_df, columns=['Sex'])

    X = data_df.drop(['Age'], axis=1)
    y = data_df['Age'].to_numpy()

    # apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    best_score = float('inf')
    best_params = None

    # Grid search for hyperparameters
    for n_estimators in range(50, 250, 50):
        for max_depth in range(10, 40, 10):
            model = train_model(X_train, y_train, X_valid, y_valid, n_estimators, max_depth)
            y_valid_pred = predict(model, X_valid)
            score = compute_metrics(y_valid, y_valid_pred)
            print(f"n_estimators: {n_estimators}, max_depth: {max_depth}, MAE: {score}")
            if score < best_score:
                best_score = score
                best_params = {'n_estimators': n_estimators, 'max_depth': max_depth}

    print(f"Best configuration: {best_params}, Best MAE: {best_score}")

    # Train the model with the best configuration
    best_model = train_model(X_train, y_train, X_valid, y_valid, best_params['n_estimators'], best_params['max_depth'])

    # submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    submission_df = pd.get_dummies(submission_df, columns=['Sex'])
    X_test = submission_df.drop(['Age'], axis=1)
    y_test_pred = predict(best_model, X_test)
    submit_predictions_for_test_set(y_test_pred)
