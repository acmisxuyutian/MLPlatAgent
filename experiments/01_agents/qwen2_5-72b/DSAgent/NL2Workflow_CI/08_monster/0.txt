import pandas as pd
from sklearn.metrics import accuracy_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
import lightgbm as lgb

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = accuracy_score(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    model = lgb.LGBMClassifier(
        n_estimators=100,
        learning_rate=0.1,
        num_leaves=31,
        max_depth=5,
        min_data_in_leaf=20,
        feature_fraction=0.9,
        lambda_l1=0.1,
        lambda_l2=0.1,
        verbose=-1,
        force_col_wise=True,
        n_jobs=-1
    )
    model.fit(X_train, y_train, eval_set=[(X_valid, y_valid)], early_stopping_rounds=10, verbose=False)
    return model


def predict(model, X):
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['type'], axis=1)
    y = data_df['type'].to_numpy()

    # apply preprocessing
    X = pd.get_dummies(X)  # Convert categorical columns to numeric using one-hot encoding
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final accuracy on validation set: ", score)

    # submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['type'], axis=1)
    X_test = pd.get_dummies(X_test)  # Convert categorical columns to numeric using one-hot encoding
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)
