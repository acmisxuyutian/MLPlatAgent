import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import OneHotEncoder
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = mean_squared_error(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    model = RandomForestRegressor(random_state=SEED)
    model.fit(X_train, y_train)
    return model


def predict(model, X):
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    # Drop duplicated rows
    data_df.drop_duplicates(inplace=True)

    # Drop specified columns
    columns_to_drop = [
        "ID", "ICO Number", "Owner", "Region", "Certification Contact", "Certification Address", "Farm Name", 
        "Lot Number", "Mill", "ICO Number", "Producer", "Company", "Expiration", "Harvest Year", "Unnamed: 0", 
        "Number of Bags", "Bag Weight", "In-Country Partner", "Country of Origin", "Processing Method", 
        "Grading Date", "Variety", "Status", "Defects", "Uniformity", "Clean Cup", "Sweetness", "Certification Body"
    ]
    data_df.drop(columns_to_drop, axis=1, inplace=True, errors='ignore')

    # One-hot encode the 'Color' column
    encoder = OneHotEncoder(sparse=False, handle_unknown='ignore')
    color_encoded = encoder.fit_transform(data_df[['Color']])
    color_encoded_df = pd.DataFrame(color_encoded, columns=encoder.get_feature_names_out(['Color']))
    data_df = pd.concat([data_df, color_encoded_df], axis=1)
    data_df.drop('Color', axis=1, inplace=True)

    X = data_df.drop(['Total Cup Points'], axis=1)
    y = data_df['Total Cup Points'].to_numpy()

    # Apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final mse on validation set: ", score)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')

    # Apply the same preprocessing steps to the test set
    submission_df.drop_duplicates(inplace=True)
    submission_df.drop(columns_to_drop, axis=1, inplace=True, errors='ignore')
    color_encoded_test = encoder.transform(submission_df[['Color']])
    color_encoded_test_df = pd.DataFrame(color_encoded_test, columns=encoder.get_feature_names_out(['Color']))
    submission_df = pd.concat([submission_df, color_encoded_test_df], axis=1)
    submission_df.drop('Color', axis=1, inplace=True)

    X_test = submission_df.drop(['Total Cup Points'], axis=1, errors='ignore')
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)
