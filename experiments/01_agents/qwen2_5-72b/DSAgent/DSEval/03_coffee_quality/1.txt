===================生成的代码========================:
import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestRegressor
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = mean_squared_error(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid, n_estimators, max_depth):
    model = RandomForestRegressor(n_estimators=n_estimators, max_depth=max_depth, random_state=SEED)
    model.fit(X_train, y_train)
    return model


def predict(model, X):
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    # Drop duplicated rows
    data_df.drop_duplicates(inplace=True)

    # Drop specified columns
    columns_to_drop = [
        "ID", "ICO Number", "Owner", "Region", "Certification Contact", "Certification Address", "Farm Name", 
        "Lot Number", "Mill", "ICO Number", "Producer", "Company", "Expiration", "Harvest Year", "Unnamed: 0", 
        "Number of Bags", "Bag Weight", "In-Country Partner", "Country of Origin", "Processing Method", 
        "Grading Date", "Variety", "Status", "Defects", "Uniformity", "Clean Cup", "Sweetness", "Certification Body"
    ]
    data_df.drop(columns_to_drop, axis=1, inplace=True, errors='ignore')

    # One-hot encode the "Color" column
    data_df = pd.get_dummies(data_df, columns=['Color'])

    X = data_df.drop(['Total Cup Points'], axis=1)
    y = data_df['Total Cup Points'].to_numpy()

    # Apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define the hyperparameters to tune
    n_estimators_values = [50, 100, 200]
    max_depth_values = [10, 20, 30]

    best_mse = float('inf')
    best_model = None
    best_params = None

    # Perform grid search
    for n_estimators in n_estimators_values:
        for max_depth in max_depth_values:
            model = train_model(X_train, y_train, X_valid, y_valid, n_estimators, max_depth)
            y_valid_pred = predict(model, X_valid)
            mse = compute_metrics(y_valid, y_valid_pred)
            print(f"n_estimators: {n_estimators}, max_depth: {max_depth}, MSE: {mse}")
            if mse < best_mse:
                best_mse = mse
                best_model = model
                best_params = {'n_estimators': n_estimators, 'max_depth': max_depth}

    print(f"Best configuration: {best_params}, MSE: {best_mse}")

    # Submit predictions for the test set using the best model
    submission_df = pd.read_csv('test.csv')

    # Apply the same preprocessing to the test set
    submission_df.drop(columns_to_drop, axis=1, inplace=True, errors='ignore')
    submission_df = pd.get_dummies(submission_df, columns=['Color'])

    X_test = submission_df.drop(['Total Cup Points'], axis=1, errors='ignore')
    y_test_pred = predict(best_model, X_test)
    submit_predictions_for_test_set(y_test_pred)
