import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.preprocessing import OneHotEncoder
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    model = RandomForestClassifier(n_estimators=100, random_state=SEED)
    model.fit(X_train, y_train)
    return model


def predict(model, X):
    y_pred = model.predict_proba(X)[:, 1]
    return y_pred


def preprocess_data(df):
    df['YearsInCompany'] = 2023 - df['JoiningYear']
    categorical_cols = df.select_dtypes(include=['object']).columns
    encoder = OneHotEncoder(sparse=False, handle_unknown='ignore')
    encoded_features = encoder.fit_transform(df[categorical_cols])
    encoded_df = pd.DataFrame(encoded_features, columns=encoder.get_feature_names_out(categorical_cols))
    df = df.drop(categorical_cols, axis=1)
    df = pd.concat([df, encoded_df], axis=1)
    return df


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    data_df = preprocess_data(data_df)

    X = data_df.drop(['LeaveOrNot'], axis=1)
    y = data_df['LeaveOrNot'].to_numpy()

    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    model = train_model(X_train, y_train, X_valid, y_valid)

    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final auc_roc on validation set: ", score)

    submission_df = pd.read_csv('test.csv')
    submission_df = preprocess_data(submission_df)
    X_test = submission_df.drop(['LeaveOrNot'], axis=1)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)

import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.preprocessing import OneHotEncoder
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid, n_estimators, max_depth):
    model = RandomForestClassifier(n_estimators=n_estimators, max_depth=max_depth, random_state=SEED)
    model.fit(X_train, y_train)
    y_valid_pred = model.predict_proba(X_valid)[:, 1]
    score = compute_metrics(y_valid, y_valid_pred)
    return model, score


def predict(model, X):
    y_pred = model.predict_proba(X)[:, 1]
    return y_pred


def preprocess_data(df):
    df['YearsInCompany'] = 2023 - df['JoiningYear']
    categorical_cols = df.select_dtypes(include=['object']).columns
    encoder = OneHotEncoder(sparse_output=False, handle_unknown='ignore')
    encoded_features = encoder.fit_transform(df[categorical_cols])
    encoded_df = pd.DataFrame(encoded_features, columns=encoder.get_feature_names_out(categorical_cols))
    df = df.drop(categorical_cols, axis=1)
    df = pd.concat([df, encoded_df], axis=1)
    return df


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    data_df = preprocess_data(data_df)

    X = data_df.drop(['LeaveOrNot'], axis=1)
    y = data_df['LeaveOrNot'].to_numpy()

    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    best_score = 0
    best_model = None
    best_params = {}

    n_estimators_values = [50, 100, 200]
    max_depth_values = [10, 20, 30]

    for n_estimators in n_estimators_values:
        for max_depth in max_depth_values:
            model, score = train_model(X_train, y_train, X_valid, y_valid, n_estimators, max_depth)
            print(f"n_estimators: {n_estimators}, max_depth: {max_depth}, ROC AUC: {score}")
            if score > best_score:
                best_score = score
                best_model = model
                best_params = {'n_estimators': n_estimators, 'max_depth': max_depth}

    print(f"Best parameters: {best_params}, Best ROC AUC: {best_score}")

    submission_df = pd.read_csv('test.csv')
    submission_df = preprocess_data(submission_df)
    X_test = submission_df.drop(['LeaveOrNot'], axis=1)
    y_test_pred = predict(best_model, X_test)
    submit_predictions_for_test_set(y_test_pred)
