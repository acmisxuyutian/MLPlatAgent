import pandas as pd
from sklearn.metrics import roc_auc_score, confusion_matrix
from sklearn.ensemble import RandomForestClassifier
import numpy as np
import random
from sklearn.model_selection import train_test_split

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score

def train_model(X_train, y_train, X_valid, y_valid):
    # Drop duplicate rows
    X_train = X_train.drop_duplicates()
    X_valid = X_valid.drop_duplicates()

    # Perform one-hot encoding on 'gender' and 'smoking_history'
    X_train = pd.get_dummies(X_train, columns=['gender', 'smoking_history'])
    X_valid = pd.get_dummies(X_valid, columns=['gender', 'smoking_history'])

    # Train a random forest classifier
    model = RandomForestClassifier(random_state=SEED)
    model.fit(X_train, y_train)
    return model

def predict(model, X):
    y_pred = model.predict_proba(X)[:, 1]
    return y_pred

if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['diabetes'], axis=1)
    y = data_df['diabetes'].to_numpy()

    # Split the data into training and validation sets
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # Make predictions on the validation set
    y_valid_pred = predict(model, X_valid)

    # Compute the area under the ROC curve
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final auc_roc on validation set: ", score)

    # Show the confusion matrix
    y_valid_pred_binary = (y_valid_pred > 0.5).astype(int)
    cm = confusion_matrix(y_valid, y_valid_pred_binary)
    print("Confusion Matrix:\n", cm)

import pandas as pd
from sklearn.metrics import roc_auc_score, confusion_matrix
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    cm = confusion_matrix(y_test, y_test_pred > 0.5)
    print("Confusion Matrix:")
    print(cm)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    model = RandomForestClassifier(n_estimators=100, random_state=SEED)
    model.fit(X_train, y_train)
    return model


def predict(model, X):
    y_pred = model.predict_proba(X)[:, 1]
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    # Drop duplicate rows
    data_df.drop_duplicates(inplace=True)

    # Perform one-hot encoding on 'gender' and 'smoking_history'
    data_df = pd.get_dummies(data_df, columns=['gender', 'smoking_history'])

    X = data_df.drop(['diabetes'], axis=1)
    y = data_df['diabetes'].to_numpy()

    # Apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final auc_roc on validation set: ", score)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    submission_df = pd.get_dummies(submission_df, columns=['gender', 'smoking_history'])
    X_test = submission_df.drop(['diabetes'], axis=1)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)
