from sklearn.linear_model import LogisticRegression
from sklearn.metrics import roc_auc_score
from sklearn.model_selection import train_test_split
import numpy as np

import torch

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

    # One-hot encode the categorical features
    encoder = OneHotEncoder(handle_unknown='ignore', sparse=False)
    X_train_encoded = encoder.fit_transform(X_train)
    X_valid_encoded = encoder.transform(X_valid)
    return X_train_encoded, X_valid_encoded

def train_model(X_train, y_train, X_valid, y_valid):
    # Train a logistic regression model
    model = LogisticRegression(random_state=SEED)
    model.fit(X_train, y_train)
    return model

def predict(model, X):
    # Predict the probabilities of the positive class
    y_pred = model.predict_proba(X)[:, 1]
    return y_pred

import pandas as pd
from sklearn.metrics import roc_auc_score
from sklearn.linear_model import LogisticRegression
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import OneHotEncoder
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    # One-hot encode the categorical features (except "Disease")
    categorical_features = [col for col in X_train.columns if col != 'Disease' and X_train[col].dtype == 'object']
    encoder = OneHotEncoder(sparse=False, handle_unknown='ignore')
    X_train_encoded = encoder.fit_transform(X_train[categorical_features])
    X_valid_encoded = encoder.transform(X_valid[categorical_features])
    
    # Combine the encoded features with the numerical features
    X_train_final = np.hstack((X_train.drop(categorical_features, axis=1).values, X_train_encoded))
    X_valid_final = np.hstack((X_valid.drop(categorical_features, axis=1).values, X_valid_encoded))
    
    # Train the logistic regression model
    model = LogisticRegression(random_state=SEED, max_iter=1000)
    model.fit(X_train_final, y_train)
    return model, encoder


def predict(model, X, encoder):
    # One-hot encode the categorical features (except "Disease")
    categorical_features = [col for col in X.columns if col != 'Disease' and X[col].dtype == 'object']
    X_encoded = encoder.transform(X[categorical_features])
    
    # Combine the encoded features with the numerical features
    X_final = np.hstack((X.drop(categorical_features, axis=1).values, X_encoded))
    
    # Predict the probabilities
    y_pred = model.predict_proba(X_final)[:, 1]
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['Outcome Variable'], axis=1)
    y = data_df['Outcome Variable'].to_numpy()

    # apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # define and train the model
    model, encoder = train_model(X_train, y_train, X_valid, y_valid)

    # evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid, encoder)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final AUC-ROC on validation set: ", score)

    # submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['Outcome Variable'], axis=1)
    y_test_pred = predict(model, X_test, encoder)
    submit_predictions_for_test_set(y_test_pred)
