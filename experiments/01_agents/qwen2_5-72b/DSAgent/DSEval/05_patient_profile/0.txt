import pandas as pd
from sklearn.metrics import accuracy_score, roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import OneHotEncoder
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred, metric='accuracy'):
    if metric == 'accuracy':
        score = accuracy_score(y_test, y_test_pred)
    elif metric == 'roc_auc':
        score = roc_auc_score(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    # One-hot encode the categorical features (except the "Disease" column)
    categorical_features = X_train.columns.difference(['Disease'])
    encoder = OneHotEncoder(sparse=False, handle_unknown='ignore')
    X_train_encoded = encoder.fit_transform(X_train[categorical_features])
    X_valid_encoded = encoder.transform(X_valid[categorical_features])
    
    # Combine the encoded features with the "Disease" column
    X_train_final = np.hstack((X_train_encoded, X_train[['Disease']].values))
    X_valid_final = np.hstack((X_valid_encoded, X_valid[['Disease']].values))
    
    # Define and train the logistic regression model
    model = LogisticRegression(random_state=SEED, max_iter=1000)
    model.fit(X_train_final, y_train)
    return model, encoder


def predict(model, X, encoder):
    # One-hot encode the categorical features (except the "Disease" column)
    categorical_features = X.columns.difference(['Disease'])
    X_encoded = encoder.transform(X[categorical_features])
    
    # Combine the encoded features with the "Disease" column
    X_final = np.hstack((X_encoded, X[['Disease']].values))
    
    # Predict the model
    y_pred = model.predict(X_final)
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['Outcome Variable'], axis=1)
    y = data_df['Outcome Variable'].to_numpy()

    # Apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model, encoder = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the model on the validation set using accuracy or AUC-ROC as the evaluation metric
    y_valid_pred = predict(model, X_valid, encoder)
    accuracy = compute_metrics(y_valid, y_valid_pred, metric='accuracy')
    roc_auc = compute_metrics(y_valid, y_valid_pred, metric='roc_auc')
    print("Final accuracy on validation set: ", accuracy)
    print("Final AUC-ROC on validation set: ", roc_auc)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['Outcome Variable'], axis=1)
    y_test_pred = predict(model, X_test, encoder)
    submit_predictions_for_test_set(y_test_pred)
