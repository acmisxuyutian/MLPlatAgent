import pandas as pd
from sklearn.metrics import accuracy_score, roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import OneHotEncoder
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    accuracy = accuracy_score(y_test, y_test_pred)
    auc_roc = roc_auc_score(y_test, y_test_pred)
    return accuracy, auc_roc


def train_model(X_train, y_train, X_valid, y_valid):
    # One-hot encode the categorical features (except the "Disease" column)
    categorical_features = X_train.columns.difference(['Disease'])
    encoder = OneHotEncoder(sparse=False, handle_unknown='ignore')
    X_train_encoded = encoder.fit_transform(X_train[categorical_features])
    X_valid_encoded = encoder.transform(X_valid[categorical_features])

    # Combine the encoded features with the "Disease" column
    X_train_final = np.hstack((X_train_encoded, X_train[['Disease']].values))
    X_valid_final = np.hstack((X_valid_encoded, X_valid[['Disease']].values))

    # Define and train the logistic regression model
    model = LogisticRegression(random_state=SEED, max_iter=1000)
    model.fit(X_train_final, y_train)
    return model, encoder


def predict(model, X, encoder):
    # One-hot encode the categorical features (except the "Disease" column)
    categorical_features = X.columns.difference(['Disease'])
    X_encoded = encoder.transform(X[categorical_features])

    # Combine the encoded features with the "Disease" column
    X_final = np.hstack((X_encoded, X[['Disease']].values))

    # Predict the model
    y_pred = model.predict(X_final)
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['Outcome Variable'], axis=1)
    y = data_df['Outcome Variable'].to_numpy()

    # Apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model, encoder = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the model on the validation set using accuracy and AUC-ROC
    y_valid_pred = predict(model, X_valid, encoder)
    accuracy, auc_roc = compute_metrics(y_valid, y_valid_pred)
    print("Final accuracy on validation set: ", accuracy)
    print("Final AUC-ROC on validation set: ", auc_roc)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['Outcome Variable'], axis=1)
    y_test_pred = predict(model, X_test, encoder)
    submit_predictions_for_test_set(y_test_pred)

import pandas as pd
from sklearn.preprocessing import OneHotEncoder
from sklearn.impute import SimpleImputer
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import accuracy_score, roc_auc_score
from sklearn.model_selection import train_test_split
import numpy as np
import random

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

def one_hot_encode_features(X):
    # One-hot encode categorical features
    encoder = OneHotEncoder(drop='first', sparse_output=False)
    categorical_features = X.select_dtypes(include=['object']).columns.tolist()
    X_encoded = pd.DataFrame(encoder.fit_transform(X[categorical_features]), columns=encoder.get_feature_names_out(categorical_features))
    X_encoded = pd.concat([X.drop(categorical_features, axis=1), X_encoded], axis=1)
    return X_encoded

def impute_missing_values(X):
    # Impute missing values
    imputer = SimpleImputer(strategy='mean')
    X_imputed = pd.DataFrame(imputer.fit_transform(X), columns=X.columns)
    return X_imputed

def train_model(X_train, y_train, X_valid, y_valid):
    # Train a logistic regression model
    model = LogisticRegression(random_state=SEED, max_iter=1000)
    model.fit(X_train, y_train)
    return model

def predict(model, X):
    # Predict using the model
    y_pred = model.predict(X)
    y_pred_proba = model.predict_proba(X)[:, 1]
    return y_pred, y_pred_proba

if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['Outcome Variable'], axis=1)
    y = data_df['Outcome Variable'].to_numpy()

    # Apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # One-hot encode the features
    X_train_encoded = one_hot_encode_features(X_train)
    X_valid_encoded = one_hot_encode_features(X_valid)

    # Impute missing values
    X_train_imputed = impute_missing_values(X_train_encoded)
    X_valid_imputed = impute_missing_values(X_valid_encoded)

    # Define and train the model
    model = train_model(X_train_imputed, y_train, X_valid_imputed, y_valid)

    # Evaluate the model on the validation set
    y_valid_pred, y_valid_pred_proba = predict(model, X_valid_imputed)
    accuracy = accuracy_score(y_valid, y_valid_pred)
    auc_roc = roc_auc_score(y_valid, y_valid_pred_proba)
    print("Accuracy on validation set: ", accuracy)
    print("AUC-ROC on validation set: ", auc_roc)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['Outcome Variable'], axis=1)
    X_test_encoded = one_hot_encode_features(X_test)
    X_test_imputed = impute_missing_values(X_test_encoded)
    y_test_pred, _ = predict(model, X_test_imputed)
    submit_predictions_for_test_set(y_test_pred)
