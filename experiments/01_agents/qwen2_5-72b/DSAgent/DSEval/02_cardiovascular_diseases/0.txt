import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    model = LogisticRegression(random_state=SEED, max_iter=1000)
    model.fit(X_train, y_train)
    return model


def predict(model, X):
    y_pred = model.predict_proba(X)[:, 1]
    return y_pred


def preprocess_data(df):
    # Drop rows with missing values
    df = df.dropna()
    # Drop duplicates
    df = df.drop_duplicates()
    
    # Create BMI_Category feature
    conditions = [
        (df['BMI'] < 18.5),
        (df['BMI'] >= 18.5) & (df['BMI'] < 25),
        (df['BMI'] >= 25) & (df['BMI'] < 30),
        (df['BMI'] >= 30)
    ]
    choices = ['Underweight', 'Normal weight', 'Overweight', 'Obesity']
    df['BMI_Category'] = np.select(conditions, choices, default='Unknown')
    
    # One-hot encode the BMI_Category feature
    df = pd.get_dummies(df, columns=['BMI_Category'], drop_first=True)
    
    return df


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    # Preprocess the data
    data_df = preprocess_data(data_df)

    X = data_df.drop(['Heart_Disease'], axis=1)
    y = data_df['Heart_Disease'].to_numpy()

    # Split the data into training and validation sets
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the model on the validation set
    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final auc_roc on validation set: ", score)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    submission_df = preprocess_data(submission_df)
    X_test = submission_df.drop(['Heart_Disease'], axis=1)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)
