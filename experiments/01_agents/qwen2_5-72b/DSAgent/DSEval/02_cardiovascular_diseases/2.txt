import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import OneHotEncoder
from submission import submit_predictions_for_test_set
from sklearn.linear_model import LogisticRegression

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

def preprocess_data(X):
    # Identify categorical columns
    categorical_cols = X.select_dtypes(include=['object']).columns
    # Initialize OneHotEncoder
    encoder = OneHotEncoder(sparse=False, handle_unknown='ignore')
    # Fit and transform the categorical columns
    X_encoded = encoder.fit_transform(X[categorical_cols])
    # Create a DataFrame from the encoded data
    X_encoded_df = pd.DataFrame(X_encoded, columns=encoder.get_feature_names_out(categorical_cols))
    # Drop the original categorical columns
    X = X.drop(categorical_cols, axis=1)
    # Concatenate the encoded DataFrame with the original DataFrame
    X = pd.concat([X, X_encoded_df], axis=1)
    return X

def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score

def train_model(X_train, y_train, X_valid, y_valid):
    model = LogisticRegression(random_state=SEED)
    model.fit(X_train, y_train)
    return model

def predict(model, X):
    y_pred = model.predict_proba(X)[:, 1]
    return y_pred

if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['Heart_Disease'], axis=1)
    y = data_df['Heart_Disease'].to_numpy()

    # Preprocess the data
    X = preprocess_data(X)

    # Apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final auc_roc on validation set: ", score)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['Heart_Disease'], axis=1)
    X_test = preprocess_data(X_test)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)

import pandas as pd

# Load the training and test datasets
train_data = pd.read_csv('train.csv')
test_data = pd.read_csv('test.csv')

# Step 1: Drop all rows with missing values
train_data.dropna(inplace=True)
test_data.dropna(inplace=True)

# Step 2: Drop duplicate rows
train_data.drop_duplicates(inplace=True)
test_data.drop_duplicates(inplace=True)

# Step 3: Construct the BMI_Category feature
def categorize_bmi(bmi):
    if bmi < 18.5:
        return 'Underweight'
    elif 18.5 <= bmi < 24.9:
        return 'Normal weight'
    elif 25 <= bmi < 29.9:
        return 'Overweight'
    else:
        return 'Obesity'

train_data['BMI_Category'] = train_data['BMI'].apply(categorize_bmi).astype('category')
test_data['BMI_Category'] = test_data['BMI'].apply(categorize_bmi).astype('category')

# Save the preprocessed datasets
train_data.to_csv('train_preprocessed.csv', index=False)
test_data.to_csv('test_preprocessed.csv', index=False)
