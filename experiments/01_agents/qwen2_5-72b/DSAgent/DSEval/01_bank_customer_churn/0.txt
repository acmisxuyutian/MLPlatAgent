import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import xgboost as xgb
from sklearn.model_selection import train_test_split

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

def preprocess_data(df):
    # Drop irrelevant columns
    df = df.drop(['CustomerId', 'Complain', 'Surname'], axis=1)
    # One-hot encode categorical columns
    df = pd.get_dummies(df, columns=['Geography', 'Card Type'])
    return df

def train_model(X_train, y_train, X_valid, y_valid):

    model = xgb.XGBClassifier(use_label_encoder=False, eval_metric='logloss', random_state=SEED)
    model.fit(X_train, y_train)
    return model

def predict(model, X):
    # Predict probabilities for the positive class
    y_pred = model.predict_proba(X)[:, 1]
    return y_pred

import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import xgboost as xgb
from sklearn.model_selection import train_test_split, GridSearchCV
from sklearn.preprocessing import LabelEncoder

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

def preprocess_data(df):
    # Drop irrelevant columns
    df = df.drop(['CustomerId', 'Complain', 'Surname'], axis=1)
    
    # Encode the 'Gender' column
    le = LabelEncoder()
    df['Gender'] = le.fit_transform(df['Gender'])
    
    # One-hot encode categorical columns
    df = pd.get_dummies(df, columns=['Geography', 'Card Type'])
    return df

def train_model(X_train, y_train, X_valid, y_valid):
    # Define the parameter grid for hyperparameter tuning
    param_grid = {
        'max_depth': [3, 5, 7, 9],
        'learning_rate': [0.01, 0.1, 0.2, 0.3]
    }
    
    # Initialize the XGBoost model
    model = xgb.XGBClassifier(use_label_encoder=False, eval_metric='logloss', random_state=SEED)
    
    # Perform grid search with cross-validation
    grid_search = GridSearchCV(estimator=model, param_grid=param_grid, scoring='roc_auc', cv=3, n_jobs=-1)
    grid_search.fit(X_train, y_train)
    
    # Get the best model
    best_model = grid_search.best_estimator_
    
    # Evaluate the best model on the validation set
    y_valid_pred = best_model.predict_proba(X_valid)[:, 1]
    score = roc_auc_score(y_valid, y_valid_pred)
    print(f"Best parameters: {grid_search.best_params_}")
    print(f"Best ROC AUC score on validation set: {score}")
    
    return best_model

def predict(model, X):
    # Predict probabilities for the positive class
    y_pred = model.predict_proba(X)[:, 1]
    return y_pred

def submit_predictions_for_test_set(y_test_pred):
    # Create a submission DataFrame
    submission_df = pd.read_csv('test.csv')
    submission_df['Exited'] = y_test_pred
    submission_df[['RowNumber', 'Exited']].to_csv('submission.csv', index=False)
    print("Predictions submitted successfully.")

if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')
    data_df = preprocess_data(data_df)

    X = data_df.drop(['Exited'], axis=1)
    y = data_df['Exited'].to_numpy()

    # Split the data into training and validation sets
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # Prepare and make predictions on the test set
    submission_df = pd.read_csv('test.csv')
    submission_df = preprocess_data(submission_df)
    X_test = submission_df.drop(['Exited'], axis=1)
    y_test_pred = predict(model, X_test)

    # Submit predictions for the test set
    submit_predictions_for_test_set(y_test_pred)
