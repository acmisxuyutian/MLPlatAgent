import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import xgboost as xgb
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

def preprocess_data(df):
    # Drop irrelevant columns
    df = df.drop(['CustomerId', 'Complain', 'Surname'], axis=1)
    # One-hot encode categorical columns
    df = pd.get_dummies(df, columns=['Geography', 'Card Type'])
    return df

def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score

def train_model(X_train, y_train, X_valid, y_valid):
    # Define the XGBoost model with basic hyperparameters
    model = xgb.XGBClassifier(max_depth=3, learning_rate=0.1, n_estimators=100, random_state=SEED)
    # Train the model
    model.fit(X_train, y_train, eval_set=[(X_valid, y_valid)], early_stopping_rounds=10, verbose=False)
    return model

def predict(model, X):
    # Predict probabilities for the positive class
    y_pred = model.predict_proba(X)[:, 1]
    return y_pred

if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')
    data_df = preprocess_data(data_df)

    X = data_df.drop(['Exited'], axis=1)
    y = data_df['Exited'].to_numpy()

    # Split the data into training and validation sets
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the model on the validation set
    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final auc_roc on validation set: ", score)

    # Make predictions on the test set and submit the results
    submission_df = pd.read_csv('test.csv')
    submission_df = preprocess_data(submission_df)
    X_test = submission_df.drop(['Exited'], axis=1)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)

import pandas as pd
from sklearn.metrics import roc_auc_score
import numpy as np
import random
import xgboost as xgb
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
np.random.seed(SEED)

def preprocess_data(df):
    # Drop irrelevant columns
    df = df.drop(['CustomerId', 'Complain', 'Surname'], axis=1)
    # One-hot encode categorical columns
    df = pd.get_dummies(df, columns=['Geography', 'Card Type'])
    # Convert 'Gender' column to categorical and then to integer
    df['Gender'] = df['Gender'].astype('category').cat.codes
    return df

def compute_metrics(y_test, y_test_pred):
    score = roc_auc_score(y_test, y_test_pred)
    return score

def train_model(X_train, y_train, X_valid, y_valid, max_depth, learning_rate):
    # Define the XGBoost model with specified hyperparameters
    model = xgb.XGBClassifier(max_depth=max_depth, learning_rate=learning_rate, n_estimators=100, random_state=SEED)
    # Train the model
    model.fit(X_train, y_train, eval_set=[(X_valid, y_valid)], verbose=False)
    return model

def predict(model, X):
    # Predict probabilities for the positive class
    y_pred = model.predict_proba(X)[:, 1]
    return y_pred

if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')
    data_df = preprocess_data(data_df)

    X = data_df.drop(['Exited'], axis=1)
    y = data_df['Exited'].to_numpy()

    # Split the data into training and validation sets
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define the grid of hyperparameters to search
    max_depths = [3, 4, 5]
    learning_rates = [0.05, 0.1, 0.2]

    best_score = 0
    best_params = None
    best_model = None

    # Perform grid search
    for max_depth in max_depths:
        for learning_rate in learning_rates:
            # Train the model with the current hyperparameters
            model = train_model(X_train, y_train, X_valid, y_valid, max_depth, learning_rate)
            # Evaluate the model on the validation set
            y_valid_pred = predict(model, X_valid)
            score = compute_metrics(y_valid, y_valid_pred)
            print(f"max_depth: {max_depth}, learning_rate: {learning_rate}, auc_roc: {score}")
            # Update the best model and hyperparameters if the current score is better
            if score > best_score:
                best_score = score
                best_params = {'max_depth': max_depth, 'learning_rate': learning_rate}
                best_model = model

    print(f"Best hyperparameters: {best_params}, Best auc_roc: {best_score}")

    # Make predictions on the test set and submit the results
    submission_df = pd.read_csv('test.csv')
    submission_df = preprocess_data(submission_df)
    X_test = submission_df.drop(['Exited'], axis=1, errors='ignore')
    y_test_pred = predict(best_model, X_test)
    submit_predictions_for_test_set(y_test_pred)
