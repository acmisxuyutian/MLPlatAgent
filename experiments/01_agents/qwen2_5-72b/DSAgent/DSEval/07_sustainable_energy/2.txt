import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = mean_squared_error(y_test, y_test_pred, squared=False)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    model = LinearRegression()
    model.fit(X_train, y_train)
    return model


def predict(model, X):
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['Renewable energy share in the total final energy consumption (%)'], axis=1)
    y = data_df['Renewable energy share in the total final energy consumption (%)'].to_numpy()

    # apply preprocessing
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final rmse on validation set: ", score)

    # submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['Renewable energy share in the total final energy consumption (%)'], axis=1)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)

import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import OneHotEncoder
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = mean_squared_error(y_test, y_test_pred, squared=False)
    return score


def preprocess_data(X):
    # Identify categorical columns
    categorical_cols = X.select_dtypes(include=['object']).columns
    # Apply one-hot encoding to categorical columns
    encoder = OneHotEncoder(sparse=False, handle_unknown='ignore')
    X_encoded = encoder.fit_transform(X[categorical_cols])
    X_encoded_df = pd.DataFrame(X_encoded, columns=encoder.get_feature_names_out(categorical_cols))
    # Combine the encoded categorical columns with the numeric columns
    X_numeric = X.drop(columns=categorical_cols)
    X_processed = pd.concat([X_numeric, X_encoded_df], axis=1)
    return X_processed


def train_model(X_train, y_train, X_valid, y_valid):
    # Define and train the linear regression model
    model = LinearRegression()
    model.fit(X_train, y_train)
    return model


def predict(model, X):
    # Make predictions using the trained model
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['Renewable energy share in the total final energy consumption (%)'], axis=1)
    y = data_df['Renewable energy share in the total final energy consumption (%)'].to_numpy()

    # Apply preprocessing
    X = preprocess_data(X)
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the model on the validation set using compute_metrics and print the results
    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final rmse on validation set: ", score)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['Renewable energy share in the total final energy consumption (%)'], axis=1)
    X_test = preprocess_data(X_test)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)
