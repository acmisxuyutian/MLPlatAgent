import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = mean_squared_error(y_test, y_test_pred, squared=False)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    model = LinearRegression()
    model.fit(X_train, y_train)
    return model


def predict(model, X):
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    X = data_df.drop(['Renewable energy share in the total final energy consumption (%)'], axis=1)
    y = data_df['Renewable energy share in the total final energy consumption (%)'].to_numpy()

    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    model = train_model(X_train, y_train, X_valid, y_valid)

    y_valid_pred = predict(model, X_valid)
    score = compute_metrics(y_valid, y_valid_pred)
    print("Final rmse on validation set: ", score)

    submission_df = pd.read_csv('test.csv')
    X_test = submission_df.drop(['Renewable energy share in the total final energy consumption (%)'], axis=1)
    y_test_pred = predict(model, X_test)
    submit_predictions_for_test_set(y_test_pred)

import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.impute import SimpleImputer
from submission import submit_predictions_for_test_set

SEED = 42
random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)


def compute_metrics(y_test, y_test_pred):
    score = mean_squared_error(y_test, y_test_pred, squared=False)
    return score


def train_model(X_train, y_train, X_valid, y_valid):
    # Impute missing values with the mean of each column
    imputer = SimpleImputer(strategy='mean')
    X_train_imputed = imputer.fit_transform(X_train)
    X_valid_imputed = imputer.transform(X_valid)

    model = LinearRegression()
    model.fit(X_train_imputed, y_train)
    return model, imputer


def predict(model, imputer, X):
    X_imputed = imputer.transform(X)
    y_pred = model.predict(X_imputed)
    return y_pred


if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    # Drop non-numeric columns
    non_numeric_columns = data_df.select_dtypes(exclude=['number']).columns

    X = X.drop(['Renewable energy share in the total final energy consumption (%)'], axis=1)
    y = data_df['Renewable energy share in the total final energy consumption (%)'].to_numpy()

    # apply preprocessing

    # define and train the model

    # evaluate the model on the valid set using compute_metrics and print the results
    y_valid_pred = predict(model, imputer, X_valid)

    submit_predictions_for_test_set(y_test_pred)
